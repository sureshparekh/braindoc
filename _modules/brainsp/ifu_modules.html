Source code for brainsp.ifu_modules

"""
Pre-pass processing modules for BRAIN's IFU mode.

A module declares its own default configuration, is handed the open HDU list
of one datacube, and returns the HDUs it produced. The runner appends them to
that same list, so the list accumulates into a megacube. Configuration layers
module defaults, then the constructor keywords, then the per-target overrides
a targets CSV supplies, and a key that was never declared as a default is
ignored rather than silently creating a setting nothing reads.

The output extension names are those of the MaNGA megacubes of Riffel et
al. (2023, MNRAS 524, 5640), so that tooling written against those megacubes
reads a BRAIN megacube unchanged. Three choices are deliberate:
dereddening uses BRAIN's own twelve-curve registry in reddening_laws.py and applies the extinction the
user asked for rather than a curve's internal R_V; the signal-to-noise map is
the mean per-pixel signal-to-noise, which is dimensionally a ratio; and the
wavelength axis is read from a WAVE extension when one is present, because
reconstructing it linearly from CD3_3 is wrong on any cube whose third axis
is not linear.

Degrading the resolution of the data is provided and warns every time it runs.
"""

import logging
import warnings

import numpy as np
from astropy.io import fits
from scipy.interpolate import interp1d

from brainsp.reddening_catalogue import LAWS

logger = logging.getLogger("BRAIN_IFU")

FWHM_CONST = 2.35482
KERNEL_TRUNCATION_SIGMA = 20.0

DATA_TYPES = ("flux", "inv_flux", "error", "inv_error",
              "variance", "inv_variance")

class IFUModule:
    """
    Base class for a pre-pass module.

    Subclasses set `name`, fill `self.default_parameters` inside
    `_set_default_parameters`, and implement `execute`. `progress_hook` is set
    by the runner to a callable taking a fraction and a message; modules that
    take a noticeable time call it, and the default does nothing so a module
    can always call it.

    `spaxel_hook` is the same idea one level finer: a stage that works through
    a cube spaxel by spaxel calls it with the flat indices it has just
    finished and the spatial shape those indices address, so that an interface
    can show the field filling in rather than a bar advancing. Only the fit
    has anything to say through it, and nothing in the pipeline depends on
    anybody listening.
    """

    name = "module"

    def __init__(self, **kwargs):
        self.default_parameters = {}
        self.received_config = dict(kwargs)
        self._set_default_parameters()
        self._config = dict(self.default_parameters)
        self.setup(**kwargs)
        self.progress_hook = lambda fraction, message="": None
        self.spaxel_hook = lambda indices, n_y, n_x: None

    @property
    def config(self):
        return dict(self._config)

    def __getitem__(self, key):
        return self._config[key]

def wavelength_axis(hdu_list, ext_name):
    """
    The wavelength of every plane of an extension, in Angstrom.

    A WAVE extension of the right length is believed in preference to the
    world coordinate system, because it is the only description that is right
    whatever the sampling. Reconstructing the axis as
    `CRVAL3 + k * CD3_3` is correct only for a linearly sampled cube, and a
    MaNGA LOGCUBE is not one: the reconstruction ends roughly 1000 A short at
    the red end, which quietly drags H-alpha and the red sky forest into a
    window the user asked to be continuum.

    Without a WAVE extension the axis is built from the header. `CTYPE3`
    decides how: a name ending in `-LOG` is logarithmic, and the two
    conventions in circulation are told apart by the size of `CRVAL3` --
    a reference wavelength in Angstrom for the FITS form, a base-ten
    logarithm for the IRAF form.
    """
    data = hdu_list[ext_name].data
    if data is None or np.ndim(data) != 3:
        raise ValueError(f"extension {ext_name!r} does not hold a datacube")
    n_wave = data.shape[0]

    if "WAVE" in hdu_list:
        wave = np.asarray(hdu_list["WAVE"].data, dtype=np.float64)
        if wave.ndim == 1 and len(wave) == n_wave:
            return wave

    header = hdu_list[ext_name].header
    delta_key = "CDELT3" if "CDELT3" in header else "CD3_3"
    if delta_key not in header or "CRVAL3" not in header:
        raise ValueError(
            f"extension {ext_name!r} carries no wavelength axis: there is no "
            f"WAVE extension of length {n_wave} and its header has no "
            f"CRVAL3/{delta_key} to build one from.")

    delta = float(header[delta_key])
    crval = float(header["CRVAL3"])
    crpix = float(header.get("CRPIX3", 1.0))
    offsets = np.arange(n_wave, dtype=np.float64) - (crpix - 1.0)

    ctype = str(header.get("CTYPE3", "")).upper().strip()
    if ctype.endswith("-LOG") or ctype == "LOG":
        if crval > 10.0:
            return crval * np.exp(offsets * delta / crval)
        return 10.0 ** (crval + offsets * delta)
    return crval + offsets * delta

def to_sigma_like(data, data_type):
    """
    Map any of the six data types onto the quantity that combines linearly.

    A flux adds like a flux and a one-sigma error adds like a flux; an inverse
    variance does not. Every module that averages, sums or convolves planes
    therefore converts to flux or to sigma first, combines, and converts back.
    Non-finite values survive the round trip as NaN so that the combination
    step can decide what to do with them, rather than having them turn into
    zeros here where nothing can tell them from real zeros.
    """
    if data_type not in DATA_TYPES:
        raise ValueError(f"unknown data type {data_type!r}; expected one of "
                         + ", ".join(DATA_TYPES))
    out = np.asarray(data, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        if data_type.startswith("inv_"):
            out = 1.0 / out
        if "variance" in data_type:
            out = np.sqrt(out)
    return np.where(np.isfinite(out), out, np.nan)

def from_sigma_like(data, data_type):
    """
    The inverse of to_sigma_like.

    An inverse-variance or inverse-error plane that comes back non-finite is
    written as zero, because zero is what an inverse variance means when there
    is no information at that pixel, and it is what MaNGA already writes there.
    A flux-like plane keeps its NaN, because there zero is a measurement.
    """
    out = np.asarray(data, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        if "variance" in data_type:
            out = out ** 2.0
        if data_type.startswith("inv_"):
            out = 1.0 / out
            out = np.where(np.isfinite(out), out, 0.0)
    return out

def inverse_variance(data, data_type):
    """
    A statistical plane expressed as the inverse variance the fit weights with.

    A cube may declare its uncertainty as an error, a variance or an
    inverse variance, and MUSE ships a variance plane where MaNGA ships an
    inverse variance, so a fitter that reads only one of them cannot fit half
    the instruments in use. The conversion is done once, here, on the way into
    the fit, so nothing downstream has to know which kind the file held.

    An inverse variance is returned untouched rather than round-tripped
    through sigma: the round trip is not the identity in floating point, and
    the weights the fit uses must be the numbers the file holds. For the other
    kinds a pixel that comes out non-finite is given zero, which is what an
    inverse variance means where there is no information and what MaNGA
    already writes there; the fit drops those pixels either way.

    'flux' and 'inv_flux' are not uncertainties and are refused rather than
    quietly interpreted as one.
    """
    if data_type in ("flux", "inv_flux"):
        raise ValueError(
            f"{data_type!r} is not a statistical plane, so it cannot be read "
            f"as an uncertainty. Expected one of error, inv_error, variance, "
            f"inv_variance.")
    out = np.asarray(data, dtype=np.float64)
    if data_type == "inv_variance":
        return out
    return from_sigma_like(to_sigma_like(out, data_type), "inv_variance")




def _stamp(header, module_name):
    header["BRNMOD"] = (module_name[:68], "BRAIN IFU module that wrote this")
    return header


def _as_written(values, like):
    """
    A processed plane cast back to the precision the input was stored in.

    Everything here is computed in float64, which is the rule BRAIN's
    numerical core is held to. What gets written is a different question: a
    cube stored as float32 carries no more than float32 of information, and
    a pre-pass stage that doubles the size of every extension it touches
    makes a megacube twice as expensive to store and to read for precision
    that was never in the data. So the output matches the input, and a
    float64 input stays float64.
    """
    dtype = np.asarray(like).dtype
    if dtype.kind != "f":
        dtype = np.dtype(np.float64)
    return np.asarray(values, dtype=np.float64).astype(dtype)

def require_extension(hdu_list, ext_name, what):
    """
    The named extension, or an error naming what is actually in the file.

    A missing extension is the failure this pipeline makes easiest to reach:
    stages rename what they produce, so one wrong name in a targets CSV column
    surfaces as a KeyError from astropy several stages later with nothing in
    it to say which stage asked or what it could have asked for instead.
    """
    try:
        return hdu_list[ext_name]
    except KeyError:
        available = [h.name for h in hdu_list if h.name]
        raise KeyError(
            f"{what} names extension {ext_name!r}, which this file does not "
            f"have. It holds: " + ", ".join(available)) from None

class Dereddening(IFUModule):
    """
    Foreground extinction removed from the flux cube. Writes `FLUX_DRD`.

    The curve comes from BRAIN's own registry, so all twelve laws the fitter
    can use are available here too, and the one used is named in `DRD_LAW`
    beside `DRD_EBV` and `DRD_RV`. Every curve in that registry is
    k(lambda) = A_lambda / A_V, so the correction is
    10 ** (0.4 * A_V * k), with A_V = R_V * E(B-V) taken from the settings
    given rather than from a value buried in the curve: a run that writes
    R_V = 3.1 into its header has had R_V = 3.1 applied to it.

    The starburst curves CAL, CCC and HZ5 were derived with R_V = 4.05 and
    that is the value to pair them with; the module does not impose it,
    because silently overriding the user's R_V while still recording it is
    exactly the trap this note exists to avoid.

    Configuration: `hdu flux` (FLUX), `ebv` (0.0), `rv` (3.1), `law` (CCM).

    The curve is the only thing in this file that is evaluated in JAX, so it
    and the double-precision switch it needs are imported here rather than at
    the top of the module. Every other stage is numpy and scipy, and a module
    list that started a JAX backend merely by being described would put one in
    the interface process, where the device has not been chosen yet.
    """

    name = "Dereddening"

    def _set_default_parameters(self):
        self.default_parameters["hdu flux"] = "FLUX"
        self.default_parameters["ebv"] = 0.0
        self.default_parameters["rv"] = 3.1
        self.default_parameters["law"] = "CCM"



[docs]
    def execute(self, hdu_list):
        import jax
        jax.config.update("jax_enable_x64", True)
        from brainsp.reddening_laws import get_k_lambda

        ext = self["hdu flux"]
        law = str(self["law"]).upper().strip()
        if law not in LAWS:
            raise ValueError(f"unknown extinction law {self['law']!r}; "
                             f"available: " + ", ".join(LAWS))

        ebv = float(self["ebv"])
        rv = float(self["rv"])
        wavelength = wavelength_axis(hdu_list, ext)
        k_lambda = np.asarray(get_k_lambda(wavelength, law, rv=rv), dtype=np.float64)
        a_lambda = rv * ebv * k_lambda
        correction = 10.0 ** (0.4 * a_lambda)

        source = require_extension(hdu_list, ext, "'hdu flux'")
        flux = np.asarray(source.data, dtype=np.float64)
        deredded = flux * correction[:, None, None]

        hdu = fits.ImageHDU(data=_as_written(deredded, source.data),
                            header=source.header.copy())
        hdu.header["EXTNAME"] = "FLUX_DRD"
        hdu.header["DRD_EBV"] = (ebv, "E(B-V) removed")
        hdu.header["DRD_RV"] = (rv, "R_V used to turn E(B-V) into A_V")
        hdu.header["DRD_LAW"] = (law, LAWS[law][0][:60])
        _stamp(hdu.header, self.name)

        logger.info(f"Dereddening: {law}, E(B-V) = {ebv:g}, R_V = {rv:g}, "
                    f"A_V = {rv * ebv:.4f} mag -> FLUX_DRD")
        return fits.HDUList([hdu])

class ButterworthFilter(IFUModule):
    """
    A low-pass Butterworth filter over the two spatial axes. Writes `FLUX_BW`.

    The transfer function is H = 1 - 1 / (1 + (range / r) ** (2 * order)) on
    the radius normalised to its largest value, with the zero frequency rolled
    to the corner to match the fast transform's ordering. Pixels that are not
    finite are set to zero before the transform, because a single NaN
    propagates to the whole plane, and restored to NaN afterwards.

    Configuration: `hdu flux` (FLUX), `order` (3), `range` (0.3).

    This filter mixes neighbouring spaxels. Any chi-squared computed after it
    is assuming pixel noise that the filter has correlated, so it is logged as
    a warning every time it runs, like the resolution degrader.

    It is spatial only: nothing is done along the wavelength axis. At the
    default order and range the kernel spreads one spaxel's light over about
    six of its neighbours -- a point source keeps roughly a sixth of its peak
    -- and that width is set by `range` in spaxels, not by the size of the
    field. On a field only a few spaxels across it is therefore the whole
    field, and light that lands on a spaxel which was not finite is dropped
    rather than returned, so a small cutout can come out of here at a small
    fraction of the flux it went in with. The fraction that survived is
    measured, written to `BW_KEPT` and logged, because a stage that quietly
    scales the data is exactly the kind of thing this code exists to catch.
    """

    name = "Butterworth filter"

    def _set_default_parameters(self):
        self.default_parameters["hdu flux"] = "FLUX"
        self.default_parameters["order"] = 3
        self.default_parameters["range"] = 0.3



[docs]
    def execute(self, hdu_list):
        ext = self["hdu flux"]
        order = float(self["order"])
        radius_range = float(self["range"])
        source = require_extension(hdu_list, ext, "'hdu flux'")
        data = np.asarray(source.data, dtype=np.float64)
        if data.ndim != 3:
            raise ValueError(f"{ext!r} must be a datacube to filter")
        _, n_y, n_x = data.shape

        logger.warning(
            "Butterworth filter: this smooths the DATA across neighbouring "
            "spaxels. Their noise is correlated afterwards, and every "
            "chi-squared BRAIN computes assumes it is not, so the error bars "
            "that follow are optimistic by an unquantified amount.")

        radial = _radial_matrix(n_y, n_x)
        transfer = 1.0 - 1.0 / (1.0 + (radius_range / radial) ** (2.0 * order))
        transfer = np.roll(transfer, -int(n_y / 2), axis=0)
        transfer = np.roll(transfer, -int(n_x / 2), axis=1)

        finite = np.isfinite(data)
        clean = np.where(finite, data, 0.0)
        filtered = np.fft.ifft2(np.fft.fft2(clean, axes=(2, 1)) * transfer).real
        filtered[~finite] = np.nan

        before = float(np.sum(np.abs(clean)))
        after = float(np.sum(np.abs(np.where(finite, filtered, 0.0))))
        kept = after / before if before > 0.0 else 1.0
        if kept < 0.8:
            logger.warning(
                f"Butterworth filter: {kept * 100:.1f} per cent of the flux "
                f"survived it on a {n_x} x {n_y} field. The kernel is a few "
                f"spaxels wide whatever the size of the field, and light it "
                f"spreads onto a spaxel that was not finite is discarded, so "
                f"a small or mostly blank field loses most of its flux here. "
                f"Whatever runs next sees the reduced flux, not the cube's.")

        hdu = fits.ImageHDU(data=_as_written(filtered, source.data),
                            header=source.header.copy())
        hdu.header["EXTNAME"] = "FLUX_BW"
        hdu.header["BW_ORDER"] = (order, "Butterworth order")
        hdu.header["BW_RANGE"] = (radius_range, "Butterworth range")
        hdu.header["BW_KEPT"] = (round(kept, 6),
                                 "fraction of |flux| surviving the filter")
        _stamp(hdu.header, self.name)
        self.summary = {"butterworth kept": kept}
        logger.info(f"Butterworth: order {order:g}, range {radius_range:g}, "
                    f"{kept * 100:.1f} per cent of the flux kept -> FLUX_BW")
        return fits.HDUList([hdu])






def _radial_matrix(n_y, n_x):
    """Distance from the centre of a plane, normalised to its largest value."""
    x_shift = 0.5 if n_x % 2 == 0 else 0.0
    y_shift = 0.5 if n_y % 2 == 0 else 0.0
    xs = (np.arange(n_x) - n_x / 2.0 + x_shift) ** 2
    ys = (np.arange(n_y) - n_y / 2.0 + y_shift) ** 2
    radial = np.sqrt(ys[:, None] + xs[None, :])
    return radial / radial.max()

class SpectralResampler(IFUModule):
    """
    A cube put onto a uniform wavelength step. Writes `<EXTNAME>_BIN`.

    The new grid runs from the first wavelength in steps of `resample size`
    Angstrom, and the header it carries describes that grid: `CRPIX3` 1,
    `CRVAL3` the first wavelength, and the step in `CDELT3` or `CD3_3`,
    whichever the input used. `RSP_SIZE` records the step.

    Resampling is linear interpolation over the pixels that are finite, which
    is what it takes to reproduce an existing analysis. It is lossy and
    it is not what BRAIN's fit does: the fit puts the flux onto its log grid
    with a flux-conserving integral, from whatever grid it is given. Run this
    stage when you are matching someone else's processing, not to help the
    fit.

    Configuration: `hdu target` (FLUX), `data type` (flux),
    `resample size` (1.0).
    """

    name = "Spectral resampler"

    def _set_default_parameters(self):
        self.default_parameters["hdu target"] = "FLUX"
        self.default_parameters["data type"] = "flux"
        self.default_parameters["resample size"] = 1.0



[docs]
    def execute(self, hdu_list):
        ext = self["hdu target"]
        data_type = str(self["data type"])
        step = float(self["resample size"])
        if step <= 0:
            raise ValueError("'resample size' must be greater than zero")

        source = require_extension(hdu_list, ext, "'hdu target'")
        wave = wavelength_axis(hdu_list, ext)
        data = to_sigma_like(source.data, data_type)
        n_wave, n_y, n_x = data.shape

        new_wave = np.arange(wave[0], wave[-1], step)
        flat = data.reshape(n_wave, n_y * n_x)
        out = np.zeros((len(new_wave), n_y * n_x), dtype=np.float64)
        for column in range(flat.shape[1]):
            spectrum = flat[:, column]
            good = np.isfinite(spectrum)
            if good.sum() < 2:
                continue
            out[:, column] = interp1d(wave[good], spectrum[good],
                                      fill_value="extrapolate")(new_wave)
            if column % 128 == 0:
                self.progress_hook(column / max(flat.shape[1], 1),
                                   "resampling in wavelength")
        out = from_sigma_like(out.reshape(len(new_wave), n_y, n_x), data_type)

        header = source.header.copy()
        source_name = str(header.get("EXTNAME", ext))
        hdu = fits.ImageHDU(data=_as_written(out, source.data), header=header)
        hdu.header["EXTNAME"] = f"{source_name}_BIN"
        hdu.header["RSP_SIZE"] = (step, "Wavelength step, Angstrom")
        delta_key = "CDELT3" if "CDELT3" in header else "CD3_3"
        hdu.header[delta_key] = step
        hdu.header["CRPIX3"] = 1
        hdu.header["CRVAL3"] = float(new_wave[0])
        hdu.header["CTYPE3"] = "WAVE"
        _stamp(hdu.header, self.name)

        logger.info(f"Spectral resampling: {source_name} {n_wave} -> "
                    f"{len(new_wave)} pixels at {step:g} A "
                    f"-> {source_name}_BIN")
        return fits.HDUList([hdu])

class SpatialResampler(IFUModule):
    """
    Square blocks of spaxels summed into one. Writes `<EXTNAME>_RSP`.

    Each output spaxel is the sum over an N by N block, taken as a NaN-
    tolerant mean multiplied by N on each axis so that a block with missing
    spaxels is scaled up to a full block rather than being biased low. A block
    with no finite spaxel at all has nothing to scale up and stays NaN; the
    empty-slice warning numpy raises for it is expected and is silenced, since
    the NaN it returns is the right answer and carries the same information.
    Blocks that do not fit whole into the array are dropped, so
    the output is floor(n / N) on each axis.

    A data-quality mask cannot be carried through this stage, since a flag is
    not an additive quantity; the runner drops the mask and says so rather
    than passing on one that no longer lines up.

    Configuration: `hdu target` (FLUX), `data type` (flux),
    `resample size` (4).
    """

    name = "Spatial resampler"

    def _set_default_parameters(self):
        self.default_parameters["hdu target"] = "FLUX"
        self.default_parameters["data type"] = "flux"
        self.default_parameters["resample size"] = 4



[docs]
    def execute(self, hdu_list):
        ext = self["hdu target"]
        data_type = str(self["data type"])
        size = int(self["resample size"])
        if size < 1:
            raise ValueError("'resample size' must be one or more")

        source = require_extension(hdu_list, ext, "'hdu target'")
        data = to_sigma_like(source.data, data_type)
        n_wave, n_y, n_x = data.shape
        out_y, out_x = n_y // size, n_x // size
        if out_y < 1 or out_x < 1:
            raise ValueError(
                f"'resample size' {size} is larger than the {n_y}x{n_x} "
                f"spatial grid of {ext!r}")

        trimmed = data[:, :out_y * size, :out_x * size]
        blocks = trimmed.reshape(n_wave, out_y, size, out_x, size)
        with warnings.catch_warnings():
            warnings.simplefilter("ignore", RuntimeWarning)
            summed = np.nanmean(blocks, axis=2) * size
            summed = np.nanmean(summed, axis=3) * size
        out = from_sigma_like(summed, data_type)

        header = source.header.copy()
        source_name = str(header.get("EXTNAME", ext))
        hdu = fits.ImageHDU(data=_as_written(out, source.data), header=header)
        hdu.header["EXTNAME"] = f"{source_name}_RSP"
        hdu.header["RSP_SIZE"] = (size, "Spaxels per side summed into one")
        _stamp(hdu.header, self.name)

        logger.info(f"Spatial resampling: {source_name} {n_y}x{n_x} -> "
                    f"{out_y}x{out_x} in {size}x{size} blocks "
                    f"-> {source_name}_RSP")
        return fits.HDUList([hdu])

class DegradeResolution(IFUModule):
    """
    The data convolved down to a coarser resolution. Writes `<EXTNAME>_DEGR`.

    The kernel width is sqrt(sigma_out ** 2 - sigma_in ** 2), clipped at zero
    where the data is already coarser than the target, and truncated at
    twenty sigma. Resolutions are given as a resolving power `R`, a `FWHM` in
    Angstrom or a `sigma` in Angstrom; an `R` with a reference wavelength is
    a constant width rather than a constant resolving power.

    This is not BRAIN's route to matching a library to an instrument. BRAIN
    convolves the templates up to the data, which leaves the pixel noise
    uncorrelated; this smooths the data, which does not. It exists so that an
    existing analysis made that way can be reproduced, and it warns every
    time it runs. Asking for it and for `instrumental_resolution` at once is
    refused before the run starts, because that counts the instrument twice.

    Configuration: `hdu target` (FLUX), `data type` (flux), `input type` (R),
    `input value` (2200), `input ref wave` (None), `output type` (R),
    `output value` (1000), `output ref wave` (None).
    """

    name = "Degrade resolution"

    def _set_default_parameters(self):
        self.default_parameters["hdu target"] = "FLUX"

    def execute(self, hdu_list):
        ext = self["hdu target"]
        data_type = str(self["data type"])
        source = require_extension(hdu_list, ext, "'hdu target'")
        wave = wavelength_axis(hdu_list, ext)

        sigma_in = _resolution_to_sigma(wave, self["input value"],
                                        self["input type"],
                                        self["input ref wave"])
        sigma_out = _resolution_to_sigma(wave, self["output value"],
                                         self["output type"],
                                         self["output ref wave"])
        difference = sigma_out ** 2 - sigma_in ** 2
        n_narrower = int(np.sum(difference < 0))
        conv_sigma = np.sqrt(np.clip(difference, 0.0, None))

        logger.warning(
            "Resolution degradation: this smooths the DATA. Neighbouring "
            "pixels become correlated, and every chi-squared in BRAIN assumes "
            "they are independent, so the fit statistic and the error bars "
            "after this stage are optimistic by an amount nobody has "
            "quantified. BRAIN's own route is 'instrumental_resolution', "
            "which convolves the templates up to the data instead.")
        if n_narrower:
            logger.warning(
                f"   {n_narrower} of {len(wave)} wavelengths ask for a "
                f"resolution finer than the data already has; the kernel is "
                f"clipped to zero there and the data passes through unchanged.")

        data = to_sigma_like(source.data, data_type)
        n_wave, n_y, n_x = data.shape
        flat = data.reshape(n_wave, n_y * n_x)
        finite = np.isfinite(flat)
        values = np.where(finite, flat, 0.0)
        weights = finite.astype(np.float64)

        out = np.empty_like(flat)
        for i in range(n_wave):
            if conv_sigma[i] <= 0.0:
                out[i] = flat[i]
                continue
            span = KERNEL_TRUNCATION_SIGMA * conv_sigma[i]
            lo = int(np.searchsorted(wave, wave[i] - span, side="left"))
            hi = int(np.searchsorted(wave, wave[i] + span, side="right"))
            kernel = np.exp(-((wave[lo:hi] - wave[i]) ** 2)
                            / (2.0 * conv_sigma[i] ** 2))
            total = kernel @ weights[lo:hi]
            numerator = kernel @ values[lo:hi]
            with np.errstate(divide="ignore", invalid="ignore"):
                out[i] = np.where(total > 0, numerator / np.where(total > 0,
                                                                  total, 1.0),
                                  np.nan)
            if i % 256 == 0:
                self.progress_hook(i / max(n_wave, 1),
                                   "degrading the resolution")

        out = from_sigma_like(out.reshape(n_wave, n_y, n_x), data_type)

        header = source.header.copy()
        source_name = str(header.get("EXTNAME", ext))
        hdu = fits.ImageHDU(data=_as_written(out, source.data), header=header)
        hdu.header["EXTNAME"] = f"{source_name}_DEGR"
        hdu.header["DGR_IN"] = (f"{self['input type']} {self['input value']}",
                                "Input resolution")
        hdu.header["DGR_OUT"] = (f"{self['output type']} "
                                 f"{self['output value']}",
                                 "Output resolution")
        hdu.header["DGR_WARN"] = ("data smoothed; chi2 assumes independent px",
                                  "see the BRAIN IFU docs")
        _stamp(hdu.header, self.name)

        logger.info(f"Resolution: {source_name} degraded, median kernel "
                    f"{np.median(conv_sigma):.3f} A "
                    f"-> {source_name}_DEGR")
        return fits.HDUList([hdu])






def _resolution_to_sigma(wave, value, kind, reference_wavelength=None):
    """
    A resolution specification as a one-sigma width in Angstrom per pixel.

    A resolving power without a reference wavelength gives a width that grows
    with wavelength; with one it gives a single width, which is the same thing
    as naming the FWHM at that wavelength.
    """
    kind = str(kind).lower().strip()
    wave = np.asarray(wave, dtype=np.float64)
    if kind == "r":
        if reference_wavelength is not None:
            return np.full_like(
                wave, float(reference_wavelength)
                / (float(value) * FWHM_CONST))
        return wave / (np.asarray(value, dtype=np.float64) * FWHM_CONST)
    if kind == "fwhm":
        return np.broadcast_to(np.asarray(value, dtype=np.float64),
                               wave.shape).astype(np.float64) / FWHM_CONST
    if kind == "sigma":
        return np.broadcast_to(np.asarray(value, dtype=np.float64),
                               wave.shape).astype(np.float64).copy()
    raise ValueError(f"unknown resolution type {kind!r}; "
                     f"expected 'R', 'FWHM' or 'sigma'")

class SNMask(IFUModule):
    """
    Which spaxels carry enough signal to fit. Writes `SN_MASKS_<threshold>`.

    One integer plane per threshold, 1 where the signal-to-noise in the window
    exceeds it, and the continuous map they were cut from as `SNMAP`, which
    costs nothing to keep and every user otherwise recomputes.

    The default is the mean over the window of the per-pixel signal-to-noise,
    `mean(flux * sqrt(ivar))`. That is the standard definition, it is what the
    fitter sees pixel by pixel, and it degrades gracefully where the inverse
    variance is zero, which is how MaNGA marks an unusable pixel. A form
    that takes the square root a second time returns a quantity with units
    of the square root of a flux, which at a threshold of 10 selects zero of
    2387 spaxels of a real MaNGA cube where this form selects 605. `median(flux * sqrt(ivar))` is offered for windows
    with unavoidable sky residuals, and the mean of the ratio to an error or
    variance plane, and the flux mean over its standard deviation, for cubes
    that carry no error at all. `SN_METH` records which was used.

    The window is rest-frame: the wavelength axis is divided by 1 + z before
    it is applied.

    Configuration: `hdu flux` (FLUX), `hdu ivar` (IVAR), `hdu var` (VAR),
    `hdu error` (ERROR), `sn window` ([4000, 6000]), `thresholds` ([10]),
    `redshift` (0.0), `method` (ivar).
    """

    name = "Signal-to-noise mask"

    def _set_default_parameters(self):
        self.default_parameters["hdu flux"] = "FLUX"
        self.default_parameters["hdu ivar"] = "IVAR"
        self.default_parameters["hdu var"] = "VAR"
        self.default_parameters["hdu error"] = "ERROR"
        self.default_parameters["sn window"] = [4000.0, 6000.0]
        self.default_parameters["thresholds"] = [10]
        self.default_parameters["redshift"] = 0.0
        self.default_parameters["method"] = "ivar"

    def execute(self, hdu_list):
        ext = self["hdu flux"]
        method = str(self["method"]).lower().strip()
        if method not in SN_METHODS:
            raise ValueError(f"unknown signal-to-noise method {method!r}; "
                             f"expected one of " + ", ".join(SN_METHODS))

        stat_key = SN_METHOD_STAT_KEY[method]
        source = require_extension(hdu_list, ext, "'hdu flux'")
        if stat_key is not None:
            require_extension(hdu_list, self[stat_key], f"{stat_key!r}")

        redshift = float(self["redshift"])
        rest_wave = wavelength_axis(hdu_list, ext) / (1.0 + redshift)
        low, high = [float(x) for x in self["sn window"]]
        inside = (rest_wave >= low) & (rest_wave <= high)
        if not np.any(inside):
            raise ValueError(
                f"the signal-to-noise window {low:g}-{high:g} A rest-frame "
                f"falls outside this cube, which covers "
                f"{rest_wave[0]:.0f}-{rest_wave[-1]:.0f} A rest-frame at "
                f"z = {redshift:g}")

        flux = np.asarray(source.data, dtype=np.float64)[inside]
        sn_map = self._sn_map(hdu_list, flux, inside, method)
        logger.info(f"S/N map: {SN_METHOD_TEXT[method]} over "
                    f"{low:g}-{high:g} A rest-frame "
                    f"({int(np.sum(inside))} of {len(rest_wave)} planes) "
                    f"-> SNMAP")

        hdus = fits.HDUList()
        window_text = f"{low:g} - {high:g}"
        for threshold in self["thresholds"]:
            threshold = float(threshold)
            plane = np.zeros(sn_map.shape, dtype=np.int32)
            plane[np.isfinite(sn_map) & (sn_map > threshold)] = 1
            hdu = fits.ImageHDU(data=plane)
            hdu.header["EXTNAME"] = f"SN_MASKS_{int(threshold)}"
            hdu.header["SN_WIND"] = (window_text, "Rest-frame window, Angstrom")
            hdu.header["THRESH"] = (threshold, "1 where S/N exceeds this")
            hdu.header["SN_METH"] = (SN_METHOD_TEXT[method],
                                     "How the S/N was measured")
            hdu.header["SN_Z"] = (redshift, "Redshift the window was moved by")
            _stamp(hdu.header, self.name)
            hdus.append(hdu)
            logger.info(f"S/N mask: {int(plane.sum())} of {plane.size} "
                        f"spaxels above {threshold:g} "
                        f"-> SN_MASKS_{int(threshold)}")

        snmap = fits.ImageHDU(data=np.where(np.isfinite(sn_map), sn_map, 0.0))
        snmap.header["EXTNAME"] = "SNMAP"
        snmap.header["SN_WIND"] = (window_text, "Rest-frame window, Angstrom")
        snmap.header["SN_METH"] = (SN_METHOD_TEXT[method],
                                   "How the S/N was measured")
        snmap.header["SN_Z"] = (redshift, "Redshift the window was moved by")
        _stamp(snmap.header, self.name)
        hdus.append(snmap)
        return hdus



    def _sn_map(self, hdu_list, flux, inside, method):
        """
        The signal-to-noise of every spaxel over the window.

        Every form here is an average of a ratio rather than a ratio of
        averages, except the last, which has no noise plane to work from. The
        difference matters: a ratio of averages divides by the arithmetic mean
        of sigma, so one unusable pixel in the window can take a whole spaxel
        to zero.
        """
        with np.errstate(divide="ignore", invalid="ignore"):
            if method in ("ivar", "ivar_median"):
                ivar = np.asarray(hdu_list[self["hdu ivar"]].data,
                                  dtype=np.float64)[inside]
                per_pixel = flux * np.sqrt(np.clip(ivar, 0.0, None))
                per_pixel = np.where(np.isfinite(per_pixel), per_pixel, np.nan)
                if method == "ivar":
                    return np.nanmean(per_pixel, axis=0)
                return np.nanmedian(per_pixel, axis=0)

            if method == "variance":
                var = np.asarray(hdu_list[self["hdu var"]].data,
                                 dtype=np.float64)[inside]
                sigma = np.sqrt(np.clip(var, 0.0, None))
                per_pixel = np.where(sigma > 0, flux / sigma, np.nan)
                return np.nanmean(per_pixel, axis=0)

            if method == "error":
                error = np.asarray(hdu_list[self["hdu error"]].data,
                                   dtype=np.float64)[inside]
                per_pixel = np.where(error > 0, flux / error, np.nan)
                return np.nanmean(per_pixel, axis=0)

            mean = np.nanmean(flux, axis=0)
            spread = np.nanstd(flux, axis=0)
            return np.where(spread > 0, mean / np.where(spread > 0, spread,
                                                        1.0), np.nan)
