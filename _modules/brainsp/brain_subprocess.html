Source code for brainsp.brain_subprocess

"""
BRAIN, the interface's end of a fit that runs somewhere else.

One class, `ChildRun`, which is a QThread whose entire job is to own a child
process, hand it a configuration and turn the lines it sends back into Qt
signals. The pages that use it subclass it, declare the signals they already
declared, and say what each kind of message means to them, so the rest of the
interface sees exactly the API it saw when the fit ran in a thread.

The reason for the process rather than the thread is JAX. It fixes its backend
the first time it is used and never changes it, so the device selector can only
be honoured by an interpreter that has not started yet. Starting one is also
what makes a Stop button able to stop anything and what keeps a crash inside
the fit from taking the interface with it.

The child is started with `subprocess.Popen`, not with a fork of this process:
BRAIN has a history of forking an interpreter that already held threaded JAX
state and deadlocking, and a fresh interpreter cannot inherit a trap it never
had.
"""

import json
import os
import shutil
import subprocess
import sys
import tempfile
import threading
import traceback
from collections import deque

from PySide6.QtCore import QThread


HERE = os.path.dirname(os.path.abspath(__file__))

# Run as a module, so its `from brainsp import ...` resolves the same way it
# does here; the folder holding the package goes on the child's path so a
# source checkout that was never pip-installed works too.
CHILD_MODULE = "brainsp.brain_child"
PACKAGE_PARENT = os.path.dirname(HERE)

PLATFORM_FOR_DEVICE = {"cpu": "cpu", "gpu": "cuda"}

CANCELLED = "Cancelled: the run was stopped and its process has ended."

class ChildRun(QThread):
    """
    One child process, owned for as long as it runs.

    `job` is whatever the child needs and nothing else; it is written to a
    temporary file outside the project and deleted afterwards, because a
    configuration is far larger than a command line should carry and because
    passing it as a file means the child reads it with json and not with an
    argument parser.

    Subclasses implement `handle`, which receives one decoded message at a
    time on this thread, and `report_failure`, which is called for the
    failures the child could not report itself: a process that died without a
    word, and a run that was stopped.
    """

    def __init__(self, job, device):
        super().__init__()
        self.job = dict(job)
        self.device = str(device)
        self.process = None
        self.cancelled = False
        self.tail = deque(maxlen=60)

    def environment(self):
        """
        This process's environment with the platform named in it.

        Named before the child starts, which is the only moment JAX reads it.
        "gpu" is not a value JAX accepts; the platform is cuda. "auto" removes
        the variable rather than setting it to something, so a selector left
        alone behaves as an interpreter with no JAX_PLATFORMS at all.
        """
        environment = dict(os.environ)
        paths = [PACKAGE_PARENT] + [p for p in environment.get(
            "PYTHONPATH", "").split(os.pathsep) if p]
        environment["PYTHONPATH"] = os.pathsep.join(paths)
        platform = PLATFORM_FOR_DEVICE.get(self.device)
        if platform:
            environment["JAX_PLATFORMS"] = platform
        else:
            environment.pop("JAX_PLATFORMS", None)
        return environment

    def cancel(self):
        """
        Stop the run by ending the process doing it.

        SIGTERM first, because a Python process ends cleanly on it and leaves
        no half-written output; SIGKILL five seconds later from a watcher, so
        a child stuck inside a compiled kernel that never returns to the
        interpreter still goes away. The watcher is a plain thread because it
        does nothing but wait.
        """
        self.cancelled = True
        process = self.process
        if process is None or process.poll() is not None:
            return
        try:
            process.terminate()
        except OSError:
            return
        threading.Thread(target=self._insist, args=(process,),
                         daemon=True).start()



    @staticmethod
    def _insist(process):
        try:
            process.wait(timeout=5.0)
        except subprocess.TimeoutExpired:
            try:
                process.kill()
            except OSError:
                pass

    def run(self):
        folder = tempfile.mkdtemp(prefix="brain_job_")
        path = os.path.join(folder, "job.json")
        try:
            with open(path, "w") as handle:
                json.dump(dict(self.job, device=self.device), handle)
            self._pump(path)
        except Exception as exc:
            self.report_failure(f"{exc}\n\n{traceback.format_exc()}")
        finally:
            shutil.rmtree(folder, ignore_errors=True)



    def _pump(self, path):
        """
        Read the child until it stops, and account for how it stopped.

        A line that is not JSON is not a message: it is whatever the runtime
        printed on its way down, so it is kept for the tail and not guessed
        at. If the child managed to describe its own failure that description
        is the one the interface gets; the exit code is only consulted when
        nothing was said.
        """
        self.process = subprocess.Popen(
            [sys.executable, "-u", "-m", CHILD_MODULE, path],
            env=self.environment(),
            stdout=subprocess.PIPE, stderr=subprocess.PIPE,
            text=True, bufsize=1)

        errors = threading.Thread(target=self._drain, daemon=True)
        errors.start()

        spoke = False
        for line in self.process.stdout:
            line = line.strip()
            if not line:
                continue
            try:
                message = json.loads(line)
            except ValueError:
                self.tail.append(line)
                continue
            if message.get("kind") == "failed":
                spoke = True
            self.handle(message)

        code = self.process.wait()
        errors.join(timeout=2.0)

        if self.cancelled:
            self.report_failure(CANCELLED)
        elif code != 0 and not spoke:
            self.report_failure(
                f"The fitting process ended with exit code {code} and said "
                f"nothing about why.\n\n" + self.recent())

    def _drain(self):
        """
        Standard error: forwarded, and kept.

        Forwarded because the fit logs there and a BRAIN started from a
        terminal used to show that log in the terminal, which it should go on
        doing now that the fit is somewhere else. Kept because when a process
        dies without a word these last lines are the only account of it there
        will ever be.
        """
        for line in self.process.stderr:
            line = line.rstrip("\n")
            if line:
                self.tail.append(line)
                sys.stderr.write(line + "\n")
        sys.stderr.flush()

    def recent(self):
        """The last of whatever the child wrote outside the protocol."""
        return "\n".join(self.tail) or "(it wrote nothing at all)"
