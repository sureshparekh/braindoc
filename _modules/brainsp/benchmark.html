#!/usr/bin/env python3
"""
BRAIN — timing benchmark.

    python benchmark.py                 all four modes, current device
    python benchmark.py spop nepop      only these modes
    python benchmark.py --reps 5        more repeats (median is reported)

Reports the time spent in every step of a fit, per mode, in seconds.

IMPORTANT — run this on an idle machine.
This server is shared. If other jobs are using the cores, BRAIN gets less CPU
and every number here inflates, sometimes by a factor of several, without a
single line of BRAIN changing. The script measures the machine load first and
refuses to pretend the numbers mean anything when it is busy.
"""

import os
import sys
import time
import json
import collections

REPS = 3
MODES = ["spop", "espop", "nepop", "photopop"]

def cpu_load(sample=2.0):
    """Per-core busy fraction, sampled over `sample` seconds."""
    def snap():
        d = {}
        for line in open("/proc/stat"):
            if line.startswith("cpu") and line[3].isdigit():
                p = line.split()
                d[int(p[0][3:])] = (sum(map(int, p[1:])), int(p[4]))
        return d
    a = snap()
    time.sleep(sample)
    b = snap()
    busy = {}
    for c in a:
        dt = b[c][0] - a[c][0]
        di = b[c][1] - a[c][1]
        busy[c] = 100.0 * (1 - di / dt) if dt else 0.0
    return busy

def report_load():
    busy = cpu_load()
    idle = sorted(c for c, u in busy.items() if u < 10.0)
    n = len(busy)
    try:
        la1, la5, la15 = os.getloadavg()
    except OSError:
        la1 = la5 = la15 = float("nan")

    print(f"  machine : {n} cores, load average {la1:.1f}")
    print(f"  free    : {len(idle)}/{n} cores below 10% use")

    if len(idle) >= 4:
        print("  -> machine is quiet; timings should be representative\n")
        return idle[:4], True
    print(f"  -> MACHINE IS BUSY. Other jobs are using {n - len(idle)} cores.")
    print("     BRAIN will be competing for CPU and every number below will")
    print("     be inflated. Re-run when the machine is free.\n")
    return (idle or sorted(busy))[:4], False

def run_mode(mode, reps):
    t0 = time.perf_counter()
    from brainsp import brain_cache
    brain_cache.enable_jax_cache(verbose=False)
    import jax
    jax.config.update("jax_enable_x64", True)
    t_import = time.perf_counter() - t0

    from brainsp import brain_settings as bs, brainv1, optimizer, io_utils
    import logging
    logging.disable(logging.WARNING)

    steps = collections.OrderedDict()

    def wrap(obj, name, label):
        orig = getattr(obj, name)

        def f(*a, **k):
            t = time.perf_counter()
            r = orig(*a, **k)
            try:
                jax.block_until_ready(r)
            except Exception:
                pass
            steps.setdefault(label, []).append(time.perf_counter() - t)
            return r
        setattr(obj, name, f)

    S = bs.load()
    S["nebular_mode"] = mode
    cfg = bs.to_config(S)

    raw = io_utils.load_ssp_templates
    memo = {}

    def cached(f, b, tw, n):
        k = brain_cache.ssp_cache_key(f, b, tw, n)
        if k not in memo:
            memo[k] = brain_cache.load_ssp_templates_cached(f, b, tw, n, raw)
        return memo[k]
    io_utils.load_ssp_templates = cached
    brainv1.load_ssp_templates = cached

    for obj, name, label in (
            (brainv1, "load_spectrum", "read spectrum"),
            (brainv1, "log_rebin", "log-rebin"),
            (brainv1, "load_mask", "build mask"),
            (brainv1, "load_ssp_templates", "load templates (cached)"),
            (optimizer.SpectralFitter, "__init__", "fitter init"),
            (optimizer.SpectralFitter, "_build_fast_constants", "build constants"),
            (optimizer.SpectralFitter, "grid_search_v0", "velocity grid search"),
            (optimizer.SpectralFitter, "_optimize_fast", "LM optimisation"),
            (io_utils, "write_summary", "write summary"),
            (io_utils, "write_fitted_spectrum", "write spectrum")):
        try:
            wrap(obj, name, label)
        except AttributeError:
            pass
    try:
        from brainsp import gas_diagnostics as gd
        wrap(gd, "measure_conditions", "gas diagnostics")
        wrap(gd, "classify_bpt", "BPT classification")
    except Exception:
        pass

    cfg.output_summary = "/tmp/_bench_sum.txt"
    cfg.output_spectrum = "/tmp/_bench_spec.txt"

    brainv1.run(cfg)

    runs = []
    for _ in range(reps):
        for k in steps:
            steps[k] = []
        t = time.perf_counter()
        brainv1.run(cfg)
        runs.append((time.perf_counter() - t,
                     {k: (len(v), sum(v)) for k, v in steps.items() if v}))
    runs.sort(key=lambda r: r[0])
    total, per_step = runs[len(runs) // 2]
    return {"mode": mode, "import": t_import, "total": total,
            "steps": per_step, "device": jax.default_backend()}

def main(argv):
    """
    Parse the command line, taking care that a flag's value is not mistaken
    for a mode name.

    Filtering on a leading "--" alone left the number after --reps in the
    positional list, so `benchmark.py --reps 5` benchmarked a mode called
    "5" as well as whatever else was asked for. An unknown mode is now
    refused by name rather than run.
    """
    global REPS
    rest = list(argv[1:])
    args = []
    while rest:
        token = rest.pop(0)
        if token == "--reps":
            if not rest:
                print("  --reps needs a number, e.g. --reps 5")
                return 2
            REPS = int(rest.pop(0))
        elif token.startswith("--"):
            continue
        else:
            args.append(token)

    unknown = [a for a in args if a not in MODES]
    if unknown:
        print(f"  not a mode: {', '.join(unknown)}")
        print(f"  choose from: {', '.join(MODES)}")
        return 2

    modes = args or MODES

    print("\n" + "=" * 74)
    print("  BRAIN timing benchmark")
    print("=" * 74)
    cores, quiet = report_load()

    if os.environ.get("BR_CHILD"):
        print(json.dumps(run_mode(modes[0], REPS)))
        return 0

    import subprocess
    results = []
    for m in modes:
        env = dict(os.environ, BR_CHILD="1")
        cmd = ["taskset", "-c", ",".join(map(str, cores)),
               sys.executable, __file__, m, "--reps", str(REPS)]
        try:
            out = subprocess.run(cmd, env=env, capture_output=True, text=True,
                                 timeout=3600).stdout
            results.append(json.loads(out.strip().splitlines()[-1]))
        except Exception as exc:
            print(f"  {m}: FAILED ({exc})")

    if not results:
        return 1

    keys = []
    for r in results:
        for k in r["steps"]:
            if k not in keys:
                keys.append(k)

    dev = results[0]["device"]
    print("=" * 74)
    print(f"  device: {dev}   |   {REPS} repeats, median   |   SECONDS")
    print("=" * 74)
    hdr = f"{'step':<32}" + "".join(f"{r['mode']:>13}" for r in results)
    print(hdr)
    print("-" * 74)
    acc = [0.0] * len(results)
    for k in keys:
        row = f"{k:<32}"
        for i, r in enumerate(results):
            s = r["steps"].get(k)
            if s:
                n, tt = s
                acc[i] += tt
                row += f"{tt:>10.3f}{('x%d' % n) if n > 1 else '':>3}"
            else:
                row += f"{'-':>13}"
        print(row)
    print("-" * 74)
    print(f"{'other (python glue)':<32}"
          + "".join(f"{r['total'] - acc[i]:>13.3f}" for i, r in enumerate(results)))
    print(f"{'FIT TOTAL (per spectrum)':<32}"
          + "".join(f"{r['total']:>13.3f}" for r in results))
    print(f"{'+ import python & JAX (once)':<32}"
          + "".join(f"{r['import']:>13.3f}" for r in results))
    print(f"{'WALL CLOCK (single spectrum)':<32}"
          + "".join(f"{r['total'] + r['import']:>13.3f}" for r in results))
    print("=" * 74)
    if not quiet:
        print("  NOTE: the machine was busy. These numbers are inflated by")
        print("        competing jobs — re-run when it is idle.")
    print("  In batch (many spectra in one run) the import row is paid once,")
    print("  so the per-spectrum cost is the FIT TOTAL row.\n")
    return 0




if __name__ == "__main__":
    sys.exit(main(sys.argv))
