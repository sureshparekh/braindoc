Source code for brainsp.brain_child

"""
BRAIN, one fit, in a process of its own.

JAX fixes its backend the first time it is used in a process and keeps it for
the life of that process. A desktop interface has already used JAX by the time
anybody presses Fit, so a device selector that writes JAX_PLATFORMS into the
running interpreter selects nothing. The only place the variable is still read
is before Python starts, which means the fit has to run somewhere Python has
not started yet: here.

The interface launches this module with `python -u brain_child.py job.json`
and an environment that already names the platform. The job file carries the
whole configuration, so nothing is inherited except the environment, and
nothing is pickled, so there is no fork and no threaded-JAX deadlock to
inherit either.

Everything travelling back to the interface goes out on standard output as one
JSON object per line. Anything the fit prints is turned into a log message
rather than written straight out, so the stream stays parseable however
talkative the fit becomes. Standard error is left alone: the parent reads it
separately and keeps the tail of it to explain a process that dies without
saying why.

The backend actually in use is compared with the one that was asked for before
any fitting starts, and a mismatch ends the run. A device selector that
accepts a request it cannot honour is worse than one that refuses, because the
result looks the same either way.
"""

import json
import os
import sys


PLATFORM_FOR_DEVICE = {"cpu": "cpu", "gpu": "cuda"}

BACKENDS_FOR_DEVICE = {"cpu": ("cpu",), "gpu": ("gpu", "cuda", "rocm")}

def plain(value):
    """
    A value JSON has no opinion about.

    Summaries come back holding numpy scalars and occasionally a numpy array,
    neither of which json knows. Anything still unrecognised becomes its own
    repr rather than killing a run that has already finished its work.
    """
    item = getattr(value, "item", None)
    if item is not None and getattr(value, "shape", None) == ():
        return item()
    tolist = getattr(value, "tolist", None)
    if tolist is not None:
        return tolist()
    return repr(value)

class LogStream:
    """
    Standard output, turned into log messages a line at a time.

    The fit writes progress with print and with logging, and both of them are
    things the interface wants to show. Buffering to the newline keeps a
    partial write from becoming half a message.
    """

    def __init__(self, channel):
        self.channel = channel
        self.pending = ""



[docs]
    def write(self, text):
        self.pending += text
        while "\n" in self.pending:
            line, self.pending = self.pending.split("\n", 1)
            self.channel.send("log", text=line)
        return len(text)

        if self.pending:

class LogBridge:
    """
    A logging handler that hands each record to the channel.

    Installed on the root logger for the length of a run, so that the fitter's
    own progress lines, which are the ones that say how many spaxels
    converged, reach the interface without any of the code that writes them
    knowing an interface exists.
    """

    def __init__(self, channel):
        import logging
        self.channel = channel
        self.handler = logging.Handler()
        self.handler.emit = self._emit
        self.handler.setLevel(logging.INFO)
        self.logging = logging

    def _emit(self, record):
        try:
            self.channel.send("log", text=record.getMessage())
        except Exception:
            pass

    def __enter__(self):
        root = self.logging.getLogger()
        self.previous = root.level
        root.addHandler(self.handler)
        if root.level > self.logging.INFO or root.level == 0:
            root.setLevel(self.logging.INFO)
        return self

    def __exit__(self, *exc):
        root = self.logging.getLogger()
        root.removeHandler(self.handler)
        root.setLevel(self.previous)
        return False

def set_platform(device):
    """
    Name the platform JAX is to use, before anything imports JAX.

    The parent already put this in the environment, which is the copy that
    matters; repeating it here makes the module runnable on its own and makes
    the mapping visible in one place. "gpu" is not a platform name JAX 0.10
    accepts: it expands to cuda and rocm, and an interpreter without rocm
    refuses the whole list. The platform is cuda and the backend it reports is
    gpu, which is why the two tables below are not the same table.
    """
    platform = PLATFORM_FOR_DEVICE.get(device)
    if platform:
        os.environ["JAX_PLATFORMS"] = platform
    else:
        os.environ.pop("JAX_PLATFORMS", None)

def confirm_backend(device, channel):
    """
    Refuse to fit on a device other than the one that was asked for.

    Reading the backend is what initialises it, so a request for a device this
    machine does not have raises here rather than quietly falling back. The
    comparison after it catches the rest: anything that leaves JAX on a
    different platform than the environment named ends the run through the
    failure channel instead of producing a fit whose provenance is wrong.
    """
    import jax
    jax.config.update("jax_enable_x64", True)
    backend = jax.default_backend()
    devices = ", ".join(str(one) for one in jax.devices())
    expected = BACKENDS_FOR_DEVICE.get(device)
    if expected is not None and backend not in expected:
        raise SystemExit(
            f"The device was set to {device} and JAX started on {backend} "
            f"({devices}). The fit was not run.")
    channel.send("backend", device=device, backend=backend, devices=devices,
                 text=f"JAX backend: {backend} [{devices}], device "
                      f"requested: {device}")
    return backend

def config_from_dict(values):
    """
    A NewBrainConfig rebuilt from the JSON the interface wrote.

    JSON has one sequence type, so every bound arrives as a list where the fit
    expects a tuple. Each field whose declared default is a tuple is put back
    into one and nothing else is touched, which round-trips the reference
    configuration field for field.
    """
    import dataclasses
    from brainsp.config import NewBrainConfig

    restored = dict(values)
    for field in dataclasses.fields(NewBrainConfig):
        if isinstance(field.default, tuple) and field.name in restored:
            value = restored[field.name]
            if isinstance(value, list):
                restored[field.name] = tuple(value)
    return NewBrainConfig(**restored)

def run_spectrum(job, channel):
    """
    One spectrum, fitted, with the interface's two caches honoured here.

    The order is the order the caching layer requires: the kernel cache is
    turned on before anything touches a device, the backend is confirmed
    next, and the template loader is replaced only after that, so a run that
    is about to be refused for the wrong device never reads a template.
    """
    channel.send("status", text="warming the engine…")

    from brainsp import brain_cache
    if job.get("cache_engine", True):
        brain_cache.enable_jax_cache(verbose=False)

    confirm_backend(job.get("device", "auto"), channel)

    from brainsp import brainv1
    from brainsp import io_utils

    uncached = io_utils.load_ssp_templates
    memo = {}

    def cached(folder, bases, target_wl, norm):
        key = brain_cache.ssp_cache_key(folder, bases, target_wl, norm)
        if key not in memo:
            memo[key] = brain_cache.load_ssp_templates_cached(
                folder, bases, target_wl, norm, uncached)
        return memo[key]

    if job.get("cache_templates", True):
        io_utils.load_ssp_templates = cached
        brainv1.load_ssp_templates = cached

    cfg = config_from_dict(job["config"])
    channel.send("status", text="fitting…")
    brainv1.run(cfg)
    channel.send("done", spectrum=cfg.output_spectrum,
                 summary=cfg.output_summary)

def run_spectra(job, channel):
    """
    A table of spectra, fitted one after another in this one process.

    The point of fitting them here rather than one process each: JAX
    compiles the fit once, for the first spectrum, and every later spectrum
    on the same wavelength grid reuses that compiled code; the templates are
    read once and kept. Each spectrum is reported as it finishes, and one
    that fails is reported and skipped, so the rest of the table still runs.
    """
    import time
    channel.send("status", text="warming the engine…")
    from brainsp import brain_cache
    if job.get("cache_engine", True):
        brain_cache.enable_jax_cache(verbose=False)
    confirm_backend(job.get("device", "auto"), channel)
    from brainsp import brainv1, io_utils
    if job.get("cache_templates", True):
        uncached = io_utils.load_ssp_templates
        memo = {}

        def cached(folder, bases, target_wl, norm):
            key = brain_cache.ssp_cache_key(folder, bases, target_wl, norm)
            if key not in memo:
                memo[key] = brain_cache.load_ssp_templates_cached(
                    folder, bases, target_wl, norm, uncached)
            return memo[key]

        io_utils.load_ssp_templates = cached
        brainv1.load_ssp_templates = cached

    configs = job["configs"]
    import jax
    if jax.default_backend() == "cpu" and len(configs) >= 16 \
            and _spectra_on_workers(configs, channel):
        return
    for i, values in enumerate(configs):
        cfg = config_from_dict(values)
        name = os.path.basename(str(cfg.obs_spectrum_file))
        channel.send("status", text=f"fitting {i + 1}/{len(configs)}: {name}")
        start = time.time()
        try:
            brainv1.run(cfg)
        except BaseException as exc:
            import traceback
            channel.send("item_failed", index=i, spectrum=cfg.obs_spectrum_file,
                         text=f"{exc}\n\n{traceback.format_exc()}")
            continue
        channel.send("item_done", index=i, spectrum=cfg.output_spectrum,
                     summary=cfg.output_summary,
                     seconds=round(time.time() - start, 3))
    channel.send("finished", summaries=[])




def _spectra_on_workers(configs, channel):
    """
    A CPU and a long table: worker processes of one thread each, every one
    keeping its own fit warm, which scales with the cores (see
    fit_megacube_jax.cpu_worker_count). Same messages as the one-process
    path, so the interface cannot tell the difference except in speed.
    """
    import pickle
    import subprocess
    import tempfile
    import threading
    import time
    from brainsp.fit_megacube_jax import cpu_worker_count
    n = cpu_worker_count(configs[0].get("cpu_workers", "auto"),
                         max(1, len(configs) // 8))
    if n <= 1:
        return None
    channel.send("status", text=f"fitting {len(configs)} spectra on {n} "
                                f"CPU workers…")
    env = dict(os.environ, JAX_PLATFORMS="cpu", OMP_NUM_THREADS="1",
               XLA_FLAGS="--xla_cpu_multi_thread_eigen=false "
                         "intra_op_parallelism_threads=1")
    folder = tempfile.mkdtemp(prefix="brainsp-table-")
    procs = []
    start = time.time()
    lock = threading.Lock()
    count = [0]
    for k in range(n):
        share = [(i, configs[i]) for i in range(k, len(configs), n)]
        path = os.path.join(folder, f"job{k}.pkl")
        with open(path, "wb") as handle:
            pickle.dump({"configs": share}, handle)
        procs.append(subprocess.Popen(
            [sys.executable, "-m", "brainsp.spectra_worker", path],
            stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, env=env,
            text=True))

    def read(proc):
        for line in proc.stdout:
            kind, _, rest = line.rstrip("\n").partition(" ")
            index, _, value = rest.partition(" ")
            if not index.isdigit():
                continue
            i = int(index)
            with lock:
                count[0] += 1
                seconds = (time.time() - start) / count[0]
                if kind == "DONE":
                    channel.send("item_done", index=i,
                                 spectrum=configs[i]["output_spectrum"],
                                 summary=value, seconds=round(seconds, 3))
                else:
                    channel.send("item_failed", index=i,
                                 spectrum=configs[i]["obs_spectrum_file"],
                                 text=value)

    threads = [threading.Thread(target=read, args=(p,), daemon=True)
               for p in procs]
    for t in threads:
        t.start()
    for p in procs:
        p.wait()
    for t in threads:
        t.join(timeout=5)
    import shutil
    shutil.rmtree(folder, ignore_errors=True)
    channel.send("finished", summaries=[])
    return True

def run_cube(job, channel):
    """
    One campaign of datacubes, with its progress reported as it happens.

    The spaxel callback runs on the fitting thread, so it does the least it
    can: one line on the channel and back. An empty index list is the fit
    saying how large the field is before it starts.
    """
    confirm_backend(job.get("device", "auto"), channel)

    import numpy as np

    from brainsp import brain_ifu

    def spaxels(indices, n_y, n_x):
        if len(indices) == 0:
            channel.send("fitting", n_y=int(n_y), n_x=int(n_x))
        else:
            flat = np.asarray(indices).ravel()
            channel.send("fitted", indices=[int(i) for i in flat])

    with LogBridge(channel):
        summaries = brain_ifu.run(
            job["config"],
            progress=lambda fraction, message="": channel.send(
                "progress", fraction=float(fraction), message=message),
            log=lambda text: channel.send("log", text=str(text)),
            spaxels=spaxels)
    channel.send("finished", summaries=list(summaries))




JOBS = {"spectrum": run_spectrum, "spectra": run_spectra, "cube": run_cube}

def main(argv):
    import traceback

    with open(argv[1]) as handle:
        job = json.load(handle)

    set_platform(job.get("device", "auto"))

    channel = Channel(os.fdopen(os.dup(sys.stdout.fileno()), "w",
                                buffering=1))
    sys.stdout = LogStream(channel)
    try:
        JOBS[job["job"]](job, channel)
    except SystemExit as exc:
        sys.stdout.flush()
        channel.send("failed", text=str(exc))
        return 1
    except BaseException as exc:
        sys.stdout.flush()
        channel.send("failed", text=f"{exc}\n\n{traceback.format_exc()}")
        return 1
    sys.stdout.flush()
    return 0




if __name__ == "__main__":
    sys.exit(main(sys.argv))
