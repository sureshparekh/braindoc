Source code for brainsp.selftest

"""
Check an installation by fitting mock data whose answer is known.

    brainsp selftest              the full check: five spectra and the
                                  16 x 16 datacube (a few minutes on a CPU)
    brainsp selftest --quick      two spectra and the cube's brightest
                                  spaxels (about a minute)
    brainsp selftest --device gpu run it on the GPU

or, from Python, ``brainsp.selftest()``.

What is checked, in order:

1. the environment: Python, JAX and its device, the other dependencies;
2. the template libraries: BasesGM and the AGN continua are installed and
   their catalogues are readable;
3. the mock data are built (:mod:`brainsp.mock`);
4. each mock spectrum is fitted and the answer compared with the truth it
   was built from: stellar velocity and dispersion, extinction, mean age,
   AGN fraction, and for the line spectra the gas velocity and the H-alpha
   flux; the observed-frame mock must give the same answer as its
   rest-frame twin once BRAIN removes the redshift and the Galactic dust;
5. the datacube is fitted through the full pipeline (S/N masks, the fit,
   the megacube) and its maps compared with the maps it was built from;
6. the megacube is read back, a map and a diagnostic diagram are drawn, and
   a run is written out as a script that validates.

Every comparison has a tolerance, printed beside it, a few times the scatter
measured on the same mocks; a check that fails prints what it got and what
it expected. The run returns a :class:`Report` whose ``passed`` is True only
when every check passed, and the command exits non-zero otherwise, so it can
gate an installation script.
"""

import json
import os
import platform
import shutil
import subprocess
import sys
import tempfile
import time

import numpy as np

__all__ = ["run", "Report"]

# name: (truth key, how to read the fit, tolerance, kind)
#   kind "abs": |fit - truth| <= tol;  "rel": |fit/truth - 1| <= tol
SPECTRUM_CHECKS = [
    ("stellar velocity [km/s]", "v", lambda r: r.v0, 10.0, "abs"),
    ("stellar dispersion", "sigma", lambda r: r.sigma, 0.08, "rel"),
    ("A_V [mag]", "av", lambda r: r.av, 0.10, "abs"),
    ("<log age>_L [dex]", "mean_log_age_light",
     lambda r: r.mean_log_age_light, 0.15, "abs"),
]

FIT_OPTIONS = {
    "mock_old_population": {},
    "mock_young_mixture": {},
    "mock_star_forming_lines": dict(nebular_mode="espop",
                                    wavelength_scale="vacuum"),
    "mock_seyfert_agn": dict(nebular_mode="espop", wavelength_scale="vacuum",
                             include_power_law=True),
    "mock_observed_frame": dict(spectrum_redshift=0.03,
                                spectrum_in_rest_frame=False,
                                galactic_ebv=0.05),
}


class Report:
    """
    The outcome of a self-test.

    Attributes
    ----------
    passed : bool
        True when every check passed.
    checks : list of dict
        One per check: ``section``, ``name``, ``ok``, ``detail``.
    folder : str
        Where the mock data and the fits were written.
    seconds : float
    """

    def __init__(self, folder):
        self.folder = folder
        self.checks = []
        self.seconds = 0.0

    @property
    def passed(self):
        return bool(self.checks) and all(c["ok"] for c in self.checks)

    def add(self, section, name, ok, detail=""):
        self.checks.append(dict(section=section, name=name, ok=bool(ok),
                                detail=detail))
        return ok

    def __repr__(self):
        n_bad = sum(not c["ok"] for c in self.checks)
        state = "PASSED" if self.passed else f"FAILED ({n_bad})"
        return (f"<brainsp self-test {state}: {len(self.checks)} checks in "
                f"{self.seconds:.0f} s>")


def _print(verbose, text=""):
    if verbose:
        print(text, flush=True)


def _line(report, section, name, ok, detail, verbose):
    report.add(section, name, ok, detail)
    _print(verbose, f"  [{'PASS' if ok else 'FAIL'}] {name:<34} {detail}")


def _compare(fit, truth, tol, kind):
    if fit is None or truth is None or not np.isfinite(fit):
        return False, "no value"
    if kind == "rel":
        err = abs(fit / truth - 1.0)
        return err <= tol, (f"{fit:9.3f}  truth {truth:9.3f}  "
                            f"off {100 * err:4.1f}%  (<= {100 * tol:.0f}%)")
    err = abs(fit - truth)
    return err <= tol, (f"{fit:9.3f}  truth {truth:9.3f}  "
                        f"off {err:6.3f}  (<= {tol:g})")


def _environment(report, device, verbose):
    from brainsp import __version__
    import brainsp
    _print(verbose, "1. Environment")
    _line(report, "environment", "brainsp", True,
          f"{__version__}, Python {platform.python_version()}, "
          f"{platform.system()}", verbose)
    for module in ("numpy", "scipy", "astropy", "matplotlib", "yaml",
                   "PySide6"):
        try:
            mod = __import__(module)
            _line(report, "environment", module, True,
                  str(getattr(mod, "__version__", "installed")), verbose)
        except ImportError as exc:
            _line(report, "environment", module, False, str(exc), verbose)
    try:
        brainsp.set_device(device)
        import jax
        jax.config.update("jax_enable_x64", True)
        backend = jax.default_backend()
        devices = ", ".join(str(d) for d in jax.devices())
        _line(report, "environment", "JAX", True,
              f"{jax.__version__} on {backend} [{devices}]", verbose)
    except Exception as exc:
        _line(report, "environment", "JAX", False, str(exc), verbose)
        return
    if backend == "gpu":
        _gpu_computes(report, verbose)
    elif device == "gpu":
        _line(report, "environment", "GPU", False,
              "asked for the GPU but JAX found none: install BRAIN with "
              "the cuda extra, pip install \"brainsp[cuda] @ git+...\", and "
              "check the NVIDIA driver with nvidia-smi", verbose)


def _gpu_computes(report, verbose):
    """
    Run a double-precision product on the GPU and check the answer.

    JAX can list a GPU it cannot compute on: a card newer than the CUDA build
    JAX was compiled with (an RTX 50-series, Blackwell, with JAX older than
    0.6) is found, and the first kernel then fails. Listing the device is not
    the test; computing on it is.
    """
    import jax
    import jax.numpy as jnp
    gpu = jax.devices()[0]
    name = getattr(gpu, "device_kind", str(gpu))
    capability = getattr(gpu, "compute_capability", None)
    label = f"{name}" + (f", compute capability {capability}" if capability else "")
    try:
        x = jnp.ones((512, 512), dtype=jnp.float64)
        value = float(jnp.sum(x @ x))
        ok = value == 512.0 ** 3
        _line(report, "environment", "GPU computes", ok,
              f"{label}: float64 product {'correct' if ok else f'WRONG ({value})'}",
              verbose)
    except Exception as exc:
        _line(report, "environment", "GPU computes", False,
              f"{label}: {str(exc).splitlines()[0][:160]} -- a GPU newer than "
              f"this JAX build; update it: pip install -U \"jax[cuda12]\"",
              verbose)


def _libraries(report, verbose):
    from brainsp import libraries
    _print(verbose, "\n2. Template libraries")
    names = libraries.available()
    _line(report, "libraries", "libraries found", "BasesGM" in names,
          ", ".join(names) or "none", verbose)
    try:
        info = libraries.info("BasesGM")
        _line(report, "libraries", "BasesGM catalogue", info["n_ssp"] > 0,
              f"{info['n_ssp']} SSPs, {len(info['ages_yr'])} ages x "
              f"{len(info['metallicities'])} Z, "
              f"{info['wavelength_min']:.0f}-{info['wavelength_max']:.0f} A",
              verbose)
    except (Exception, SystemExit) as exc:
        _line(report, "libraries", "BasesGM catalogue", False, str(exc),
              verbose)
    try:
        info = libraries.info("M13")
        _line(report, "libraries", "M13 catalogue",
              info["n_ssp"] == 47 and len(info["agn"]) == 10,
              f"{info['n_ssp']} SSPs, {len(info['agn'])} AGN components "
              f"({', '.join(info['agn'][:3])}, ...)", verbose)
    except (Exception, SystemExit) as exc:
        _line(report, "libraries", "M13 catalogue", False, str(exc), verbose)
    from brainsp.brain_settings import read_catalogue
    try:
        _, agn = read_catalogue(libraries.agn_folder())
        _line(report, "libraries", "AGN continua", len(agn) == 7,
              f"{len(agn)} templates", verbose)
    except (Exception, SystemExit) as exc:
        _line(report, "libraries", "AGN continua", False, str(exc), verbose)


def _spectra(report, folder, names, verbose):
    import brainsp
    _print(verbose, "\n4. Mock spectra")
    rest_twin = None
    for name in names:
        path = os.path.join(folder, "spectra", f"{name}.spec")
        truth = json.load(open(os.path.join(folder, "spectra",
                                            f"{name}.truth.json")))
        t0 = time.time()
        try:
            r = brainsp.fit_spectrum(path, output_dir=os.path.join(
                folder, "fits"), quiet=True, make_plot=False,
                **FIT_OPTIONS[name])
        except (Exception, SystemExit) as exc:
            _line(report, name, f"{name}: fit", False, str(exc)[:200],
                  verbose)
            continue
        _print(verbose, f"  {name}  ({time.time() - t0:.1f} s, "
                        f"chi2/N {r.chi2:.2f})")
        for label, key, read, tol, kind in SPECTRUM_CHECKS:
            ok, detail = _compare(read(r), truth[key], tol, kind)
            _line(report, name, f"  {label}", ok, detail, verbose)
        if truth["agn_fraction"]:
            ok, detail = _compare(r.agn_fraction,
                                  100 * truth["agn_fraction"], 8.0, "abs")
            _line(report, name, "  AGN light fraction [%]", ok, detail,
                  verbose)
        if truth["lines"]:
            ok, detail = _compare(r.v_gas, truth["v_gas"], 15.0, "abs")
            _line(report, name, "  gas velocity [km/s]", ok, detail, verbose)
            observed = r.line_fluxes(units="normalised", observed=True)
            ha = observed.get("Halpha_6563", (None,))[0]
            ok, detail = _compare(ha, truth["lines"]["Halpha_6563"], 0.15,
                                  "rel")
            _line(report, name, "  H-alpha flux (observed)", ok, detail,
                  verbose)
        if name == "mock_old_population":
            rest_twin = r
        if name == "mock_observed_frame" and rest_twin is not None:
            ok, detail = _compare(r.v0, rest_twin.v0, 5.0, "abs")
            _line(report, name, "  same v as the rest-frame twin", ok,
                  detail, verbose)
            ok, detail = _compare(r.av, rest_twin.av, 0.05, "abs")
            _line(report, name, "  same A_V as the rest-frame twin", ok,
                  detail, verbose)


def _cube(report, folder, quick, verbose):
    import brainsp
    from astropy.io import fits
    _print(verbose, "\n5. Mock datacube (16 x 16)")
    cube = os.path.join(folder, "cube", "mock_cube.fits")
    out = os.path.join(folder, "cube_fit")
    threshold = 20 if quick else 3
    settings = brainsp.Settings(
        nebular_mode="spop" if quick else "espop", make_plot=False)
    t0 = time.time()
    try:
        written = brainsp.CubePipeline(
            cube, redshift=0.02, settings=settings, wavelength_scale="vacuum",
            sn_mask=[3, 10, 20], fit_sn_threshold=threshold,
            output_dir=out).run(log=lambda text: None)
    except (Exception, SystemExit) as exc:
        _line(report, "cube", "pipeline", False, str(exc)[:300], verbose)
        return None
    ok = len(written) == 1 and os.path.isfile(written[0])
    _line(report, "cube", "megacube written", ok,
          f"{time.time() - t0:.0f} s, {os.path.basename(written[0]) if ok else ''}",
          verbose)
    if not ok:
        return None
    c = brainsp.read_megacube(written[0])
    truth = fits.open(cube.replace(".fits", ".truth.fits"))
    # The mask BRAIN measured on the noisy cube is what the fit used; the
    # true S/N differs from it near the threshold, so it is the mask the
    # count is checked against.
    selected = np.asarray(c.h[f"SN_MASKS_{threshold}"].data) > 0
    fitted = np.isfinite(c.plane("vrot_star"))
    _line(report, "cube", "spaxels fitted = S/N mask",
          int(fitted.sum()) == int(selected.sum()),
          f"{int(fitted.sum())} fitted, {int(selected.sum())} in "
          f"SN_MASKS_{threshold}", verbose)
    both = fitted & (truth["INSIDE"].data > 0)
    for label, plane, key, tol in (
            ("velocity map, median offset", "vrot_star", "V", 8.0),
            ("dispersion map, median offset", "Sigma_star", "SIGMA", 12.0),
            ("A_V map, median offset", "Av", "AV", 0.08)):
        d = (c.plane(plane) - truth[key].data)[both]
        med = float(np.median(d)) if d.size else float("nan")
        _line(report, "cube", label, abs(med) <= tol,
              f"{med:7.3f}  (|.| <= {tol:g}), rms "
              f"{float(np.sqrt(np.mean(d ** 2))):.3f}", verbose)
    v = c.plane("vrot_star")[both]
    tv = truth["V"].data[both]
    corr = float(np.corrcoef(v, tv)[0, 1]) if v.size > 2 else float("nan")
    _line(report, "cube", "velocity field correlation", corr > 0.97,
          f"r = {corr:.4f}  (> 0.97)", verbose)
    return c


def _outputs(report, folder, cube, verbose):
    import brainsp
    _print(verbose, "\n6. Outputs")
    plots = os.path.join(folder, "plots")
    try:
        if cube is not None:
            brainsp.plots.map(cube, "vrot_star",
                              os.path.join(plots, "velocity.png"))
            written = brainsp.plots.export_maps(
                cube, os.path.join(plots, "maps"), ["vrot_star", "Av"],
                formats=("png", "fits"))
            ok = all(os.path.isfile(p) for p in written)
            _line(report, "outputs", "map export (png, fits)", ok,
                  f"{len(written)} files", verbose)
        fits_dir = os.path.join(folder, "fits")
        summary = os.path.join(fits_dir, "mock_old_population_summary.txt")
        if os.path.isfile(summary):
            r = brainsp.read_result(summary)
            brainsp.plots.spectrum(r, os.path.join(plots, "spectrum.png"))
            brainsp.plots.sfh(r, os.path.join(plots, "sfh.png"))
            _line(report, "outputs", "spectrum and SFH figures", True,
                  plots, verbose)
    except Exception as exc:
        _line(report, "outputs", "figures", False, str(exc)[:200], verbose)
    try:
        script = os.path.join(folder, "run_mock.py")
        brainsp.write_script(script, spectra=[os.path.join(
            folder, "spectra", "mock_old_population.spec")])
        done = subprocess.run([sys.executable, script, "--check"],
                              capture_output=True, text=True, timeout=300)
        _line(report, "outputs", "exported run script validates",
              done.returncode == 0,
              (done.stdout or done.stderr).strip().splitlines()[-1][:120]
              if (done.stdout or done.stderr) else "", verbose)
    except Exception as exc:
        _line(report, "outputs", "exported run script", False, str(exc),
              verbose)

def run(quick=False, device="auto", folder=None, keep=None, verbose=True):
    """
    Run the self-test.

    Parameters
    ----------
    quick : bool
        Two spectra and the cube's brightest spaxels, in ``spop``, instead
        of everything.
    device : {"auto", "cpu", "gpu"}
        Where to fit. Must be chosen before anything else in this Python
        session has used JAX.
    folder : str, optional
        Where to write the mock data and the fits; a temporary folder by
        default.
    keep : bool, optional
        Keep the folder afterwards. By default a temporary folder is removed
        when everything passed and kept when something failed, so the
        evidence is there to look at.
    verbose : bool
        Print each check as it runs.

    Returns
    -------
    Report
    """
    import logging
    import brainsp
    start = time.time()
    temporary = folder is None
    folder = os.path.abspath(folder or tempfile.mkdtemp(prefix="brainsp-"))
    report = Report(folder)
    _print(verbose, f"brainsp self-test ({'quick' if quick else 'full'}), "
                    f"working in {folder}\n")
    quiet = logging.getLogger("NEWBRAIN")
    level = quiet.level
    quiet.setLevel(logging.ERROR)
    try:
        _environment(report, device, verbose)
        _libraries(report, verbose)
        _print(verbose, "\n3. Mock data")
        t0 = time.time()
        brainsp.mock.write_examples(folder, cube=True)
        _line(report, "mock", "mock spectra and 16 x 16 cube written", True,
              f"{time.time() - t0:.1f} s", verbose)
        names = (["mock_old_population", "mock_star_forming_lines"] if quick
                 else list(FIT_OPTIONS))
        _spectra(report, folder, names, verbose)
        cube = _cube(report, folder, quick, verbose)
        _outputs(report, folder, cube, verbose)
    finally:
        quiet.setLevel(level)
    report.seconds = time.time() - start
    n_bad = sum(not c["ok"] for c in report.checks)
    _print(verbose, "\n" + "=" * 70)
    if report.passed:
        _print(verbose, f"  ALL {len(report.checks)} CHECKS PASSED in "
                        f"{report.seconds:.0f} s. BRAIN is working.")
    else:
        _print(verbose, f"  {n_bad} OF {len(report.checks)} CHECKS FAILED. "
                        f"The data and fits are in {folder}")
        for c in report.checks:
            if not c["ok"]:
                _print(verbose, f"    {c['section']}: {c['name'].strip()}: "
                                f"{c['detail']}")
    _print(verbose, "=" * 70)
    remove = (keep is False) or (keep is None and temporary and report.passed)
    if remove:
        shutil.rmtree(folder, ignore_errors=True)
    return report
