"""
BRAIN: which extinction curves exist, and what dust each one describes.

The registry and the curves are separated because they are asked for by
different things. A fit asks for k(lambda) and needs the numerics, which are
JAX; a menu asks what it may offer and needs only the names. Keeping the
names here lets the desktop interface fill its menu without starting a JAX
backend in the process that draws it, which is what makes the device selector
able to select a device at all.

`reddening_laws` imports these names and re-exports them, so
`from reddening_laws import LAWS` continues to mean exactly what it meant.
There is one registry, and it is this one.
"""

LAWS = {
    "CCM":  ("Cardelli, Clayton & Mathis (1989)",
             "Milky Way average, R_V = 3.1. The safe default."),
    "HZ1":  ("Allen (1976)",
             "Milky Way, tabulated and interpolated."),
    "HZ2":  ("Seaton (1979), fit by Fitzpatrick (1986)",
             "Milky Way."),
    "CAL":  ("Calzetti et al. (2000)",
             "Starburst galaxies, R_V = 4.05."),
    "CCC":  ("Calzetti (2000) + Leitherer (2002)",
             "Starburst, with a far-UV extension below 1846 A."),
    "HZ5":  ("Calzetti (2000), starburst (SB) form",
             "Starburst, identical curve to CAL."),
    "HZ3":  ("Fitzpatrick (1986)",
             "Large Magellanic Cloud."),
    "GD3":  ("Gordon et al. (2003)",
             "LMC average."),
    "GD2":  ("Gordon et al. (2003)",
             "LMC2 supershell, near 30 Doradus."),
    "HZ4":  ("Prevot (1984) & Bouchet (1985)",
             "Small Magellanic Cloud."),
    "GD1":  ("Gordon et al. (2003)",
             "SMC bar. Steep far-UV rise, no 2175 A bump."),
    "GD24": ("Gordon et al. (2024)",
             "Small Magellanic Cloud."),
}

LAW_NAMES = tuple(LAWS)

def describe(key):
    """One-line summary of a law, for menus and error messages."""
    ref, what = LAWS[key]
    return f"{what}  -  {ref}"
