Source code for brainsp.scripts

"""
Runs written out as one Python file each.

A BRAIN run is completely described by one ``.py`` file: every setting, with
its documentation as comments, every spectrum or cube with its own
corrections, every preparation stage of a datacube. ``python run.py`` runs
it, on any machine with brainsp installed. The desktop interface's "Export
run script" writes these files, and so does ``brainsp init``.

Paths inside the folder the script is saved in are written relative to that
folder, and the script finds them from its own location, wherever it is run
from. So a folder holding the data and the script can be copied to a server
as it is and run there with ``python run.py``. A path outside that folder is
kept as it is and listed at the top of the script, because it will not
travel with the folder.

Each script understands three options::

    python run.py                 run it
    python run.py --check         check everything, fit nothing
    python run.py --device gpu    choose the processor (auto, cpu, gpu)

and a spectrum script also takes file names, to fit those instead::

    python run.py more/*.spec
"""

import datetime
import os

from brainsp import brain_settings as _bs

__all__ = ["settings_block", "spectrum_script", "cube_script",
           "write_script"]

HEADER = '''"""
{title}

Written by brainsp {version} on {when}. Every setting of the run is in this
file; edit it here and run it again:

    python {script}                  run it
    python {script} --check          check everything, fit nothing
    python {script} --device gpu     choose the processor: auto, cpu or gpu
{extra}
Each setting is documented where it is set. The same text, and more, is in
the BRAIN documentation (Settings reference), and in Python:
    import brainsp; print(brainsp.Settings.describe("dust_range"))
"""

import os
import sys

import brainsp

HERE = os.path.dirname(os.path.abspath(__file__))


def here(path):
    """A path written relative to this file's folder, made usable from any
    working directory. Absolute paths are left as they are."""
    return path if os.path.isabs(path) else os.path.join(HERE, path)
'''

# Settings that may name a file, made relative like the data.
PATH_SETTINGS = ("instrumental_resolution", "library_resolution",
                 "base_folder")


def _portable(value, base, outside):
    """
    `value` as a script saved in folder `base` should write it.

    A path inside `base` becomes relative to it, with forward slashes, so
    the folder can be copied anywhere. A path outside it stays absolute and
    is added to `outside`. Anything that is not a path to an existing file
    or folder, and is not absolute -- a library name, "2.5 A", an output
    folder not made yet -- is left as it is.
    """
    if not isinstance(value, (str, os.PathLike)):
        return value
    text = str(value)
    if not text or not (os.path.isabs(text) or os.path.exists(text)):
        return text
    full = os.path.abspath(text)
    try:
        relative = os.path.relpath(full, base)
    except ValueError:              # another drive, on Windows
        relative = None
    if relative is not None and relative != os.pardir \
            and not relative.startswith(os.pardir + os.sep):
        return relative.replace(os.sep, "/")
    outside.append(full)
    return full


def _portable_output(value, base, outside):
    """An output folder: a relative one already means beside the script."""
    text = str(value)
    return text if not os.path.isabs(text) else _portable(text, base, outside)


def _outside_note(outside):
    """The warning at the top of a script that names paths outside it."""
    if not outside:
        return ""
    lines = ["", "# NOTE: these paths are outside this script's folder and do "
             "not travel",
             "# with it. Copy them along and edit the paths below, or move "
             "them into", "# the folder and export again:"]
    lines += [f"#   {p}" for p in dict.fromkeys(outside)]
    return "\n".join(lines) + "\n"

# Settings a script sets for itself, so they are not repeated in SETTINGS.
_SET_BY_SCRIPT = ("spectrum", "output_folder")


def _comment(text, indent):
    out = []
    for line in text.splitlines():
        out.append((indent + "# " + line).rstrip())
    return "\n".join(out)

def settings_block(settings, name="SETTINGS", skip=_SET_BY_SCRIPT):
    """
    Python source building a :class:`~brainsp.Settings` with every setting,
    each under its documentation, grouped by section.
    """
    docs = _bs.setting_docs()
    lines = [f"{name} = brainsp.Settings("]
    section = None
    for key in _bs.DEFAULTS:
        if key in skip:
            continue
        doc = docs.get(key, {"section": "", "text": ""})
        if doc["section"] and doc["section"] != section:
            section = doc["section"]
            lines.append("")
            lines.append(f"    # {'=' * 70}")
            lines.append(f"    # {section}")
            lines.append(f"    # {'=' * 70}")
        if doc["text"]:
            lines.append("")
            lines.append(_comment(doc["text"], "    "))
        value = settings.get(key, _bs.DEFAULTS[key])
        default = _bs.DEFAULTS[key]
        mark = "" if value == default else f"   # default: {default!r}"
        lines.append(f"    {key}={value!r},{mark}")
    lines.append(")")
    return "\n".join(lines)

def spectrum_script(settings, spectra, output_dir="brain_out", device="auto",
                    title="A BRAIN run: spectra.", script_dir=None,
                    outside=None):
    """
    The source of a script that fits `spectra` with `settings`.

    Parameters
    ----------
    settings : Settings or dict
    spectra : list of (str, dict)
        Each spectrum's path and its own settings (typically the pre-fit
        corrections: ``spectrum_redshift``, ``spectrum_in_rest_frame``,
        ``galactic_ebv``, ``galactic_rv``, ``galactic_law``).
    output_dir : str
        Relative paths are taken from the script's folder.
    device : str
    script_dir : str, optional
        The folder the script will be saved in (default: the working
        directory). Paths inside it are written relative to it.
    outside : list, optional
        Collects the paths that are outside `script_dir`.
    """
    from brainsp import __version__
    base = os.path.abspath(script_dir or os.getcwd())
    outside = [] if outside is None else outside
    rows = []
    for path, extra in spectra:
        rows.append(f"    {_portable(path, base, outside)!r}: {dict(extra)!r},")
    mask = settings.get("mask")
    portable = {k: _portable(settings.get(k), base, outside)
                for k in PATH_SETTINGS if settings.get(k)}
    output_dir = _portable_output(output_dir, base, outside)
    text = HEADER.format(
        title=title, version=__version__,
        when=datetime.datetime.now().strftime("%Y-%m-%d %H:%M"),
        script="run.py", extra="    python run.py more/*.spec       fit "
        "these files instead\n")
    text += _outside_note(outside)
    # The mask travels in this file (MASK below), not as a path.
    text += "\n" + settings_block(dict(settings, mask=None, **portable)) + "\n\n"
    text += ("# Each spectrum, with the settings that are its own: the redshift\n"
             "# BRAIN removes (spectrum_in_rest_frame False) or that was already\n"
             "# removed (True), and the Milky Way's reddening, E(B-V), R_V and law.\n")
    text += ("# Paths relative to this file's folder.\n"
             "SPECTRA = {\n" + "\n".join(rows) + "\n}\n\n")
    text += ("# Where the results go: <name>_summary.txt, <name>_fitted.txt and,\n"
             "# with make_plot, <name>_fit.pdf. Relative to this file's folder.\n")
    text += f"OUTPUT_DIR = {output_dir!r}\n\n"
    text += "# The processor: auto (a GPU when JAX sees one), cpu or gpu.\n"
    text += f"DEVICE = {device!r}\n"
    if mask and os.path.isfile(str(mask)):
        from brainsp.io_utils import read_mask_regions
        regions, _ = read_mask_regions(str(mask))
        text += ("\n# The wavelength mask: (start, end, weight) in Angstrom; "
                 "weight 0\n# leaves the pixels out, above 1 counts them for "
                 "more.\nMASK = [\n" + "\n".join(
                     f"    ({lo:.2f}, {hi:.2f}, {w:g}),"
                     + (f"   # {label}" if label else "")
                     for lo, hi, w, label in regions) + "\n]\n")
    else:
        text += "\n# The wavelength mask, as (start, end, weight); none.\nMASK = []\n"
    text += '''

def main(argv):
    flags = [a for a in argv[1:] if a.startswith("--")]
    files = [a for a in argv[1:] if not a.startswith("--")]
    device = DEVICE
    if "--device" in argv:
        device = argv[argv.index("--device") + 1]
        files = [f for f in files if f != device]
    out = here(OUTPUT_DIR)
    settings = SETTINGS.copy()
    for key in ("instrumental_resolution", "library_resolution",
                "base_folder"):
        value = settings.get(key)
        if isinstance(value, str) and value and os.path.exists(here(value)):
            settings[key] = here(value)
    if MASK:
        os.makedirs(out, exist_ok=True)
        mask_file = os.path.join(out, "mask.sm")
        brainsp.write_mask(mask_file, MASK)
        settings["mask"] = mask_file
    listed = {here(f): extra for f, extra in SPECTRA.items()}
    spectra = [os.path.abspath(f) for f in files] or list(listed)
    corrections = {f: listed.get(f, next(iter(listed.values()), {}))
                   for f in spectra}
    missing = [f for f in spectra if not os.path.isfile(f)]
    if missing:
        print("These spectra are not here:\\n  " + "\\n  ".join(missing))
        return 1
    if "--check" in flags:
        for f in spectra:
            settings.copy(spectrum=f, **corrections[f]).validate()
        print(f"{len(spectra)} spectra, all valid. Output: {out}")
        return 0
    results = brainsp.fit_spectra(spectra, settings, corrections=corrections,
                                  output_dir=out, device=device)
    failed = [r for r in results if r.error_message]
    print(f"\\nDone: {len(results) - len(failed)} fitted, {len(failed)} "
          f"failed. Results in {out}")
    for r in results:
        print(f"  {r!r}")
    return 1 if failed else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
'''
    return text

def cube_script(pipeline, device="auto", title="A BRAIN run: datacubes.",
                script_dir=None, outside=None):
    """
    The source of a script that runs a :class:`~brainsp.CubePipeline`.

    Paths inside `script_dir` (default: the working directory), the folder
    the script will be saved in, are written relative to it; `outside`, if
    given, collects the ones that are not.
    """
    from brainsp import __version__
    base = os.path.abspath(script_dir or os.getcwd())
    outside = [] if outside is None else outside
    settings = dict(pipeline.settings)
    settings.update({k: _portable(settings.get(k), base, outside)
                     for k in PATH_SETTINGS if settings.get(k)})
    values = {}
    for key, _ in _PIPELINE_DOCS:
        value = getattr(pipeline, key)
        if key in ("cubes",):
            value = [_portable(c, base, outside) for c in value]
        elif key == "output_dir":
            value = _portable_output(value, base, outside)
        elif key == "per_cube" and isinstance(value, (str, os.PathLike)):
            value = _portable(value, base, outside)
        elif key == "per_cube" and value:
            value = [{k: (_portable(v, base, outside)
                          if k in ("target", "mask file") else v)
                      for k, v in dict(entry).items()} for entry in value]
        values[key] = value
    text = HEADER.format(
        title=title, version=__version__,
        when=datetime.datetime.now().strftime("%Y-%m-%d %H:%M"),
        script="run.py", extra="")
    text += _outside_note(outside)
    text += "\n" + settings_block(settings) + "\n\n"
    text += "PIPELINE = dict(\n"
    for key, doc in _PIPELINE_DOCS:
        value = values[key]
        if key == "wavelength_mask" and isinstance(value, (str, os.PathLike)) \
                and os.path.isfile(str(value)):
            from brainsp.io_utils import read_mask_regions
            regions, _ = read_mask_regions(str(value))
            value = [(round(lo, 2), round(hi, 2), w) for lo, hi, w, _ in regions]
        text += "\n" + _comment(doc, "    ") + "\n"
        if isinstance(value, list) and len(value) > 3:
            text += f"    {key}=[\n" + "".join(
                f"        {v!r},\n" for v in value) + "    ],\n"
        else:
            text += f"    {key}={value!r},\n"
    text += ")\n\n"
    text += "# The processor: auto (a GPU when JAX sees one), cpu or gpu.\n"
    text += f"DEVICE = {device!r}\n"
    text += '''

def main(argv):
    device = DEVICE
    if "--device" in argv:
        device = argv[argv.index("--device") + 1]
    options = dict(PIPELINE)
    options["output_dir"] = here(options["output_dir"])
    options["cubes"] = [here(c) for c in options["cubes"]]
    per_cube = options.get("per_cube")
    if isinstance(per_cube, str):
        options["per_cube"] = here(per_cube)
    elif per_cube:
        options["per_cube"] = [
            {k: (here(v) if k in ("target", "mask file") and isinstance(v, str)
                 and v else v) for k, v in entry.items()}
            for entry in per_cube]
    settings = SETTINGS.copy()
    for key in ("instrumental_resolution", "library_resolution",
                "base_folder"):
        value = settings.get(key)
        if isinstance(value, str) and value and os.path.exists(here(value)):
            settings[key] = here(value)
    pipeline = brainsp.CubePipeline(settings=settings, **options)
    if "--check" in argv:
        pipeline.validate()
        print(f"{len(pipeline.cubes)} cube(s), valid. Output: "
              f"{options['output_dir']}")
        return 0
    written = pipeline.run(device=device)
    print("\\nMegacubes written:\\n  " + "\\n  ".join(written))
    return 0 if len(written) == len(pipeline.cubes) else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv))
'''
    return text

def write_script(path, kind="spectrum", settings=None, spectra=(),
                 pipeline=None, device="auto", **pipeline_options):
    """
    Write a run as one Python file.

    Parameters
    ----------
    path : str
        The ``.py`` file to write.
    kind : {"spectrum", "cube"}
    settings : Settings, dict or str, optional
    spectra : list of str or (str, dict)
        For ``kind="spectrum"``: the spectra, optionally with their own
        settings.
    pipeline : CubePipeline, optional
        For ``kind="cube"``; or give its options as keyword arguments
        (``cubes=``, ``redshift=`` ...).
    device : str

    Returns
    -------
    str
        `path`.

    Examples
    --------
    >>> brainsp.write_script("run_spectra.py", spectra=["a.spec", "b.spec"],
    ...                      settings=brainsp.Settings(nebular_mode="espop"))
    >>> brainsp.write_script("run_cube.py", kind="cube", cubes=["c.fits"],
    ...                      redshift=0.02)
    """
    from brainsp.api import Settings, CubePipeline
    s = settings if isinstance(settings, Settings) else Settings(settings)
    if kind == "spectrum":
        entries = [(e, {}) if isinstance(e, (str, os.PathLike)) else e
                   for e in spectra]
        source = spectrum_script(s, entries, device=device,
                                 script_dir=os.path.dirname(
                                     os.path.abspath(path)))
    elif kind == "cube":
        if pipeline is None:
            pipeline = CubePipeline(settings=s, **pipeline_options)
        source = cube_script(pipeline, device=device,
                             script_dir=os.path.dirname(os.path.abspath(path)))
    else:
        raise ValueError("kind is 'spectrum' or 'cube'")
    with open(path, "w") as handle:
        handle.write(source)
    return path
