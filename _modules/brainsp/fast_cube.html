Source code for brainsp.fast_cube

"""
A faster batch fitter for datacubes, kept apart from the one in use.

fit_megacube_jax._fit_batch fits one batch of spaxels. This module offers a
drop-in replacement, fit_batch_fast, with the same arguments and the same
return values, and changes only how the work is done, not what is computed.
It is a separate file so that it can be measured against the reference on the
same spaxels before anything in the pipeline depends on it.

Three changes, each aimed at a cost measured on a real MaNGA batch:

1. THE NNLS IS WARM-STARTED. Every forward evaluation solved the
   non-negative least squares from an empty active set, rediscovering the
   ~16 templates with weight, one per sequential step. Between successive LM
   iterations, and between the perturbed evaluations around one point, that
   set barely moves. solver.solve_active_set accepts a starting set and is
   documented, and tested, to reach the same vertex from any start: the
   start changes the path, not the answer. Here the set is carried through
   the LM loop, the clipping pass and the final evaluation. Measured on 16
   spaxels: 23.6 ms cold, 2.7 ms warm, weights identical to 0.0.

2. THE JACOBIAN IS ANALYTIC (VARIABLE PROJECTION). The LM differentiates
   the projected residual  rho(theta) = b - A(theta) w*(theta), where w* is
   the NNLS solution. The reference does it by central differences: 2 x 8
   perturbed evaluations per iteration, each a full template build and a
   full NNLS. On the active set P the NNLS solution is the least-squares
   solution on the columns in P, so the derivative has a closed form
   (Golub & Pereyra 1973, the full form, not Kaufman's approximation):

       d rho / d theta_k = -(I - Pi_P) (dA_k w)  -  A_P G^-1 [dA_k^T rho]_P

   with G = A_P^T A_P and Pi_P = A_P G^-1 A_P^T. dA_k w and dA_k^T rho come
   from one forward-mode pass of JAX through the model, and no NNLS is
   solved at all. This is the derivative central differences approximate,
   without their truncation error. It is exact while P is fixed, which is
   what central differences also assume unless a step happens to move a
   template in or out of P. It covers the unconstrained problem only; with
   the simplex constraint there is an extra multiplier term, so a simplex
   fit keeps central differences, warm-started.

3. THE VELOCITY GRID RUNS AS A SCAN. The initial-velocity search evaluated
   all 47 grid points at once with vmap, so a batch held 47 copies of the
   template matrix per spaxel: that is what ran a 12 GB GPU out of memory at
   32 spaxels and forced 16. The points are now visited in order with
   lax.scan, which carries the active set from one velocity to the next as
   well. The chi-squared at each point is the same number, so the argmin is
   the same point.

Nothing here touches the physics: the model matrix is built by the same
functions in the same order as optimizer._forward_full, the LM keeps the
reference's damping, acceptance and convergence rules line for line, and the
clipping pass is the reference's.
"""

class Templates(NamedTuple):
    """
    The stellar template matrix with its Fourier transform, computed once.

    Broadening is a product in the Fourier domain, and the templates are the
    same for every spaxel and every evaluation: their transform was computed
    afresh each time, per spaxel. It is computed once per batch here, and an
    evaluation only multiplies by the kernel and transforms back. n_left and
    n_fft are the padding and transform length, fixed for the batch.
    """
    log: jax.Array
    fft: jax.Array
    n_left: int
    n_fft: int

    @property
    def shape(self):
        return self.log.shape




def _templates(M_star, pad_pixels):
    """Pad and transform the templates once (spectral_engine's padding)."""
    n_left, n_right, n_fft = _broadening_pad(M_star.shape[0], pad_pixels)
    return Templates(M_star, rfft(_edge_pad(M_star, n_left, n_right), axis=0),
                     n_left, n_fft)


def _broaden(T, sigma_v, v0, velscale):
    """
    spectral_engine.fourier_broaden_matrix, from the precomputed transform:
    the same kernel, the same inverse transform and crop, the same shortcut
    for a kernel narrower than 0.05 pixel.
    """
    n_pix = T.log.shape[0]
    sigma_pix = sigma_v / velscale
    shift_pix = v0 / velscale
    omega = 2.0 * jnp.pi * rfftfreq(T.n_fft)
    kernel = jnp.exp(-0.5 * (omega * sigma_pix) ** 2 - 1j * omega * shift_pix)
    M_broad = irfft(T.fft * kernel[:, jnp.newaxis], n=T.n_fft,
                    axis=0)[T.n_left:T.n_left + n_pix]
    return jnp.where((sigma_pix < 0.05) & (jnp.abs(shift_pix) < 0.05),
                     T.log, M_broad)


def _model_matrix(params, M_star, lam, velscale, dk, yav_flags,
                  rest_wl, ratios, n_comp, gids, max_comp):
    """The unweighted model matrix, built exactly as optimizer builds it."""
    av = params[0]
    yav = params[1]
    sigma_star = params[2]
    v0_star = params[3]
    gas_p = params[4:]
    sigma_gas = gas_p[::2]
    v0_gas = gas_p[1::2]

    M_broad = _broaden(M_star, sigma_star, v0_star, velscale)
    atten = 10.0 ** (-0.4 * av * dk)
    atten_yav = 10.0 ** (-0.4 * (av + yav) * dk)
    M_star_dust = jnp.where(
        yav_flags[jnp.newaxis, :] > 0,
        M_broad * atten_yav[:, jnp.newaxis],
        M_broad * atten[:, jnp.newaxis],
    )
    M_gas = _build_emission_jit(
        lam, velscale, sigma_gas, v0_gas,
        rest_wl, ratios, n_comp, gids, max_comp,
    )
    M_gas_dust = M_gas * atten[:, jnp.newaxis]
    return jnp.hstack([M_star_dust, M_gas_dust]), M_star_dust, M_gas_dust


def _augmented(M_full, flux, isig, D_full):
    A = M_full * isig[:, jnp.newaxis]
    b = flux * isig
    A_aug = jnp.vstack([A, D_full])
    b_aug = jnp.concatenate([b, jnp.zeros(D_full.shape[0])])
    return A_aug, b_aug


def _cvec(n, n_star, simplex):
    return (jnp.where(jnp.arange(n) < n_star, 1.0, 0.0) if simplex
            else jnp.ones(n))


def _solve(A_aug, b_aug, n_star, simplex, P, ATA=None):
    """
    NNLS on the augmented system, starting from P.

    An all-False P is the cold start: solve_active_set falls back to its own
    cold set when the set it is handed is empty. ATA may be passed in when the
    caller has it already.
    """
    if ATA is None:
        ATA = A_aug.T @ A_aug
    ATb = A_aug.T @ b_aug
    return solve_active_set(ATA, ATb, _cvec(ATb.shape[0], n_star, simplex),
                            P_init=P, simplex=simplex)


def _evaluate(params, P, flux, isig, M_star, lam, velscale, dk, D_full,
              yav_flags, rest_wl, ratios, n_comp, gids, max_comp, simplex):
    """
    One forward evaluation: rho = b - A w, the weights, and the new active set.

    rho differs from optimizer._forward_residual only in the sign of the
    regularisation rows (b - A w gives -D w there, the reference returns
    +D w). The sum of squares is the same, and since the Jacobian is taken of
    this same rho, every LM step is the same.
    """
    rho, w, P_new, _ = _evaluate_full(params, P, flux, isig, M_star, lam,
                                      velscale, dk, D_full, yav_flags, rest_wl,
                                      ratios, n_comp, gids, max_comp, simplex)
    return rho, w, P_new


def _evaluate_full(params, P, flux, isig, M_star, lam, velscale, dk, D_full,
                   yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
                   simplex):
    """
    _evaluate, also returning (M_full, A^T A), which the Jacobian at the same
    parameters needs: the LM keeps them for the point it accepts, rather than
    building the model and its Gram matrix a second time there.
    """
    M_full, _, _ = _model_matrix(params, M_star, lam, velscale, dk, yav_flags,
                                 rest_wl, ratios, n_comp, gids, max_comp)
    A_aug, b_aug = _augmented(M_full, flux, isig, D_full)
    ATA = A_aug.T @ A_aug
    w, P_new = _solve(A_aug, b_aug, M_star.shape[1], simplex, P, ATA=ATA)
    return b_aug - A_aug @ w, w, P_new, (M_full, ATA)


def _varpro_jacobian(params, w, rho, P, flux, isig, M_star, lam, velscale,
                     dk, D_full, yav_flags, rest_wl, ratios, n_comp, gids,
                     max_comp, cache=None):
    """
    d rho / d params on the fixed active set P, by Golub-Pereyra.

    The two products the formula needs, dA_k w (holding the weights) and
    dA_k^T rho (holding the residual), are contracted from the few
    derivatives of the model that exist, rather than from a full derivative
    of the model matrix along every parameter:

      A_V, Y_V   scale columns, so d/dA_V of a column is c dk times it
                 (c = -0.4 ln 10); Y_V only the flagged templates'.
      sigma*, v* one forward pass of the broadening along each.
      gas        a line column depends on its own group's (sigma, v) only,
                 so ONE pass along sigma and one along v, applied to every
                 group at once, gives each column's derivative with respect
                 to its own group; the per-group products follow by summing
                 over that group's columns.

    Pushing the model forward along all 4 + 2 x groups directions built a
    full (pixels x columns) derivative of the matrix 16 times in espop with
    six groups; this builds four, and the Jacobian is the same numbers.
    The regularisation rows do not depend on the parameters.

    The projections are solves with the passive block of A^T A, set up the
    way solve_active_set sets up its own KKT system: off-passive rows and
    columns replaced by the identity scaled to the matrix, right-hand sides
    zeroed off P, so the solution is the passive-set solution and exactly
    zero elsewhere.
    """
    n_ssp = M_star.shape[1]
    n_pix = flux.shape[0]
    av, yav, sigma_star, v0_star = params[0], params[1], params[2], params[3]
    sigma_gas, v0_gas = params[4::2], params[5::2]
    n_groups = sigma_gas.shape[0]
    c = -0.4 * jnp.log(10.0)
    flag = yav_flags > 0

    atten = 10.0 ** (-0.4 * av * dk)
    atten_yav = 10.0 ** (-0.4 * (av + yav) * dk)
    a_cols = jnp.where(flag[jnp.newaxis, :], atten_yav[:, jnp.newaxis],
                       atten[:, jnp.newaxis])

    dB2 = jax.jacfwd(lambda q: _broaden(M_star, q[0], q[1], velscale))(
        jnp.array([sigma_star, v0_star]))
    dB, dB_v = dB2[..., 0], dB2[..., 1]
    w_s = w[:n_ssp]

    def gas(shift):
        return _build_emission_jit(lam, velscale, sigma_gas + shift[0],
                                   v0_gas + shift[1], rest_wl, ratios, n_comp,
                                   gids, max_comp)
    dG2 = jax.jacfwd(gas)(jnp.zeros(2))
    dG_s, dG_v = dG2[..., 0], dG2[..., 1]
    if cache is None:
        M_full, _, _ = _model_matrix(params, M_star, lam, velscale, dk,
                                     yav_flags, rest_wl, ratios, n_comp,
                                     gids, max_comp)
        ATA = None
    else:
        M_full, ATA = cache
    M_s, M_g = M_full[:, :n_ssp], M_full[:, n_ssp:]
    w_g = w[n_ssp:]
    onehot = (gids[:, jnp.newaxis] == jnp.arange(n_groups)[jnp.newaxis, :])

    q = isig * rho[:n_pix]                     # data rows of the residual
    model = M_s @ w_s + M_g @ w_g

    # dA_k w, on the data rows: one column per parameter, in params order.
    u_av = c * dk * model
    u_yav = c * dk * (M_s @ jnp.where(flag, w_s, 0.0))
    u_sig = (dB * a_cols) @ w_s
    u_v = (dB_v * a_cols) @ w_s
    u_sg = atten[:, jnp.newaxis] * (dG_s @ (w_g[:, jnp.newaxis] * onehot))
    u_vg = atten[:, jnp.newaxis] * (dG_v @ (w_g[:, jnp.newaxis] * onehot))
    u_gas = jnp.stack([u_sg, u_vg], axis=2).reshape(n_pix, 2 * n_groups)
    U = jnp.concatenate([jnp.stack([u_av, u_yav, u_sig, u_v], axis=1),
                         u_gas], axis=1) * isig[:, jnp.newaxis]
    dAw = jnp.concatenate([U, jnp.zeros((D_full.shape[0], U.shape[1]))], 0)

    # dA_k^T rho: one column per parameter, a value per model column.
    qc = c * dk * q
    z_av = jnp.concatenate([M_s.T @ qc, M_g.T @ qc])
    z_yav = jnp.concatenate([jnp.where(flag, M_s.T @ qc, 0.0),
                             jnp.zeros(w_g.shape[0])])
    z_sig = jnp.concatenate([(dB * a_cols).T @ q, jnp.zeros(w_g.shape[0])])
    z_v = jnp.concatenate([(dB_v * a_cols).T @ q, jnp.zeros(w_g.shape[0])])
    g_s = onehot * ((dG_s * atten[:, jnp.newaxis]).T @ q)[:, jnp.newaxis]
    g_v = onehot * ((dG_v * atten[:, jnp.newaxis]).T @ q)[:, jnp.newaxis]
    z_gas = jnp.concatenate([jnp.zeros((n_ssp, 2 * n_groups)),
                             jnp.stack([g_s, g_v], axis=2).reshape(
                                 w_g.shape[0], 2 * n_groups)], axis=0)
    dATrho = jnp.concatenate([jnp.stack([z_av, z_yav, z_sig, z_v], axis=1),
                              z_gas], axis=1)

    A_aug, _ = _augmented(M_full, flux, isig, D_full)
    if ATA is None:
        ATA = A_aug.T @ A_aug
    n = ATA.shape[0]
    scl = jnp.maximum(jnp.trace(ATA) / n, 1e-30)
    H = jnp.where(P[:, None] & P[None, :], ATA, scl * jnp.eye(n))

    def passive_solve(rhs):
        return jnp.linalg.solve(H, jnp.where(P[:, None], rhs, 0.0))

    projected = dAw - A_aug @ passive_solve(A_aug.T @ dAw)
    correction = A_aug @ passive_solve(dATrho)
    return -(projected + correction)


def _central_jacobian(params, P, diff_step, evaluate):
    """The reference's central differences, every evaluation warm-started."""
    eps = diff_step * jnp.maximum(jnp.abs(params), diff_step)
    E = jnp.diag(eps)
    r_plus = jax.vmap(lambda dp: evaluate(params + dp, P)[0])(E)
    r_minus = jax.vmap(lambda dp: evaluate(params - dp, P)[0])(E)
    return ((r_plus - r_minus) / (2.0 * eps[:, jnp.newaxis])).T


def _lm_parts(bounds_lo, bounds_hi, flux, isig, M_star, lam, velscale, dk,
              D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
              max_iter, ftol, xtol, diff_step, simplex, analytic):
    """
    The LM of _lm in its three parts -- start, continue-test, one iteration --
    so that the work queue (fit_queue_fast) can run the same iterations a
    few at a time. _lm runs them to the end in one loop.
    """
    n_params = bounds_lo.shape[0]

    def evaluate_full(p, P):
        return _evaluate_full(p, P, flux, isig, M_star, lam, velscale, dk,
                              D_full, yav_flags, rest_wl, ratios, n_comp, gids,
                              max_comp, simplex)

    def evaluate(p, P):
        return evaluate_full(p, P)[:3]

    def jacobian(p, w, rho, P, cache):
        if simplex or not analytic:
            return _central_jacobian(p, P, diff_step, evaluate)
        return _varpro_jacobian(p, w, rho, P, flux, isig, M_star, lam,
                                velscale, dk, D_full, yav_flags, rest_wl,
                                ratios, n_comp, gids, max_comp, cache=cache)

    def start(x0, P0):
        r0, w0, P_start, cache = evaluate_full(x0, P0)
        chi2_0 = jnp.sum(r0 ** 2)
        return (x0, jnp.array(1e-3), chi2_0, r0, 0, 1, False,
                jnp.ones(n_params), 0, 0, chi2_0, w0, P_start, cache)

    def cond(state):
        niter, conv, n_rej = state[4], state[6], state[8]
        return (~conv) & (niter < max_iter) & (n_rej < 10)

    def body(state):
        (x, lam_lm, chi2, r, niter, nfev, _, scale, n_rej, n_stag, chi2_ref,
         w, P, cache) = state

        J = jacobian(x, w, r, P, cache)

        col_norms = jnp.sqrt(jnp.sum(J ** 2, axis=0))
        scale_new = jnp.maximum(scale, col_norms)
        scale_safe = jnp.maximum(scale_new, 1e-10)

        JTJ = J.T @ J
        JTr = J.T @ r
        H = JTJ + lam_lm * jnp.diag(scale_safe ** 2)
        dp = jnp.linalg.solve(H, -JTr)

        x_cand = jnp.clip(x + dp, bounds_lo, bounds_hi)

        r_cand, w_cand, P_cand, cache_cand = evaluate_full(x_cand, P)
        chi2_cand = jnp.sum(r_cand ** 2)

        dp_actual = x_cand - x
        dr_pred = J @ dp_actual
        pred_chi2 = jnp.sum((r + dr_pred) ** 2)
        pred_red = chi2 - pred_chi2
        actual_red = chi2 - chi2_cand
        gain = jnp.where(jnp.abs(pred_red) > 1e-30,
                         actual_red / pred_red, 0.0)

        accept = chi2_cand < chi2
        x_out = jnp.where(accept, x_cand, x)
        chi2_out = jnp.where(accept, chi2_cand, chi2)
        r_out = jnp.where(accept, r_cand, r)
        w_out = jnp.where(accept, w_cand, w)
        P_out = jnp.where(accept, P_cand, P)
        cache_out = jax.tree_util.tree_map(
            lambda c, o: jnp.where(accept, c, o), cache_cand, cache)
        n_rej_out = jnp.where(accept, 0, n_rej + 1)

        lam_new = jnp.where(
            accept & (gain > 0.75), jnp.maximum(lam_lm / 3.0, 1e-12),
            jnp.where(accept & (gain > 0.25), lam_lm,
                      jnp.minimum(lam_lm * 3.0, 1e10)),
        )

        rel_chi2 = jnp.where(accept,
                             jnp.abs(actual_red) / jnp.maximum(chi2, 1e-30),
                             1.0)
        step_norm = jnp.max(
            jnp.abs(dp_actual) / jnp.maximum(bounds_hi - bounds_lo, 1e-30))
        classical_conv = (accept & (rel_chi2 < ftol) & (step_norm < xtol)
                          & (niter > 3))

        window_rel_change = (jnp.abs(chi2_ref - chi2_out)
                             / jnp.maximum(chi2_ref, 1e-30))
        big_improvement = window_rel_change > ftol * 100
        n_stag_out = jnp.where(accept,
                               jnp.where(big_improvement, 0, n_stag + 1),
                               n_stag)
        chi2_ref_out = jnp.where(big_improvement, chi2_out, chi2_ref)
        stagnation_conv = (n_stag_out >= 5) & (niter > 3)

        conv = classical_conv | stagnation_conv | (n_rej_out >= 10)
        new_nfev = nfev + 2 * n_params + 1

        return (x_out, lam_new, chi2_out, r_out, niter + 1, new_nfev, conv,
                scale_new, n_rej_out, n_stag_out, chi2_ref_out, w_out, P_out,
                cache_out)

    return start, cond, body


def _lm(x0, P0, bounds_lo, bounds_hi, flux, isig, M_star, lam, velscale, dk,
        D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
        max_iter, ftol, xtol, diff_step, simplex, analytic):
    """
    optimizer._lm_optimize, with the active set carried in the state and the
    Jacobian taken analytically. Damping, acceptance, the gain ratio, the
    scaling and all three convergence tests are the reference's, unchanged.
    """
    start, cond, body = _lm_parts(
        bounds_lo, bounds_hi, flux, isig, M_star, lam, velscale, dk, D_full,
        yav_flags, rest_wl, ratios, n_comp, gids, max_comp, max_iter, ftol,
        xtol, diff_step, simplex, analytic)
    final = jax.lax.while_loop(cond, body, start(x0, P0))
    return final[0], final[2], final[4], final[5], final[6], final[12]


def _grid_chi2(v0, P, flux, isig, M_star, lam, velscale, sigma_star, dk, av,
               sigma_gas_arr, rest_wl, ratios, n_comp, gids, max_comp, D_full,
               simplex):
    """optimizer._grid_point_chi2, warm-started, returning the active set."""
    atten = jnp.where(av > 0, 10.0 ** (-0.4 * av * dk), 1.0)
    M_broad = _broaden(M_star, sigma_star, v0, velscale)
    M_star_dust = M_broad * atten[:, jnp.newaxis]
    n_groups = sigma_gas_arr.shape[0]
    M_gas = _build_emission_jit(
        lam, velscale, sigma_gas_arr, jnp.full(n_groups, v0),
        rest_wl, ratios, n_comp, gids, max_comp,
    )
    M_gas_dust = M_gas * atten[:, jnp.newaxis]
    M_full = jnp.hstack([M_star_dust, M_gas_dust])
    A_aug, b_aug = _augmented(M_full, flux, isig, D_full)
    w, P_new = _solve(A_aug, b_aug, M_star.shape[1], simplex, P)
    residuals = (flux - M_full @ w) * isig
    return jnp.sum(residuals ** 2), P_new

def outlier_nsigma(resid, istat, isig, scaled):
    """
    Each pixel's residual in the units outliers are judged in.

    scaled False (outlier_scale 'error', the default): its own error, as
    STARLIGHT's NSIGMA clipping -- a pixel beyond the threshold is clipped
    however badly the rest of the spectrum fits. scaled True ('typical'):
    the larger of its error and the spectrum's typical residual (1.4826 MAD
    of residual/error), so that where the model cannot match a very high-S/N
    spectrum anywhere the clip does not take most of the pixels.
    """
    nsigma = jnp.abs(resid) * istat
    if scaled:
        typical = 1.4826 * jnp.nanmedian(jnp.where(isig > 0, nsigma, jnp.nan))
        nsigma = nsigma / jnp.maximum(1.0, typical)
    return nsigma

def fit_batch_fast(flux_b, isig_b, istat_b, M_star, lam, velscale, dk,
                   D_full, yav_flags, rest_wl, ratios, n_comp, gids,
                   max_comp, n_groups, blo, bhi, v0_grid, av0, sig_s0,
                   sig_g0, solver="active_set", simplex=False,
                   max_iter=200, ftol=1e-7, xtol=1e-7, n_clip=0,
                   clip_threshold=3.0, have_errors=True,
                   analytic_jacobian=True, pad_pixels=None,
                   clip_scaled=False):
    """
    fit_megacube_jax._fit_batch, faster. Same arguments, same return tuple:
    (params, weights, fitted, stellar, gas, chi2, n_eff, converged, n_iter,
    inverse_sigma) per spaxel.

    analytic_jacobian=False keeps the reference's central differences, still
    warm-started and with the scanned grid. It exists to separate the three
    changes when measuring them: with it, any difference from the reference
    is down to the warm start and the scan alone.

    Only the active-set solver has an active set to carry. Any other solver
    is refused rather than quietly run through a path it was not measured on.

    pad_pixels is the padding of the broadening transform at each end, a
    fixed number for the batch: the pipeline sets it from the fit's own
    bounds (8 sigma_max + the largest velocity, plus a margin), which the kernel never
    reaches past. None pads by the full spectrum length, as
    spectral_engine does when it is not told the bounds: a transform three
    times as long, for a model the same to 1e-13 (to 2e-6 at the outermost
    pixels for a kernel under a pixel and a half, which rings).
    """
    if solver != "active_set":
        raise ValueError(
            f"fit_batch_fast carries the active set between solves, and "
            f"solver={solver!r} has none. Use fit_megacube_jax._fit_batch.")
    n_ssp = M_star.shape[1]
    M_star = _templates(M_star, pad_pixels)
    diff_step = 1e-3
    statics = dict(max_iter=max_iter, ftol=ftol, xtol=xtol,
                   diff_step=diff_step, simplex=simplex,
                   analytic=analytic_jacobian)

    def fit_one(flux, isig, istat):
        sga = jnp.full(n_groups, sig_g0)

        def grid_step(P, v0):
            c2, P_next = _grid_chi2(v0, P, flux, isig, M_star, lam, velscale,
                                    sig_s0, dk, av0, sga, rest_wl, ratios,
                                    n_comp, gids, max_comp, D_full, simplex)
            return P_next, c2

        n_cols = jax.eval_shape(
            lambda p: _model_matrix(p, M_star, lam, velscale, dk, yav_flags,
                                    rest_wl, ratios, n_comp, gids,
                                    max_comp)[0],
            jnp.zeros(4 + 2 * n_groups)).shape[1]
        P_cold = jnp.zeros(n_cols, dtype=bool)
        P_grid, all_c2 = jax.lax.scan(grid_step, P_cold, v0_grid)
        bv0 = v0_grid[jnp.argmin(all_c2)]

        x0 = jnp.concatenate([jnp.array([av0, 0.0, sig_s0, bv0]),
                              jnp.tile(jnp.array([sig_g0, bv0]), n_groups)])
        bp, bc2, ni, nf, conv, P = _lm(
            x0, P_cold, blo, bhi, flux, isig, M_star, lam, velscale, dk,
            D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
            **statics)

        def final(p, P, isig):
            M_full, Msd, Mgd = _model_matrix(p, M_star, lam, velscale, dk,
                                             yav_flags, rest_wl, ratios,
                                             n_comp, gids, max_comp)
            A_aug, b_aug = _augmented(M_full, flux, isig, D_full)
            w, P_new = _solve(A_aug, b_aug, n_ssp, simplex, P)
            return w, M_full @ w, Msd, Mgd, P_new

        w, fitted, Msd, Mgd, P = final(bp, P, isig)

        for _ in range(n_clip):
            resid = flux - fitted
            if have_errors:
                nsigma = outlier_nsigma(resid, istat, isig, clip_scaled)
            else:
                rms = jnp.std(jnp.where(isig > 0, resid, 0.0))
                nsigma = jnp.abs(resid) / jnp.maximum(rms, 1e-30)
            outliers = (nsigma > clip_threshold) & (isig > 0)
            clipped = jnp.any(outliers)
            isig = jnp.where(outliers, 0.0, isig)
            bp2, bc22, ni2, nf2, conv2, P2 = _lm(
                bp, P, blo, bhi, flux, isig, M_star, lam, velscale, dk,
                D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
                **statics)
            bp = jnp.where(clipped, bp2, bp)
            bc2 = jnp.where(clipped, bc22, bc2)
            ni = jnp.where(clipped, ni2, ni)
            conv = jnp.where(clipped, conv2, conv)
            P = jnp.where(clipped, P2, P)
            w, fitted, Msd, Mgd, P = final(bp, P, isig)

        stellar = Msd @ w[:n_ssp]
        gas = Mgd @ w[n_ssp:]
        neff = jnp.sum(isig > 0)
        return bp, w, fitted, stellar, gas, bc2, neff, conv, ni, isig

    return jax.vmap(fit_one)(flux_b, isig_b, istat_b)




# ---------------------------------------------------------------------------
# The work queue: the same fit, with no lane left waiting
# ---------------------------------------------------------------------------
#
# fit_batch_fast fits a batch of spaxels in lockstep: the batch runs until its
# slowest spaxel has converged, and every other lane computes nothing useful
# in the meantime. On MaNGA 8626-12704 (espop, batches of 16) the batches ran
# for 2125 LM iterations against the 821 their spaxels needed.
#
# fit_queue_fast keeps a fixed number of lanes busy instead. Each lane holds
# one spaxel and moves it through the steps fit_batch_fast takes -- grid
# search, LM from a cold active set, final solve, outlier clipping and the
# refit, final solve -- a few LM iterations at a time. A spaxel that finishes
# leaves its lane and the next one takes it. Every step is the same function
# on the same inputs; only when it runs changes. An LM run interrupted after
# k iterations and resumed from its state takes exactly the iterations it
# would have taken in one loop.

_Q_STATIC = ('n_left', 'n_fft', 'max_comp', 'n_groups', 'simplex', 'max_iter',
             'ftol', 'xtol', 'analytic', 'have_errors', 'clip_threshold',
             'clip_scaled')


def _q_parts(flux, isig, M_log, M_fft, n_left, n_fft, lam, velscale, dk,
             D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp, blo,
             bhi, simplex, max_iter, ftol, xtol, analytic):
    T = Templates(M_log, M_fft, n_left, n_fft)
    return T, _lm_parts(blo, bhi, flux, isig, T, lam, velscale, dk, D_full,
                        yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
                        max_iter, ftol, xtol, 1e-3, simplex, analytic)


def _q_final(T, p, P, flux, isig, lam, velscale, dk, D_full, yav_flags,
             rest_wl, ratios, n_comp, gids, max_comp, simplex):
    """fit_batch_fast's final(): the solve at the converged parameters."""
    M_full, Msd, Mgd = _model_matrix(p, T, lam, velscale, dk, yav_flags,
                                     rest_wl, ratios, n_comp, gids, max_comp)
    A_aug, b_aug = _augmented(M_full, flux, isig, D_full)
    w, P_new = _solve(A_aug, b_aug, T.shape[1], simplex, P)
    return w, M_full @ w, Msd, Mgd, P_new


@partial(jax.jit, static_argnames=_Q_STATIC)
def _q_start(flux_b, isig_b, M_log, M_fft, lam, velscale, dk, D_full,
             yav_flags, rest_wl, ratios, n_comp, gids, blo, bhi, v0_grid, av0,
             sig_s0, sig_g0, n_left, n_fft, max_comp, n_groups, simplex,
             max_iter, ftol, xtol, analytic, have_errors, clip_threshold,
             clip_scaled):
    """Grid search and the LM's starting state, per lane (fit_batch_fast's)."""
    def one(flux, isig):
        T, (start, _, _) = _q_parts(
            flux, isig, M_log, M_fft, n_left, n_fft, lam, velscale, dk,
            D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp, blo,
            bhi, simplex, max_iter, ftol, xtol, analytic)
        sga = jnp.full(n_groups, sig_g0)

        def grid_step(P, v0):
            c2, P_next = _grid_chi2(v0, P, flux, isig, T, lam, velscale,
                                    sig_s0, dk, av0, sga, rest_wl, ratios,
                                    n_comp, gids, max_comp, D_full, simplex)
            return P_next, c2

        n_cols = T.shape[1] + rest_wl.shape[0]
        P_cold = jnp.zeros(n_cols, dtype=bool)
        _, all_c2 = jax.lax.scan(grid_step, P_cold, v0_grid)
        bv0 = v0_grid[jnp.argmin(all_c2)]
        x0 = jnp.concatenate([jnp.array([av0, 0.0, sig_s0, bv0]),
                              jnp.tile(jnp.array([sig_g0, bv0]), n_groups)])
        return start(x0, P_cold)
    return jax.vmap(one)(flux_b, isig_b)


@partial(jax.jit, static_argnames=_Q_STATIC + ('chunk',))
def _q_advance(state, flux_b, isig_b, M_log, M_fft, lam, velscale, dk, D_full,
               yav_flags, rest_wl, ratios, n_comp, gids, blo, bhi, n_left,
               n_fft, max_comp, n_groups, simplex, max_iter, ftol, xtol,
               analytic, have_errors, clip_threshold, clip_scaled, chunk):
    """Up to `chunk` LM iterations on every lane; a finished lane stays put."""
    def one(s, flux, isig):
        _, (_, cond, body) = _q_parts(
            flux, isig, M_log, M_fft, n_left, n_fft, lam, velscale, dk,
            D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp, blo,
            bhi, simplex, max_iter, ftol, xtol, analytic)
        out, _ = jax.lax.while_loop(lambda sk: cond(sk[0]) & (sk[1] < chunk),
                                    lambda sk: (body(sk[0]), sk[1] + 1),
                                    (s, 0))
        return out, ~cond(out)
    return jax.vmap(one)(state, flux_b, isig_b)


@partial(jax.jit, static_argnames=_Q_STATIC)
def _q_clip(state, flux_b, isig_b, istat_b, M_log, M_fft, lam, velscale, dk,
            D_full, yav_flags, rest_wl, ratios, n_comp, gids, blo, bhi,
            n_left, n_fft, max_comp, n_groups, simplex, max_iter, ftol, xtol,
            analytic, have_errors, clip_threshold,
            clip_scaled):
    """One pass of fit_batch_fast's clipping: outliers, and the refit's start."""
    def one(s, flux, isig, istat):
        T = Templates(M_log, M_fft, n_left, n_fft)
        bp, P = s[0], s[12]
        _, fitted, _, _, P = _q_final(T, bp, P, flux, isig, lam, velscale,
                                      dk, D_full, yav_flags, rest_wl, ratios,
                                      n_comp, gids, max_comp, simplex)
        resid = flux - fitted
        if have_errors:
            nsigma = outlier_nsigma(resid, istat, isig, clip_scaled)
        else:
            rms = jnp.std(jnp.where(isig > 0, resid, 0.0))
            nsigma = jnp.abs(resid) / jnp.maximum(rms, 1e-30)
        outliers = (nsigma > clip_threshold) & (isig > 0)
        isig_new = jnp.where(outliers, 0.0, isig)
        _, (start, _, _) = _q_parts(
            flux, isig_new, M_log, M_fft, n_left, n_fft, lam, velscale, dk,
            D_full, yav_flags, rest_wl, ratios, n_comp, gids, max_comp, blo,
            bhi, simplex, max_iter, ftol, xtol, analytic)
        return jnp.any(outliers), isig_new, start(bp, P)
    return jax.vmap(one)(state, flux_b, isig_b, istat_b)


@partial(jax.jit, static_argnames=_Q_STATIC)
def _q_outputs(state, flux_b, isig_b, M_log, M_fft, lam, velscale, dk, D_full,
               yav_flags, rest_wl, ratios, n_comp, gids, blo, bhi, n_left,
               n_fft, max_comp, n_groups, simplex, max_iter, ftol, xtol,
               analytic, have_errors, clip_threshold, clip_scaled):
    """fit_batch_fast's return tuple, per lane."""
    def one(s, flux, isig):
        T = Templates(M_log, M_fft, n_left, n_fft)
        bp, P = s[0], s[12]
        w, fitted, Msd, Mgd, _ = _q_final(T, bp, P, flux, isig, lam, velscale,
                                          dk, D_full, yav_flags, rest_wl,
                                          ratios, n_comp, gids, max_comp,
                                          simplex)
        n_ssp = T.shape[1]
        return (bp, w, fitted, Msd @ w[:n_ssp], Mgd @ w[n_ssp:], s[2],
                jnp.sum(isig > 0), s[6], s[4], isig)
    return jax.vmap(one)(state, flux_b, isig_b)


@jax.jit
def _q_place(old, new, src, take):
    """Lanes with take=True receive row src of `new`; the rest keep `old`."""
    def put(o, n):
        picked = n[src]
        mask = take.reshape((-1,) + (1,) * (o.ndim - 1))
        return jnp.where(mask, picked, o)
    return jax.tree_util.tree_map(put, old, new)

def fit_queue_fast(flux_all, isig_all, istat_all, M_star, lam, velscale, dk,
                   D_full, yav_flags, rest_wl, ratios, n_comp, gids,
                   max_comp, n_groups, blo, bhi, v0_grid, av0, sig_s0,
                   sig_g0, solver="active_set", simplex=False,
                   max_iter=200, ftol=1e-7, xtol=1e-7, n_clip=0,
                   clip_threshold=3.0, have_errors=True,
                   analytic_jacobian=True, pad_pixels=None, lanes=16,
                   chunk=4, progress=None, clip_scaled=False):
    """
    fit_batch_fast for any number of spaxels, scheduled as a work queue.

    Takes every spaxel at once (flux_all etc. are (n_spaxels, n_pix)) and
    returns fit_batch_fast's tuple with n_spaxels rows. `lanes` spaxels are
    in flight at a time; each lane is advanced `chunk` LM iterations per
    step, and a lane whose spaxel is done is refilled from the queue, a
    quarter of the lanes at a time so the grid search runs in groups.
    progress(n_done, n_total, finished) is called as spaxels finish, with
    the indices (rows of flux_all) of the ones that just did.
    """
    import numpy as np
    if solver != "active_set":
        raise ValueError(
            f"fit_queue_fast carries the active set between solves, and "
            f"solver={solver!r} has none. Use fit_megacube_jax._fit_batch.")
    flux_all = np.asarray(flux_all)
    isig_all = np.asarray(isig_all)
    istat_all = np.asarray(istat_all)
    n, npx = flux_all.shape
    T = _templates(jnp.asarray(M_star), pad_pixels)
    statics = dict(n_left=T.n_left, n_fft=T.n_fft, max_comp=max_comp,
                   n_groups=n_groups, simplex=simplex, max_iter=max_iter,
                   ftol=ftol, xtol=xtol, analytic=analytic_jacobian,
                   have_errors=have_errors, clip_threshold=clip_threshold,
                   clip_scaled=clip_scaled)
    consts = dict(M_log=T.log, M_fft=T.fft, lam=lam, velscale=velscale,
                  dk=dk, D_full=D_full, yav_flags=yav_flags, rest_wl=rest_wl,
                  ratios=ratios, n_comp=n_comp, gids=gids, blo=blo, bhi=bhi)
    lanes = int(max(1, min(lanes, n)))
    group = max(1, lanes // 4)

    def rows(idx, size):
        """The spaxels idx as device arrays padded to `size` rows."""
        out = []
        for arr in (flux_all, isig_all, istat_all):
            block = np.zeros((size, npx))
            block[:len(idx)] = arr[idx]
            out.append(jnp.asarray(block))
        return out

    def start(idx, size):
        f, s, _ = rows(idx, size)
        return _q_start(f, s, v0_grid=v0_grid, av0=av0, sig_s0=sig_s0,
                        sig_g0=sig_g0, **consts, **statics)

    # Fill every lane, mark the empty ones done.
    lane_spaxel = np.full(lanes, -1)
    first = np.arange(min(lanes, n))
    lane_spaxel[:len(first)] = first
    next_i = len(first)
    flux_l, isig_l, istat_l = rows(first, lanes)
    state = start(first, lanes)
    phase = np.zeros(lanes, dtype=int)
    active = lane_spaxel >= 0
    state = state[:6] + (state[6] | ~jnp.asarray(active),) + state[7:]
    results = None
    n_done = 0

    while n_done < n:
        state, done = _q_advance(state, flux_l, isig_l, chunk=chunk,
                                 **consts, **statics)
        done = np.asarray(done) & (lane_spaxel >= 0)
        if not done.any():
            continue
        finish = done.copy()
        if n_clip and (done & (phase < n_clip)).any():
            clipped, isig_new, restart = _q_clip(state, flux_l, isig_l,
                                                 istat_l, **consts, **statics)
            refit = done & (phase < n_clip) & np.asarray(clipped)
            if refit.any():
                take = jnp.asarray(refit)
                src = jnp.arange(lanes)
                state = _q_place(state, restart, src, take)
                isig_l = jnp.where(take[:, None], isig_new, isig_l)
                phase[refit] += 1
                finish &= ~refit
        if finish.any():
            outs = [np.asarray(o) for o in
                    _q_outputs(state, flux_l, isig_l, **consts, **statics)]
            if results is None:
                results = [np.zeros((n,) + o.shape[1:], dtype=o.dtype)
                           for o in outs]
            finished = lane_spaxel[finish].copy()
            for lane in np.flatnonzero(finish):
                k = lane_spaxel[lane]
                for r, o in zip(results, outs):
                    r[k] = o[lane]
                lane_spaxel[lane] = -1
                phase[lane] = 0
            n_done += int(finish.sum())
            if progress is not None:
                progress(n_done, n, finished)
        # Refill free lanes, a group at a time (or whatever is left).
        free = np.flatnonzero(lane_spaxel < 0)
        while next_i < n and (len(free) >= group or
                              len(free) >= n - next_i or
                              (lane_spaxel < 0).all()):
            k = min(group, len(free), n - next_i)
            idx = np.arange(next_i, next_i + k)
            slots = free[:k]
            new_state = start(idx, group)
            src = np.zeros(lanes, dtype=int)
            src[slots] = np.arange(k)
            take = np.zeros(lanes, dtype=bool)
            take[slots] = True
            take_d, src_d = jnp.asarray(take), jnp.asarray(src)
            state = _q_place(state, new_state, src_d, take_d)
            f, s, e = rows(idx, group)
            flux_l = jnp.where(take_d[:, None], f[src_d], flux_l)
            isig_l = jnp.where(take_d[:, None], s[src_d], isig_l)
            istat_l = jnp.where(take_d[:, None], e[src_d], istat_l)
            lane_spaxel[slots] = idx
            phase[slots] = 0
            next_i += k
            free = np.flatnonzero(lane_spaxel < 0)
        # Lanes with nothing in them are marked done so they do no work.
        idle = jnp.asarray(lane_spaxel < 0)
        state = state[:6] + (state[6] | idle,) + state[7:]
    return tuple(results)
