"""
NewBrain Spectral Engine — JAX Backend
Core computational routines: log-rebinning, Fourier broadening, dust attenuation.
All operations designed for batch/matrix form → GPU-accelerated via JAX.
"""

import jax
import jax.numpy as jnp
from functools import partial
import numpy as np
from jax.numpy.fft import rfft, irfft, rfftfreq

from brainsp import reddening_laws

C_KMS = 299792.458


def _smooth_length(n, factors=(2, 3, 5, 7)):
    """Smallest m >= n whose only prime factors are in `factors`."""
    while True:
        m = n
        for p in factors:
            while m % p == 0:
                m //= p
        if m == 1:
            return n
        n += 1


def _broadening_pad(n_pix, pad_pixels):
    """
    How many pixels of padding a broadening transform needs at each end.

    An FFT convolution is periodic, so without padding the broadened model at
    one end of the window is built partly out of the template at the other.
    Measured on the reference grid at sigma = 150 km/s and v0 = 0, with no
    padding, 18 pixels were wrong by more than 0.1%, the last by 315% and the
    first by 77%.

    The pad has to be a static number of pixels, because these functions are
    traced with sigma and v0 as values and a shape cannot depend on a value.
    `pad_pixels` is how a caller that knows its own bounds says so. With none
    given the pad is the full length of the input, which makes the convolution
    linear for any shift and any width the array can represent -- the only
    choice that is safe without being told what the bounds are, at the cost of
    a transform twice as long.

    Returns (n_left, n_right, n_fft).
    """
    n_left = int(n_pix if pad_pixels is None else max(int(pad_pixels), 0))
    n_fft = _smooth_length(n_pix + 2 * n_left)
    return n_left, n_fft - n_pix - n_left, n_fft


def _edge_pad(values, n_left, n_right):
    """
    Extend an array at both ends by holding its first and last value.

    Zero-filling instead darkens the ends of the model: a template padded with
    zeros is dragged toward zero by the convolution, and the result looks like
    a continuum slope and is fitted as one. Holding the end values says the
    template is flat beyond the window, which leaves a flat template exactly
    flat end to end and is the more honest answer about a region nothing was
    measured in.
    """
    if n_left <= 0 and n_right <= 0:
        return values
    left = jnp.repeat(values[:1], n_left, axis=0)
    right = jnp.repeat(values[-1:], n_right, axis=0)
    return jnp.concatenate([left, values, right], axis=0)


def _bin_edges(centers):
    """Compute bin edges from bin centers using midpoints."""
    n = len(centers)
    edges = jnp.empty(n + 1)
    edges = edges.at[1:-1].set(0.5 * (centers[:-1] + centers[1:]))
    edges = edges.at[0].set(centers[0] - 0.5 * (centers[1] - centers[0]))
    edges = edges.at[-1].set(centers[-1] + 0.5 * (centers[-1] - centers[-2]))
    return edges

def flux_conserving_resample(wl_old, flux_old, wl_new):
    """
    Resample flux density from one wavelength grid to another,
    conserving total flux via cumulative integral interpolation.

    Pixels outside the input range receive zero flux.

    Parameters
    ----------
    wl_old : 1D array
        Input wavelength grid (Angstroms).
    flux_old : 1D array
        Input flux density array.
    wl_new : 1D array
        Target wavelength grid (Angstroms).

    Returns
    -------
    flux_new : 1D array
        Flux density on the target grid.
    """
    edges_old = _bin_edges(wl_old)
    h_old = jnp.diff(edges_old)
    cum = jnp.zeros(len(wl_old) + 1)
    cum = cum.at[1:].set(jnp.cumsum(flux_old * h_old))

    edges_new = _bin_edges(wl_new)
    cum_new = jnp.interp(edges_new, edges_old, cum,
                         left=0.0, right=cum[-1])
    h_new = jnp.diff(edges_new)
    return jnp.diff(cum_new) / h_new

def clean_fraction(wl_lin, usable_lin, lam_new):
    """
    Fraction of each log-rebinned pixel that came from usable input pixels.

    Log-rebinning is a flux-conserving redistribution: every output pixel is
    an integral of the input over its own bin, so where the output bin is
    wider than the input bin it blends two or more input pixels into one
    value. Toward the red end of a linear-lambda spectrum the output bins are
    the wider ones, and an output pixel adjacent to a rejected input pixel is
    then assembled partly from data that was rejected for a reason.

    The quality mask does not follow, because it is carried across by a
    nearest-neighbour lookup, which asks only which input pixel is closest to
    the output pixel centre and never notices that the output pixel's *value*
    reaches further than that. The result is a pixel marked good, fitted at
    full weight, holding flux from a cosmic ray or a dead column. On the
    example MaNGA spectra this affects 21-30 of ~3130 fitted pixels, with as
    little as 0.50 of the flux coming from good data.

    Passing the indicator function of the usable input pixels through the
    identical resampling operator answers the question directly: the result
    is the length-weighted fraction of each output bin that came from usable
    input. Anything below 1 is contaminated. Output bins that hang over the
    ends of the input grid score below 1 too, which is also correct -- the
    missing part was filled with zero.
    """
    return flux_conserving_resample(
        jnp.asarray(wl_lin),
        jnp.asarray(np.asarray(usable_lin, dtype=np.float64)),
        jnp.asarray(lam_new))

def log_rebin(wavelength, flux, velscale=None):
    """
    Rebin spectrum from uniform linear-λ grid to uniform ln(λ) grid.

    Uses flux-conserving cumulative integral interpolation to preserve
    spectral features and total flux exactly.

    Parameters
    ----------
    wavelength : 1D array
        Linear wavelength array (Angstroms). Must be uniformly spaced.
    flux : 1D array
        Flux array on the linear grid.
    velscale : float or None
        Desired velocity scale per pixel (km/s). If None, computed to
        approximately preserve the number of pixels.

    Returns
    -------
    flux_log : 1D array
        Flux on the log-lambda grid.
    log_lam : 1D array
        Natural-log wavelength array.
    velscale : float
        Velocity scale per pixel (km/s).
    """
    wl_np = np.asarray(wavelength)
    lam_range = (wl_np[0], wl_np[-1])

    if velscale is None:
        velscale = C_KMS * np.mean(np.diff(wl_np) / wl_np[:-1])

    dln = velscale / C_KMS
    n_pix = int(np.log(lam_range[1] / lam_range[0]) / dln)

    log_lam = np.log(lam_range[0]) + np.arange(n_pix) * dln
    lam_new = np.exp(log_lam)

    flux_log = flux_conserving_resample(
        jnp.asarray(wavelength), jnp.asarray(flux), jnp.asarray(lam_new)
    )

    return flux_log, jnp.asarray(log_lam), velscale

@partial(jax.jit, static_argnames=('pad_pixels',))
def fourier_broaden_single(spec_log, sigma_v, v0, velscale, pad_pixels=None):
    """
    Broaden and shift a single log-rebinned spectrum via analytic FFT kernel.

    The spectrum is edge-padded before the transform so the convolution is
    linear rather than periodic; see _broadening_pad and _edge_pad for why
    both of those matter and what `pad_pixels` buys.

    Parameters
    ----------
    spec_log : 1D array
        Spectrum on uniform log-lambda grid.
    sigma_v : float
        Velocity dispersion (km/s).
    v0 : float
        Radial velocity (km/s). Positive = redshift.
    velscale : float
        Velocity per pixel (km/s).
    pad_pixels : int or None
        Padding at each end, in pixels. Static. None pads by the full length
        of the input, which is safe for any sigma and any shift.

    Returns
    -------
    broadened : 1D array
        Convolved + shifted spectrum.
    """
    n = spec_log.shape[0]
    n_left, n_right, n_fft = _broadening_pad(n, pad_pixels)
    sigma_pix = sigma_v / velscale
    shift_pix = v0 / velscale

    padded = _edge_pad(spec_log[:, jnp.newaxis], n_left, n_right)[:, 0]
    spec_fft = rfft(padded)
    omega = 2.0 * jnp.pi * rfftfreq(n_fft)

    kernel = jnp.exp(-0.5 * (omega * sigma_pix)**2 - 1j * omega * shift_pix)
    result = irfft(spec_fft * kernel, n_fft)[n_left:n_left + n]

    return jnp.where(
        (sigma_pix < 0.05) & (jnp.abs(shift_pix) < 0.05),
        spec_log,
        result
    )

@partial(jax.jit, static_argnames=('pad_pixels',))
def fourier_broaden_matrix(matrix_log, sigma_v, v0, velscale, pad_pixels=None):
    """
    Broaden and shift ALL columns of a template matrix simultaneously.
    Each column is an SSP template on the log-lambda grid.

    Every column is edge-padded before the transform so the convolution is
    linear rather than periodic. Without it the model at the blue end of the
    window was built partly out of the template at the red end, which on the
    reference grid put 18 pixels more than 0.1% wrong and the outermost ones
    by factors of several.

    Parameters
    ----------
    matrix_log : 2D array (N_pix, N_templates)
        SSP template matrix on uniform log-lambda grid.
    sigma_v : float
        Velocity dispersion (km/s).
    v0 : float
        Radial velocity (km/s).
    velscale : float
        Velocity per pixel (km/s).
    pad_pixels : int or None
        Padding at each end, in pixels. Static. None pads by the full length
        of the input, which is safe for any sigma and any shift.

    Returns
    -------
    broadened : 2D array (N_pix, N_templates)
        All templates broadened + shifted.
    """
    n_pix = matrix_log.shape[0]
    n_left, n_right, n_fft = _broadening_pad(n_pix, pad_pixels)
    sigma_pix = sigma_v / velscale
    shift_pix = v0 / velscale

    M_fft = rfft(_edge_pad(matrix_log, n_left, n_right), axis=0)
    omega = 2.0 * jnp.pi * rfftfreq(n_fft)

    kernel = jnp.exp(-0.5 * (omega * sigma_pix)**2 - 1j * omega * shift_pix)

    M_fft_broad = M_fft * kernel[:, jnp.newaxis]

    M_broad = irfft(M_fft_broad, n=n_fft, axis=0)[n_left:n_left + n_pix]

    return jnp.where(
        (sigma_pix < 0.05) & (jnp.abs(shift_pix) < 0.05),
        matrix_log,
        M_broad
    )



def _get_reddening_curve(wavelength, law_name, norm_wavelength):
    """
    Compute the reddening differential: k(λ) - k(λ_norm).
    Removed python dict caching to support JAX Tracers inside JIT.
    """
    k_lam = reddening_laws.get_k_lambda(wavelength, law_name)
    k_norm = reddening_laws.get_k_lambda(jnp.atleast_1d(norm_wavelength), law_name)
    return k_lam - k_norm[0]

def dust_vector(wavelength, av, law_name, norm_wavelength):
    """
    Compute the dust attenuation vector: 10^(-0.4 * Av * [k(λ) - k(λ_norm)]).

    Parameters
    ----------
    wavelength : 1D array
        Wavelength grid (Angstroms).
    av : float
        V-band extinction (Av).
    law_name : str
        Reddening law identifier (e.g., 'CCM', 'CAL', 'GD24').
    norm_wavelength : float
        Normalization wavelength (Angstroms).

    Returns
    -------
    atten : 1D array
        Multiplicative attenuation factor at each wavelength.
    """
    dk = _get_reddening_curve(wavelength, law_name, norm_wavelength)
    atten = 10.0 ** (-0.4 * av * dk)
    return jnp.where(av <= 0.0, jnp.ones(len(wavelength)), atten)

def apply_dust_matrix(matrix, wavelength, av, law_name, norm_wavelength):
    """
    Apply dust attenuation to all columns of a template matrix.

    Parameters
    ----------
    matrix : 2D array (N_pix, N_templates)
        Template matrix.
    wavelength : 1D array
        Wavelength grid.
    av : float
        V-band extinction.
    law_name : str
        Reddening law name.
    norm_wavelength : float
        Normalization wavelength.

    Returns
    -------
    attenuated : 2D array
        Dust-attenuated template matrix.
    """
    atten = dust_vector(wavelength, av, law_name, norm_wavelength)
    return matrix * atten[:, jnp.newaxis]

def normalize_at_wavelength(flux, wavelength, norm_wl, window=10):
    """
    Normalize flux using a median window around norm_wl.

    Parameters
    ----------
    flux : 1D array
        Flux array.
    wavelength : 1D array
        Wavelength array.
    norm_wl : float
        Normalization wavelength.
    window : int
        Half-window in pixels for median.

    Returns
    -------
    norm_flux : 1D array
        Normalized flux.
    factor : float
        Normalization factor applied.
    """
    idx = jnp.argmin(jnp.abs(wavelength - norm_wl))
    i0 = jnp.maximum(0, idx - window)
    i1 = jnp.minimum(len(flux), idx + window + 1)
    factor = float(jnp.median(jax.lax.dynamic_slice(flux, (i0,), (i1 - i0,))))

    if factor <= 0:
        pos = flux[flux > 0]
        factor = float(jnp.median(pos)) if len(pos) > 0 else 1.0

    return flux / factor, factor
