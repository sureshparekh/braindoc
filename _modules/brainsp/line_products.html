Source code for brainsp.line_products

"""
BRAIN: per-line emission products for a datacube, in the megacube layout.

The published MaNGA megacubes (Riffel et al. 2023, MNRAS 524, 5640) carry an
emission-line fit. BRAIN fits the lines together with the stars, and used to
keep only each kinematic group's velocity and dispersion: the line fluxes
were computed and thrown away. This writes them, with the extensions, names
and definitions of those megacubes, so a BRAIN megacube reads the same way.

    SOLUTION  (3N + 1, ny, nx)  A, v, sigma per line; the last plane is the
                                spaxel's chi2 / Nl_eff
    DISP      (3N, ny, nx)      uncertainty of SOLUTION. NaN: not computed,
                                as in the published megacubes
    INIGUESS  (3N, ny, nx)      the starting sigma; A and v are NaN, because
                                BRAIN starts from the velocity grid search
                                and an amplitude is never guessed (it is
                                solved linearly)
    FLUX_M    (N, ny, nx)       line flux from the model
    FLUX_D    (N, ny, nx)       line flux measured directly on the data
    EQW_M     (N, ny, nx)       equivalent width from the model
    EQW_D     (N, ny, nx)       equivalent width measured on the data
    MASK2D    (ny, nx)          0 where a spaxel was fitted, 1 elsewhere
    PARNAMES  table             (component, parameter): parameter A, v or s
    SPECIDX   table             (row, column) of every fitted spaxel
    FITCONFIG table             (parameters, values): rest wavelengths,
                                kinematic groups, continuum windows

THE DEFINITIONS, each checked against the published 8626-12704 megacube by
rebuilding its numbers from its own SOLUTION and spectra:

    FLUX_M = integral of the line model over lambda_c +- 5 sigma
             (reproduced to 2e-5)
    EQW_M  = integral of -(line model) / C over the same window
             (reproduced to 3e-5)
    EQW_D  = integral of 1 - data / C over the same window
             (reproduced to 1e-15)
    FLUX_D = integral of data - stellar - (every other line) over it

where the window is (lambda_c - 5 s, lambda_c + 5 s) with s = sigma
lambda_rest / c, and C is a straight
line fitted by least squares to the stellar model in the line's two
continuum windows (blue0, blue1, red0, red1). A line with no windows falls
back to a degree-7 polynomial over the whole
fitting range, three passes of 3-sigma clipping. EQW is negative for an
emission line.

UNITS AND FRAME

Fluxes are in the cube's own flux unit times Angstrom, and OBSERVED: the
line as it is in the data, attenuated by the fitted dust like everything
else in the model. That is the published megacubes' FLUX_M, and it is not BRAIN's fitted weight, which is the line before attenuation.
The wavelength frame is the frame the fit ran in: the rest frame when BRAIN
removed a redshift, with the flux density not rescaled by (1 + z).

A is the peak of the Gaussian whose integral is the line's total observed
flux, so FLUX_M = A sqrt(2 pi) s lambda_c / c up to the 5.7e-7 of a Gaussian
that lies beyond 5 sigma; v and s are the line's kinematic group's, observed
(no instrumental correction).

THE LINE MODEL is BRAIN's own: the pixel-integrated Gaussian of
emission_lines._build_emission_jit, rebuilt here per component in numpy, so
the products are integrals of exactly the profile that was fitted.
"""

import numpy as np

C_KMS = 299792.458
SIGMA_FACTOR = 5.0

CONTINUUM_WINDOWS = {
    "Hbeta_4861": (4450.0, 4600.0, 5300.0, 5380.0),
    "[O III]_4959": (4900.0, 4920.0, 5050.0, 5100.0),
    "[O III]_5007": (4900.0, 4920.0, 5050.0, 5100.0),
    "He I_5876": (5760.0, 5840.0, 5920.0, 6000.0),
    "[O I]_6300": (6200.0, 6250.0, 6480.0, 6530.0),
    "[N II]_6548": (6480.0, 6530.0, 6600.0, 6670.0),
    "Halpha_6563": (6480.0, 6530.0, 6600.0, 6670.0),
    "[N II]_6583": (6480.0, 6530.0, 6600.0, 6650.0),
    "[S II]_6716": (6650.0, 6700.0, 6745.0, 6790.0),
    "[S II]_6731": (6650.0, 6700.0, 6745.0, 6790.0),
}
"""
Continuum windows, (blue0, blue1, red0, red1) in Angstrom, from the
configuration the published 8626-12704 megacube was made with (its
FITCONFIG). Lines not listed use the no-window fallback.
"""

def components(table, doublets):
    """
    Every line BRAIN fits, one entry per emitting line, in wavelength order.

    A tied doublet is one fitted column and one weight, the stronger line's
    flux; it is two lines here, each with its own flux, because every reader of a line map want [O III] 4959 and 5007 separately.
    """
    out = []
    for j, name in enumerate(table["names"]):
        rest = np.asarray(table["rest_wavelengths"][j], dtype=np.float64)
        ratio = np.asarray(table["ratios"][j], dtype=np.float64)
        group = int(table["group_indices"][j])
        label = int(table["group_labels"][j])
        names = ([c[0] for c in doublets[name]] if name in doublets
                 else [name])
        for k in range(len(rest)):
            if rest[k] <= 0.0:
                continue
            out.append(dict(name=names[k] if k < len(names) else f"{name}_{k}",
                            entry=j, rest=float(rest[k]), ratio=float(ratio[k]),
                            group=group, label=label))
    out.sort(key=lambda c: c["rest"])
    return out

def unit_profile(lam, velscale, rest, sigma, v0):
    """
    One line's profile on the log-lambda grid, unit area in Angstrom.

    The same construction as emission_lines._build_emission_jit: the
    analytic transform of a Gaussian times that of a one-pixel top hat,
    transformed back over N_pix and divided by the pixel width in Angstrom.
    """
    lam = np.asarray(lam, dtype=np.float64)
    n = lam.size
    dx = velscale / C_KMS
    omega = 2.0 * np.pi * np.arange(n // 2 + 1) / n
    obs = rest * (1.0 + v0 / C_KMS)
    x0 = (np.log(obs) - np.log(lam[0])) / dx
    xs = sigma / velscale
    transform = (np.exp(-0.5 * (omega * xs) ** 2 - 1j * omega * x0)
                 * np.sinc(omega / (2.0 * np.pi)))
    return np.fft.irfft(transform, n=n) / (obs * dx)




def _legendre_continuum(lam, y, degree=7, n_iterate=3, clip=3.0):
    keep = np.isfinite(y)
    x = (lam - lam.mean()) / (0.5 * (lam.max() - lam.min()))
    coef = None
    for _ in range(n_iterate):
        coef = np.polynomial.legendre.legfit(x[keep], y[keep], degree)
        resid = y - np.polynomial.legendre.legval(x, coef)
        s = np.std(resid[keep])
        keep = keep & (np.abs(resid) < clip * s)
    return lambda w: np.polynomial.legendre.legval(
        (w - lam.mean()) / (0.5 * (lam.max() - lam.min())), coef)

def continuum(lam, stellar, name, fallback=None):
    """
    C(lambda): the continuum an equivalent width is measured against.

    `fallback` is the no-window continuum, which depends only on the stellar
    model, so a caller measuring many lines fits it once and passes it in.
    """
    win = CONTINUUM_WINDOWS.get(name)
    if win is not None:
        m = (((lam > win[0]) & (lam < win[1])) | ((lam > win[2]) & (lam < win[3])))
        m &= np.isfinite(stellar)
        if m.sum() >= 2:
            p = np.polyfit(lam[m], stellar[m], 1)
            return lambda w: np.polyval(p, w)
    return fallback if fallback is not None else _legendre_continuum(lam, stellar)

def measure(lam, velscale, data, stellar, params, gas_weights, n_groups,
            dk, comps):
    """
    Every product for one spaxel, in the spaxel's normalised units.

    Returns arrays over `comps`: A, v, s, flux_m, flux_d, eqw_m, eqw_d.
    Fluxes are per normalised unit times Angstrom; the caller multiplies by
    the spaxel's normalisation to reach the cube's units. EQWs are unitless
    ratios integrated over Angstrom and need no scaling.
    """
    av = float(params[0])
    atten = 10.0 ** (-0.4 * av * np.asarray(dk, dtype=np.float64))
    models = []
    for c in comps:
        g = c["group"]
        sigma, v0 = float(params[4 + 2 * g]), float(params[5 + 2 * g])
        w = float(gas_weights[c["entry"]]) * c["ratio"]
        models.append(w * atten * unit_profile(lam, velscale, c["rest"], sigma, v0))
    models = np.array(models) if models else np.zeros((0, lam.size))
    total = models.sum(axis=0) if len(models) else np.zeros_like(lam)

    n = len(comps)
    A, V, S, FM, FD, EM, ED = (np.full(n, np.nan) for _ in range(7))
    fallback = None
    if any(c["name"] not in CONTINUUM_WINDOWS for c in comps):
        fallback = _legendre_continuum(lam, stellar)
    for i, c in enumerate(comps):
        g = c["group"]
        sigma, v0 = float(params[4 + 2 * g]), float(params[5 + 2 * g])
        lam_c = c["rest"] * (1.0 + v0 / C_KMS)
        s_win = sigma * c["rest"] / C_KMS
        cond = (lam > lam_c - SIGMA_FACTOR * s_win) & (lam < lam_c + SIGMA_FACTOR * s_win)
        V[i], S[i] = v0, sigma
        if cond.sum() < 2:
            continue
        x = lam[cond]
        line = models[i]
        total_flux = float(gas_weights[c["entry"]]) * c["ratio"] * float(
            np.interp(lam_c, lam, atten))
        A[i] = total_flux / (np.sqrt(2.0 * np.pi) * sigma * lam_c / C_KMS) \
            if sigma > 0 else np.nan
        FM[i] = np.trapezoid(line[cond], x)
        others = total - line
        FD[i] = np.trapezoid((data - stellar - others)[cond], x)
        C = continuum(lam, stellar, c["name"], fallback)(x)
        if np.all(C != 0.0):
            EM[i] = np.trapezoid(-line[cond] / C, x)
            ED[i] = np.trapezoid(1.0 - data[cond] / C, x)
    return A, V, S, FM, FD, EM, ED

def hdus(lam, velscale, data, stellar, params, weights, n_ssp, norm, chi2,
         n_groups, dk, table, doublets, vi, ny, nx, wavelength_scale=None,
         sigma_start=np.nan, progress=None):
    """
    The line products of a fitted cube as a list of HDUs.

    data, stellar: (n_spaxel, n_pix) on the log grid, normalised.
    params: (n_spaxel, 4 + 2 n_groups). weights: (n_spaxel, n_ssp + n_gas).
    norm: (n_spaxel,) the factor that turns normalised flux into cube flux.
    vi: flat spaxel index of each fitted spaxel. Returns [] with no lines.
    """
    from astropy.io import fits

    comps = components(table, doublets)
    if not comps:
        return []
    n, ns = len(comps), len(vi)
    planes = {k: np.full((ns, n), np.nan) for k in ("A", "v", "s", "fm", "fd", "em", "ed")}
    for i in range(ns):
        res = measure(lam, velscale, data[i], stellar[i], params[i],
                      weights[i, n_ssp:], n_groups, dk, comps)
        for key, val in zip(("A", "v", "s", "fm", "fd", "em", "ed"), res):
            planes[key][i] = val
        if progress is not None and i % 64 == 0:
            progress(i / ns)
    for key in ("A", "fm", "fd"):
        planes[key] *= norm[:, None]

    yc, xc = vi // nx, vi % nx

    def cube(values, depth):
        out = np.full((depth, ny, nx), np.nan, dtype=np.float64)
        out[:, yc, xc] = values.T
        return out

    sol = np.empty((ns, 3 * n + 1))
    sol[:, 0:3 * n:3], sol[:, 1:3 * n:3], sol[:, 2:3 * n:3] = (
        planes["A"], planes["v"], planes["s"])
    sol[:, -1] = chi2
    ini = np.full((ns, 3 * n), np.nan)
    ini[:, 2::3] = sigma_start

    def header(obj, comment):
        h = fits.Header()
        h["OBJECT"] = obj
        h["FUNCTION"] = ("gaussian", "BRAIN's pixel-integrated Gaussian")
        h["NFUNC"] = n
        h["COMMENT"] = comment
        return h

    out = [
        fits.ImageHDU(cube(sol, 3 * n + 1), header("parameters",
            "A, v, s per line; last plane chi2/Nl_eff"), name="SOLUTION"),
        fits.ImageHDU(cube(np.full((ns, 3 * n), np.nan), 3 * n),
                      header("dispersion", "uncertainties not computed"), name="DISP"),
        fits.ImageHDU(cube(ini, 3 * n), header("initial guess",
                      "A, v: not guessed; see SOLUTION docs"), name="INIGUESS"),
        fits.ImageHDU(cube(planes["fd"], n), header("flux_direct",
            "observed, cube flux unit x Angstrom"), name="FLUX_D"),
        fits.ImageHDU(cube(planes["fm"], n), header("flux_model",
            "observed, cube flux unit x Angstrom"), name="FLUX_M"),
        fits.ImageHDU(cube(planes["em"], n), header("eqw_model",
            "Angstrom; negative for emission"), name="EQW_M"),
        fits.ImageHDU(cube(planes["ed"], n), header("eqw_direct",
            "Angstrom; negative for emission"), name="EQW_D"),
    ]
    mask = np.ones((ny, nx), dtype=np.int16)
    mask[yc, xc] = 0
    out.append(fits.ImageHDU(mask, header("spatial mask", "0 = fitted"), name="MASK2D"))

    names = [c["name"] for c in comps]
    out.append(fits.BinTableHDU.from_columns([
        fits.Column(name="component", format="32A",
                    array=np.repeat(names, 3)),
        fits.Column(name="parameter", format="1A",
                    array=np.tile(["A", "v", "s"], n))], name="PARNAMES"))
    out.append(fits.BinTableHDU.from_columns([
        fits.Column(name="row", format="J", array=yc.astype(np.int32)),
        fits.Column(name="column", format="J", array=xc.astype(np.int32))],
        name="SPECIDX"))
    keys, values = ["fit.function", "equivalent_width.sigma_factor"], [
        "gaussian", f"{SIGMA_FACTOR:g}"]
    if wavelength_scale:
        keys.append("brain.wavelength_scale"); values.append(str(wavelength_scale))
    for c in comps:
        keys += [f"{c['name']}.rest_wavelength", f"{c['name']}.k_group"]
        values += [f"{c['rest']:.4f}", str(c["label"])]
        win = CONTINUUM_WINDOWS.get(c["name"])
        keys.append(f"{c['name']}.continuum_windows")
        values.append(", ".join(f"{w:g}" for w in win) if win
                      else "none: degree-7 continuum over the fitting range")
    out.append(fits.BinTableHDU.from_columns([
        fits.Column(name="parameters", format="64A", array=np.array(keys)),
        fits.Column(name="values", format="64A", array=np.array(values))],
        name="FITCONFIG"))
    return out
