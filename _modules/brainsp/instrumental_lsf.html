Source code for brainsp.instrumental_lsf

"""
BRAIN — instrumental resolution and the line-spread function.

The data and the template library almost never share a line-spread function,
and the difference between them lands directly on the fitted stellar velocity
dispersion. This module measures that difference, brings the library up to the
data's line-spread function once at load time, and refuses when the library is
the broader of the two.

Observed frame against rest frame. The templates live at rest wavelengths, and
so does a spectrum that was de-redshifted before BRAIN saw it, but the
spectrograph smeared the light at the wavelength it was recorded at. So
sigma_inst is looked up at lambda_rest * (1 + z) while sigma_lib is looked up
at lambda_rest, and the two are compared in km/s, a currency in which the
(1 + z) factor cancels. `spectrum_redshift` is the redshift that was *removed*
from the spectrum before BRAIN saw it.

The data is never smoothed. The other way to match two line-spread functions is
to degrade the better one, which when the library is broader means convolving
the spectrum. Every chi2 in this code is formed with a diagonal inverse
variance; smoothing correlates neighbouring noise samples, so chi2 drops for a
reason that is not a better fit. When the library is broader BRAIN fits at the
library's native resolution and says in the summary what that costs.
"""

import logging
import os
import re

import numpy as np

logger = logging.getLogger("NEWBRAIN")

C_KMS = 299792.458

def removed_redshift(cfg):
    """
    The redshift taken out of the data before the fit, whoever took it out.

    A spectrum's is spectrum_redshift, whether you removed it before handing
    the spectrum over or BRAIN removed it (spectrum_in_rest_frame: false). A
    cube's is cube_redshift, which BRAIN always removes itself. brain_settings
    refuses both at once, so at most one is non-zero.

    Only spectrum_redshift used to be read here. A cube fitted at z > 0 has
    spectrum_redshift = 0, so the instrument's resolution was looked up at the
    rest wavelength instead of the observed one: harmless for a constant R,
    wrong by a factor (1 + z) for a resolution given in Angstrom, and at the
    wrong place on a resolution curve.
    """
    z_spectrum = float(getattr(cfg, "spectrum_redshift", 0.0) or 0.0)
    z_cube = float(getattr(cfg, "cube_redshift", 0.0) or 0.0)
    return z_spectrum if z_spectrum else z_cube


FWHM_TO_SIGMA = 1.0 / (2.0 * np.sqrt(2.0 * np.log(2.0)))

NODE_SPACING_PIXELS = 0.05
MAX_NODES = 256
COVERAGE_FLOOR = 1e-8
UNIFORM_TOLERANCE = 1e-9
MAX_WORK_PIXELS = 1 << 20

_HEADER_FWHM = re.compile(
    r"^\s*#\s*spectral\s+resolution\s*:\s*([0-9.eE+-]+)\s*\(?\s*(a|aa|angstrom)",
    re.IGNORECASE)
_HEADER_VD = re.compile(
    r"^\s*#\s*smoothed.*vd\s*\(km/?s\)\s*=\s*([0-9.eE+-]+)",
    re.IGNORECASE)
_SCALAR = re.compile(
    r"^\s*([-+]?[0-9]*\.?[0-9]+(?:[eE][-+]?[0-9]+)?)\s*"
    r"(km\s*/?\s*s|kms|a|aa|angstrom|)\s*"
    r"(fwhm|sigma|)\s*$", re.IGNORECASE)

def parse_resolution(value, setting_name):
    """
    Turn one setting into a resolution descriptor.

    Notes
    -----
    Accepted forms:

    None
        No information. Returns None.
    a number
        Full width at half maximum in Angstrom, constant with wavelength.
    "2.51 A" / "2.51 A FWHM"
        The same thing written out.
    "2.51 A sigma"
        A dispersion in Angstrom rather than a full width.
    "45 km/s"
        Constant in velocity. Read as a dispersion, because that is the
        convention every library that quotes one uses: template headers
        say `vd (km/s)`, and vd is a sigma. Write "45 km/s FWHM" if you mean
        a full width.
    a path to a file
        Two columns, wavelength in Angstrom and full width at half maximum in
        Angstrom, interpolated onto the fitted grid.

    The frame the tabulated wavelengths are read in is decided by the setting,
    not by this parser, which serves both. `instrumental_resolution` is
    evaluated at observed-frame wavelengths, lambda_rest * (1 + z), because a
    line-spread function is a property of the spectrograph and the spectrograph
    saw the light there. `library_resolution` is evaluated at rest wavelengths,
    because that is where the library lives.

    A unit is required on any string form. "2.51" and "2.51 FWHM" are refused
    rather than assumed to be Angstrom, because the same two characters of
    ambiguity between Angstrom and km/s is a factor of twenty on the answer.

    Returns
    -------
    dict or None
        Descriptor with a 'kind' of 'angstrom', 'kms' or 'table', plus a
        'text' field naming it for the log and the summary.
    """
    if value is None:
        return None

    if isinstance(value, (int, float)) and not isinstance(value, bool):
        fwhm = float(value)
        if not np.isfinite(fwhm) or fwhm <= 0:
            raise SystemExit(
                f"\n'{setting_name}' is {value!r}; a resolution must be a "
                f"positive number of Angstrom.\n")
        return {"kind": "angstrom", "sigma": fwhm * FWHM_TO_SIGMA,
                "text": f"{fwhm:.4g} A FWHM"}

    text = str(value).strip()
    if not text:
        return None

    m = _SCALAR.match(text)
    if m:
        number = float(m.group(1))
        unit = m.group(2).lower().replace(" ", "")
        shape = m.group(3).lower()
        if number <= 0:
            raise SystemExit(
                f"\n'{setting_name}' is {value!r}; a resolution must be "
                f"positive.\n")
        to_sigma = 1.0 if shape == "sigma" else FWHM_TO_SIGMA
        if unit in ("km/s", "kms"):
            to_sigma = FWHM_TO_SIGMA if shape == "fwhm" else 1.0
            return {"kind": "kms", "sigma_kms": number * to_sigma,
                    "bare_velocity": shape == "",
                    "text": f"{number:.4g} km/s "
                            f"{'FWHM' if shape == 'fwhm' else 'sigma'}"}
        if unit in ("a", "aa", "angstrom"):
            return {"kind": "angstrom", "sigma": number * to_sigma,
                    "text": f"{number:.4g} A "
                            f"{'sigma' if shape == 'sigma' else 'FWHM'}"}
        raise SystemExit(
            f"\n'{setting_name}' is {value!r}, which states a number"
            f"{' and a shape' if shape else ''} but no unit, and BRAIN will "
            f"not guess between Angstrom and km/s.\n"
            f"  Write {number:.4g} as  {number:.4g} A "
            f"{shape if shape else 'FWHM'}  for Angstrom, or as  "
            f"{number:.4g} km/s {shape if shape else 'sigma'}  for a "
            f"velocity.\n"
            f"  An unquoted plain number, {setting_name}: {number:.4g}, is "
            f"read as a FWHM in Angstrom.\n")

    if os.path.isfile(text):
        table = np.loadtxt(text)
        if table.ndim != 2 or table.shape[1] < 2:
            raise SystemExit(
                f"\n'{setting_name}' points at {text!r}, which needs two "
                f"columns: wavelength in Angstrom, then FWHM in Angstrom.\n")
        order = np.argsort(table[:, 0])
        lam = np.asarray(table[order, 0], dtype=np.float64)
        fwhm = np.asarray(table[order, 1], dtype=np.float64)
        if np.any(~np.isfinite(fwhm)) or np.any(fwhm <= 0):
            raise SystemExit(
                f"\n'{setting_name}' file {text!r} contains a FWHM that is "
                f"zero, negative or not a number.\n")
        return {"kind": "table", "lam": lam, "sigma": fwhm * FWHM_TO_SIGMA,
                "path": text,
                "text": f"{os.path.basename(text)} "
                        f"({fwhm.min():.3g}-{fwhm.max():.3g} A FWHM over "
                        f"{lam[0]:.0f}-{lam[-1]:.0f} A)"}

    raise SystemExit(
        f"\n'{setting_name}' is {value!r}, which BRAIN cannot read.\n"
        f"  Give a number of Angstrom (FWHM), a string like '45 km/s', or "
        f"the path to a two-column file (wavelength, FWHM in Angstrom).\n"
        f"  It does not parse as a number with a unit, and there is no file "
        f"at {text!r}.\n")

def sigma_angstrom(descriptor, lam, lookup_lam=None):
    """
    Evaluate a resolution descriptor as a dispersion in Angstrom.

    Parameters
    ----------
    descriptor : dict
        From parse_resolution or read_library_headers.
    lam : 1D array
        The wavelengths the returned widths belong to. Widths come back in the
        units of this axis.
    lookup_lam : 1D array or None
        The wavelengths at which the descriptor is evaluated, when that differs
        from `lam`. This is how a line-spread function tabulated in the
        observed frame is used on a rest-frame grid: look up at
        lambda_rest * (1 + z), and the width returned is then rescaled by
        lam / lookup_lam to land back on the rest-frame axis.
    """
    lam = np.asarray(lam, dtype=np.float64)
    look = lam if lookup_lam is None else np.asarray(lookup_lam,
                                                     dtype=np.float64)
    scale = lam / look

    kind = descriptor["kind"]
    if kind == "angstrom":
        return np.full_like(lam, descriptor["sigma"]) * scale
    if kind == "kms":
        return descriptor["sigma_kms"] / C_KMS * look * scale
    if kind == "table":
        return np.interp(look, descriptor["lam"], descriptor["sigma"],
                         left=descriptor["sigma"][0],
                         right=descriptor["sigma"][-1]) * scale
    if kind == "quadrature":
        total = np.zeros_like(lam)
        for part in descriptor["parts"]:
            total = total + sigma_angstrom(part, lam, lookup_lam) ** 2
        return np.sqrt(total)
    raise ValueError(f"unknown resolution descriptor kind {kind!r}")

def sigma_kms(descriptor, lam, lookup_lam=None):
    """
    The same width expressed as a velocity dispersion.

    Velocity widths are frame-invariant, so this is the currency in which the
    data and the library are compared and in which every reported number is
    quoted.
    """
    lam = np.asarray(lam, dtype=np.float64)
    return C_KMS * sigma_angstrom(descriptor, lam, lookup_lam) / lam

def read_library_headers(ssp_folder, bases_info):
    """
    Read the resolution each template file states about itself.

    Two header lines are understood, and a file may carry both:

        # Spectral resolution: 2.51 (A) FWHM
        # smoothed by a vd (km/s) = 45.0

    The first is a full width at half maximum in Angstrom, constant with
    wavelength. The second is a velocity dispersion applied on top of it, and
    is constant in km/s. A file carrying both has been broadened twice, so the
    two add in quadrature.

    A file that states neither is returned as None. It is not guessed at: a
    library's resolution is a measurable property of the library, and inventing
    a plausible number for it would produce exactly the kind of fit that looks
    right and is not. The caller is expected to say so and to name the
    `library_resolution` setting.

    Returns
    -------
    list
        One descriptor (or None) per entry in bases_info, in the same order.
    """
    out = []
    for entry in bases_info:
        path = os.path.join(ssp_folder, entry["filename"])
        fwhm_a, vd_kms = None, None
        if os.path.isfile(path):
            with open(path) as fh:
                for line in fh:
                    if not line.lstrip().startswith("#"):
                        break
                    hit = _HEADER_FWHM.match(line)
                    if hit:
                        fwhm_a = float(hit.group(1))
                        continue
                    hit = _HEADER_VD.match(line)
                    if hit:
                        vd_kms = float(hit.group(1))

        parts = []
        if fwhm_a is not None and fwhm_a > 0:
            parts.append({"kind": "angstrom", "sigma": fwhm_a * FWHM_TO_SIGMA,
                          "text": f"{fwhm_a:.4g} A FWHM"})
        if vd_kms is not None and vd_kms > 0:
            parts.append({"kind": "kms", "sigma_kms": vd_kms,
                          "text": f"{vd_kms:.4g} km/s sigma"})

        if not parts:
            out.append(None)
        elif len(parts) == 1:
            out.append(parts[0])
        else:
            out.append({"kind": "quadrature", "parts": parts,
                        "text": " + ".join(p["text"] for p in parts)})
    return out

def gaussian_filter1d(values, sigma_pixels):
    """
    Convolve along the first axis with a Gaussian whose dispersion varies pixel
    by pixel, exactly, including for widths well below one pixel.

    Sub-pixel is the ordinary case here, not an edge case: a 2.51 A library
    against a 2.9 A spectrograph asks for 1.45 A of extra width, which is a
    third of a pixel where the BC03 grid opens out to 4 A. A kernel built by
    point-sampling a Gaussian at integer offsets and renormalising cannot
    deliver that, and neither can one built by integrating it over the pixel:
    the first collapses below about 0.7 pixels and reaches zero near 0.1, and
    the second converges to the pixel's own top hat, delivering a variance
    wrong by dlam^2/12 in one direction or the other at every width. Either way
    the broadening that was asked for is silently not the one applied, and the
    difference lands on the fitted stellar dispersion.

    The construction here works in Fourier space, where a Gaussian of any
    width, undersampled or not, is the exact multiplier exp(-sigma^2 k^2 / 2).
    One transform cannot serve a width that changes with wavelength, so a set
    of sigma nodes is laid across the range of widths required, the whole array
    is convolved at each node, and every output pixel is a blend of the two
    nodes that bracket its own width.

    The blend is linear in VARIANCE, not in sigma. A mixture of two Gaussians
    has the weighted mean of their variances as its second moment, so weights
    f and 1 - f chosen from sigma^2 make the delivered second moment equal to
    the requested one exactly, at every pixel, at every width. Blending
    linearly in sigma would instead deliver a variance too large by
    f (1 - f) (sigma_1 - sigma_0)^2, which is a fraction of a per cent at a
    pixel of width but a factor of two at a twentieth of one.

    Nodes are spaced NODE_SPACING_PIXELS apart so the residual difference in
    SHAPE between the mixture and a true Gaussian stays negligible, and the
    count is capped at MAX_NODES. Only one node's transform exists at a time,
    so the working set stays at the size of the input.

    The array is zero-padded on both sides by more than the widest kernel
    before the transform, so the periodic wrap-around an FFT convolution
    implies cannot reach any pixel of the input. Zero padding also darkens the
    ends, which is deliberate: the caller divides by this function applied to
    an array of ones, and that ratio renormalises the ends in one step.

    Parameters
    ----------
    values : array, (n_pix,) or (n_pix, n_column)
        Spectra to convolve.
    sigma_pixels : 1D array, (n_pix,)
        Gaussian dispersion at each pixel, in pixels of this grid.
    """
    values = np.asarray(values, dtype=np.float64)
    flat = values.ndim == 1
    if flat:
        values = values[:, None]

    n_pix, n_col = values.shape
    sig = np.asarray(sigma_pixels, dtype=np.float64)
    if sig.shape != (n_pix,):
        raise ValueError("sigma_pixels must have one entry per pixel")
    sig = np.where(np.isfinite(sig), np.maximum(sig, 0.0), 0.0)

    sig_max = float(np.max(sig))
    if sig_max <= 0.0:
        out = np.array(values, copy=True)
        return out[:, 0] if flat else out

    sig_min = float(np.min(sig))
    n_nodes = int(np.ceil((sig_max - sig_min) / NODE_SPACING_PIXELS)) + 1
    n_nodes = int(np.clip(n_nodes, 2, MAX_NODES))
    nodes = np.linspace(sig_min, sig_max, n_nodes)
    node_var = nodes ** 2

    lower = np.clip(np.searchsorted(nodes, sig, side="right") - 1,
                    0, n_nodes - 2)
    span = node_var[lower + 1] - node_var[lower]
    frac = np.where(span > 0.0,
                    (sig ** 2 - node_var[lower]) / np.where(span > 0.0, span,
                                                            1.0),
                    0.0)
    frac = np.clip(frac, 0.0, 1.0)

    blend = np.zeros((n_nodes, n_pix), dtype=np.float64)
    columns = np.arange(n_pix)
    blend[lower, columns] = 1.0 - frac
    blend[lower + 1, columns] += frac

    pad = int(np.ceil(6.0 * sig_max)) + 4
    n_fft = int(2 ** int(np.ceil(np.log2(n_pix + 2 * pad))))
    padded = np.zeros((n_fft, n_col), dtype=np.float64)
    padded[pad:pad + n_pix] = values

    spectrum = np.fft.rfft(padded, axis=0)
    omega = 2.0 * np.pi * np.fft.rfftfreq(n_fft)

    out = np.zeros_like(values)
    for node in range(n_nodes):
        weight = blend[node]
        if not np.any(weight > 0.0):
            continue
        multiplier = np.exp(-0.5 * (nodes[node] * omega) ** 2)
        smoothed = np.fft.irfft(spectrum * multiplier[:, None], n=n_fft,
                                axis=0)
        out += weight[:, None] * smoothed[pad:pad + n_pix]

    return out[:, 0] if flat else out




def _uniform_step(axis):
    """
    The step of a uniformly spaced axis, or None if it is not uniform.

    The tolerance is relative to the mean step and is tight, because the
    question being asked is whether an exactly uniform sampling theorem
    applies, not whether the grid looks tidy.
    """
    axis = np.asarray(axis, dtype=np.float64)
    if axis.size < 3:
        return None
    step = np.diff(axis)
    mean = float(np.mean(step))
    if mean <= 0.0:
        return None
    if float(np.max(np.abs(step - mean))) > UNIFORM_TOLERANCE * mean:
        return None
    return mean

def convolve_on_native_grid(lam, block, sigma_lam):
    """
    Convolve templates at the sampling the library wrote them on.

    The width that separates two line-spread functions is quoted in Angstrom
    on the template's own wavelength axis, so that is where it belongs. The
    return value carries the axis it was computed on, because for one class of
    library grid that axis cannot be the native one.

    Three grids, three answers:

    Uniform in wavelength -- the shipped BasesGM, a clean 1 A step -- and
    uniform in log wavelength are both handled on the native axis itself, with
    the requested width converted to pixels of that axis. Nothing is
    interpolated and nothing is moved.

    Everything else has no uniform sampling for the Fourier kernel to be exact
    on. BC03 is the example that matters: 1 A through the blue, opening out to
    2, 3, 5 and eventually 2e5 A in the infrared. Those are convolved on a
    uniform log-wavelength axis whose velocity step is the FINEST the native
    grid uses anywhere in its range, so the working axis never undersamples
    what the file contains, and the caller resamples from that axis instead of
    from the native one. That is one flux-conserving mapping onto a grid at
    least as fine as the input, which is a different thing from convolving on
    the coarse grid the fit runs on.

    The ends are renormalised rather than darkened: gaussian_filter1d
    zero-pads, so an array of ones convolved with the identical kernel is
    exactly the fraction of the kernel that landed inside the array, and
    dividing by it makes the operator preserve a constant end to end.

    Returns
    -------
    axis : 1D array
        The wavelengths the convolved block belongs to.
    convolved : 2D array
        One column per input column.
    """
    lam = np.asarray(lam, dtype=np.float64)
    block = np.asarray(block, dtype=np.float64)
    if block.ndim == 1:
        block = block[:, None]
    sigma_lam = np.asarray(sigma_lam, dtype=np.float64)

    step = _uniform_step(lam)
    if step is not None:
        return lam, _renormalised_filter(block, sigma_lam / step)

    log_step = _uniform_step(np.log(lam))
    if log_step is not None:
        return lam, _renormalised_filter(block, sigma_lam / lam / log_step)

    from brainsp.spectral_engine import flux_conserving_resample

    velocity_step = float(np.min(np.diff(np.log(lam))))
    n_work = int(np.ceil(np.log(lam[-1] / lam[0]) / velocity_step)) + 1
    if n_work > MAX_WORK_PIXELS:
        velocity_step = np.log(lam[-1] / lam[0]) / (MAX_WORK_PIXELS - 1)
        n_work = MAX_WORK_PIXELS
        logger.warning(
            f"   The template grid needs {n_work} pixels to be sampled as "
            f"finely as its own finest step; the convolution axis was capped "
            f"there, so features narrower than "
            f"{C_KMS * velocity_step:.1f} km/s in the library are smoothed by "
            f"the mapping onto it.")
    work = lam[0] * np.exp(np.arange(n_work) * velocity_step)

    moved = np.zeros((n_work, block.shape[1]), dtype=np.float64)
    for column in range(block.shape[1]):
        moved[:, column] = flux_conserving_resample(lam, block[:, column],
                                                    work)
    sigma_work = np.interp(work, lam, sigma_lam)
    return work, _renormalised_filter(moved, sigma_work / work / velocity_step)




def _renormalised_filter(block, sigma_pixels):
    """
    gaussian_filter1d with the array ends divided back up to full weight.

    The kernel that reaches past the first or last pixel has nowhere to draw
    from, and zero-padding silently answers "zero" for the missing part. The
    same kernel applied to an array of ones measures exactly how much of it
    landed inside, and dividing by that is the same Gaussian truncated at the
    boundary and renormalised -- which is what a convolution of a signal that
    is simply unmeasured beyond a point should be. A flat template therefore
    comes back flat at both ends instead of dropping by half.
    """
    numerator = gaussian_filter1d(block, sigma_pixels)
    denominator = gaussian_filter1d(np.ones(block.shape[0]), sigma_pixels)
    safe = denominator > COVERAGE_FLOOR
    denominator = np.where(safe, denominator, 1.0)
    return numerator / denominator[:, None]

def convolve_group_to_data_lsf(ssp_folder, bases_info, indices, lam_grid,
                               i_ref, data_res, library_res, redshift):
    """
    Rebuild one resolution group from its files: convolve, resample, normalise.

    This is the order the correction has to happen in. The kernel that carries
    a library up to a spectrograph's line-spread function is a width in
    Angstrom on the grid the library was written on, so it is applied there,
    at the library's own sampling, before anything has been interpolated.
    Only then is the result mapped onto the fitted log-lambda grid, and only
    then is it divided by its own value at the normalisation wavelength.

    Doing it the other way round -- resample, then convolve -- applies the
    kernel on a grid whose Angstrom-per-pixel is not the one the width was
    quoted on, to flux that has already been through one interpolation, and it
    normalises before the operation that changes the number being normalised.

    The columns are built from the files rather than from the matrix that is
    already on the log grid, because the native sampling is the whole point and
    it does not survive that matrix. Every column the caller replaces is
    therefore resampled exactly once; the first resampling is discarded, not
    composed with a second.

    Each file is trimmed to the fitted range plus ten kernel dispersions before
    anything is convolved. The trim never cuts inside the fitted range, so
    which target pixels a template covers is exactly what it was, and the
    pixels it removes are ones the Gaussian tail reaches at the 1e-22 level.
    It matters for a library like BC03 that runs to 1.6e6 A: without it the
    working axis for a non-uniform grid would be laid across five decades of
    wavelength to serve a fit that uses one octave of it.

    `library_res` is evaluated at rest wavelengths and `data_res` at
    lambda_rest * (1 + redshift), for the reason at the top of this module: the
    spectrograph smeared the light where it recorded it. Where the library is
    the broader of the two the difference is clipped to zero rather than
    sharpened. The caller has already refused any group for which that happens
    at a fitted pixel, so the clip can only bite outside the fitted window --
    on the BC03 infrared, say, where a library broadened in velocity overtakes
    an instrument of constant width in Angstrom.

    Returns
    -------
    columns : 2D array, (n_grid, len(indices))
        Normalised to exactly 1.0 at lam_grid[i_ref].
    factors : 1D array, (len(indices),)
        What each column was divided by. This is the column's entry in
        fbase_norms: it is measured on the template that is actually fitted,
        because the convolution happened before it was measured.
    usable : 1D bool array, (len(indices),)
        False where the convolved column is zero or not finite at the
        normalisation wavelength and the 1.0 convention could not be restored.
    delta_lam : 1D array
        The broadening applied, in Angstrom, over the trimmed native grid of
        the first file in the group. Reported, not used.
    """
    from brainsp.io_utils import load_native_template, resample_template

    lam_grid = np.asarray(lam_grid, dtype=np.float64)
    indices = np.asarray(indices, dtype=int)

    grids = {}
    order = []
    for position, j in enumerate(indices):
        path = os.path.join(ssp_folder, bases_info[j]["filename"])
        wavelength, flux = load_native_template(path)
        wavelength = np.asarray(wavelength, dtype=np.float64)
        key = (wavelength.size, float(wavelength[0]), float(wavelength[-1]),
               float(np.sum(wavelength)))
        if key not in grids:
            grids[key] = [wavelength, [], []]
            order.append(key)
        grids[key][1].append(position)
        grids[key][2].append(np.asarray(flux, dtype=np.float64))

    columns = np.zeros((lam_grid.size, indices.size), dtype=np.float64)
    factors = np.ones(indices.size, dtype=np.float64)
    usable = np.ones(indices.size, dtype=bool)
    delta_lam = None

    for key in order:
        wavelength, positions, fluxes = grids[key]
        sigma_data = sigma_angstrom(data_res, wavelength,
                                    lookup_lam=wavelength * (1.0 + redshift))
        sigma_library = sigma_angstrom(library_res, wavelength)
        delta = np.sqrt(np.maximum(sigma_data ** 2 - sigma_library ** 2, 0.0))

        keep = _fitting_window(wavelength, delta, lam_grid[0], lam_grid[-1])
        wavelength, delta = wavelength[keep], delta[keep]
        block = np.column_stack([column[keep] for column in fluxes])
        if delta_lam is None:
            delta_lam = delta

        axis, smoothed = convolve_on_native_grid(wavelength, block, delta)
        for k, position in enumerate(positions):
            column = resample_template(axis, smoothed[:, k], lam_grid)
            factor = float(column[i_ref])
            if not np.isfinite(factor) or factor <= 0.0:
                usable[position] = False
                factor = 1.0
            columns[:, position] = column / factor
            factors[position] = factor

    return columns, factors, usable, delta_lam




def _fitting_window(lam, sigma_lam, lam_low, lam_high):
    """
    Which native pixels the fitted range needs, with room for the kernel.

    Ten dispersions past each end of the fitted range, and never less than
    twenty native pixels, so a Gaussian truncated at the trim is truncated
    where it is below 1e-22 of its peak. The cut is taken outside the fitted
    range in both directions, which is what keeps the template's coverage of
    that range exactly what it was: `resample_template` decides coverage from
    the first and last wavelength it is handed, and neither moves inside the
    range.
    """
    lam = np.asarray(lam, dtype=np.float64)
    if lam.size < 3:
        return np.ones(lam.size, dtype=bool)

    step = np.gradient(lam)
    at_low = int(np.argmin(np.abs(lam - lam_low)))
    at_high = int(np.argmin(np.abs(lam - lam_high)))
    reach_low = max(10.0 * sigma_lam[at_low], 20.0 * step[at_low])
    reach_high = max(10.0 * sigma_lam[at_high], 20.0 * step[at_high])
    return (lam >= lam_low - reach_low) & (lam <= lam_high + reach_high)


def _crossover_text(lam, delta_var):
    """
    Describe where a signed variance difference changes sign, in words.

    delta_var is sigma_inst^2 - sigma_lib^2 per pixel. Where it is negative the
    library is the broader of the two and no real convolution exists. Both
    arrays are expected to have been restricted to the pixels that are actually
    fitted, because a statement about pixels the fit never looks at is not a
    statement about the fit.
    """
    lam = np.asarray(lam, dtype=np.float64)
    delta_var = np.asarray(delta_var, dtype=np.float64)
    if lam.size == 0:
        return None, "there are no fitted pixels to compare over"

    negative = delta_var < 0
    lo, hi = float(lam[0]), float(lam[-1])
    if not np.any(negative):
        return None, f"the library is narrower than the data across {lo:.0f}-{hi:.0f} A"
    if np.all(negative):
        return None, (f"the library is broader than the data across the whole "
                      f"fitted window, {lo:.0f}-{hi:.0f} A, and the two never "
                      f"cross inside it")

    flips = np.nonzero(np.diff(negative.astype(int)) != 0)[0]
    crossings = []
    for i in flips:
        y0, y1 = delta_var[i], delta_var[i + 1]
        frac = 0.0 if y1 == y0 else y0 / (y0 - y1)
        crossings.append(float(lam[i] + frac * (lam[i + 1] - lam[i])))

    bad = lam[negative]
    frac_bad = 100.0 * float(np.sum(negative)) / len(lam)
    where = ", ".join(f"{c:.0f} A" for c in crossings)
    return crossings, (f"the library is broader than the data over "
                       f"{bad.min():.0f}-{bad.max():.0f} A ({frac_bad:.0f}% of "
                       f"the fitted window); the two line-spread functions "
                       f"cross at {where}")


def _worst_deficit(lam, delta_var, fitted):
    """
    The largest broadening the library asks for and cannot have, and where.

    Returns (kms, wavelength). The quantity is sqrt(max(-delta_var)) over the
    fitted pixels only, which is the honest floor on a stellar dispersion this
    library can support: quoting the value at the normalisation wavelength
    instead reports 0.0 km/s in the common case where the deficit sits at the
    blue end and the two line-spread functions have already crossed by 5500 A.
    """
    deficit = np.where(fitted, np.asarray(delta_var, dtype=np.float64), np.inf)
    if not np.any(np.isfinite(deficit)):
        return 0.0, float(lam[0])
    k = int(np.argmin(deficit))
    if not np.isfinite(deficit[k]) or deficit[k] >= 0:
        return 0.0, float(lam[k])
    return float(np.sqrt(-deficit[k])), float(lam[k])


def _is_featureless(entry):
    """
    Whether a base component has no spectral features to be broadened.

    The template catalogue's KIND column marks a power-law component as
    something other than 'ssp', and brain_settings carries that word through
    onto the entry. A power law has no lines and its curvature over the
    fraction of a pixel these kernels are wide is nothing, so convolving it is
    a no-operation: it is excluded from the convolution, and it is not counted
    as a template of unknown resolution, because there is no resolution for it
    to state and requiring one would let it veto the correction for the whole
    library.
    """
    return str(entry.get("kind", "ssp")).lower() != "ssp"

def prepare_templates(matrix, lam_grid, cfg, bases_info, fit_pixels=None,
                      fbase_norms=None):
    """
    Bring the templates to the data's line-spread function, or explain why not.

    Called once, immediately after the templates have been mapped onto the
    log-lambda grid and after the pixels outside the template coverage have
    been dropped from the fit. The returned matrix has exactly the shape it
    came in with, so nothing downstream recompiles.

    With `instrumental_resolution` unset there is nothing to match to, the
    matrix and the normalisation factors are returned as the very objects they
    came in as, and the fit is bit-for-bit what it was.

    `fit_pixels` is the boolean mask of pixels the fit will actually use, which
    is where the comparison is made. Judging it over the whole grid instead
    binds the decision to pixels no template covers and that are about to be
    thrown away: on the reference setup, widening `wavelength_max` from 6900 to
    9000 A flipped the module from correcting to refusing on the strength of
    379 pixels beyond 8445 A that are identically zero in every template.

    The decision is taken once per resolution group, not once for the library.
    A heterogeneous library is the normal case — the shipped one is 56 files at
    2.51 A FWHM and 28 at 0.3 A FWHM plus 45 km/s — and a global refusal lets
    one group that crosses the data's line-spread function leave every other
    group uncorrected. Groups whose difference is real at every fitted pixel
    are convolved; the rest are left at their native resolution; which is which
    is recorded, reported, and carried into `dispersion_report`.

    A group that is corrected is rebuilt from its files by
    `convolve_group_to_data_lsf`: convolved at the library's own sampling,
    then resampled onto lam_grid, then divided by its own value at the
    normalisation wavelength. The 1.0 at that wavelength and the matching
    entry in `fbase_norms` therefore hold by construction rather than by a
    correction applied afterwards — the factor is measured on the template
    that is actually fitted, because the convolution has already happened when
    it is measured. Everything downstream — the light fractions, Mini, Mcor,
    Q(H) — is written against that convention, and the physical SSP the column
    and its factor jointly describe is untouched.

    `fbase_norms` may be None, in which case the returned array carries 1.0 for
    every group that was left alone and the measured factor for every group
    that was rebuilt.

    Pixels that were exactly zero before the correction are zero after it: a
    template covers the span its file covers, convolving does not change that
    span, and `resample_template` leaves everything outside it at zero. The
    pixels at the edge of that span are renormalised rather than darkened; see
    convolve_native_gaussian.

    Returns
    -------
    matrix : 2D array
        Corrected templates, or the input array itself when no correction was
        applied.
    fbase_norms : 1D array
        Normalisation factors, matching the columns that came back.
    info : dict
        Everything the summary needs, including the per-pixel dispersions and
        the per-group verdicts that survive into the reported numbers.
    """
    lam = np.asarray(lam_grid, dtype=np.float64)
    redshift = removed_redshift(cfg)
    lam_obs = lam * (1.0 + redshift)

    bases_info = [e for e in bases_info
                  if os.path.isfile(os.path.join(cfg.ssp_folder,
                                                 e["filename"]))]
    n_template = len(bases_info)
    featureless = np.array([_is_featureless(e) for e in bases_info],
                           dtype=bool)

    norms_in = fbase_norms
    norms = (np.ones(n_template) if fbase_norms is None
             else np.array(fbase_norms, dtype=np.float64, copy=True))

    if fit_pixels is None:
        fitted = np.any(np.abs(np.asarray(matrix)) > 0, axis=1)
    else:
        fitted = np.asarray(fit_pixels, dtype=bool)
    if not np.any(fitted):
        fitted = np.ones(lam.shape, dtype=bool)

    info = {
        "lam": lam,
        "fitted": fitted,
        "redshift": redshift,
        "data_text": "not set",
        "library_text": "not determined",
        "library_source": "not determined",
        "library_setting_text": "not set",
        "applied": False,
        "refused": False,
        "state_text": "none",
        "note": None,
        "sigma_inst_kms": None,
        "library_groups": None,
        "pre_convolution_kms": None,
        "n_templates": n_template,
        "n_featureless": int(np.sum(featureless)),
        "n_unknown": 0,
        "unknown_indices": np.zeros(0, dtype=int),
        "n_applied": 0,
        "n_refused": 0,
        "floor_kms": 0.0,
        "floor_wavelength": float(lam[0]),
    }

    data_res = parse_resolution(getattr(cfg, "instrumental_resolution", None),
                               "instrumental_resolution")
    lib_setting = parse_resolution(getattr(cfg, "library_resolution", None),
                                   "library_resolution")

    for name, descriptor in (("instrumental_resolution", data_res),
                             ("library_resolution", lib_setting)):
        if descriptor is not None and descriptor.get("bare_velocity"):
            logger.info(
                f"   '{name}' is {descriptor['text']}: a velocity resolution "
                f"written without the word FWHM or sigma is read as a 1-SIGMA "
                f"width. Spectrograph documentation usually quotes a FWHM "
                f"instead, and reading one as the other inflates the variance "
                f"by 5.55. Write '... km/s FWHM' if that is what you have.")

    if lib_setting is not None:
        descriptors = [None if featureless[j] else lib_setting
                       for j in range(n_template)]
        info["library_source"] = "library_resolution setting"
        info["library_text"] = lib_setting["text"]
        info["library_setting_text"] = lib_setting["text"]
    else:
        descriptors = read_library_headers(cfg.ssp_folder, bases_info)
        descriptors = [None if featureless[j] else d
                       for j, d in enumerate(descriptors)]
        known = [d for d in descriptors if d is not None]
        if known:
            info["library_source"] = "template file headers"
            labels = sorted({d["text"] for d in known})
            counts = {t: sum(1 for d in known if d["text"] == t)
                      for t in labels}
            info["library_text"] = "; ".join(
                f"{t} x{counts[t]}" for t in labels)
        else:
            info["library_source"] = "unknown"
            info["library_text"] = "not stated by the library"

    unknown_indices = np.asarray(
        [j for j in range(n_template)
         if descriptors[j] is None and not featureless[j]], dtype=int)
    info["unknown_indices"] = unknown_indices
    info["n_unknown"] = int(unknown_indices.size)

    if info["n_unknown"] and info["library_source"] == "template file headers":
        info["library_text"] += f"; unknown x{info['n_unknown']}"
    if info["n_featureless"]:
        info["library_text"] += (f"; {info['n_featureless']} featureless "
                                 f"component(s), no resolution needed")

    groups = {}
    for j, d in enumerate(descriptors):
        if d is None:
            continue
        groups.setdefault(d["text"], [d, []])[1].append(j)
    group_list = [(np.asarray(idx, dtype=int), d, label)
                  for label, (d, idx) in groups.items()]
    info["library_groups"] = [
        (idx, sigma_kms(d, lam), label, False) for idx, d, label in group_list]

    if data_res is None:
        if info["library_source"] != "unknown":
            logger.info(f"   Library resolution: {info['library_text']} "
                        f"(from {info['library_source']}). "
                        f"'instrumental_resolution' is not set, so no "
                        f"line-spread-function correction is applied and the "
                        f"reported dispersion is an observed one.")
        return matrix, norms_in if norms_in is not None else norms, info

    info["data_text"] = data_res["text"]
    sigma_inst_A = sigma_angstrom(data_res, lam, lookup_lam=lam_obs)
    info["sigma_inst_kms"] = C_KMS * sigma_inst_A / lam

    if info["n_unknown"]:
        logger.warning(
            f"   {info['n_unknown']}/{n_template} template files state no "
            f"spectral resolution in their headers, so BRAIN does not know "
            f"what to correct from and will not guess. Set "
            f"'library_resolution' to the library's FWHM in Angstrom (or to "
            f"something like '45 km/s' for a library broadened in velocity) "
            f"and run again. Fitting at the library's native resolution; "
            f"the reported dispersion is an observed one, and sigma_star is "
            f"reported as unknown rather than computed from the "
            f"{n_template - info['n_unknown']} templates that do state one.")
        return matrix, norms_in if norms_in is not None else norms, info

    i_ref = int(np.argmin(np.abs(lam - cfg.norm_wavelength)))
    matrix_in = np.asarray(matrix, dtype=np.float64)
    out = np.array(matrix_in, copy=True)
    applied_kms = np.zeros(lam.shape)

    verdicts = []
    n_applied = 0
    n_refused = 0
    worst = None
    unscaled = 0

    for idx, descriptor, label in group_list:
        sig_lib_kms = sigma_kms(descriptor, lam)
        delta_var = info["sigma_inst_kms"] ** 2 - sig_lib_kms ** 2
        floor, floor_lam = _worst_deficit(lam, delta_var, fitted)

        if floor > 0.0:
            _, where = _crossover_text(lam[fitted], delta_var[fitted])
            verdicts.append((idx, sig_lib_kms, label, False))
            n_refused += len(idx)
            names = [bases_info[j]["short_name"] for j in idx]
            logger.warning(
                f"   Instrumental resolution: {len(idx)} of {n_template} "
                f"templates are stated as {label}, which is BROADER than the "
                f"data ({data_res['text']} in the observed frame, z = "
                f"{redshift:.4f}) somewhere in the fitted window, so BRAIN is "
                f"not convolving them. Over the fitted pixels, {where}. The "
                f"broadening the fit would need there is imaginary. BRAIN is "
                f"also not smoothing your spectrum to match: chi2 here is "
                f"formed with a diagonal inverse variance, and smoothing the "
                f"data correlates neighbouring noise samples, which makes "
                f"chi2 drop for a reason that is not a better fit. These "
                f"templates enter the fit at their native resolution, and "
                f"with them in it a stellar dispersion below {floor:.2f} km/s "
                f"— the worst deficit, at {floor_lam:.0f} A — is not "
                f"measurable. Affected templates begin with "
                f"{', '.join(names[:3])}{' ...' if len(names) > 3 else ''}.")
            if worst is None or floor > worst[0]:
                worst = (floor, floor_lam, where)
            continue

        delta_kms = np.sqrt(np.maximum(delta_var, 0.0))
        columns, factors, usable, delta_native = convolve_group_to_data_lsf(
            cfg.ssp_folder, bases_info, idx, lam, i_ref, data_res, descriptor,
            redshift)
        unscaled += int(np.sum(~usable))

        previous = norms[idx]
        scale = factors / np.where(previous > 0.0, previous, 1.0)

        out[:, idx] = columns
        norms[idx] = factors
        applied_kms = np.maximum(applied_kms, delta_kms)
        verdicts.append((idx, sig_lib_kms, label, True))
        n_applied += len(idx)

        logger.info(
            f"   Pre-convolved {len(idx)} template(s) stated as {label} up to "
            f"the data line-spread function ({data_res['text']}, z = "
            f"{redshift:.4f}), on the library's own wavelength grid and before "
            f"the resampling: {delta_kms[i_ref]:.2f} km/s at "
            f"{lam[i_ref]:.0f} A ({delta_kms.min():.2f}-{delta_kms.max():.2f} "
            f"km/s across the fitted window, "
            f"{delta_native.min():.3f}-{delta_native.max():.3f} A across the "
            f"template's own grid); normalising after the convolution rather "
            f"than before it moved the normalisation factors by "
            f"{scale.min():.5f}-{scale.max():.5f}, and every column is exactly "
            f"1.0 at {lam[i_ref]:.0f} A.")

    if unscaled:
        logger.warning(
            f"   {unscaled} convolved template(s) are zero or not finite at "
            f"the normalisation wavelength {lam[i_ref]:.0f} A, so the 1.0 "
            f"convention could not be restored for them. Their light "
            f"fractions and masses are as reliable as they were before the "
            f"convolution and no more.")

    info["library_groups"] = verdicts
    info["applied"] = n_applied > 0
    info["refused"] = n_refused > 0
    info["n_applied"] = n_applied
    info["n_refused"] = n_refused

    if worst is not None:
        info["floor_kms"], info["floor_wavelength"], info["note"] = worst

    if n_applied and n_refused:
        info["state_text"] = (f"partial: {n_applied} pre-convolved, "
                              f"{n_refused} left at library resolution")
        logger.warning(
            f"   Instrumental resolution, PARTIAL outcome: {n_applied} of "
            f"{n_template} templates were pre-convolved to the data "
            f"line-spread function and {n_refused} were left at their native "
            f"resolution because the library is broader than the data over "
            f"part of the fitted window for them. sigma_star below is "
            f"corrected for the difference that remains, weighted by how much "
            f"light each group carries.")
    elif n_applied:
        info["state_text"] = "pre-convolved to the data LSF"
    elif n_refused:
        info["state_text"] = "refused, library broader than the data"

    if not n_applied:
        return matrix, norms_in if norms_in is not None else norms, info

    info["pre_convolution_kms"] = applied_kms
    return out, norms, info

def on_bound(value, lower, upper=None):
    """
    Is a fitted dispersion sitting on the lower or the upper end of its range?

    Returns (on_lower, on_upper). A parameter at a bound is not a measurement
    of anything, and the stellar and the gas dispersions have to agree on when
    that has happened, so the test lives here once rather than in each of the
    two reports.

    The test is made to a tolerance of 1e-4 of the WIDTH of the allowed range
    rather than to an absolute one in km/s, because a range starting at 0
    makes any absolute tolerance either meaningless or arbitrary: the
    optimiser returns values like 5.6e-05 km/s on a railed fit, which an
    absolute 1e-6 km/s test misses and then reports as a measurement of 0.00.
    A range of zero width -- the case where no upper bound was supplied --
    falls back to a relative test on the bound itself.

    A non-finite value is on neither bound; it is a failed fit, and the caller
    has to say so in its own words.
    """
    low = float(lower)
    high = float(upper) if upper is not None else low
    width = high - low
    tolerance = (1e-4 * width if width > 0
                 else 1e-6 * max(abs(low), 1.0))
    if not np.isfinite(value):
        return False, False
    return (bool(abs(float(value) - low) <= tolerance),
            bool(width > 0 and abs(float(value) - high) <= tolerance))

def instrumental_dispersion_at(info, wavelengths):
    """
    The instrumental dispersion in km/s at a set of wavelengths, or None.

    `info["sigma_inst_kms"]` is the data's line-spread function expressed as a
    velocity dispersion, sampled on `info["lam"]`, which is the REST grid the
    fit runs on; the observed-frame lookup and the (1 + z) that cancels in
    km/s are already inside it. Interpolating it onto the rest wavelengths of
    a set of emission lines is therefore the whole of the operation, and it is
    written here rather than at the call site so that nothing else has to know
    which frame that array is indexed in.

    None when `instrumental_resolution` was not set, which is the state in
    which BRAIN knows nothing about the instrument and says so rather than
    guessing.
    """
    if info is None or info.get("sigma_inst_kms") is None:
        return None
    lam = np.asarray(info["lam"], dtype=np.float64)
    sigma = np.asarray(info["sigma_inst_kms"], dtype=np.float64)
    return np.interp(np.asarray(wavelengths, dtype=np.float64), lam, sigma)

def dispersion_report(info, sigma_obs, weights_star, model_flux, good,
                      lower_bound, upper_bound=None,
                      reference_wavelength=5500.0):
    """
    Turn the fitted dispersion into the three numbers that belong in a summary.

    sigma_obs
        What the kinematic kernel actually applied to the library. This is the
        number BRAIN has always reported as vd_min. On its own it is a
        statement about the templates, not about the galaxy.

    delta_sigma_inst
        The signed quadrature difference between the data's line-spread
        function and the library's, as the library entered the fit, averaged
        over the fitted window. Positive means the data is the broader of the
        two. A group of templates that was pre-convolved contributes zero,
        because the library was brought to the data's line-spread function
        before the fit ever saw it and the correction is not counted twice; a
        group that was refused contributes sigma_inst^2 - sigma_lib^2, which is
        what is still there.

        Westfall et al. 2019 (AJ 158, 231) Eq. 6 is the quadrature relation
        this rests on, sigma_obs^2 = sigma_star^2 + delta_sigma_inst^2, and
        Eq. 7 averages it over the fitted channels. That average is an
        unweighted mean, 1/N_fit, and the paper names uniform weighting as its
        simplifying assumption. The weighting used here — by the model flux,
        and by each template group's share of the light — is BRAIN's own
        choice, on the grounds that a pixel with no light in it carries no
        information about the line-spread function the fit saw. It was checked
        against the unweighted mean and moves the answer by 0.4%.

    sigma_star
        sign(x) * sqrt(abs(x)) with x = sigma_obs^2 - delta_sigma_inst^2, where
        the subtracted quantity keeps the sign of the variance difference. A
        negative x means the fit applied less broadening than the difference in
        line-spread functions alone demands, which no astrophysical dispersion
        can produce; the magnitude is then an upper limit, not a measurement.

    Both are reported as unknown, never as a number, when any template in the
    fit states no resolution. A number averaged over the subset that does state
    one is not a property of the fit that was run, and publishing it would
    contradict the log line that says no correction was applied. A power law or
    other featureless component is not a template of unknown resolution and
    does not trigger this; see _is_featureless.

    An upper limit is also reported whenever sigma_obs comes back sitting on
    the lower bound of its allowed range, because a parameter at a bound is not
    a measurement of anything. See on_bound for the test.
    """
    out = {
        "sigma_obs": float(sigma_obs),
        "delta_sigma_inst": None,
        "sigma_star": None,
        "upper_limit": False,
        "railed": False,
        "reason": None,
        "data_text": info["data_text"],
        "library_text": info["library_text"],
        "library_source": info["library_source"],
        "redshift": info["redshift"],
        "applied": info["applied"],
        "refused": info["refused"],
        "state_text": info.get("state_text", "none"),
        "n_applied": info.get("n_applied", 0),
        "n_refused": info.get("n_refused", 0),
    }

    lam = info["lam"]
    good = np.asarray(good, dtype=bool)
    weight = np.where(good, np.abs(np.asarray(model_flux, dtype=np.float64)),
                      0.0)
    if weight.sum() <= 0:
        weight = good.astype(np.float64)

    low = float(lower_bound)
    high = float(upper_bound) if upper_bound is not None else low
    out["railed"] = on_bound(sigma_obs, low, high)[0]

    if info["sigma_inst_kms"] is None or not info["library_groups"]:
        if out["railed"]:
            out["upper_limit"] = True
            out["reason"] = (
                f"sigma_obs came back on the lower bound of "
                f"dispersion_range ({low:.2f} km/s), so it is a bound, "
                f"not a measurement. BRAIN cannot say how much of that is "
                f"instrumental because 'instrumental_resolution' is not set "
                f"and the library "
                f"{'states ' + info['library_text'] if info['library_source'] != 'unknown' else 'states no resolution'}"
                f". Set 'instrumental_resolution' to the data line-spread "
                f"function and 'spectrum_redshift' to the redshift removed "
                f"from the spectrum, and re-run.")
        return out

    star_w = np.asarray(weights_star, dtype=np.float64)
    if star_w.sum() <= 0:
        star_w = np.ones_like(star_w)

    unknown = np.asarray(info.get("unknown_indices", np.zeros(0, dtype=int)),
                         dtype=int)
    if unknown.size:
        with_light = int(np.sum(star_w[unknown] > 0))
        out["reason"] = (
            f"sigma_star is not reported: {unknown.size} of the "
            f"{info['n_templates']} templates in this fit state no spectral "
            f"resolution ({with_light} of them carry light), so BRAIN does "
            f"not know what the library's line-spread function was and "
            f"applied no correction. A number averaged over the "
            f"{info['n_templates'] - unknown.size} templates that do state "
            f"one would not describe the fit that actually ran, and would "
            f"contradict the log line saying nothing was corrected. "
            f"'library_resolution' is {info['library_setting_text']}; set it "
            f"to the library's FWHM in Angstrom (or to something like "
            f"'45 km/s') and re-run. The reported vd_min "
            f"{float(sigma_obs):.2f} km/s is an observed dispersion and "
            f"includes the instrument.")
        return out

    shares = [(float(star_w[idx].sum()), sig_lib_kms, corrected)
              for idx, sig_lib_kms, _label, corrected
              in info["library_groups"]]
    total = float(sum(share for share, _g, _c in shares))
    if total <= 0:
        shares = [(float(len(idx)), sig_lib_kms, corrected)
                  for idx, sig_lib_kms, _label, corrected
                  in info["library_groups"]]
        total = float(sum(share for share, _g, _c in shares))

    delta_var_pix = np.zeros_like(lam)
    for share, sig_lib_kms, corrected in shares:
        if share <= 0:
            continue
        residual = (np.zeros_like(lam) if corrected
                    else info["sigma_inst_kms"] ** 2 - sig_lib_kms ** 2)
        delta_var_pix = delta_var_pix + share * residual
    delta_var_pix = delta_var_pix / total

    delta_var = float(np.sum(weight * delta_var_pix) / np.sum(weight))
    out["delta_sigma_inst"] = float(np.sign(delta_var)
                                    * np.sqrt(abs(delta_var)))
    out["delta_variance"] = delta_var

    x = float(sigma_obs) ** 2 - delta_var
    out["sigma_star"] = float(np.sign(x) * np.sqrt(abs(x)))

    floor, floor_lam = _worst_deficit(lam, delta_var_pix, good)
    out["floor_kms"] = floor
    out["floor_wavelength"] = floor_lam

    if x < 0:
        out["upper_limit"] = True
        out["reason"] = (
            f"sigma_obs {sigma_obs:.2f} km/s is smaller than the difference "
            f"between the two line-spread functions alone "
            f"({abs(out['delta_sigma_inst']):.2f} km/s, data broader than "
            f"library), so no real stellar dispersion reproduces it. "
            f"|sigma_star| = {abs(out['sigma_star']):.2f} km/s is an upper "
            f"limit.")
    elif out["railed"] and info["refused"]:
        out["upper_limit"] = True
        out["reason"] = (
            f"sigma_star railed at the lower bound of dispersion_range "
            f"({low:.2f} km/s). The template library "
            f"({info['library_text']}) exceeds the data line-spread function "
            f"({info['data_text']} in the observed frame, z = "
            f"{info['redshift']:.4f}) for "
            f"{info.get('n_refused', 0)} of {info['n_templates']} templates: "
            f"{info['note']}. The broadening the fit would need there is "
            f"imaginary, and with those templates in the fit a stellar "
            f"dispersion below {floor:.2f} km/s — the worst deficit, at "
            f"{floor_lam:.0f} A — is not measurable. The reported sigma_star "
            f"{out['sigma_star']:.2f} km/s is an UPPER LIMIT: the fit could "
            f"not go any lower, not that the stars are this cold.")
    elif out["railed"]:
        out["upper_limit"] = True
        out["reason"] = (
            f"sigma_star railed at the lower bound of dispersion_range "
            f"({low:.2f} km/s), so it is a bound and not a "
            f"measurement. The line-spread functions were matched before the "
            f"fit ({info['data_text']} data against {info['library_text']} "
            f"library), so this is the galaxy asking for less broadening than "
            f"you allowed. Lower the first number of dispersion_range and "
            f"re-run. The reported sigma_star {out['sigma_star']:.2f} km/s is "
            f"an UPPER LIMIT.")

    return out
