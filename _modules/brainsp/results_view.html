Source code for brainsp.results_view

"""
BRAIN GUI: the Results workspace, a viewer for a fitted megacube.

For a megacube BRAIN wrote: every map the file holds, two at a time side by side;
a click on a spaxel shows its spectrum with the stellar fit and each
emission line, its star-formation history in light and in mass, and its
numbers; the BPT [N II], BPT [S II] and WHAN classifications as maps and as
diagrams; and every figure can be exported.

Nothing here fits anything or imports JAX: it reads the file.

The diagnostic curves, with their sources:

    Kewley et al. 2001     [N II] BPT maximum starburst
    Kauffmann et al. 2003  [N II] BPT pure star formation, to its asymptote
                           at log [N II]/Halpha = 0.05. No extra cut at
                           -0.25 is applied, which would leave the spaxels
                           between -0.25 and 0.05 below the curve in no
                           class at all.
    Cid Fernandes et al. 2010   LINER / Seyfert in the [N II] BPT
    Kewley et al. 2006     [S II] BPT, star forming and Seyfert / LINER
    Cid Fernandes et al. 2011   WHAN

A diagnostic is drawn only where every line it needs has a positive model
flux. BRAIN does not yet write line-flux uncertainties, so there is no S/N
cut on the lines: a weak line that the fit placed at a small positive flux
is classified like a strong one. The classification of a faint spaxel is
therefore no stronger than its weakest line.
"""

        v.addWidget(self.canvas, 1)

    def draw(self, marker=None):
        self.ax.clear(); self.cax.clear()
        self._style()
        C = self.C
        if self.data is None:
            self.cax.set_visible(False)
            self.ax.set_xticks([]); self.ax.set_yticks([])
            self.ax.text(0.5, 0.5, "Choose a map from the list", ha="center",
                         va="center", color=C["text_mute"], fontsize=10,
                         transform=self.ax.transAxes)
            self.canvas.draw_idle()
            return
        self.cax.set_visible(True)
        d = self.data
        if self.name in CLASS_STYLES:
            names, cols = CLASS_STYLES[self.name]
            cmap = mcolors.ListedColormap(cols)
            cmap.set_bad(C["void"])
            norm = mcolors.BoundaryNorm(np.arange(0.5, len(cols) + 1), cmap.N)
            im = self.ax.imshow(d, origin="lower", cmap=cmap, norm=norm,
                                interpolation="nearest")
            cb = self.fig.colorbar(im, cax=self.cax, ticks=range(1, len(cols) + 1))
            cb.ax.set_yticklabels([SHORT.get(n, n) for n in names])
        else:
            finite = d[np.isfinite(d)]
            lo, hi = ((np.nanpercentile(finite, 1), np.nanpercentile(finite, 99))
                      if self.cb_scale.currentIndex() == 0 and finite.size else
                      (np.nanpercentile(finite, 5), np.nanpercentile(finite, 95))
                      if self.cb_scale.currentIndex() == 1 and finite.size else
                      (np.nanmin(finite), np.nanmax(finite)) if finite.size
                      else (0, 1))
            import matplotlib
            cm = matplotlib.colormaps[self.cb_cmap.currentText()].copy()
            cm.set_bad(C["void"])
            im = self.ax.imshow(d, origin="lower", cmap=cm, vmin=lo, vmax=hi,
                                interpolation="nearest")
            self.fig.colorbar(im, cax=self.cax)
            if self.ck_contour.isChecked() and finite.size:
                self.ax.contour(np.where(np.isfinite(d), d, np.nan),
                                levels=np.linspace(lo, hi, 8)[1:-1],
                                colors=C["text"], linewidths=0.6, alpha=0.7)
        self.cax.tick_params(colors=C["text_mute"], labelsize=8)
        if marker is not None:
            self.marker = marker
        if self.marker is not None:
            y, x = self.marker
            self.ax.plot([x], [y], marker="s", ms=9, mfc="none",
                         mec=C["accent"], mew=1.6)
        n_y, n_x = d.shape
        full = (-0.5, n_x - 0.5, -0.5, n_y - 0.5)
        self._zoom.set_full(full)
        self.ax.set_xlim(full[0], full[1])
        self.ax.set_ylim(full[2], full[3])
        self._zoom.apply()
        self.canvas.draw_idle()



    def _click(self, ev):
        if ev.inaxes is not self.ax or ev.xdata is None or self.data is None:
            return
        x, y = int(round(ev.xdata)), int(round(ev.ydata))
        if 0 <= y < self.data.shape[0] and 0 <= x < self.data.shape[1]:
            self.picked.emit(y, x)

    def export(self):
        if self.data is None:
            return

    def open(self, path):
        try:
            mc = Megacube(path)
        except Exception as exc:
            QMessageBox.warning(self, "BRAIN", f"Cannot open this file:\n{exc}")
            return False
        if self.mc is not None:
            self.mc.close()
        self.mc = mc
        n_fit = int(np.sum(mc.status > 0))
        self.lbl_file.setText(
            f"{os.path.basename(path)}\n{mc.shape[0]} × {mc.shape[1]} spaxels, "
            f"{n_fit} fitted, {len(mc.lines)} emission lines")
        self._fill()
        first = next((n for n in ("Mage_L", "Av") if n in mc.maps), None)
        if first:
            self.panes[0].show(first, mc.plane(first))
        second = next((n for n in ("BPT [N II]", "Halpha_6563  flux", "Sigma_star")
                       if n in mc.maps), None)
        if second:
            self.panes[1].show(second, mc.plane(second))
        self.spaxel = None
        self._draw_right()
        self.status.emit(f"{os.path.basename(path)} open")
        return True



    def _fill(self):
        self.lst.clear()
        if self.mc is None:
            return
        q = self.ed_find.text().lower()
        order = ["Populations", "Kinematics", "Emission lines", "Diagnostics",
                 "Fit quality"]
        for group in order + [g for g in self.mc.groups if g not in order]:
            names = [n for n in self.mc.groups.get(group, []) if q in n.lower()]
            if not names:
                continue
            head = QListWidgetItem(group.upper())
            head.setFlags(Qt.NoItemFlags)
            self.lst.addItem(head)
            for n in names:
                item = QListWidgetItem("   " + n)
                item.setData(Qt.UserRole, n)
                self.lst.addItem(item)

    def _chosen(self, item):
        name = item.data(Qt.UserRole)
        if name and self.mc is not None:
            self.panes[self.active].show(name, self.mc.plane(name))
            if self.spaxel:
                self.panes[self.active].draw(marker=self.spaxel)

    def pick(self, y, x):
        if self.mc is None:
            return
        self.spaxel = (y, x)
        for p in self.panes:
            p.draw(marker=(y, x))
        self._draw_right()



    def _style(self, ax):
        C = self.C
        ax.set_facecolor(C["void"])
        ax.tick_params(colors=C["text_mute"], labelsize=8, length=3)
        for s in ax.spines.values():
            s.set_color(C["border_2"])
        ax.xaxis.label.set_color(C["text_dim"])
        ax.yaxis.label.set_color(C["text_dim"])
        ax.xaxis.label.set_size(9)
        ax.yaxis.label.set_size(9)

    def _draw_right(self):
        C = self.C
        self.fig.set_facecolor(C["surface"])
        ax = self.ax_spec
        ax.clear()
        self._style(ax)
        if self.mc is None or self.spaxel is None:
            ax.set_xticks([]); ax.set_yticks([])
            ax.text(0.5, 0.5, "Open a megacube, then click a spaxel" if self.mc
                    is None else "Click a spaxel on either map", ha="center",
                    va="center", color=C["text_mute"], transform=ax.transAxes)
            self.lbl_numbers.setText("")
            self._draw_lower()
            return
        y, x = self.spaxel
        st = int(self.mc.status[y, x])
        self.lbl_spaxel.setText(f"Spaxel  row {y}, column {x}"
                                + ("" if st else ", not fitted"))
        obs, syn, lines = self.mc.spectrum(y, x)
        w = self.mc.wave
        ax.plot(w, obs, color=C["p_obs"], lw=0.7, label="observed")
        ax.plot(w, syn, color=C["p_star"], lw=0.9, label="stars")
        total = syn + sum((l for _, l in lines), np.zeros_like(w))
        if lines:
            ax.plot(w, total, color=C["p_fit"], lw=0.9, label="stars + lines")
        ax.set_xlabel("rest wavelength  [Å]")
        ax.set_ylabel(self.mc.units or "flux")
        leg = ax.legend(frameon=False, fontsize=8, loc="upper right")
        for t in leg.get_texts():
            t.set_color(C["text_dim"])
        nums = self.mc.numbers(y, x)
        keys = [("Mage_L", "⟨log age⟩_L", "{:.2f}"), ("MZ_L", "⟨Z⟩_L", "{:.4f}"),
                ("Av", "A_V", "{:.3f}"), ("Sigma_star", "σ★", "{:.1f}"),
                ("vrot_star", "v★", "{:.1f}"), ("ChiSqrt", "χ²/N", "{:.3f}"),
                ("Adev", "Adev %", "{:.2f}"), ("AGN_light", "x_AGN %", "{:.1f}")]
        parts = [f"{lab} {fmt.format(nums[k])}" for k, lab, fmt in keys if k in nums]
        self.lbl_numbers.setText("    ".join(parts))
        self._plot_zoom.remember(ax)
        self._draw_lower()

    def _draw_lower(self):
        C = self.C
        ax = self.ax_low
        ax.clear()
        self._style(ax)
        choice = self.cb_diagram.currentIndex()
        if self.mc is None:
            ax.set_xticks([]); ax.set_yticks([])
            self.canvas.draw_idle()
            return
        if choice == 0:
            if self.spaxel is None:
                ax.set_xticks([]); ax.set_yticks([])
                self.canvas.draw_idle()
                return
            light, mass = self.mc.populations(*self.spaxel)
            ages = self.mc.ages
            keep = ages > 0
            uniq = np.unique(ages[keep])
            L = np.array([light[keep][ages[keep] == a].sum() for a in uniq])
            ax.bar(np.log10(uniq) - 0.03, L, width=0.06, color=C["p_fit"],
                   label="light")
            if mass is not None:
                M = np.array([mass[keep][ages[keep] == a].sum() for a in uniq])
                ax.bar(np.log10(uniq) + 0.03, M, width=0.06, color=C["p_star"],
                       label="mass")
            ax.set_xlabel("log age  [yr]")
            ax.set_ylabel("per cent")
            leg = ax.legend(frameon=False, fontsize=8)
            for t in leg.get_texts():
                t.set_color(C["text_dim"])
        else:
            n2, o3, s2, ew = self.mc.diagnostic_inputs()
            name = ["", "BPT [N II]", "BPT [S II]", "WHAN"][choice]
            xs, ys = {1: (n2, o3), 2: (s2, o3), 3: (n2, ew)}[choice]
            if xs is None or ys is None:
                ax.text(0.5, 0.5, "This megacube lacks the lines this diagram "
                        "needs", ha="center", va="center", color=C["text_mute"],
                        transform=ax.transAxes)
            else:
                cls = self.mc._diagnostic(name)
                names, cols = CLASS_STYLES[name]
                for k, (n, c) in enumerate(zip(names, cols), start=1):
                    m = cls == k
                    ax.scatter(xs[m], ys[m], s=6, color=c, alpha=0.7, label=n,
                               linewidths=0)
                if choice == 1:
                    t = np.linspace(-1.5, 0.0, 200); ax.plot(t, kauffmann03(t), "--", color=C["text_dim"], lw=1)
                    t = np.linspace(-1.5, 0.4, 200); ax.plot(t, kewley01(t), "-", color=C["text_dim"], lw=1)
                    t = np.linspace(-0.2, 0.75, 200); ax.plot(t, cid10(t), ":", color=C["text_dim"], lw=1)
                    ax.set_xlim(-1.5, 0.6); ax.set_ylim(-1.2, 1.5)
                    ax.set_xlabel("log [N II]/Hα"); ax.set_ylabel("log [O III]/Hβ")
                elif choice == 2:
                    t = np.linspace(-1.5, 0.3, 200); ax.plot(t, 0.72 / (t - 0.32) + 1.30, "-", color=C["text_dim"], lw=1)
                    t = np.linspace(-0.3, 1.0, 200); ax.plot(t, 1.89 * t + 0.76, "-", color=C["text_dim"], lw=1)
                    ax.set_xlim(-1.5, 0.8); ax.set_ylim(-1.2, 1.5)
                    ax.set_xlabel("log [S II]/Hα"); ax.set_ylabel("log [O III]/Hβ")
                else:
                    ax.set_yscale("log")
                    ax.axhline(3, ls="--", color=C["text_dim"], lw=1)
                    ax.axhline(6, xmin=0.44, ls="--", color=C["text_dim"], lw=1)
                    ax.axvline(-0.4, ls="--", color=C["text_dim"], lw=1)
                    ax.set_xlim(-1.5, 1.0); ax.set_ylim(0.1, 300)
                    ax.set_xlabel("log [N II]/Hα"); ax.set_ylabel("EW(Hα)  [Å]")
                if self.spaxel is not None:
                    y, x = self.spaxel
                    if np.isfinite(xs[y, x]) and np.isfinite(ys[y, x]):
                        ax.plot([xs[y, x]], [ys[y, x]], marker="o", ms=9,
                                mfc="none", mec=C["text"], mew=1.5)
                counts = [int(np.sum(cls == k)) for k in range(1, len(names) + 1)]
                total = max(sum(counts), 1)
                leg = ax.legend([f"{n}  {c} ({100 * c / total:.0f}%)"
                                 for n, c in zip(names, counts)],
                                frameon=False, fontsize=8, loc="lower left")
                for t in leg.get_texts():
                    t.set_color(C["text_dim"])
        self._plot_zoom.remember(ax)
        self.canvas.draw_idle()
