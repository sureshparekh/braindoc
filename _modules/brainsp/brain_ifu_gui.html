"""
BRAIN — the datacube page of the desktop interface.

One scrolling page of panels with a sidebar that jumps to each of them:
targets and output, the input extensions, the optional pre-pass stages, and
the fit. An optional stage is a checkable group box, and ticking it puts that
stage into the chain; the chain itself is built by brain_ifu, not here, so
the interface and the command line cannot drift apart.

There is no control on this page that does not drive something that runs. A
capability BRAIN's IFU mode does not have has no panel, because a panel that
looks like a setting and changes nothing is worse than an absent feature: it
produces a run the user believes was configured one way and was not.

Progress is a real fraction. The number of batches is known before the first
one is launched, so the bar reports stages finished and batches finished
rather than reporting that something, somewhere, is happening.
"""

"""
The per-batch convergence figure, and deliberately not the run's total.

The fitter logs `conv n/m` twice over: once per batch, where it counts that
batch, and once more in the closing `Done:` line, where it counts the whole
run. The viewer accumulates what it is told, so a pattern that matched both
would add the total to the sum of its own parts and report twice the work
that was done, at the moment the run ends and the number is finally looked
at. Anchoring on the `Batch` the line opens with is what keeps the account
true; `Done:` is already carried by the run finishing.
"""

class RunThread(ChildRun):
    """
    The pipeline, in a process of its own, watched from a thread of its own.

    It used to run in this thread, with JAX_PLATFORMS set just before it,
    which selected nothing: JAX fixes its backend the first time it is used in
    a process and the interface had used it long before any cube was fitted.
    The selector now reaches a fresh interpreter, whose environment names the
    platform before it starts and which refuses to fit on anything else.

    Two things come with the process that the thread could not offer. Stop
    ends the run, because ending a process is something an operating system
    can do and interrupting a compiled kernel is not. And a fit that dies
    inside JAX now dies on its own, leaving the interface holding a message
    instead of going down with it.
    """

    logged = Signal(str)
    advanced = Signal(float, str)
    fitting = Signal(int, int)
    fitted = Signal(object)
    finished_ok = Signal(list)
    failed = Signal(str)

    def __init__(self, cfg, device):
        super().__init__({"job": "cube", "config": cfg}, device)
        self.cfg = cfg

    def handle(self, message):
        """
        One message from the run, on its way to the panel that shows it.

        A queued signal is a copy into the receiving thread's event queue and
        nothing more, so the process is never waiting on a redraw.
        """
        kind = message.get("kind")
        if kind in ("log", "backend"):
            self.logged.emit(message.get("text", ""))
        elif kind == "progress":
            self.advanced.emit(float(message.get("fraction", 0.0)),
                               message.get("message", ""))
        elif kind == "fitting":
            self.fitting.emit(int(message["n_y"]), int(message["n_x"]))
        elif kind == "fitted":
            self.fitted.emit(message.get("indices", []))
        elif kind == "finished":
            self.finished_ok.emit(list(message.get("summaries", [])))
        elif kind == "failed":
            self.failed.emit(message.get("text", ""))

    def _add_column(self):
        """
        A column the chain as configured will actually read.

        The list offered is derived from the stages that are switched on, so
        a column for a stage that is off cannot be added by accident. It is
        still editable, because a pipeline file may be edited afterwards to
        turn that stage on; the runner refuses an unread column either way.
        """
        name, ok = QInputDialog.getItem(
            self, "Add a column",
            "Override key, one of the settings the enabled stages read:",
            self.accepted(), 0, True)
        name = (name or "").strip()
        if not ok or not name:
            return
        if name in self.headers():
            QMessageBox.warning(self, APP_NAME,
                                f"There is already a {name!r} column.")
            return
        c = self.table.columnCount()
        self.table.insertColumn(c)
        self.table.setHorizontalHeaderLabels(self.headers()[:c] + [name])
        for r in range(self.table.rowCount()):
            self.table.setItem(r, c, QTableWidgetItem(""))

    def _remove_column(self):
        for c in sorted({i.column() for i in self.table.selectedIndexes()},
                        reverse=True):
            if c == 0:
                QMessageBox.warning(
                    self, APP_NAME,
                    "The first column names the cube and cannot be removed.")
                continue
            self.table.removeColumn(c)

    def _add_files(self):
        """
        Pick cubes from disk instead of typing their names.

        A row holds the name relative to the targets folder, so the folder is
        filled in from the first selection when it is still empty, and a cube
        from somewhere else is stored by its full path -- which the runner
        joins correctly, and which is better than a relative name that points
        at nothing.
        """
        paths, _ = QFileDialog.getOpenFileNames(
            self, "Add datacubes", self.folder() or "",
            "FITS (*.fits *.fits.gz *.fit);;All files (*)")
        if not paths:
            return
        if not self.folder().strip():
            self.set_folder(os.path.dirname(paths[0]))
        base = os.path.abspath(self.folder() or ".")
        first = None
        for path in paths:
            inside = os.path.dirname(os.path.abspath(path)) == base
            row = self.add_row([os.path.basename(path) if inside
                                else os.path.abspath(path)])
            first = row if first is None else first
        if first is not None:
            self.select_row(first)

    def _load_csv(self):
        path, _ = QFileDialog.getOpenFileName(
            self, "Load a targets CSV", "", "CSV (*.csv);;All files (*)")
        if not path:
            return
        try:
            columns, rows = brain_ifu.read_targets_csv(path)
        except Exception as exc:
            QMessageBox.critical(self, APP_NAME, str(exc))
            return
        self.set_contents(columns, rows)

    def _save_csv(self):
        path, _ = QFileDialog.getSaveFileName(
            self, "Save the targets CSV", "targets.csv",
            "CSV (*.csv);;All files (*)")
        if not path:
            return
        brain_ifu.write_targets_csv(path, self.headers(), self.rows())

    def bins(self):
        """
        The table as the runner wants it, with unreadable cells left as text.

        A cell that is not a number is passed through rather than dropped or
        silently replaced, so that validation refuses the run and names the
        bin, instead of the interface quietly fitting a different set of bins
        from the one on the screen.
        """
        out = []
        for r in range(self.table.rowCount()):
            cells = []
            for c in range(3):
                item = self.table.item(r, c)
                cells.append(item.text().strip() if item else "")
            if not any(cells):
                continue
            row_values = [cells[0]]
            for text in cells[1:]:
                try:
                    row_values.append(float(text))
                except ValueError:
                    row_values.append(text)
            out.append(row_values)
        return out

    def _update_steps(self):
        """
        Each step's one line of state, read from the widgets themselves.

        Refreshed on a short timer rather than wired to every control: it is
        a few dozen reads, and a summary that missed one signal would say
        something the run would not do.
        """
        from brainsp.brain_shell import READY, TODO, ATTENTION, OFF
        if not self.isVisible():
            return
        ws = self.ws
        rows = self.targets.rows()
        names = [r[0] for r in rows if r and r[0]]
        if names:
            ws.set_state(0, READY, names[0] if len(names) == 1
                         else f"{len(names)} cubes in the table")
        else:
            ws.set_state(0, TODO, "no cube chosen")
        headers = [h.lower() for h in self.targets.headers()]
        z = self.ed_fit_z.text().strip() or "0"
        zpart = "z per cube" if "redshift" in headers else f"z {z}"
        if self.gb_dereddening.isChecked():
            epart = ("E(B−V) per cube" if "ebv" in headers
                     else f"E(B−V) {self.sp_ebv.value():g}")
        else:
            epart = "no dereddening"
        if "redshift" in headers or self.ed_fit_z.text().strip():
            ws.set_state(1, READY, f"{zpart}, {epart}")
        else:
            ws.set_state(1, ATTENTION, "enter the redshift")
        stages = [n for n, g in (("binning", self.gb_spatial),
                                 ("resolution", self.gb_resolution),
                                 ("resampling", self.gb_spectral),
                                 ("filter", self.gb_filter)) if g.isChecked()]
        ws.set_state(2, READY if stages else OFF,
                     ", ".join(stages) if stages else "nothing to do")
        n_mask = len(getattr(self, "cube_masks", []))
        wpart = f", {n_mask} λ region{'' if n_mask == 1 else 's'}" \
            if n_mask else ""
        if self.gb_sn.isChecked():
            ws.set_state(3, READY, "S/N ≥ " + ", ".join(
                f"{t:g}" for t in self._thresholds()) + wpart)
        else:
            ws.set_state(3, READY if n_mask else OFF, "every spaxel" + wpart)
        if self._model_settings is not None:
            m = self._model_settings()
            n = len(m.get("ages") or []) * len(m.get("metallicities") or [])
            needs_scale = m.get("nebular_mode") != "spop"
            scale_box = dict(self._panels).get("Wavelength scale")
            if scale_box is not None:
                scale_box.set_tag("required" if needs_scale else "unused",
                                  None if needs_scale else "Not needed in spop")
            if needs_scale and self.seg_wave_scale.value() is None:
                ws.set_state(0, ATTENTION, "say whether the cube is air or vacuum")
            ws.set_state(4, READY, f"{m.get('nebular_mode')}, {n} templates"
                         + (", AGN" if m.get("include_power_law") else ""))
        ws.set_state(5, *({"running": (READY, "running…"),
                           "done": (READY, "finished")}.get(
                               self._run_state, (TODO, "ready to run"))))

        """

    def _open_cube(self):
        """
        Pick one datacube and show it, without going through the table first.

        The panel's prominent control used to be the targets folder, which
        opens a directory chooser, so a user looking for a cube met a dialog
        that would not let them select one and had to find "Add cubes" inside
        the table's own toolbar. Picking a file is the thing people come here
        to do, so it is the first button and it is a file dialog.

        is what the runner reads: the cube becomes a row, the row is selected,
        and selecting a row is already what opens the viewer.

            self, "Open a datacube",
            self.ed_targets_dir.text().strip() or os.path.expanduser("~"),
            "FITS datacubes (*.fits *.fits.gz *.fit *.fts);;All files (*)")
        if not paths:
            return

            self.ed_targets_dir.setText(os.path.dirname(paths[0]))

        first = None

            inside = os.path.dirname(os.path.abspath(path)) == base
            row = self.targets.add_row(
                [os.path.basename(path) if inside else os.path.abspath(path)])
            first = row if first is None else first
        if first is not None:
            self.targets.select_row(first)

    def _adopt_extensions(self):
        """
        Point the Input fields at extensions the open cube actually has.

        The viewer opens whatever 3-D extension a cube holds, so a JWST cube
        (SCI, ERR) displayed while the fields still said FLUX and IVAR, and
        the run then stopped on the missing name. A field naming something
        this file lacks is set to what the file has, and the panel says so.
        A raw cube that is not a megacube is the only kind read here.
        """
        path = getattr(self.viewer, "path", "")
        if not path or not os.path.isfile(path):
            return
        try:
            with fits.open(path, memmap=True) as hdul:
                names = [h.name for h in hdul]
                cubes = [h.name for h in hdul
                         if h.header.get("NAXIS") == 3]
        except Exception:
            return
        if "FLXOBS" in names:
            return
        changed = []
        flux = self.ed_hdu_flux.text().strip()
        if flux not in names and cubes:
            pick = next((n for n in ("FLUX", "SCI", "DATA") if n in cubes),
                        cubes[0])
            self.ed_hdu_flux.setText(pick)
            changed.append(f"flux {flux or '(blank)'} → {pick}")
        stat = self.ed_hdu_stat.text().strip()
        if stat not in names:
            found = next(((n, t) for n, t in self.STAT_NAMES if n in names),
                         None)
            if found:
                self.ed_hdu_stat.setText(found[0])
                self.cb_stat_type.setCurrentText(found[1])
                changed.append(f"uncertainty {stat or '(blank)'} → "
                               f"{found[0]} ({found[1]})")
        mask = self.ed_hdu_mask.text().strip()
        if mask and mask not in names:
            self.ed_hdu_mask.setText("")
            changed.append(f"quality mask {mask} → none")
        if changed:
            text = (f"Set from {os.path.basename(path)}, which holds "
                    f"{', '.join(names)}: " + "; ".join(changed) + ". Check "
                    f"them, above all what the uncertainty is.")
            self.lbl_ext_note.setText(text)
            self.lbl_ext_note.setVisible(True)
            self.status.emit(text)

    def _filter_field_warning(self, n_y=None, n_x=None):
        """
        Say when this cube is too small a field for the filter to survive.

        The kernel is a few spaxels wide however large the field is, so on a
        field of a few tens of spaxels it covers most of it and the flux that
        leaves through the blank spaxels does not come back. The warning is
        driven by the shape of the cube the viewer has actually opened, so it
        appears for the cube in front of you and not as a general caution.
        """
        if n_y is not None:
            self._field_shape = (int(n_y), int(n_x))
        shape = getattr(self, "_field_shape", None)
        show = bool(shape) and self.gb_filter.isChecked() and min(shape) < 24
        if show:
            self.lbl_bw_field.setText(
                f"This cube is {shape[1]} \u00d7 {shape[0]} spaxels. The "
                f"filter's kernel is a few spaxels wide whatever the field "
                f"size, so on a field this small it covers most of it: expect "
                f"the flux to come out well below what went in, and the S/N "
                f"mask after it to select far fewer spaxels, or none.")
        self.lbl_bw_field.setVisible(show)

    def _match_sn_method(self, stat_type=None):
        """
        Measure the S/N with the formula for the uncertainty the cube has.

        The method reads the uncertainty extension as an inverse variance, a
        variance or an error; read as the wrong one, the S/N of every spaxel
        is nonsense (for 1-sigma errors under the ivar formula, about zero).
        So the method follows "Uncertainty is a"; mean_std, which reads no
        uncertainty, is left alone.
        """
        from brainsp.brain_ifu import SN_METHOD_FOR_STAT
        allowed = SN_METHOD_FOR_STAT.get(stat_type or
                                         self.cb_stat_type.currentText())
        current = self.cb_sn_method.currentData()
        if allowed and current != "mean_std" and current not in allowed:
            self.cb_sn_method.setCurrentIndex(
                self.cb_sn_method.findData(allowed[0]))

    def _sn_window_edited(self, _=None):
        if not self._sn_window_setting:
            self._sn_window_auto = False

    def _cube_rest_range(self):
        """The open cube's wavelength range in the rest frame, or None."""
        observed = getattr(self.viewer, "wavelength_range", None)
        if not observed:
            return None
        overrides = getattr(self, "_shown_overrides", None) or {}
        redshift, _ = self._frame_for(overrides)
        z = float(redshift or 0.0)
        return observed[0] / (1.0 + z), observed[1] / (1.0 + z)

    def _fit_sn_window(self):
        """
        Put the S/N window inside the open cube's rest-frame range.

        4000-6000 A, the optical continuum window, is kept whenever the cube
        covers it, so an optical cube is measured exactly as before. A cube
        that does not cover it -- a near-infrared one -- gets the middle 40
        per cent of its own rest-frame range, away from the edges, where the
        throughput falls and the noise rises. A window the user has typed is
        never replaced; if it misses the cube, the panel says so.
        """
        rest = self._cube_rest_range()
        if rest is None:
            return
        low, high = rest
        path = getattr(self.viewer, "path", "")
        self.cube_range.emit(float(low), float(high),
                             path != getattr(self, "_range_path", None))
        self._range_path = path
        if self._sn_window_auto:
            if low <= SN_WINDOW_OPTICAL[0] and SN_WINDOW_OPTICAL[1] <= high:
                window = SN_WINDOW_OPTICAL
            else:
                span = high - low
                window = (round(low + 0.3 * span), round(low + 0.7 * span))
            self._sn_window_setting = True
            self.sp_sn_lo.setValue(float(window[0]))
            self.sp_sn_hi.setValue(float(window[1]))
            self._sn_window_setting = False
        self._preview_sn_soon()

    def _sn_window_problem(self):
        """Why the typed window cannot be measured on the open cube, or None."""
        rest = self._cube_rest_range()
        if rest is None:
            return None
        lo, hi = float(self.sp_sn_lo.value()), float(self.sp_sn_hi.value())
        if hi <= lo:
            return "The S/N window's end must be above its start."
        if hi < rest[0] or lo > rest[1]:
            return (f"The S/N window {lo:.0f}-{hi:.0f} Å is outside this cube, "
                    f"which covers {rest[0]:.0f}-{rest[1]:.0f} Å in the rest "
                    f"frame. Choose a stretch of continuum inside it.")
        return None

    def _chip_clicked(self, threshold):
        """A chip is both a choice to write that mask and a request to see it."""
        if self.sn_chips[threshold].isChecked():
            self._sn_shown = threshold
        self._refresh_thresholds()
        self._preview_sn_soon()

    def _preview_sn_soon(self):
        self._sn_preview_timer.start(150)

    def _preview_sn(self):
        """
        Show on the map which spaxels the chosen threshold keeps, before any
        run, by running the pipeline's own S/N stage on the open cube.

        The stage is plain numpy, so the interface stays free of JAX, and it
        is the same code the run uses, so the preview cannot disagree with
        the SN_MASKS the run writes.
        """
        if not self.gb_sn.isChecked():
            self.lbl_sn_preview.setText("Off: every spaxel with data is fitted.")
            return
        problem = self._sn_window_problem()
        if problem:
            self.lbl_sn_preview.setText(problem)
            return
        thresholds = self._thresholds()
        if not thresholds:
            self.lbl_sn_preview.setText("Choose at least one threshold.")
            return
        if self._sn_shown not in thresholds:
            self._sn_shown = thresholds[-1]
        settings = {
            "hdu flux": self.ed_hdu_flux.text().strip() or "FLUX",
            # The uncertainty under the name this method reads, as the run
            # passes it (brain_ifu.build_chain).
            SN_METHOD_STAT_KEY.get(self.cb_sn_method.currentData())
            or "hdu ivar": self.ed_hdu_stat.text().strip() or "IVAR",
            "sn window": [float(self.sp_sn_lo.value()),
                          float(self.sp_sn_hi.value())],
            "thresholds": thresholds,

    def _show_target(self, index):
        """
        Show the cube the selected row names, and its megacube if it has one.

        Everything the viewer needs to describe what it is showing comes from
        the pipeline as configured -- which extension the fit will read, which
        redshift it will remove, which emission-line mask it will apply -- and
        a per-target column in the table overrides those for that row alone,
        exactly as it does for the run. So the spectrum the viewer draws is
        the spectrum the fit would see, in the frame the fit would see it in.
        """
        values = self.targets.row_at(index)
        if not values or not values[0]:
            return
        name, overrides = brain_ifu.target_overrides(
            self.targets.headers(), values)
        if not name:
            return
        folder = self.ed_targets_dir.text().strip()
        path = os.path.join(folder, name)
        megacube = brain_ifu.output_path(
            self.ed_out_dir.text().strip() or "brain_ifu_out", path)
        redshift, mask = self._frame_for(overrides)
        self.viewer.set_target(
            path, megacube if os.path.isfile(megacube) else None,
            flux_ext=self.ed_hdu_flux.text().strip() or "FLUX",
            redshift=redshift, mask_file=mask,
            redshift_known=self._redshift_known(overrides) or bool(redshift))
        self._shown_overrides = overrides
        if not overrides.get("mask file"):
            # The cube's own mask is the list on the Masks step, editable
            # from the spectrum; a per-cube file in the table is shown as is.
            self.viewer.set_mask_regions(self.cube_masks, editable=True)

    def _redshift_known(self, overrides):
        """True when this cube's redshift was given, 0 included."""
        return overrides.get("redshift") is not None \
            or _optional_float(self.ed_fit_z.text()) is not None

    def _redshift_typed(self):
        """The redshift field changed: move the viewer's spectrum with it."""
        overrides = getattr(self, "_shown_overrides", None)
        if overrides is None:
            return
        redshift, _ = self._frame_for(overrides)
        self.viewer.set_redshift(
            redshift, self._redshift_known(overrides) or bool(redshift))
        self._fit_sn_window()

    def _frame_for(self, overrides):
        """
        The redshift and the emission-line mask that apply to one target.

        Same precedence the runner uses: a per-target column beats the fit
        panel, and the fit panel beats the settings file. Reading the settings
        file here is what lets the viewer show a rest-frame axis for a cube
        whose redshift was never typed into this page because it lives in the
        YAML.
        """
        settings_z, settings_mask = 0.0, None
        try:
            from brainsp import brain_settings
            loaded = brain_settings.load(
                self.ed_settings.text().strip() or "brain_setup.yaml")
            settings_z = float(loaded.get("cube_redshift") or 0.0)
            settings_mask = loaded.get("mask")
        except Exception:
            pass

        redshift = overrides.get("redshift")
        if redshift is None:
            redshift = _optional_float(self.ed_fit_z.text())
        if redshift is None:
            redshift = settings_z

        mask = overrides.get("mask file") or settings_mask
        return float(redshift or 0.0), mask

    def _scale_changed(self):
        """
        Say what the wavelength scale is for, and what leaving it blank means.

        The control lives beside the redshift because it is the same kind of
        thing: a fact about the instrument that produced this cube, not about
        the physics being fitted, and the one setting of its kind that
        differs between two surveys in a single sweep. It is an override, so
        unanswered means the settings file's own value stands rather than
        meaning unset -- the settings file is where it is normally declared,
        and that file's own validator refuses to run a line-fitting mode
        without it.

        The caption says what a wrong answer costs, because nothing in the
        output reveals it: air and vacuum differ by about 83 km/s, very
        nearly constant in velocity across the optical, so the difference is
        absorbed into the fitted gas velocity and chi2 barely moves. In
        'spop' no emission line is ever built and the setting cannot reach a
        number, which is why it is not demanded here.
        """
        chosen = self.seg_wave_scale.value()
        if chosen is None:
            self.lbl_wave_scale.setText(
                "Blank: whatever the settings file declares. Set it here "
                "only to override that file, or to answer it for a file "
                "that leaves it out. It is required by espop, nepop and "
                "photopop, which place emission lines in your spectrum and "
                "have to know which scale it is on, and it is ignored by "
                "spop, which builds no lines.")
            return
        other = "vacuum" if chosen == "air" else "air"
        self.lbl_wave_scale.setText(
            f"Every cube in this run is read as {chosen}, whatever the "
            f"settings file says. If it is really {other}, the fit still "
            f"runs to completion and every line flux still looks ordinary: "
            f"the only trace is a gas-to-stars velocity offset of about "
            f"83 km/s that is not real. SDSS and MaNGA are vacuum; most "
            f"ground-based optical spectrographs are air.")

    def _set_cube_masks(self, regions):

        n = len(self.cube_masks)
        self.lbl_cube_mask.setText(
            "No regions: every wavelength is fitted." if not n else
            f"{n} region{'' if n == 1 else 's'}, applied to every spaxel.")

    def _cube_mask_erased(self, at):
        """Remove one region under the pointer: the narrowest, where several
        overlap, since that is the one drawn on top of the others."""
        under = [i for i, r in enumerate(self.cube_masks) if r[0] <= at <= r[1]]
        if under:
            i = min(under, key=lambda k: self.cube_masks[k][1]
                    - self.cube_masks[k][0])
            self._set_cube_masks(self.cube_masks[:i] + self.cube_masks[i + 1:])

    def _remove_cube_mask(self):
        i = self.lst_cube_mask.currentRow()
        if 0 <= i < len(self.cube_masks):
            self._set_cube_masks(self.cube_masks[:i] + self.cube_masks[i + 1:])

    def _standard_cube_mask(self):
        have = {(round(r[0], 1), round(r[1], 1)) for r in self.cube_masks}
        from brainsp.io_utils import STANDARD_OPTICAL_MASK

                 if (round(lo, 1), round(hi, 1)) not in have]
        self._set_cube_masks(self.cube_masks + extra)

    def _load_cube_mask_from(self, path, replace=False):
        from brainsp.io_utils import read_mask_regions
        try:
            regions, skipped = read_mask_regions(path)
        except (OSError, UnicodeDecodeError) as exc:
            QMessageBox.warning(self, APP_NAME, f"Cannot read {path}:\n{exc}")
            return
        have = set() if replace else {(round(r[0], 2), round(r[1], 2))
                                      for r in self.cube_masks}
        new = [r for r in regions if (round(r[0], 2), round(r[1], 2))
               not in have]
        self._set_cube_masks(([] if replace else self.cube_masks) + new)
        self.status.emit(f"{len(regions)} regions read from "
                         f"{os.path.basename(path)}"
                         + (f", {skipped} lines skipped" if skipped else ""))

    def _refresh_thresholds(self, *_):
        """
        The fit can only select on a threshold the mask stage actually writes.

        Rebuilt from the thresholds field rather than typed in separately, so
        the mismatch the runner refuses cannot usually be reached from here
        at all -- and the runner still refuses it, because the pipeline file
        can be edited by hand.
        """
        current = self.cb_flag.currentData()
        self.cb_flag.blockSignals(True)
        self.cb_flag.clear()
        self.cb_flag.addItem("every spaxel with data", None)
        if self.gb_sn.isChecked():
            for value in self._thresholds():
                self.cb_flag.addItem(f"S/N > {value:g}", value)
        index = self.cb_flag.findData(current)
        self.cb_flag.setCurrentIndex(max(index, 0))
        self.cb_flag.blockSignals(False)

    def _thresholds(self):
        """The chosen chips, and any others typed under Also, in order."""
        chosen = {float(t) for t, b in getattr(self, "sn_chips", {}).items()
                  if b.isChecked()}
        extra = self.ed_thresholds.text().replace(";", ",")
        for part in extra.split(","):
            part = part.strip()
            if part:
                try:
                    chosen.add(float(part))
                except ValueError:
                    pass
        return sorted(chosen)

    def _required_errors(self, cfg):
        """
        What the interface asks for that the pipeline would default.

        A blank redshift runs at the settings file's cube_redshift, which
        from this window is 0: an observed-frame cube is then fitted as if it
        were at rest, and nothing fails. So the window requires an answer,
        here or in a 'redshift' column covering every cube.
        """
        errors = []
        if cfg["fit"]["enabled"] and cfg["fit"]["redshift"] is None:
            headers = [h.lower() for h in cfg["targets"]["columns"]]
            rows = [r for r in cfg["targets"]["rows"] if r and str(r[0]).strip()]
            col = headers.index("redshift") if "redshift" in headers else None
            covered = col is not None and all(
                len(r) > col and str(r[col]).strip() for r in rows)
            if not covered:
                errors.append((
                    "redshift",
                    "Enter the galaxy's redshift (0 if the cube is already in "
                    "the rest frame), or give every cube one in a 'redshift' "
                    "column of the targets table."))
        return errors

    def _check(self):
        cfg = self.config()
        errors = self._required_errors(cfg) + brain_ifu.validate_detailed(cfg)
        self._show_errors(errors)
        self.console.clear()
        self.console.append("Pipeline:")
        self.console.append(brain_ifu.describe(cfg))
        if errors:
            self.console.append("\nThis pipeline will not run:")
            for _, error in errors:
                self.console.append(f"  * {error}")
            self.status.emit(f"{len(errors)} problem"
                             f"{'' if len(errors) == 1 else 's'} to fix")
            self._jump_to_first_error(errors)
        else:
            self.console.append("\nValid.")
            self.status.emit("ready to run")

    def _show_errors(self, errors):
        """
        Put each refusal inside the panel it is about, and mark that panel.

        A panel with something wrong in it gets a critical border and its
        title in the critical colour as well as the message, so the state is
        not carried by the colour of a line of text alone, and clearing is
        total: a panel with nothing wrong loses both.
        """
        by_section = {}
        for section, message in errors:
            by_section.setdefault(section, []).append(message)
        for section, message_label in self._errors.items():
            messages = by_section.get(section, [])
            message_label.setText("\n".join(f"•  {m}" for m in messages))
            message_label.setVisible(bool(messages))
            self._sections[section].set_error(bool(messages))

    def _jump_to_first_error(self, errors):
        for section, _ in errors:
            widget = self._sections.get(section)
            if widget is not None:
                self._jump_to(widget)
                return

    def _export_script(self):
        path, _ = QFileDialog.getSaveFileName(
            self, "Export a runnable script",
            os.path.join(os.getcwd(), "run_brain_ifu.py"),
            "Python (*.py);;All files (*)")
        if not path:
            return
        from brainsp.ifu_script_export import export
        cfg = self.config()
        required = self._required_errors(cfg)
        if required:
            self._show_errors(required)
            self._jump_to_first_error(required)
            QMessageBox.warning(self, APP_NAME, "\n".join(
                m for _, m in required))
            return
        try:
            written, outside = export(cfg, path)
        except SystemExit as exc:
            QMessageBox.warning(self, APP_NAME, str(exc))
            return
        except Exception as exc:
            QMessageBox.critical(self, APP_NAME, str(exc))
            return

        note = ("\n\nPaths inside the script's folder are written relative to "
                "it, so the folder can be copied to a server and run there.")

                     "with it; copy them along or move them into the folder:"

        QMessageBox.information(
            self, APP_NAME, "Wrote:\n  " + "\n  ".join(written)
            + f"\n\nRun it from a terminal with:\n  python {path}" + note)

    def _run(self):
        if self.thread is not None and self.thread.isRunning():
            return
        cfg = self.config()
        errors = self._required_errors(cfg) + brain_ifu.validate_detailed(cfg)
        self._show_errors(errors)
        if errors:
            self._jump_to_first_error(errors)
            QMessageBox.warning(
                self, APP_NAME,
                "This pipeline will not run. Each problem is also marked in "
                "the panel it belongs to:\n\n"
                + "\n".join(f"  •  {e}" for _, e in errors))
            return
        self.console.clear()
        self.console.append("Pipeline:")
        self.console.append(brain_ifu.describe(cfg))
        self.console.append("")
        self.bar.setValue(0)
        self.btn_run.setEnabled(False)
        self.btn_run.setText("  Running…")
        self.btn_stop.setEnabled(True)
        self.status.emit("running")
        self._run_state = "running"

        self.thread = RunThread(cfg, self._device())
        self.thread.logged.connect(self._log)
        self.thread.advanced.connect(self._advance)
        self.thread.fitting.connect(self.viewer.begin_run)
        self.thread.fitted.connect(self.viewer.mark_spaxels)
        self.thread.finished_ok.connect(self._done)
        self.thread.failed.connect(self._failed)
        self.thread.start()

    def _log(self, text):
        """
        One line of the run's own log, shown, and read for what it carries.

        The fit tells the interface which spaxels each batch touched and not
        which of them converged, so the only convergence figure available
        while a run is in flight is the one the fitter writes into its log.
        It is read with two strict patterns and nothing is inferred when they
        do not match: a run whose log changes shape leaves the progress map
        with no convergence line under it rather than a guessed one.
        """
        self.console.append(text)
        bar = self.console.verticalScrollBar()
        bar.setValue(bar.maximum())
        found = VALID_SPAXELS.match(text)
        if found:
            self.viewer.note_valid(int(found.group(1)))
            return
        found = BATCH_CONVERGED.search(text)
        if found:
            self.viewer.note_batch(int(found.group(1)), int(found.group(2)))

    def _advance(self, fraction, message):
        self.bar.setValue(int(max(0.0, min(1.0, fraction)) * 1000))
        if message:
            self.bar.setFormat(f"{message}   %p%")
            self.status.emit(message)

    def _stop(self):
        """
        End the run, and say so before the process has finished ending.

        The button goes dead on the first press so that a second press cannot
        arrive while the child is still being asked politely; the run is only
        reported over when the process has actually gone.
        """
        if self.thread is None or not self.thread.isRunning():
            return
        self.btn_stop.setEnabled(False)
        self._log("\nStopping the run.")
        self.thread.cancel()

    def _reset(self, reason="finished"):
        self.btn_run.setEnabled(True)
        self.btn_run.setText("  Run the pipeline")
        self.btn_stop.setEnabled(False)
        self.bar.setFormat("%p%")
        self.viewer.end_run(reason)
        QTimer.singleShot(1800, self._reopen_target)

    def _reopen_target(self):
        """
        Put the viewer back on the data once the progress map has settled.

        Re-selecting the target rather than merely clearing the progress map,
        because the megacube the run has just written did not exist when the
        target was first shown, and reopening is what makes its maps and its
        fitted spectra appear without anybody having to ask.
        """
        self.viewer.clear_progress()
        self._show_target(self.targets.table.currentRow())

    def _done(self, summaries):
        self._run_state = "done"
        self._reset()
        self.bar.setValue(1000)
        done = [s for s in summaries if s.get("state") == "done"]
        skipped = [s for s in summaries if s.get("state") == "skipped"]
        missing = [s for s in summaries if s.get("state") == "missing"]
        self._log(f"\nFinished: {len(done)} fitted, {len(skipped)} skipped, "
                  f"{len(missing)} missing.")
        for summary in done:
            if "n_valid" in summary:
                self._log(f"  {os.path.basename(summary['target'])}: "
                          f"{summary['n_conv']}/{summary['n_valid']} spaxels "
                          f"converged, median chi2/Nl_eff "
                          f"{summary['med_chi2']:.4f}")
        self.status.emit("done")
        outputs = [s.get("output") for s in done if s.get("output")]
        if outputs and os.path.isfile(outputs[-1]):
            self._last_output = outputs[-1]
            self.btn_results.setEnabled(True)
            self.btn_results.setText(
                f"View results: {os.path.basename(outputs[-1])}")

    def _failed(self, message):
        """
        The run ended without finishing, and why.

        A stopped run is not a failure and does not get a failure's dialog,
        but it travels the same channel because it leaves the interface in
        the same state: no summaries, and a Run button that has to come back.
        """
        self._run_state = None
        stopped = message == CANCELLED
        self._reset("stopped" if stopped else "failed")
        self._log("\n" + message)
        self.status.emit("stopped" if stopped else "failed")
        if stopped:
            QMessageBox.information(self, APP_NAME, message + STOPPED_OUTPUT)
        else:
            failure_dialog(self, message, self.ed_out_dir.text().strip()
                           or "brain_ifu_out")

def _number_text(value):
    """A number as a short string, and anything else as itself."""
    try:
        return f"{float(value):g}"
    except (TypeError, ValueError):
        return str(value)
