Source code for brainsp.fit_datacube

"""
Parallel Datacube Fitter for NewBrain
Fits an entire MEGACUBE or FITS datacube in parallel.

The templates and the log-lambda grid are built once and shared by every
spaxel, so the line-spread-function correction is applied once, to that one
matrix, before the pool is launched. It is not a per-spaxel cost. The mask the
decision is taken over is the union of the pixels the fitted spaxels will use;
cube_fitted_pixels says why.

Each spaxel's own dispersion is then reported twice: sigma_star as the kernel
applied it to the library, and sigma_star_corrected with the remaining
difference in line-spread functions taken out of it in quadrature. When
'instrumental_resolution' is not set the corrected value is None rather than a
copy of the observed one, so nothing downstream can mistake an observed
dispersion for an intrinsic one.
"""

import os
import time
import logging
import numpy as np
from astropy.io import fits
import multiprocessing as mp
from multiprocessing.pool import ThreadPool
from functools import partial

os.environ["XLA_PYTHON_CLIENT_PREALLOCATE"] = "false"
os.environ["OMP_NUM_THREADS"] = "1"
os.environ["MKL_NUM_THREADS"] = "1"
os.environ["OPENBLAS_NUM_THREADS"] = "1"

import jax
jax.config.update('jax_enable_x64', True)
jax.config.update('jax_compilation_cache_dir', '/tmp/jax_cache')

from brainsp.config import NewBrainConfig, NEBULAR_MODES

logger = logging.getLogger("DATACUBE")

MANGA_DONOTUSE = 1 << 10

def spaxel_pool(num_threads):
    """
    A pool of threads, not of processes.

    multiprocessing.Pool forks, and JAX is already multithreaded by the time
    the fork happens, so the child inherits locks held by threads that do not
    exist in it. The pool then deadlocks with no error and no traceback: this
    module used to hang indefinitely on four spaxels, with Python's own
    "os.fork() was called" warning as the only sign that anything was wrong.

    Threads have no such problem and lose nothing here, because the work is
    inside compiled XLA kernels, which release the interpreter lock for their
    whole duration.
    """
    return ThreadPool(num_threads)

def bases_info_for(cfg):
    """
    The template list, chosen the way brainv1 chooses it.

    A configuration built by brain_settings.to_config carries bases_entries,
    the (age, Z) selection read from the library's CATALOGUE.txt. The cube
    fitters used to skip that and fall through to a filename scan for
    '_baseFe', which matches nothing in the shipped library, so driving them
    from brain_setup.yaml produced zero templates and an IndexError on an
    empty matrix.
    """
    if getattr(cfg, 'bases_entries', None):
        return cfg.bases_entries
    if getattr(cfg, 'bases_dir_file', None):
        return load_bases_dir(cfg.bases_dir_file)
    return scan_ssp_folder(cfg.ssp_folder)

def cube_fitted_pixels(flux_cube, ivar_cube, wl_lin, lam_grid):
    """
    The pixels the line-spread-function decision is taken over, for a cube.

    One template matrix is shared by every spaxel, so the decision to convolve
    a group of templates or to leave it at its native resolution is taken once
    and then binds all of them. A single spaxel's good-pixel mask is therefore
    the wrong unit: a convolution justified on one spaxel's window would be
    pushed onto spaxels that fit pixels where it is not justified.

    The union of the good pixels over the spaxels that will actually be fitted
    is used instead, intersected with the template coverage by the caller. Of
    the three masks on offer it is the conservative one: prepare_templates
    refuses a group of templates when the data is narrower than the library
    anywhere in the mask, so widening the mask can only turn a convolution into
    a refusal, never a refusal into a convolution, and a refusal leaves the
    templates untouched and says so in the log. The intersection would let a
    group be convolved on the strength of a window narrower than any spaxel
    really fits; the coverage mask on its own would carry pixels that every
    spaxel has already thrown away.

    The union is exact, not an approximation. A spaxel fits the pixels where
    the interpolation of its error array onto the log grid comes out positive,
    and that interpolation is a convex combination of the two bracketing
    samples, so interpolating the per-wavelength indicator "some fitted spaxel
    has a usable inverse variance here" is positive on exactly the union.
    """
    lam = np.asarray(lam_grid, dtype=np.float64)
    if ivar_cube is None:
        return np.ones(lam.shape, dtype=bool)

    with np.errstate(invalid='ignore'):
        spaxel_used = ((np.sum(flux_cube, axis=0) > 0)
                       & ~np.all(np.isnan(flux_cube), axis=0))
    if not np.any(spaxel_used):
        return np.ones(lam.shape, dtype=bool)

    ivar_used = ivar_cube[:, spaxel_used]
    with np.errstate(invalid='ignore'):
        usable = np.any(np.isfinite(ivar_used) & (ivar_used > 0), axis=1)
    return np.interp(lam, np.asarray(wl_lin, dtype=np.float64),
                     usable.astype(np.float64)) > 0.0

def to_rest_frame(wl, cfg, logger):
    """
    Put an observed-frame cube onto rest wavelengths.

    The two entry points take different frames, and that asymmetry is the
    kind of thing this code fails silently on, so it is stated here and in
    brain_setup.yaml rather than left implicit. The scalar path expects a
    spectrum you de-redshifted yourself; a datacube arrives straight from the
    telescope in the observed frame, so the cube paths do it before any
    spaxel is fitted.

    Applied to the wavelength axis rather than to the flux, and before the
    log grid is built, so every template, mask and line window downstream is
    in one frame. Nothing is interpolated and no flux is touched.

    cube_redshift and spectrum_redshift do opposite jobs and setting both is
    refused in brain_settings.validate, so reaching here with a non-zero
    value means exactly one correction is wanted.
    """
    z = float(getattr(cfg, 'cube_redshift', 0.0) or 0.0)
    if z == 0.0:
        return wl, 0.0
    logger.info(f"Rest frame: dividing the wavelength axis by 1+z with "
                f"z = {z:g} ({wl[0]:.0f}-{wl[-1]:.0f} A observed becomes "
                f"{wl[0] / (1.0 + z):.0f}-{wl[-1] / (1.0 + z):.0f} A rest)")
    return wl / (1.0 + z), z

def spaxel_inputs(flux_lin, ivar_lin, mask_lin, wl_lin, lam_grid,
                  file_weight_lin, norm_wavelength):
    """
    One spaxel's flux and inverse sigma on the log grid.

    Named and separate so that there is exactly one definition of what a
    spaxel looks like by the time it reaches the fitter, and so that it can
    be compared against the scalar path without running a fit. It follows
    brainv1 step for step, because any step where it does not is a place the
    same spectrum gets two different answers depending on which entry point
    was used.

    The error array keeps its values at rejected pixels rather than being
    zeroed there, because the rejection is carried by the weight and by the
    clean fraction, not by the error. Zeroing it instead would interpolate a
    real error against a zero at every pixel next to a rejected one, which
    changes the weight of pixels that were never in question. Where the
    inverse variance carries no information at all the error is filled the
    way load_spectrum fills it, ten times the median, so that the two entry
    points agree pixel for pixel.

    The two mask carriers are both needed and do different jobs. The
    nearest-neighbour weight says which input pixel an output pixel sits on
    top of. The clean fraction says how much of the output pixel's *value*
    came from input that was worth keeping, which is the question
    nearest-neighbour cannot answer: log-rebinning is a flux-conserving
    redistribution, so where the output bin is wider than the input bin it
    blends neighbours, and a pixel whose nearest input was good can still
    hold flux from a rejected one next door.
    """
    flux_norm, norm_fac = normalize_at_wavelength(flux_lin, wl_lin,
                                                  norm_wavelength)
    flux_log, log_lam, velscale = log_rebin(wl_lin, flux_norm)

    dq_good_lin = np.ones(len(wl_lin), dtype=bool)
    if ivar_lin is not None:
        dq_good_lin &= np.isfinite(ivar_lin) & (ivar_lin > 0)
    if mask_lin is not None:
        dq_good_lin &= (np.asarray(mask_lin) & MANGA_DONOTUSE) == 0

    file_weight_lin = np.asarray(file_weight_lin)
    weight_lin = file_weight_lin * dq_good_lin
    usable_lin = dq_good_lin & (file_weight_lin > 0)
    uncontaminated = np.asarray(
        clean_fraction(wl_lin, usable_lin, lam_grid)) >= 1.0 - 1e-8

    weight_log = np.clip(
        interp1d(np.asarray(wl_lin), weight_lin, kind='nearest',
                 bounds_error=False, fill_value=0.0)(np.asarray(lam_grid)),
        0.0, None)
    good_log = (weight_log > 0) & uncontaminated
    sqrt_w = np.sqrt(weight_log)

    if ivar_lin is not None:
        have_ivar = np.isfinite(ivar_lin) & (ivar_lin > 0)
        err_lin = np.where(have_ivar,
                           1.0 / np.sqrt(np.where(have_ivar, ivar_lin, 1.0)),
                           0.0)
        if not np.all(have_ivar):
            filler = np.median(err_lin[have_ivar]) * 10.0 if np.any(have_ivar) else 1.0
            err_lin = np.where(have_ivar, err_lin, filler)
        err_log = np.asarray(np.interp(lam_grid, wl_lin, err_lin / norm_fac))
        valid = good_log & (err_log > 0)
        inv_sigma = np.where(valid, sqrt_w / np.where(err_log > 0, err_log, 1.0),
                             0.0)
    else:
        err_log = None
        inv_sigma = sqrt_w * good_log

    return flux_log, inv_sigma, err_log, float(norm_fac), float(velscale)

def process_spaxel(args, shared_data):
    """
    Worker function to fit a single spaxel.
    """
    (y, x, flux_lin, ivar_lin, mask_lin) = args
    (wl_lin, lam_grid, M_log, cfg, ssp_yav_flags, qh_ratios, nebular_model,
     n_ssp, lsf_info, file_weight_lin) = shared_data

    flux_log, inv_sigma, err_log, norm_fac, velscale = spaxel_inputs(
        flux_lin, ivar_lin, mask_lin, wl_lin, lam_grid, file_weight_lin,
        cfg.norm_wavelength)

    fitter = SpectralFitter(
        M_log, lam_grid, flux_log, inv_sigma, velscale, cfg,
        ssp_yav_flags=ssp_yav_flags,
        nebular_model=nebular_model,
        qh_ratios=qh_ratios,
    )

    if cfg.photoionization_mode == "photopop":
        fitter._photoionization = 'nepop'

    result = fitter.optimize()

    if (cfg.photoionization_mode in NEBULAR_MODES
            and 'Halpha_6563' in result['gas_names']):
        from brainsp.nebular import Av_neb_from_balmer_decrement, extract_line_flux
        gas_names = result['gas_names']
        w = result['weights']
        F_Ha_obs = extract_line_flux(gas_names, w[n_ssp:], 'Halpha')
        F_Hb_obs = extract_line_flux(gas_names, w[n_ssp:], 'Hbeta')
        
        if F_Ha_obs > 0 and F_Hb_obs > 0:
            Av_neb = Av_neb_from_balmer_decrement(F_Ha_obs, F_Hb_obs, cfg.reddening_law, cfg.electron_temperature)
            Av_neb = np.clip(Av_neb, cfg.av_nebular_bounds[0], cfg.av_nebular_bounds[1])
            fitter.update_Av_neb(Av_neb)
            result = fitter.optimize(initial_guess=result['params'])

            if cfg.photoionization_mode == "photopop" and result['Q_eff'] > 0:
                fitter._photoionization = 'photopop'
                for _ in range(cfg.max_sc_iterations):
                    result = fitter.optimize(initial_guess=result['params'])

    M_star_dust = result['M_star_dust']
    M_gas_dust = result['M_gas_dust']
    neb_fit = result.get('nebular_spectrum', np.zeros_like(flux_log))
    flux_for_mc = (flux_log - neb_fit
                   if cfg.photoionization_mode in NEBULAR_MODES else flux_log)
    
    w_med, w_err = mc_weight_errors(
        M_star_dust, M_gas_dust, flux_for_mc, inv_sigma, err_log,
        cfg.reg_strength, cfg.reg_order, cfg.n_mc_resamples,
        solver=cfg.linear_solver, simplex=cfg.enforce_simplex
    )

    best_params = result['params']

    w_fit = np.asarray(result['weights'])
    model_flux = np.asarray(M_star_dust @ w_fit[:n_ssp])
    n_gas = len(result['gas_names'])
    if M_gas_dust is not None and n_gas > 0:
        model_flux = model_flux + np.asarray(M_gas_dust @ w_fit[n_ssp:n_ssp + n_gas])
    if cfg.photoionization_mode in NEBULAR_MODES and neb_fit is not None:
        model_flux = model_flux + np.asarray(neb_fit)

    lsf_report = dispersion_report(
        lsf_info, float(best_params[2]), w_fit[:n_ssp], model_flux,
        np.asarray(inv_sigma) > 0, cfg.sigma_star_bounds[0],
        upper_bound=cfg.sigma_star_bounds[1],
        reference_wavelength=cfg.norm_wavelength)

    return {
        'y': y,
        'x': x,
        'Av': best_params[0],
        'YAV': best_params[1],
        'sigma_star': best_params[2],
        'v0_star': best_params[3],
        'chi2': result['chi2'],
        'weights': w_med,
        'weights_err': w_err,
        'Q_eff': result.get('Q_eff', 0.0),
        'Av_neb': result.get('Av_neb', 0.0),
        'sigma_star_corrected': lsf_report['sigma_star'],
        'delta_sigma_inst': lsf_report['delta_sigma_inst'],
        'sigma_star_upper_limit': lsf_report['upper_limit'],
        'lsf_reason': lsf_report['reason'],
    }

def run_datacube(cfg, datacube_path, output_path, num_threads=None):
    t0 = time.time()
    logger.info(f"Loading datacube: {datacube_path}")
    
    with fits.open(datacube_path) as hdul:
        flux_cube = hdul['FLUX'].data.astype(np.float64)
        ivar_cube = hdul['IVAR'].data.astype(np.float64) if 'IVAR' in hdul else None
        mask_cube = hdul['MASK'].data if 'MASK' in hdul else None
        wl_lin = hdul['WAVE'].data.astype(np.float64)
        header = hdul['PRIMARY'].header

    wl_lin, z_applied = to_rest_frame(wl_lin, cfg, logger)

    if cfg.wavelength_min is not None or cfg.wavelength_max is not None:
        lo = cfg.wavelength_min if cfg.wavelength_min is not None else wl_lin[0]
        hi = cfg.wavelength_max if cfg.wavelength_max is not None else wl_lin[-1]
        keep = (wl_lin >= lo) & (wl_lin <= hi)
        if not np.any(keep):
            raise SystemExit(
                f"\n'wavelength_min'/'wavelength_max' ({lo}-{hi} A) select no "
                f"part of this cube, which spans "
                f"{wl_lin[0]:.0f}-{wl_lin[-1]:.0f} A.\n")
        wl_lin = wl_lin[keep]
        flux_cube = flux_cube[keep]
        ivar_cube = ivar_cube[keep] if ivar_cube is not None else None
        mask_cube = mask_cube[keep] if mask_cube is not None else None
        logger.info(f"Clipped to {lo:.0f}-{hi:.0f} A: {len(wl_lin)} wavelengths")

    n_wave, n_y, n_x = flux_cube.shape
    logger.info(f"Cube shape: {n_y}x{n_x} spaxels, {n_wave} wavelengths")

    logger.info("Pre-loading SSP library...")
    flux_dummy, log_lam, velscale = log_rebin(wl_lin, flux_cube[:, n_y//2, n_x//2])
    lam_grid = np.exp(log_lam)
    
    bases_info = bases_info_for(cfg)

    M_log, ssp_names, ssp_ages, ssp_metals, ssp_short, ssp_mstars, ssp_yav_flags, ssp_fbase_norms = load_ssp_templates(
        cfg.ssp_folder, bases_info, lam_grid, cfg.norm_wavelength
    )
    n_ssp = M_log.shape[1]

    covered = np.any(np.abs(np.asarray(M_log)) > 0, axis=1)
    fitted_pixels = covered & cube_fitted_pixels(flux_cube, ivar_cube,
                                                 wl_lin, lam_grid)
    n_bare = int(np.sum(~covered))
    if n_bare:
        bare_wl = np.asarray(lam_grid)[~covered]
        logger.warning(
            f"   {n_bare} pixels lie outside the template coverage "
            f"({bare_wl.min():.0f}-{bare_wl.max():.0f} A). They carry no "
            f"template flux in any spaxel and are excluded from the "
            f"line-spread-function decision.")

    M_log, ssp_fbase_norms, lsf_info = prepare_templates(
        M_log, lam_grid, cfg, bases_info, fitted_pixels, ssp_fbase_norms)

    if getattr(cfg, 'instrumental_resolution', None) is not None:
        logger.info(
            f"   Line-spread function, cube-wide: {lsf_info['state_text']} "
            f"({lsf_info['n_applied']} templates pre-convolved, "
            f"{lsf_info['n_refused']} left at library resolution), decided "
            f"over the {int(np.sum(fitted_pixels))}/{len(lam_grid)} pixels "
            f"that at least one fitted spaxel uses and at least one template "
            f"covers. The same templates are used for every spaxel.")

    nebular_model = None
    qh_ratios = None
    if cfg.photoionization_mode in NEBULAR_MODES:
        from brainsp.nebular import NebularModel, compute_qh_ratios
        nebular_model = NebularModel(
            lam_grid, wavelength_scale=cfg.wavelength_scale,
            Te=cfg.electron_temperature, ne=cfg.electron_density,
            Y_He=cfg.helium_abundance, Z_gas=cfg.gas_metallicity, f_esc=cfg.lyman_escape_fraction,
        )
        qh_ratios = compute_qh_ratios(
            ssp_ages, ssp_metals, ssp_fbase_norms,
            ssp_flux_unit=cfg.ssp_flux_unit,
            ssp_folder=cfg.ssp_folder,
            bases_info=bases_info,
            allow_partial=cfg.allow_partial_ionising_coverage,
        )

    shared_data = (wl_lin, lam_grid, M_log, cfg, ssp_yav_flags, qh_ratios,
                   nebular_model, n_ssp, lsf_info, load_mask(cfg.mask_file, wl_lin))

    tasks = []
    for y in range(n_y):
        for x in range(n_x):
            f = flux_cube[:, y, x]
            if np.sum(f) > 0 and not np.all(np.isnan(f)):
                tasks.append((y, x, f,
                              ivar_cube[:, y, x] if ivar_cube is not None else None,
                              mask_cube[:, y, x] if mask_cube is not None else None))

    logger.info(f"Total valid spaxels to fit: {len(tasks)}")

    logger.info("Compiling JAX graph on the first spaxel...")
    process_spaxel(tasks[0], shared_data)
    logger.info("Compilation complete. Launching parallel pool...")

    num_threads = num_threads or max(1, mp.cpu_count() - 2)
    logger.info(f"Using {num_threads} parallel threads/processes.")

    worker = partial(process_spaxel, shared_data=shared_data)
    
    results = []
    with spaxel_pool(num_threads) as pool:
        for i, res in enumerate(pool.imap_unordered(worker, tasks)):
            results.append(res)
            if (i + 1) % 10 == 0 or (i + 1) == len(tasks):
                logger.info(f"Progress: {i + 1}/{len(tasks)} spaxels completed.")

    logger.info(f"Finished fitting entire datacube in {time.time() - t0:.1f}s!")

    corrected = np.array([r['sigma_star_corrected'] for r in results
                          if r['sigma_star_corrected'] is not None])
    if corrected.size:
        deltas = np.array([r['delta_sigma_inst'] for r in results
                           if r['delta_sigma_inst'] is not None])
        n_limit = int(sum(1 for r in results if r['sigma_star_upper_limit']))
        logger.info(
            f"   sigma_star corrected for the line-spread function in "
            f"{corrected.size}/{len(results)} spaxels: median "
            f"{np.median(np.abs(corrected)):.2f} km/s, median "
            f"delta_sigma_inst {np.median(deltas):+.2f} km/s, "
            f"{n_limit} upper limits.")
    else:
        reasons = [r['lsf_reason'] for r in results if r['lsf_reason']]
        logger.warning(
            f"   sigma_star is an OBSERVED dispersion in all "
            f"{len(results)} spaxels, not an intrinsic one"
            f"{': ' + reasons[0] if reasons else '.'}")

    logger.info(f"Saving results to {output_path} (Placeholder for FITS writer)")

    return results



if __name__ == "__main__":
    """
    Every physics setting comes from brain_setup.yaml, as it does everywhere
    else. The command line names the cube, the output and the pool size, and
    nothing else: a second place to declare the template library, the fitting
    range or the nebular mode is a second answer to a question that has one.
    """
    import argparse
    ap = argparse.ArgumentParser(
        description='BRAIN parallel datacube fitter. Settings come from '
                    'brain_setup.yaml; the command line names the cube.')
    ap.add_argument('datacube')
    ap.add_argument('-o', '--output', default=None)
    ap.add_argument('-j', '--threads', type=int, default=None)
    ap.add_argument('-s', '--settings', default='brain_setup.yaml')
    args = ap.parse_args()

    from brainsp import brain_settings
    _S = brain_settings.validate(
        brain_settings.load(args.settings), args.settings)
    cfg = brain_settings.to_config(_S, spectrum_file=args.datacube)
    out = args.output or (
        os.path.splitext(args.datacube)[0].replace('-RAW', '')
        + '-BRAIN_FIT.fits')
    run_datacube(cfg, args.datacube, out, num_threads=args.threads)
