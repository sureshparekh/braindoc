"""
What this machine has, and how much of it a fit uses.

BRAIN sizes its work from the machine it finds, not from fixed numbers:

* CPU: every core it may run on but two (fit_megacube_jax.cpu_budget), one
  worker per physical core of those, no more than the free memory allows
  (fit_megacube_jax.cpu_worker_count, worker_cores).
* GPU: the queue's lanes (spaxels in flight) from the card itself,
  gpu_lanes: about two per streaming multiprocessor, as many as its memory
  holds.
* Several GPUs, several cubes or lists of spectra: brainsp.scheduler.

`describe()` gives the lines the fit logs. Nothing here changes what a fit
computes: spaxels are fitted independently, and how many are in flight at
once changes only the order XLA sums in (about 1e-5 at most, the same as a
different batch size).
"""

import os
import shutil
import subprocess

# Lanes per streaming multiprocessor at which the queue's throughput stops
# rising. Measured on the RTX 3060 (28 SMs), 180-spaxel espop mock cube:
# 32 lanes 4.8 spaxels/s, 64 lanes 5.2, 128 lanes 4.8.
LANES_PER_SM = 2
# Without the SM count (a device that does not report it), the 3060's value.
DEFAULT_LANES = 64
# Never more lanes than this, whatever the card: a queue's tail (its last
# spaxels, in lanes of their own) grows with it.
MAX_LANES = 1024

# NVIDIA compute capabilities whose double precision runs at half the
# single-precision rate (P100, V100, A100, H100/H200, B100/B200); every
# other card runs it at 1/32 or 1/64. BRAIN fits in double precision, so on
# these cards a fit is far faster than their memory alone suggests.
FULL_RATE_FP64 = ("6.0", "7.0", "8.0", "9.0", "10.0")

def gpus():
    """
    The NVIDIA GPUs visible to this process, from nvidia-smi (so without
    starting JAX): a list of dicts with index, name, memory_gb and
    compute_capability. Empty without nvidia-smi or GPUs. CUDA_VISIBLE_DEVICES
    is honoured: only the cards it names, in its order.
    """
    if shutil.which("nvidia-smi") is None:
        return []
    try:
        r = subprocess.run(
            ["nvidia-smi", "--query-gpu=index,name,memory.total,compute_cap",
             "--format=csv,noheader,nounits"],
            capture_output=True, text=True, timeout=10)
    except (OSError, subprocess.SubprocessError):
        return []
    if r.returncode != 0:
        return []
    out = []
    for line in r.stdout.strip().splitlines():
        parts = [p.strip() for p in line.split(",")]
        if len(parts) < 4:
            continue
        try:
            out.append(dict(index=int(parts[0]), name=parts[1],
                            memory_gb=float(parts[2]) / 1024.0,
                            compute_capability=parts[3]))
        except ValueError:
            continue
    visible = os.environ.get("CUDA_VISIBLE_DEVICES")
    if visible is not None and visible.strip() != "":
        by_index = {g["index"]: g for g in out}
        chosen = []
        for v in visible.split(","):
            v = v.strip()
            if v.isdigit() and int(v) in by_index:
                chosen.append(by_index[int(v)])
        out = chosen
    return out

def full_rate_fp64(compute_capability):
    return str(compute_capability) in FULL_RATE_FP64

def gpu_lanes(device, memory_lanes):
    """
    Lanes for the work queue on `device` (a JAX GPU device): LANES_PER_SM per
    streaming multiprocessor, rounded to a power of two, never more than
    `memory_lanes` (what the card's memory holds, estimate_lanes_fast) or
    MAX_LANES. On the RTX 3060 this is 64, as before it was computed.
    """
    sms = getattr(device, "core_count", None)
    if not sms:
        want = DEFAULT_LANES
    else:
        import math
        want = 1 << max(0, int(round(math.log2(LANES_PER_SM * int(sms)))))
    return int(max(1, min(want, int(memory_lanes), MAX_LANES)))

def describe(backend, device=None, lanes=None, workers=None):
    """Lines describing the machine and what this fit uses of it."""
    from brainsp.fit_megacube_jax import (allowed_cores, available_memory_gb,
                                          cpu_budget, physical_cores)
    cores = allowed_cores()
    phys = len(physical_cores(cores))
    mem = available_memory_gb()
    lines = [f"Machine: {len(cores)} CPU threads on {phys} cores available, "
             f"{cpu_budget()} usable (all but two)"
             + (f", {mem:.0f} GB memory free" if mem is not None else "")]
    if backend == "gpu" and device is not None:
        sms = getattr(device, "core_count", None)
        cc = getattr(device, "compute_capability", None)
        rate = ("" if cc is None else
                (", full-rate FP64" if full_rate_fp64(cc)
                 else ", reduced-rate FP64"))
        others = len(gpus())
        lines.append(
            f"GPU: {getattr(device, 'device_kind', device)}"
            + (f", {sms} SMs" if sms else "") + rate
            + (f"; {others} GPUs visible" if others > 1 else "")
            + (f"; {lanes} lanes" if lanes else ""))
    elif workers:
        lines.append(f"CPU: {workers} worker processes, one core each")
    return lines
