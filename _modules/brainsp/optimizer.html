"""
NewBrain Optimizer Module — 100% JAX Backend
Direct Pixel-Fitting via grid-search initialization + Levenberg-Marquardt.

ALL computation is in JAX — no scipy dependency for the standard mode.
  - FFT broadening, dust attenuation, emission lines: JAX matrix ops
  - NNLS inner solver: JIT-compiled FISTA
  - LM outer optimizer: jax.lax.while_loop + vmap'd numerical Jacobian
  - Grid search: jax.vmap across velocity grid

The Jacobian is computed via vmap'd central differences — all 2*n_params
perturbations evaluate in a single fused parallel batch.

Supports three photoionization modes:
  "spop"     — standard fitting (100% JAX LM)
  "nepop"    — nebular continuum from Q(H) subtracted iteratively
  "photopop" — self-consistent: predicted H lines replace empirical Gaussians
"""

import jax
import jax.numpy as jnp
import numpy as np
from functools import partial
import logging

from brainsp.spectral_engine import (
    fourier_broaden_matrix,
    _get_reddening_curve,
)
from brainsp.solver import (
    solve_nnls_regularized, compute_residual_vector,
    build_regularization_matrix, _fista_nnls, solve_active_set,
)
from brainsp.emission_lines import (
    build_emission_matrix_grouped,
    get_line_catalog,
    assign_kinematic_groups,
    group_positions,
    precompile_line_table,
    _build_emission_jit,
)

logger = logging.getLogger("NEWBRAIN")


def _nnls(ATA, ATb, n_star, solver, simplex):
    """
    The non-negative solve, chosen the way fast_engine chooses it.

    Exists so that the batched cube path and the scalar path cannot disagree
    about which solver they use. The three forward models below called
    _fista_nnls directly, so linear_solver reached the scalar engine and
    never reached a cube: every weight, population vector and map a cube
    produced came from the iterative solver that does not converge on this
    problem, whatever the settings file said.

    solver and simplex are static, resolved when the enclosing function is
    traced, so this is a compile-time branch and costs nothing.
    """
    if solver == "active_set":
        n = ATb.shape[0]
        cvec = (jnp.where(jnp.arange(n) < n_star, 1.0, 0.0) if simplex
                else jnp.ones(n))
        return solve_active_set(ATA, ATb, cvec, simplex=simplex)[0]
    return _fista_nnls(ATA, ATb)


def _forward_residual(params, M_star, lam, flux, inv_sigma, velscale,
                      dk, D_full, ssp_yav_flags,
                      rest_wl, ratios, n_comp, group_indices, max_comp,
                      solver="active_set", simplex=False):
    """
    Full forward model evaluation — JIT-compatible, no Python control flow.

    params : [Av, YAV, sigma_star, v0_star, sig_g0, v0_g0, ...]

    Returns the augmented residual vector [data_residual, reg_residual].
    """
    av = params[0]
    yav = params[1]
    sigma_star = params[2]
    v0_star = params[3]
    gas_p = params[4:]
    sigma_gas = gas_p[::2]
    v0_gas = gas_p[1::2]

    M_broad = fourier_broaden_matrix(M_star, sigma_star, v0_star, velscale)

    atten = 10.0 ** (-0.4 * av * dk)
    atten_yav = 10.0 ** (-0.4 * (av + yav) * dk)
    M_star_dust = jnp.where(
        ssp_yav_flags[jnp.newaxis, :] > 0,
        M_broad * atten_yav[:, jnp.newaxis],
        M_broad * atten[:, jnp.newaxis],
    )

    M_gas = _build_emission_jit(
        lam, velscale, sigma_gas, v0_gas,
        rest_wl, ratios, n_comp, group_indices, max_comp,
    )
    M_gas_dust = M_gas * atten[:, jnp.newaxis]

    M_full = jnp.hstack([M_star_dust, M_gas_dust])
    A = M_full * inv_sigma[:, jnp.newaxis]
    b = flux * inv_sigma
    A_aug = jnp.vstack([A, D_full])
    b_aug = jnp.concatenate([b, jnp.zeros(D_full.shape[0])])
    ATA = A_aug.T @ A_aug
    ATb = A_aug.T @ b_aug
    weights = _nnls(ATA, ATb, M_star.shape[1], solver, simplex)

    fitted = M_full @ weights
    data_resid = (flux - fitted) * inv_sigma
    reg_resid = D_full @ weights

    return jnp.concatenate([data_resid, reg_resid])


def _forward_full(params, M_star, lam, flux, inv_sigma, velscale,
                  dk, D_full, ssp_yav_flags,
                  rest_wl, ratios, n_comp, group_indices, max_comp,
                  solver="active_set", simplex=False):
    """
    Full forward model returning all components for post-processing.
    Same computation as _forward_residual but returns intermediate results.
    """
    av = params[0]
    yav = params[1]
    sigma_star = params[2]
    v0_star = params[3]
    gas_p = params[4:]
    sigma_gas = gas_p[::2]
    v0_gas = gas_p[1::2]

    M_broad = fourier_broaden_matrix(M_star, sigma_star, v0_star, velscale)

    atten = 10.0 ** (-0.4 * av * dk)
    atten_yav = 10.0 ** (-0.4 * (av + yav) * dk)
    M_star_dust = jnp.where(
        ssp_yav_flags[jnp.newaxis, :] > 0,
        M_broad * atten_yav[:, jnp.newaxis],
        M_broad * atten[:, jnp.newaxis],
    )

    M_gas = _build_emission_jit(
        lam, velscale, sigma_gas, v0_gas,
        rest_wl, ratios, n_comp, group_indices, max_comp,
    )
    M_gas_dust = M_gas * atten[:, jnp.newaxis]

    M_full = jnp.hstack([M_star_dust, M_gas_dust])
    A = M_full * inv_sigma[:, jnp.newaxis]
    b = flux * inv_sigma
    A_aug = jnp.vstack([A, D_full])
    b_aug = jnp.concatenate([b, jnp.zeros(D_full.shape[0])])
    ATA = A_aug.T @ A_aug
    ATb = A_aug.T @ b_aug
    weights = _nnls(ATA, ATb, M_star.shape[1], solver, simplex)

    fitted = M_full @ weights
    data_resid = (flux - fitted) * inv_sigma
    reg_resid = D_full @ weights
    residual = jnp.concatenate([data_resid, reg_resid])

    return weights, fitted, M_star_dust, M_gas_dust, residual


def _grid_point_chi2(v0_test, M_star, lam, flux, inv_sigma, velscale,
                     sigma_star, dk, av,
                     sigma_gas_arr,
                     rest_wavelengths, ratios, n_components, group_indices,
                     max_comp, D_full, solver="active_set", simplex=False):
    """
    Chi2 for one velocity grid point. Used inside jax.vmap.
    """
    atten = jnp.where(av > 0, 10.0 ** (-0.4 * av * dk), 1.0)
    M_broad = fourier_broaden_matrix(M_star, sigma_star, v0_test, velscale)
    M_star_dust = M_broad * atten[:, jnp.newaxis]

    n_groups = sigma_gas_arr.shape[0]
    v0_gas_arr = jnp.full(n_groups, v0_test)
    M_gas = _build_emission_jit(
        lam, velscale, sigma_gas_arr, v0_gas_arr,
        rest_wavelengths, ratios, n_components, group_indices, max_comp,
    )
    M_gas_dust = M_gas * atten[:, jnp.newaxis]

    M_full = jnp.hstack([M_star_dust, M_gas_dust])
    A = M_full * inv_sigma[:, jnp.newaxis]
    b = flux * inv_sigma
    A = jnp.vstack([A, D_full])
    b = jnp.concatenate([b, jnp.zeros(D_full.shape[0])])
    ATA = A.T @ A
    ATb = A.T @ b
    weights = _nnls(ATA, ATb, M_star.shape[1], solver, simplex)

    fitted = M_full @ weights
    residuals = (flux - fitted) * inv_sigma
    return jnp.sum(residuals ** 2)


@partial(jax.jit, static_argnames=('max_iter', 'max_comp', 'solver', 'simplex'))
def _lm_optimize(x0, bounds_lo, bounds_hi,
                 M_star, lam, flux, inv_sigma, velscale,
                 dk, D_full, ssp_yav_flags,
                 rest_wl, ratios, n_comp, group_indices,
                 max_comp,
                 max_iter=200, ftol=1e-8, xtol=1e-8, diff_step=1e-3,
                 solver="active_set", simplex=False):
    """
    Projected Levenberg-Marquardt — 100% JAX, fully JIT-compiled.

    Uses vmap'd central differences for the Jacobian: all 2*n_params
    perturbations evaluate in a single fused parallel batch.
    Bounds handled by projected step (parameter clipping).
    Diagonal Jacobian scaling (x_scale='jac') for numerical stability
    across parameters of different magnitudes.

    Parameters
    ----------
    x0 : (n_params,)  Initial parameter vector.
    bounds_lo, bounds_hi : (n_params,)  Parameter bounds.
    max_comp : int (static)
    max_iter : int (static)

    Returns
    -------
    best_params, best_chi2, niter, nfev, converged
    """
    n_params = x0.shape[0]

    def resid(p):
        return _forward_residual(
            p, M_star, lam, flux, inv_sigma, velscale,
            dk, D_full, ssp_yav_flags,
            rest_wl, ratios, n_comp, group_indices, max_comp,
            solver=solver, simplex=simplex,
        )

    def jacobian(p):
        eps = diff_step * jnp.maximum(jnp.abs(p), diff_step)
        E = jnp.diag(eps)
        r_plus = jax.vmap(lambda dp: resid(p + dp))(E)
        r_minus = jax.vmap(lambda dp: resid(p - dp))(E)
        return ((r_plus - r_minus) / (2.0 * eps[:, jnp.newaxis])).T

    r0 = resid(x0)
    chi2_0 = jnp.sum(r0 ** 2)

    scale_init = jnp.ones(n_params)
    init = (x0, jnp.array(1e-3), chi2_0, r0, 0, 1, False, scale_init,
            0, 0, chi2_0)

    def cond(state):
        _, _, _, _, niter, _, conv, _, n_rej, _, _ = state
        return (~conv) & (niter < max_iter) & (n_rej < 10)

    def body(state):
        x, lam_lm, chi2, r, niter, nfev, _, scale, n_rej, n_stag, chi2_ref = state

        J = jacobian(x)

        col_norms = jnp.sqrt(jnp.sum(J ** 2, axis=0))
        scale_new = jnp.maximum(scale, col_norms)
        scale_safe = jnp.maximum(scale_new, 1e-10)

        JTJ = J.T @ J
        JTr = J.T @ r
        H = JTJ + lam_lm * jnp.diag(scale_safe ** 2)
        dp = jnp.linalg.solve(H, -JTr)

        x_cand = jnp.clip(x + dp, bounds_lo, bounds_hi)

        r_cand = resid(x_cand)
        chi2_cand = jnp.sum(r_cand ** 2)

        dp_actual = x_cand - x
        dr_pred = J @ dp_actual
        pred_chi2 = jnp.sum((r + dr_pred) ** 2)
        pred_red = chi2 - pred_chi2
        actual_red = chi2 - chi2_cand
        gain = jnp.where(jnp.abs(pred_red) > 1e-30,
                         actual_red / pred_red, 0.0)

        accept = chi2_cand < chi2
        x_out = jnp.where(accept, x_cand, x)
        chi2_out = jnp.where(accept, chi2_cand, chi2)
        r_out = jnp.where(accept, r_cand, r)
        n_rej_out = jnp.where(accept, 0, n_rej + 1)

        lam_new = jnp.where(
            accept & (gain > 0.75), jnp.maximum(lam_lm / 3.0, 1e-12),
            jnp.where(accept & (gain > 0.25), lam_lm,
                       jnp.minimum(lam_lm * 3.0, 1e10)),
        )

        rel_chi2 = jnp.where(accept,
                             jnp.abs(actual_red) / jnp.maximum(chi2, 1e-30),
                             1.0)
        step_norm = jnp.max(
            jnp.abs(dp_actual) / jnp.maximum(bounds_hi - bounds_lo, 1e-30)
        )
        classical_conv = accept & (rel_chi2 < ftol) & (step_norm < xtol) & (niter > 3)

        window_rel_change = jnp.abs(chi2_ref - chi2_out) / jnp.maximum(chi2_ref, 1e-30)
        big_improvement = window_rel_change > ftol * 100
        n_stag_out = jnp.where(accept,
                               jnp.where(big_improvement, 0, n_stag + 1),
                               n_stag)
        chi2_ref_out = jnp.where(big_improvement, chi2_out, chi2_ref)
        stagnation_conv = (n_stag_out >= 5) & (niter > 3)

        conv = classical_conv | stagnation_conv | (n_rej_out >= 10)

        new_nfev = nfev + 2 * n_params + 1

        return (x_out, lam_new, chi2_out, r_out, niter + 1, new_nfev, conv,
                scale_new, n_rej_out, n_stag_out, chi2_ref_out)

    final = jax.lax.while_loop(cond, body, init)
    return final[0], final[2], final[4], final[5], final[6]

class SpectralFitter:
    """
    Direct Pixel-Fitting engine — 100% JAX.

    Wraps the nonlinear optimization of kinematic + dust parameters,
    with internal NNLS solution of linear SSP/gas weights at each step.

    All broadening, dust, emission, NNLS, and LM optimization run in
    JIT-compiled JAX — no scipy dependency (standard mode).

    The nonlinear parameter vector is:
        [Av, YAV, sigma_star, v0_star, sig_g0, v0_g0, sig_g1, v0_g1, ...]
    where g0, g1, ... are the kinematic groups.
    """

    def __init__(self, matrix_star_log, lam_grid, flux_obs, inv_sigma,
                 velscale, config, ssp_yav_flags=None,
                 nebular_model=None, qh_ratios=None):
        self.M_star = jnp.asarray(matrix_star_log)
        self.lam = jnp.asarray(lam_grid)
        self.flux = jnp.asarray(flux_obs)
        self.inv_sigma = jnp.asarray(inv_sigma)
        self.velscale = velscale
        self.cfg = config
        self.n_star = matrix_star_log.shape[1]
        self.ssp_yav_flags = jnp.asarray(
            ssp_yav_flags if ssp_yav_flags is not None
            else np.zeros(self.n_star, dtype=int)
        )

        self.nebular_model = nebular_model
        self.qh_ratios = jnp.asarray(
            qh_ratios if qh_ratios is not None else np.zeros(self.n_star)
        )
        from brainsp.config import NEBULAR_MODES
        self._photoionization = (
            config.photoionization_mode
            if config.photoionization_mode in NEBULAR_MODES
            else 'spop'
        )

        if self._photoionization != 'spop':
            from brainsp import reddening_laws
            self._k_norm = float(reddening_laws.get_k_lambda(
                jnp.array([config.norm_wavelength]), config.reddening_law
            )[0])
        else:
            self._k_norm = 0.0

        if config.fit_emission_lines:
            self._lines, self._doublets = get_line_catalog(
                (float(lam_grid[0]), float(lam_grid[-1])),
                config.wavelength_scale
            )
        else:
            self._lines, self._doublets = {}, {}
        if self._lines or self._doublets:
            from brainsp.emission_lines import drop_masked_lines
            self._lines, self._doublets, masked = drop_masked_lines(
                self._lines, self._doublets, lam_grid, inv_sigma,
                config.sigma_gas_init)
            if masked:
                logger.warning(
                    f"   {len(masked)} emission line(s) lie under the mask "
                    f"or in a gap of the data and are NOT fitted or "
                    f"reported: {', '.join(masked)}. Unmask them to fit "
                    f"them.")

        self._line_groups, self._n_groups = assign_kinematic_groups(
            self._lines, self._doublets, config.kinematic_groups
        )
        self._group_labels = list(group_positions(self._line_groups))
        self._n_groups = len(self._group_labels)

        logger.info(f"   Kinematic groups: {self._n_groups} groups")
        for label in self._group_labels:
            members = [n for n, g in self._line_groups.items() if g == label]
            logger.info(
                f"     Group {label}: {', '.join(members[:5])}"
                + (f" ... +{len(members)-5} more" if len(members) > 5 else "")
            )

        self._line_table = precompile_line_table(
            lam_grid, self._lines, self._doublets, self._line_groups,
            config.v0_gas_bounds, config.sigma_gas_bounds[1]
        )
        self._max_comp = (
            int(self._line_table['rest_wavelengths'].shape[1])
            if self._line_table['names'] else 1
        )
        self._n_gas_cols = len(self._line_table['names'])

        self._dk = _get_reddening_curve(
            lam_grid, config.reddening_law, config.norm_wavelength
        )

        n_total = self.n_star + self._n_gas_cols
        D = build_regularization_matrix(
            self.n_star, config.reg_strength, config.reg_order
        )
        self._has_D = D is not None
        if self._has_D:
            self._D_full = jnp.zeros((D.shape[0], n_total))
            self._D_full = self._D_full.at[:, :self.n_star].set(D)
        else:
            self._D_full = jnp.zeros((0, n_total))

        self._hydrogen_line_keys = {
            'Halpha_6563', 'Hbeta_4861', 'Hgamma_4340', 'Hdelta_4102',
            'Hepsilon_3970', 'H8_3889', 'H9_3835', 'H10_3798',
            'Padelta_10049', 'Pagamma_10938', 'Pabeta_12820', 'Paalpha_18750',
        }

        self._last_M_star_dust = None
        self._last_M_gas_dust = None
        self._last_weights = None
        self._last_gas_names = None
        self._last_gas_group_labels = None
        self._last_Q_eff = 0.0
        self._last_nebular_spectrum = jnp.zeros(len(lam_grid))
        self._last_Av_neb = 0.0

    @property
    def inv_sigma(self):
        """
        The per-pixel inverse sigma the fit is weighted by.

        Read through a property so that writing to it cannot be separated
        from invalidating the cached fast_engine constants. Those constants
        embed the weights, and outlier clipping works by assigning a new
        array here between two optimize() calls: before this existed the
        cache key held only the mode and Av_neb, so the second call was
        handed constants built from the pre-clip weights and every clipped
        pixel went on being fitted at full weight. The failure was silent --
        the re-fit ran, converged and returned the same parameters.

        A setter is used rather than a hash of the array in the cache key.
        Hashing costs a pass over the spectrum on every optimize() call, and
        it still misses an in-place mutation of the array already stored,
        which is the case that is hardest to see. Invalidating on assignment
        cannot be bypassed by a caller that does not know the cache exists.
        """
        return self._inv_sigma

    @inv_sigma.setter
    def inv_sigma(self, value):
        self._inv_sigma = jnp.asarray(value)
        self._fast_key = None

    def _params_to_components(self, params):
        """Unpack flat parameter vector into named components."""
        av = float(params[0])
        yav = float(params[1])
        sig_star = float(params[2])
        v0_star = float(params[3])
        group_kinematics = {}
        for i, gid in enumerate(self._group_labels):
            idx = 4 + 2 * i
            group_kinematics[gid] = (float(params[idx]), float(params[idx + 1]))
        return av, yav, sig_star, v0_star, group_kinematics

    def _compute_Q_eff(self, weights, av, yav):
        """Compute the effective Q(H) parameter from stellar weights."""
        from brainsp.nebular import compute_Q_eff
        return compute_Q_eff(
            weights[:self.n_star], self.qh_ratios, self.ssp_yav_flags,
            av, yav, self._k_norm
        )

    def _evaluate(self, params, return_components=False):
        """
        Model evaluation — compatibility wrapper.

        For photoionization_mode='spop': calls pure-JAX forward model.
        For 'nepop'/'photopop': falls back to iterative nebular loop.
        """
        lt = self._line_table
        params_jax = jnp.asarray(params)

        if self._photoionization == 'spop':
            weights, fitted, M_star_dust, M_gas_dust, residual = _forward_full(
                params_jax, self.M_star, self.lam, self.flux, self.inv_sigma,
                self.velscale, self._dk, self._D_full, self.ssp_yav_flags,
                lt['rest_wavelengths'], lt['ratios'],
                lt['n_components'], lt['group_indices'], self._max_comp,
            )

            self._last_M_star_dust = M_star_dust
            self._last_M_gas_dust = M_gas_dust
            self._last_weights = weights
            self._last_gas_names = list(lt['names'])
            self._last_gas_group_labels = list(lt['group_labels'])

            if return_components:
                return M_star_dust, M_gas_dust, weights, list(lt['names'])
            return np.asarray(residual)

        av, yav, sig_star, v0_star, group_kin = self._params_to_components(params)

        M_broad = fourier_broaden_matrix(
            self.M_star, sig_star, v0_star, self.velscale
        )
        atten_base = 10.0 ** (-0.4 * av * self._dk)
        if yav > 0 and jnp.any(self.ssp_yav_flags > 0):
            atten_yav = 10.0 ** (-0.4 * (av + yav) * self._dk)
            M_star_dust = jnp.where(
                self.ssp_yav_flags[jnp.newaxis, :] > 0,
                M_broad * atten_yav[:, jnp.newaxis],
                M_broad * atten_base[:, jnp.newaxis],
            )
        else:
            M_star_dust = M_broad * atten_base[:, jnp.newaxis]

        M_gas, gas_names, gas_group_labels = build_emission_matrix_grouped(
            self.lam, self.velscale, group_kin, self._lines, self._doublets,
            self._line_groups
        )
        M_gas = jnp.asarray(M_gas)
        M_gas_dust = M_gas * atten_base[:, jnp.newaxis] if M_gas.shape[1] > 0 else None

        if not jnp.all(jnp.isfinite(M_star_dust)):
            if return_components:
                return M_star_dust, M_gas_dust, jnp.zeros(self.n_star), gas_names
            return jnp.full(len(self.flux), 1e10)

        flux_for_nnls = self.flux
        extra_rows, extra_rhs = None, None
        Q_eff = self._last_Q_eff
        Av_neb = self._last_Av_neb
        neb_atten = 10.0 ** (-0.4 * Av_neb * self._dk) if Av_neb > 0 else None

        for neb_iter in range(4):
            neb_cont = self.nebular_model.nebular_continuum(Q_eff)
            if neb_atten is not None:
                neb_cont = neb_cont * neb_atten
            flux_for_nnls = self.flux - neb_cont

            weights, fitted, chi2 = solve_nnls_regularized(
                M_star_dust, M_gas_dust, flux_for_nnls, self.inv_sigma,
                self.cfg.reg_strength, self.cfg.reg_order,
                extra_rows=extra_rows, extra_rhs=extra_rhs,
            )

            Q_eff_new = self._compute_Q_eff(weights, av, yav)
            if Q_eff > 0 and abs(Q_eff_new - Q_eff) / max(Q_eff, 1e-30) < 0.01:
                Q_eff = Q_eff_new
                break
            if neb_iter > 0 and Q_eff > 0:
                Q_eff = (self.cfg.sc_damping * Q_eff_new
                         + (1 - self.cfg.sc_damping) * Q_eff)
            else:
                Q_eff = Q_eff_new

        self._last_Q_eff = Q_eff
        self._last_nebular_spectrum = neb_cont
        self._last_M_star_dust = M_star_dust
        self._last_M_gas_dust = M_gas_dust
        self._last_weights = weights
        self._last_gas_names = list(gas_names)
        self._last_gas_group_labels = list(gas_group_labels)

        if return_components:
            return M_star_dust, M_gas_dust, weights, list(gas_names)

        residual = compute_residual_vector(
            weights, M_star_dust, M_gas_dust,
            flux_for_nnls, self.inv_sigma,
            self.cfg.reg_strength, self.cfg.reg_order,

    def grid_search_v0(self):
        """
        Coarse grid search over v0_star — jax.vmap evaluates ALL grid points
        in a single fused kernel.
        """
        cfg = self.cfg
        v0_lo, v0_hi = cfg.v0_star_bounds
        n_points = int(getattr(cfg, 'v0_grid_points', 48))
        v0_grid = jnp.linspace(v0_lo, v0_hi, n_points)

        logger.info(f"  Grid search: {n_points} velocity points "
                     f"[{v0_lo:.0f}, {v0_hi:.0f}] km/s")

        if n_points == 0:
            return cfg.v0_star_init

        sigma_gas_arr = jnp.full(max(self._n_groups, 1), cfg.sigma_gas_init)
        lt = self._line_table

        if getattr(cfg, 'use_fast_engine', True):
            from brainsp import fast_engine as fe
            C, S = self._build_fast_constants()

            def scan(grid):
                c = fe.grid_chi2(C, grid, cfg.sigma_star_init, cfg.av_init,
                                 cfg.sigma_gas_init, S, max(self._n_groups, 1))
                i = int(jnp.argmin(c))
                return float(grid[i]), float(c[i])

            n_coarse = max(n_points // 4, 8)
            n_fine = max(n_points // 3, 8)
            if getattr(cfg, 'coarse_to_fine_grid', True):
                g1 = jnp.linspace(v0_lo, v0_hi, n_coarse)
                v1, _ = scan(g1)
                half = (v0_hi - v0_lo) / max(n_coarse - 1, 1)
                g2 = jnp.linspace(max(v1 - half, v0_lo),
                                  min(v1 + half, v0_hi), n_fine)
                best_v0, best_chi2 = scan(g2)
                n_points = n_coarse + n_fine
            else:
                best_v0, best_chi2 = scan(v0_grid)
                n_points = len(v0_grid)

            logger.info(f"  Best velocity basin: v0 = {best_v0:.1f} km/s "
                        f"(chi2 = {best_chi2:.2f}, {n_points} points)")
            return best_v0

        def eval_one(v0):
            return _grid_point_chi2(
                v0, self.M_star, self.lam, self.flux, self.inv_sigma,
                self.velscale, cfg.sigma_star_init, self._dk, cfg.av_init,
                sigma_gas_arr,
                lt['rest_wavelengths'], lt['ratios'],
                lt['n_components'], lt['group_indices'], self._max_comp,
                self._D_full,
            )

        all_chi2 = jax.vmap(eval_one)(v0_grid)

        best_idx = jnp.argmin(all_chi2)
        best_v0 = float(v0_grid[best_idx])
        best_chi2 = float(all_chi2[best_idx])

        logger.info(f"  Best velocity basin: v0 = {best_v0:.1f} km/s "
                     f"(chi2 = {best_chi2:.2f})")
        return best_v0

    def optimize(self, initial_guess=None):
        """
        Full optimization: grid-search + Levenberg-Marquardt.

        For photoionization_mode='spop': 100% JAX (custom LM optimizer).
        For 'nepop'/'photopop': uses scipy.optimize.least_squares fallback.

        THE GAS DISPERSION LOWER BOUND IS THE USER'S, AND NOTHING RAISES IT

        This routine used to raise the lower end of sigma_gas_bounds to the
        velocity scale, one pixel, behind the user's back. That existed for
        one reason: the emission-line profile was a bare Gaussian sampled at
        pixel centres, which loses flux as soon as the line is narrower than a
        pixel -- 35.7 per cent of it retained at sigma_gas = 10 km/s on a
        58.39 km/s grid -- so a narrow line was silently deleted rather than
        fitted. The profile is now integrated over the pixel in the Fourier
        domain and is exact at every width including exactly zero, so the
        reason is gone and the floor is only harm: the pixel-integrated line
        is broader at a given sigma_gas than the centre-sampled one was, and
        with the floor in place both kinematic groups of the reference
        spectrum sat on it, unable to reach the narrower dispersion that fits
        better. A parameter pinned at a bound is not a measurement, and a
        bound nobody asked for makes every sigma_gas the code reported a
        statement about the grid rather than about the gas.

        What is left is the bound the user sets, gas_dispersion_range, whose
        shipped lower end is 1.0 km/s. That is below the thermal width of
        every ion in the line catalogue at T_e = 10^4 K -- 9.1 km/s for
        hydrogen, 1.2 km/s for iron, sqrt(k T_e / A m_u) -- so no fit that
        reaches it is measuring a line width. It is positive rather than zero
        because the profile depends on sigma only through sigma^2, so the
        gradient vanishes at zero and a fit that arrives there has nowhere to
        go; and because a printed 0.00 km/s invites being read as a dispersion
        of zero, which Westfall et al. 2019 Section 7.6.3 is explicit that a
        sub-instrumental width is not. A bound of this size (velscale/100
        would be 0.58 km/s on this grid) is the usual defensive choice;
        1.0 km/s is of that order and, unlike a fraction of the velocity
        scale, is a number the user can read
        in their settings file and change.

        The megacube path builds its bounds straight from sigma_gas_bounds
        and never had the floor, so the two paths now search the same box.
        """
        cfg = self.cfg
        n_params = 4 + 2 * self._n_groups

        if initial_guess is None:
            best_v0 = self.grid_search_v0()
            initial_guess = self._build_param_vector(
                cfg.av_init, 0.0, cfg.sigma_star_init, best_v0,
                cfg.sigma_gas_init, best_v0,
            )
        else:
            initial_guess = list(initial_guess)
            if len(initial_guess) == 6 and self._n_groups > 1:
                sig_g, v0_g = initial_guess[4], initial_guess[5]
                initial_guess = self._build_param_vector(
                    initial_guess[0], initial_guess[1],
                    initial_guess[2], initial_guess[3], sig_g, v0_g,
                )
            elif len(initial_guess) != n_params:
                while len(initial_guess) < n_params:
                    initial_guess.extend([cfg.sigma_gas_init, cfg.v0_gas_init])
                initial_guess = initial_guess[:n_params]

        yav_bounds = getattr(cfg, 'yav_bounds', (0.0, 5.0))
        bounds_lo = [cfg.av_bounds[0], yav_bounds[0],
                     cfg.sigma_star_bounds[0], cfg.v0_star_bounds[0]]
        bounds_hi = [cfg.av_bounds[1], yav_bounds[1],
                     cfg.sigma_star_bounds[1], cfg.v0_star_bounds[1]]
        for _ in self._group_labels:
            bounds_lo.extend([cfg.sigma_gas_bounds[0], cfg.v0_gas_bounds[0]])
            bounds_hi.extend([cfg.sigma_gas_bounds[1], cfg.v0_gas_bounds[1]])

        av0, yav0, ss0, vs0, gk0 = self._params_to_components(initial_guess)
        gas_str = ", ".join(
            f"g{gid}=(sig={s:.1f}, v0={v:.1f})" for gid, (s, v) in gk0.items()
        ) if cfg.fit_emission_lines else ""
        logger.info(f"  LM optimization from: Av={av0:.2f}, YAV={yav0:.2f}, "
                     f"sig_star={ss0:.1f}, v0_star={vs0:.1f}"
                     + (f", {gas_str}" if gas_str else ""))

        if getattr(cfg, 'use_fast_engine', True):
            return self._optimize_fast(initial_guess, bounds_lo, bounds_hi)
        if self._photoionization == 'spop':
            return self._optimize_jax(initial_guess, bounds_lo, bounds_hi)
        else:
            return self._optimize_scipy_fallback(
                initial_guess, bounds_lo, bounds_hi
            )




    def _build_fast_constants(self):
        """
        Assemble fast_engine constants for the current mode.

        Cached on (mode, Av_neb) — Av_neb is the only thing the outer
        self-consistency loop changes between optimize() calls, so a re-fit
        after update_Av_neb() correctly rebuilds while repeated calls at the
        same Av_neb reuse the precomputed FFT and nebular template.

        The weights are the other thing that changes, and they are not in the
        key: assigning to inv_sigma clears the key instead, so a re-fit after
        outlier clipping rebuilds. See the inv_sigma property.
        """
        from brainsp import fast_engine as fe

        key = (self._photoionization, float(self._last_Av_neb))
        if getattr(self, '_fast_key', None) == key:
            return self._fast_C, self._fast_S

        cfg = self.cfg
        lt = self._line_table
        mode = self._photoionization

        kw = dict(mode=mode)
        if mode != 'spop':
            if self.nebular_model is None:
                raise ValueError(
                    f"photoionization_mode='{mode}' requires a NebularModel"
                )
            nm = self.nebular_model
            from brainsp import reddening_laws
            k_norm = float(reddening_laws.get_k_lambda(
                jnp.array([cfg.norm_wavelength]), cfg.reddening_law)[0])

            from brainsp.emission_lines import species_of
            H_idx, H_rat, H_dk = [], [], []
            for i, gname in enumerate(lt['names']):
                short = species_of(gname)
                if short in nm._H_ratios:
                    wl_L, ratio = nm._H_ratios[short]
                    H_idx.append(i)
                    H_rat.append(ratio)
                    H_dk.append(float(reddening_laws.get_k_lambda(
                        jnp.array([wl_L]), cfg.reddening_law)[0]) - k_norm)

            kw.update(
                qh_ratios=self.qh_ratios,
                neb_template=nm.template,
                k_norm=k_norm,
                av_neb=float(self._last_Av_neb),
                f_capture=nm._f_capture,
                H_gas_idx=H_idx, H_ratios=H_rat, H_dk=H_dk,
                C_Hbeta=(nm._alpha_Hbeta / nm._alpha_B)
                        * nm._hnu_Hbeta * nm._f_capture,
                sc_weight=cfg.sc_penalty_weight,
                sc_damping=cfg.sc_damping,
                n_neb_iter=getattr(cfg, 'n_nebular_iterations', 4),
            )

        kw.update(
            solver=getattr(cfg, 'linear_solver', 'active_set'),
            simplex=getattr(cfg, 'enforce_simplex', False),
            fit_lines=bool(cfg.fit_emission_lines),
        )
        C, S = fe.build_constants(
            self.M_star, self.lam, self.flux, self.inv_sigma, self.velscale,
            self._dk, self._D_full, self.ssp_yav_flags, lt, **kw)
        self._fast_key, self._fast_C, self._fast_S = key, C, S
        return C, S

    def _optimize_fast(self, initial_guess, bounds_lo, bounds_hi):
        """Unified JAX LM — serves 'spop', 'nepop' and 'photopop' alike."""
        from brainsp import fast_engine as fe

        cfg = self.cfg
        lt = self._line_table
        C, S = self._build_fast_constants()
        n_params = len(initial_guess)

        best_params, best_chi2, niter, nfev, converged = fe.lm_optimize(
            C, jnp.array(initial_guess), jnp.array(bounds_lo),
            jnp.array(bounds_hi), S, n_params,
            max_iter=int(getattr(cfg, 'lm_max_iter', 200)),
            ftol=cfg.lm_ftol, xtol=cfg.lm_xtol,
            diff_step=1e-3,
            central=getattr(cfg, 'central_differences', False),
        )

        residual, weights, Q_eff, M_full, _P = fe.forward(C, best_params, S)
        M_star_dust = M_full[:, :self.n_star]
        if S.fit_lines:
            M_gas_dust = M_full[:, self.n_star:self.n_star + S.n_gas]
            gas_names = list(lt['names'])
            gas_group_labels = list(lt['group_labels'])
        else:
            M_gas_dust = M_full[:, self.n_star:self.n_star]
            gas_names, gas_group_labels = [], []
        neb_spec = C.neb_shape * Q_eff

        self._last_M_star_dust = M_star_dust
        self._last_M_gas_dust = M_gas_dust
        self._last_weights = weights
        self._last_gas_names = gas_names
        self._last_gas_group_labels = gas_group_labels
        self._last_Q_eff = float(Q_eff)
        self._last_nebular_spectrum = neb_spec

        _, _, _, _, group_kin = self._params_to_components(
            np.asarray(best_params))

        return {
            'params': np.asarray(best_params),
            'weights': weights,
            'chi2': float(best_chi2),
            'M_star_dust': M_star_dust,
            'M_gas_dust': M_gas_dust,
            'gas_names': gas_names,
            'gas_group_labels': gas_group_labels,
            'group_kinematics': group_kin,
            'n_groups': self._n_groups,
            'success': bool(converged),
            'message': f'fast-engine LM ({self._photoionization}) '
                       f'{"converged" if converged else "max iter"} '
                       f'in {int(niter)} iterations',
            'nfev': int(nfev),
            'Q_eff': float(Q_eff),
            'nebular_spectrum': neb_spec,
            'Av_neb': self._last_Av_neb,
        }

    def _optimize_jax(self, initial_guess, bounds_lo, bounds_hi):
        """100% JAX Levenberg-Marquardt optimization."""
        cfg = self.cfg
        lt = self._line_table
        n_params = len(initial_guess)

        x0 = jnp.array(initial_guess)
        lo = jnp.array(bounds_lo)
        hi = jnp.array(bounds_hi)

        max_lm_iter = int(getattr(cfg, 'lm_max_iter', 200))

        best_params, best_chi2, niter, nfev, converged = _lm_optimize(
            x0, lo, hi,
            self.M_star, self.lam, self.flux, self.inv_sigma, self.velscale,
            self._dk, self._D_full, self.ssp_yav_flags,
            lt['rest_wavelengths'], lt['ratios'],
            lt['n_components'], lt['group_indices'],
            self._max_comp,
            max_iter=max_lm_iter,
            ftol=cfg.lm_ftol, xtol=cfg.lm_xtol, diff_step=1e-3,
        )

        weights, fitted, M_star_dust, M_gas_dust, residual = _forward_full(
            best_params, self.M_star, self.lam, self.flux, self.inv_sigma,
            self.velscale, self._dk, self._D_full, self.ssp_yav_flags,
            lt['rest_wavelengths'], lt['ratios'],
            lt['n_components'], lt['group_indices'], self._max_comp,
        )

        chi2 = float(jnp.sum(residual ** 2))

        self._last_M_star_dust = M_star_dust
        self._last_M_gas_dust = M_gas_dust
        self._last_weights = weights
        self._last_gas_names = list(lt['names'])
        self._last_gas_group_labels = list(lt['group_labels'])

        _, _, _, _, group_kin_best = self._params_to_components(
            np.asarray(best_params)
        )

            'weights': weights,
            'chi2': chi2,
            'M_star_dust': M_star_dust,
            'M_gas_dust': M_gas_dust,

    def _optimize_scipy_fallback(self, initial_guess, bounds_lo, bounds_hi):
        """Scipy fallback for photoionization modes (iterative nebular)."""
        from scipy.optimize import least_squares

        cfg = self.cfg

        result = least_squares(
            self._evaluate,
            initial_guess,
            bounds=(bounds_lo, bounds_hi),
            method='trf',
            x_scale='jac',
            loss='linear',
            diff_step=1e-3,
            ftol=cfg.lm_ftol,
            xtol=cfg.lm_xtol,
            max_nfev=cfg.lm_max_nfev,
        )

        best_params = result.x
        M_star_dust, M_gas_dust, weights, gas_names = self._evaluate(
            best_params, return_components=True
        )
        resid = self._evaluate(best_params)
        chi2 = np.sum(resid ** 2)

        _, _, _, _, group_kin_best = self._params_to_components(best_params)

        return {
            'params': best_params,
            'weights': weights,
            'chi2': chi2,
            'M_star_dust': M_star_dust,
            'M_gas_dust': M_gas_dust,
            'gas_names': gas_names,
            'gas_group_labels': self._last_gas_group_labels,
            'group_kinematics': group_kin_best,
            'n_groups': self._n_groups,
            'success': result.success,
            'message': result.message,
            'nfev': result.nfev,
            'Q_eff': self._last_Q_eff,
            'nebular_spectrum': self._last_nebular_spectrum,
            'Av_neb': self._last_Av_neb,
        }
