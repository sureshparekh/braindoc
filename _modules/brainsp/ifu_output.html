"""
BRAIN — the megacube's extensions, and the compatibility contract.

A megacube is the cube that was fitted with the fit appended to it as extra
extensions. The names, the mixed-case spelling, the DATA<n> cards inside
PoPBins and the linear wavelength axis on FLXOBS follow the published MaNGA
megacubes of Riffel et al. 2023 (MNRAS 524, 5640), because tooling written against those files reads
extensions by name and takes its wavelength axis from FLXOBS's CRVAL3 and
CD3_3 with no fallback. A BRAIN megacube that spells one of them differently,
or leaves the WCS off, does not open at all.

Two consequences run through this module. Extension names are set as header
cards rather than through astropy's `name=`, which upper-cases them.  And
FLXOBS and FLXSYN are written on a linear grid -- the only place BRAIN
resamples an observed flux, for the output copy alone and never for the fit
-- while BRAIN's own spectral extensions stay on the grid the cube arrived
on, where no resolution is thrown away.

The map quantities themselves are computed from the same per-template light
and mass numbers io_utils.population_statistics gives the summary file, so a
map value and that spaxel's summary line are the same arithmetic.
"""

import os
import re

import numpy as np
from astropy.io import fits

AUTHORS = "Suresh Parekh (suresh.parekh@ufrgs.br)"
SOFTWARE = "BRAIN v1.0"
FORMAT_AFTER = "Megacube layout of Riffel et al. 2023, MNRAS 524, 5640"

LINEAR_WAVELENGTH_STEP = 1.0

DEFAULT_AGE_BINS = (
    ("xyy", 9.0e5, 1.0e7),
    ("xyo", 1.4e7, 5.6e7),
    ("xiy", 1.0e8, 5.0e8),
    ("xii", 6.3e8, 8.0e8),
    ("xio", 8.9e8, 2.0e9),
    ("xo", 5.0e9, 1.3e10),
)

def named(hdu, extname, summary=""):
    """
    Give an HDU a name with its capitalisation intact.

    astropy's `name=` argument upper-cases what it is given, so an HDU built
    that way is called POPBINS and not PoPBins. Lookup by name is
    case-insensitive and would not notice, but anything that compares the
    string -- a listing, a filter over extension names, a reader written
    against the published megacubes -- would.
    """
    hdu.header["EXTNAME"] = extname
    if summary:
        hdu.header["SUMMARY"] = summary
    hdu.header["AUTHORS"] = AUTHORS
    hdu.header["SOFTWARE"] = SOFTWARE
    hdu.header["FORMAT"] = FORMAT_AFTER
    return hdu

def linear_wavelength_grid(wavelength, step=LINEAR_WAVELENGTH_STEP):
    """
    The linear grid the observed and synthetic spectra are written on.

    It starts at the first whole multiple of the step inside the fitted
    range and ends at the last one, so that CRVAL3 is a round number and
    every point of the grid is covered by data. The step defaults to 1 A,
    which is what the published megacubes use.
    """
    step = float(step)
    if step <= 0:
        raise ValueError("The linear wavelength step must be positive.")
    lo = float(np.min(wavelength))
    hi = float(np.max(wavelength))
    first = np.ceil(lo / step) * step
    n = int(np.floor((hi - first) / step)) + 1
    if n < 2:
        raise ValueError(
            f"A step of {step} A leaves {n} points between {lo:.1f} and "
            f"{hi:.1f} A, which is not a spectrum.")
    return first + step * np.arange(n, dtype=np.float64)

def wcs_cards(hdu, grid):
    """The three cards megacube viewers build their wavelength axis from."""
    step = float(grid[1] - grid[0])
    hdu.header["CRPIX3"] = 1
    hdu.header["CRVAL3"] = float(grid[0])
    hdu.header["CD3_3"] = step
    hdu.header["WAT3_001"] = "wtype=linear axtype=wave"
    return hdu

def resample_batch(wavelength, spectra, grid):
    """
    A batch of spectra moved onto another wavelength grid, flux-conserving.

    The same integral the fit itself uses to reach its log grid, not a linear
    interpolation: between grids of unequal spacing the two are not
    approximations of each other, and on this repository's reference spectrum
    interpolation was wrong by 18.9 per cent at its worst pixel.

    JAX is imported here rather than at the top of the module so that naming
    an extension, reading a set of age bins or validating a pipeline costs
    nothing: the runner and the desktop interface both do all three long
    before any fitting starts, and importing JAX to do them would add seconds
    to every `check`.
    """
    import jax
    import jax.numpy as jnp
    from brainsp.spectral_engine import flux_conserving_resample

    source = jnp.asarray(np.asarray(wavelength, dtype=np.float64))
    target = jnp.asarray(np.asarray(grid, dtype=np.float64))
    batch = jnp.asarray(np.asarray(spectra, dtype=np.float64))
    resample = jax.vmap(lambda y: flux_conserving_resample(source, y, target))
    return np.asarray(resample(batch))

def age_bin_membership(ages_yr, bins, is_ssp=None):
    """
    Which templates belong to each age bin, as a boolean (n_bin, n_ssp).

    The convention is the published one: min exclusive, max inclusive,
    `min < age <= max`. It matters, because the standard bin sets put edges
    exactly on template ages, and a `>=` on the low edge would count such a
    template in two bins at once.

    A template that is not a stellar population -- the power law, when
    `include_power_law` is on -- is never a member of any bin. Its light
    belongs under its own card, and adding it to a population bin would
    report accretion-disc light as stars.
    """
    ages = np.asarray(ages_yr, dtype=np.float64)
    stellar = (np.ones(ages.shape, dtype=bool) if is_ssp is None
               else np.asarray(is_ssp, dtype=bool))
    return np.array([(ages > float(lo)) & (ages <= float(hi)) & stellar
                     for _, lo, hi in bins], dtype=bool)

def binned(percentages, membership):
    """Per-bin sums of a (n_spec, n_ssp) percentage array, as (n_bin, n_spec)."""
    values = np.asarray(percentages, dtype=np.float64)
    return np.array([np.sum(values[:, member], axis=1)
                     for member in membership])

def featureless_parameter(short_name):
    """
    What a featureless component's name says it is: ("FC", exponent) for a
    power law F_nu ~ nu^-alpha, ("BB", temperature in K) for a blackbody, or
    (None, None).

    FC0.25 and FC1.50 give the exponent as written; the shipped library's
    Power_150 writes it in hundredths; BB800 is an 800 K blackbody.
    """
    name = str(short_name).strip().upper()
    if name.startswith("POWER"):
        digits = "".join(c for c in name if c.isdigit())
        return ("FC", int(digits) / 100.0) if digits else (None, None)
    match = re.match(r"^(FC|BB)[_ ]?([0-9]*\.?[0-9]+(?:E[+-]?[0-9]+)?)$",
                     name)
    if match:
        return match.group(1), float(match.group(2))
    return None, None

SOLAR_LUMINOSITY = 3.826e33          # erg/s
MEGAPARSEC_CM = 3.08567758e24

def distance_factor(distance_mpc):
    """4 pi d^2 / L_sun: an apparent mass (flux / template flux per Msun)
    times this, and times the flux unit, is a mass in solar masses."""
    return 4.0 * np.pi * (float(distance_mpc) * MEGAPARSEC_CM) ** 2 \
        / SOLAR_LUMINOSITY

def flux_unit_scale(bunit):
    """
    The number a flux unit carries: 1e-17 for MaNGA's '1E-17 erg/s/cm^2/
    Angstrom/spaxel', 1 for a JWST cube's 'erg/s/cm^2/Angstrom'.
    """
    match = re.match(r"^\s*([0-9]*\.?[0-9]+(?:[eE][+-]?[0-9]+)?)\s*[*x]?\s*",
                     str(bunit or ""))
    return float(match.group(1)) if match else 1.0

def unbinned_percent(percentages, membership, is_ssp=None):
    """
    How much of each spaxel's light or mass no age bin claims.

    Age bins as published have gaps between them, and a template library
    whose ages differ from the one the bins were drawn for can drop a lot of
    light into those gaps. That is not visible in the maps themselves --
    every bin looks reasonable -- so the number is computed here and
    reported.
    """
    values = np.asarray(percentages, dtype=np.float64)
    unclaimed = ~np.any(membership, axis=0)
    if is_ssp is not None:
        # Featureless (AGN) light is in no bin by design, not a gap.
        unclaimed &= np.asarray(is_ssp, dtype=bool)
    return np.sum(values[:, unclaimed], axis=1)

def mean_log_age(percentages, ages_yr, is_ssp=None, over_all=False):
    """
    The weighted mean of log10(age), which is not the log of the mean age.

    Published maps report <log age>, so
    a mean age computed the other way is a different quantity that looks like
    the same one. Templates with a non-positive age -- a power law has no age
    -- carry no term.

    over_all chooses the normalisation. False divides by the weight of the
    stellar templates alone, the mean age of the stars. True divides by the
    weight of every component, featureless ones included, which is how the
    megacubes of Riffel et al. (2023) define Mage_L: there 10 per cent of AGN
    light lowers the map by about 0.94 dex.
    """
    values = np.asarray(percentages, dtype=np.float64)
    ages = np.asarray(ages_yr, dtype=np.float64)
    usable = ages > 0
    if is_ssp is not None:
        usable &= np.asarray(is_ssp, dtype=bool)
    weight = values[:, usable]
    total = np.sum(values if over_all else weight, axis=1)
    logs = np.log10(ages[usable])
    return np.where(total > 0,
                    np.sum(weight * logs[None, :], axis=1)
                    / np.where(total > 0, total, 1.0), 0.0)

def mean_metallicity(percentages, metallicities, is_ssp=None,
                     over_all=False):
    """
    The weighted mean of Z as a mass fraction, not of [Z/H].

    Linear Z is what the published maps carry, and the two differ by more
    than a constant because the average of a logarithm is not the logarithm
    of the average. over_all as in mean_log_age.
    """
    values = np.asarray(percentages, dtype=np.float64)
    metals = np.asarray(metallicities, dtype=np.float64)
    usable = (np.ones(metals.shape, dtype=bool) if is_ssp is None
              else np.asarray(is_ssp, dtype=bool))
    weight = values[:, usable]
    total = np.sum(values if over_all else weight, axis=1)
    return np.where(total > 0,
                    np.sum(weight * metals[usable][None, :], axis=1)
                    / np.where(total > 0, total, 1.0), 0.0)

def signal_to_noise(flux, inverse_sigma, window):
    """
    S/N per pixel averaged over the normalisation window.

    flux and inverse_sigma are the fit's own arrays, so this is the
    signal-to-noise the fitter sees rather than a second estimate of it made
    from the file. Pixels the fit excluded carry inverse_sigma 0 and are left
    out; a spaxel with no usable pixel in the window reports 0.
    """
    f = np.asarray(flux, dtype=np.float64)[:, window]
    s = np.asarray(inverse_sigma, dtype=np.float64)[:, window]
    used = s > 0
    n = np.sum(used, axis=1)
    return np.where(n > 0, np.sum(f * s * used, axis=1) / np.maximum(n, 1), 0.0)




def _fits_comment(index, name, comment):
    """
    A comment cut to what is left of the 80 columns after the value.

    astropy truncates an over-long comment and warns, which is a warning
    nobody reads and a card that says something slightly different from what
    the code asked for. Cutting it here keeps the two the same.
    """
    field = max(len(f"'{name:<8}'" if len(name) <= 8 else f"'{name}'"), 20)
    room = 80 - len(f"DATA{index}".ljust(8)) - len("= ") - field - len(" / ")
    return comment[:max(room, 0)]

def text_value(value, comment):
    """
    A string card value cut so that the card keeps its comment.

    A FITS card is eighty columns and holds the keyword, the value and the
    comment; a value cut to what fits the value field alone leaves the
    comment to be truncated instead, which is how a card ends up saying
    "see the BRAIN IFU documentati". The value is the part that varies, so
    it is the part that gives way.
    """
    text = str(value)
    room = max(65 - len(comment), 8)
    if len(text) <= room:
        return text
    cut = text[:room - 3]
    space = cut.rfind(" ")
    if space > room // 2:
        cut = cut[:space]
    return cut.rstrip(" ,;") + "..."

def pop_bins_hdu(planes, shape, positions, unfitted=None):
    """
    The map stack, as one 3D extension addressed by DATA<n> cards.

    planes is an ordered sequence of (name, comment, values), where values is
    one number per fitted spaxel; shape is (n_y, n_x) and positions is the
    (y, x) of each fitted spaxel.

    A spaxel that was not fitted is 0, which is what the published megacubes
    carry and what the viewers written against them scale their colour bars
    to -- NaN limits break those viewers outright. Since 0 is also a
    plausible value for several of these quantities, the stack carries a
    BadSpaxel plane of its own, so that the question "is this spaxel a
    measurement" can be answered inside the same extension as the maps, and
    not only from BADSPXL and STATUS.

    That plane is the one place where 0 is the wrong fill, and `unfitted`
    names the planes whose background is something else: a spaxel nobody
    fitted is a bad spaxel, so its BadSpaxel is 1 and not the 0 that would
    otherwise be left there by never writing to it. Getting this wrong is
    invisible -- the plane exists, it is full of zeros, and every unfitted
    spaxel reads as a converged fit.

    The card value is the quantity's name and the card's comment describes
    it, which is the convention the analysis tools index the stack by.
    """
    n_y, n_x = shape
    y, x = positions
    background = dict(unfitted or {})
    cube = np.zeros((len(planes), n_y, n_x), dtype=np.float64)
    hdu = fits.ImageHDU(data=cube)
    for index, (name, comment, values) in enumerate(planes):
        if name in background:
            cube[index] = float(background[name])
        cube[index, y, x] = np.asarray(values, dtype=np.float64)
        hdu.header[f"DATA{index}"] = (name, _fits_comment(index, name, comment))
    hdu.data = cube
    return named(hdu, "PoPBins",
                 "Synthesis Parameters & Binned Population Vectors")

def power_law_card(short_name):
    """
    The card a non-stellar component is reported under.

    The published cubes call an F_nu ~ nu^-1.5 power law FC1.50, and the
    shipped library calls the same component Power_150, so a power law named
    that way has its exponent read out of the name. Every other name is used
    unchanged: a blackbody BB800 is not a power law of exponent 8.00, and
    reading the digits out of it filed hot dust under FC. Either way the
    component is reported on its own card and never inside a population bin.
    """
    name = str(short_name)
    digits = "".join(c for c in name if c.isdigit())
    if name.upper().startswith("POWER") and digits:
        return f"FC{int(digits) / 100.0:.2f}"
    return name[:16]




TRUST_GOOD = 0

TRUST_SN_CUT = 1

TRUST_NOT_CONVERGED = 2

TRUST_NO_VALUE = 4

TRUST_MEANINGS = (
    (TRUST_SN_CUT, "below the chosen signal-to-noise threshold"),
    (TRUST_NOT_CONVERGED, "not a converged fit"),
    (TRUST_NO_VALUE, "the plane holds no finite value here"),
)

SPATIAL_WCS_CARDS = (
    "WCSAXES", "CRPIX1", "CRPIX2", "CRVAL1", "CRVAL2", "CDELT1", "CDELT2",
    "CD1_1", "CD1_2", "CD2_1", "CD2_2", "PC1_1", "PC1_2", "PC2_1", "PC2_2",
    "CTYPE1", "CTYPE2", "CUNIT1", "CUNIT2", "RADESYS", "EQUINOX", "LONPOLE",
    "LATPOLE", "MJDREF",
)

def trust_mask(values, sn_keep=None, converged=None):
    """
    Why each spaxel of a map is or is not a measurement, as a bit field.

    0 is a spaxel worth reading and every other value names a reason it is
    not: the chosen signal-to-noise threshold threw it away, its fit did not
    converge, or the plane carries nothing finite there. The reasons are bits
    rather than an enumeration because a spaxel can fail all three at once
    and a map that reported only the first would understate the damage.

    This is the layer the maps need and did not have. A megacube writes 0
    into a spaxel nobody fitted, and 0 is a plausible value for most of these
    quantities, so a map without a companion mask cannot be read safely by
    anything -- including a person.
    """
    data = np.asarray(values, dtype=np.float64)
    mask = np.zeros(data.shape, dtype=np.uint8)
    if sn_keep is not None:
        mask |= np.where(np.asarray(sn_keep, dtype=bool), 0,
                         TRUST_SN_CUT).astype(np.uint8)
    if converged is not None:
        mask |= np.where(np.asarray(converged, dtype=bool), 0,
                         TRUST_NOT_CONVERGED).astype(np.uint8)
    mask |= np.where(np.isfinite(data), 0, TRUST_NO_VALUE).astype(np.uint8)
    return mask

def trusted_values(values, mask):
    """The map with every untrustworthy spaxel blanked, as NaN."""
    data = np.array(np.asarray(values, dtype=np.float64))
    data[np.asarray(mask, dtype=np.uint8) != TRUST_GOOD] = np.nan
    return data

def spatial_wcs(header):
    """
    The celestial part of a cube's world coordinate system, as plain cards.

    Only the two spatial axes are carried over: the third axis of a cube is
    wavelength and a two-dimensional map that kept CRVAL3 would claim an axis
    it does not have. A header with no celestial solution yields nothing,
    which is the honest answer and is what the exported file then says.
    """
    out = {}
    if header is None:
        return out
    for card in SPATIAL_WCS_CARDS:
        if card in header:
            out[card] = header[card]
    return out




def _settings_cards(hdu, settings):
    """Record what produced a map, one card each, comments intact."""
    for key, (value, comment) in settings.items():
        hdu.header[key] = (text_value(value, comment), comment)
    return hdu

def plane_hdus(values, mask, *, plane, cube, units="", wcs=None,
               settings=None):
    """
    One map plane as a FITS file: the values, and the mask that qualifies them.

    The primary array is the map as the interface draws it, which means
    NaN wherever the mask says the spaxel is not a measurement, so a reader
    that does nothing but `imshow` the primary sees what the interface saw.
    The TRUST extension carries the reason bits, so a reader that wants the
    excluded spaxels back knows exactly which they are and why each one went.

    The header records which plane, which cube and which settings produced
    the file, because a map in isolation is a picture of nothing: the same
    array means different things under a different signal-to-noise cut, and
    a file that does not say which cut it was written under cannot be
    checked against the cube it came from.
    """
    import datetime

    data = trusted_values(values, mask)
    primary = fits.PrimaryHDU(data=np.asarray(data, dtype=np.float64))
    for card, value in (wcs or {}).items():
        primary.header[card] = value
    primary.header["BUNIT"] = (str(units), "unit of the map")
    primary.header["PLANE"] = (text_value(plane, "the plane this map is"),
                               "the plane this map is")
    primary.header["CUBEFILE"] = (text_value(os.path.basename(cube),
                                             "the file it was read from"),
                                  "the file it was read from")
    primary.header["CUBEPATH"] = (text_value(cube, "where that file was"),
                                  "where that file was")
    primary.header["DATEEXP"] = (
        datetime.datetime.now(datetime.timezone.utc).strftime(
            "%Y-%m-%dT%H:%M:%S"), "UTC, when this map was written")
    primary.header["BLANKED"] = (
        "NaN", "untrusted spaxels are NaN here, see TRUST")
    _settings_cards(primary, settings or {})
    primary.header["AUTHORS"] = AUTHORS
    primary.header["SOFTWARE"] = SOFTWARE

    trust = fits.ImageHDU(data=np.asarray(mask, dtype=np.uint8))
    trust.header["EXTNAME"] = "TRUST"
    trust.header["BUNIT"] = ("flag", "bit field, 0 is a measurement")
    for bit, meaning in TRUST_MEANINGS:
        trust.header[f"BIT{bit}"] = (meaning, f"set when bit {bit} is 1")
    trust.header["NGOOD"] = (int(np.count_nonzero(mask == TRUST_GOOD)),
                             "spaxels with no reason against them")
    trust.header["NTOTAL"] = (int(np.asarray(mask).size), "spaxels in the map")
    for card, value in (wcs or {}).items():
        trust.header[card] = value
    trust.header["AUTHORS"] = AUTHORS
    trust.header["SOFTWARE"] = SOFTWARE
    return fits.HDUList([primary, trust])

def write_plane_fits(path, values, mask, **cards):

def map_figure(values, mask, *, plane, cube, units="", colour_map="viridis",
               vmin=None, vmax=None, scale="", cut="", orientation="",
               figsize=(5.0, 4.4)):
    """
    One map as a figure meant for print rather than for the screen.

    White ground, black text and a stated unit on every axis, because the
    interface's dark palette is a property of the interface and a figure
    dropped into a paper is read on paper. The colour maps on offer are all
    perceptually uniform, and the spaxels that are not measurements are grey
    AND hatched AND counted in the caption, so none of the three states is
    carried by colour alone.

    Nothing is drawn that is not data or the scale it is read against: no
    grid, no box around the legend, no frame around the figure.
    """
    import matplotlib
    from matplotlib.figure import Figure

    data = trusted_values(values, mask)
    n_y, n_x = data.shape
    extent = (-0.5, n_x - 0.5, -0.5, n_y - 0.5)

    figure = Figure(figsize=figsize, dpi=300)
    figure.set_facecolor("white")
    axes = figure.add_axes([0.13, 0.17, 0.70, 0.76])
    axes.set_facecolor("white")

    cmap = matplotlib.colormaps[str(colour_map)].copy()
    cmap.set_bad("#d9d9d9")
    image = axes.imshow(np.ma.masked_invalid(data), origin="lower", cmap=cmap,
                        vmin=vmin, vmax=vmax, extent=extent,
                        interpolation="nearest", aspect="equal")

    blanked = np.asarray(mask, dtype=np.uint8) != TRUST_GOOD
    if np.any(blanked) and not np.all(blanked):
        hatched = axes.contourf(np.arange(n_x), np.arange(n_y),
                                blanked.astype(np.float64), levels=[0.5, 1.5],
                                colors="none", hatches=["///"])
        hatched.set_edgecolor("#6e6e6e")
        hatched.set_linewidth(0.0)

    axes.set_xlabel("column   [spaxel]", fontsize=8, color="black")
    axes.set_ylabel("row   [spaxel]", fontsize=8, color="black")
    axes.tick_params(labelsize=7, colors="black", length=2.5, width=0.6)
    for spine in axes.spines.values():
        spine.set_color("black")
        spine.set_linewidth(0.6)
    axes.set_title(str(plane), fontsize=9, color="black", pad=6)

    bar_axes = figure.add_axes([0.855, 0.17, 0.035, 0.76])
    bar = figure.colorbar(image, cax=bar_axes)
    bar.set_label(units or "map value", fontsize=8, color="black")
    bar.outline.set_linewidth(0.6)
    bar_axes.tick_params(labelsize=7, colors="black", length=2.5, width=0.6)
    for text in bar_axes.get_yticklabels():
        text.set_fontfamily("monospace")
    for text in axes.get_xticklabels() + axes.get_yticklabels():
        text.set_fontfamily("monospace")

    blanked_n = int(np.count_nonzero(blanked))
    note = [os.path.basename(str(cube))]
    if scale:
        note.append(str(scale))
    if cut:
        note.append(str(cut))
    if blanked_n:
        plural = "" if blanked_n == 1 else "s"
        note.append(f"{blanked_n} spaxel{plural} blanked and hatched, "
                    f"not a measurement")
    if orientation:
        note.append(str(orientation))
