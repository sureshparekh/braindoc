"""
BRAIN v1 — Pure JAX Datacube Fitter
Fits all valid spaxels using jax.vmap batches.
Runs on one device, GPU or CPU, chosen with --device; a batch of spaxels is one
vmapped call on that device.
Outputs a MEGACUBE-format FITS file.

One template matrix is shared by the whole cube, so the line-spread-function
correction is applied to it once, after preprocessing and before the first
batch is compiled: it costs nothing per spaxel and does not change any array
shape, so nothing recompiles. lsf_decision_pixels says which pixels the
decision is taken over.

The KINEMATICS plane for sigma_star is what the kernel applied to the library,
an observed dispersion. When 'instrumental_resolution' is set, an LSFCORR
extension carries the dispersion with the remaining difference in line-spread
functions taken out in quadrature, the difference itself, and a per-spaxel flag
saying whether that is a measurement, an upper limit, or not reportable. When
it is not set, no LSFCORR extension is written and the file is what it has
always been: there is no plane of corrected dispersions that is secretly a copy
of the uncorrected one.
"""

def select_device(device):
    """
    Put JAX on the device asked for, and refuse to go on if it is not there.

    This module imports JAX at the top, so by the time the command line has
    been parsed JAX has already read JAX_PLATFORMS from the environment, and
    writing the variable then does nothing. That is what --device cpu used to
    do: it logged "CPU mode enabled" and fitted on the GPU. The platform is
    set through JAX's own configuration instead, which is honoured as long as
    no device has been touched yet, and the backend that actually starts is
    compared with the request, as brain_child does for the desktop interface.

    None means auto: whatever JAX picks, which is the GPU when one is visible.
    """
    if device in PLATFORM_FOR_DEVICE:
        jax.config.update("jax_platforms", PLATFORM_FOR_DEVICE[device])
    backend = jax.default_backend()
    if device in BACKENDS_FOR_DEVICE and backend not in BACKENDS_FOR_DEVICE[device]:
        raise SystemExit(
            f"--device {device} was asked for and JAX started on {backend}. "
            f"Nothing was fitted.")
    return backend

def get_device_info():
    dev = jax.devices()[0]
    backend = dev.platform
    mem_gb = 0.0
    if backend == 'gpu':
        try:
            s = dev.memory_stats()
            if s: mem_gb = s.get('bytes_limit', 0) / 1e9
        except: pass
        if mem_gb == 0:
            try:
                r = subprocess.run(['nvidia-smi','--query-gpu=memory.total','--format=csv,noheader,nounits'],
                                   capture_output=True, text=True, timeout=5)
                if r.returncode == 0: mem_gb = float(r.stdout.strip().split('\n')[0]) / 1024
            except: mem_gb = 8.0
    return backend, mem_gb, dev

def out_of_memory(exc):
    """
    Whether a failure is the device running out of memory, and not a fault.

    XLA reports it as RESOURCE_EXHAUSTED from whichever kernel asked last, so
    the only thing that distinguishes it from a real error is the text. The
    test is deliberately narrow: anything it does not recognise is re-raised
    rather than retried at half the batch size, because retrying a genuine
    fault would turn one loud failure into a sequence of quiet ones.
    """
    text = f"{type(exc).__name__}: {exc}"
    return "RESOURCE_EXHAUSTED" in text or "Out of memory" in text




# On a CPU a batch buys no parallelism XLA does not already find inside each
# spaxel's linear algebra, and every spaxel of a batch iterates until the
# slowest has converged, so small batches are fastest. Measured on 36 spaxels
# of the mock cube, spop, 64 cores: batch 4 1.16 s/spaxel (compile 8.8 s),
# 8 1.22 (12.7 s), 16 1.33 (16.9 s), 55 1.63 (45.9 s); the fits identical.
CPU_BATCH = 4


# ---------------------------------------------------------------------------
# CPU: one worker process per core
# ---------------------------------------------------------------------------
#
# One process cannot use a many-core CPU for this problem: XLA's threads
# speed up each spaxel's linear algebra a little, and a vmapped batch makes
# every spaxel wait for the slowest. Measured on 36 spaxels of the mock cube
# (spop): one process gave 0.9 spaxels/s whether it had 4, 16 or 64 cores,
# and a single core, fitting one spaxel at a time, gave 0.55. So the cube is
# split over worker processes, each pinned to one thread, which scales with
# the cores there are.

# Memory one worker needs, in GB: the interpreter, JAX, the constants and the
# compiled fit. Measured at about 0.7 GB; this leaves room.
WORKER_GB = 1.2

def available_memory_gb():
    """Memory the operating system says is available, in GB (or None)."""
    try:
        with open("/proc/meminfo") as handle:
            for line in handle:
                if line.startswith("MemAvailable:"):
                    return int(line.split()[1]) / 1024 ** 2
    except OSError:
        pass
    return None

def cpu_worker_count(requested, n_spaxels):
    """
    How many CPU workers to start.

    "auto": every core but one (leaving the machine usable), no more than
    the memory available allows at WORKER_GB each, and no more than there are
    spaxels. An integer is taken as given (1 means no workers: the fit runs
    in this process).
    """
    # The cores this process may run on: what SLURM, taskset or a container
    # gave it, not every core the machine has.
    try:
        cores = len(os.sched_getaffinity(0))
    except AttributeError:
        cores = os.cpu_count() or 1
    if str(requested).lower() != "auto":
        n = int(requested)
    else:
        n = max(1, cores - 1)
        memory = available_memory_gb()
        if memory is not None:
            n = min(n, max(1, int(memory / WORKER_GB)))
    return max(1, min(n, int(n_spaxels)))




def _fit_cpu_workers(n_workers, fl, isl, ist, constants, options, fitter,
                     on_result):
    """
    Fit every spaxel on `n_workers` single-threaded worker processes.

    Spaxels are dealt out in turn (worker k gets spaxels k, k + n, ...), so
    neighbouring spaxels, which cost about the same, spread over all the
    workers and they finish together. `on_result(index, outputs)` is called
    here, in this process, as each spaxel comes back.
    """
    import pickle
    import queue
    import struct
    import tempfile
    import threading

    env = dict(os.environ, JAX_PLATFORMS="cpu", OMP_NUM_THREADS="1",
               XLA_FLAGS="--xla_cpu_multi_thread_eigen=false "
                         "intra_op_parallelism_threads=1")
    package_parent = os.path.dirname(os.path.dirname(os.path.abspath(
        __file__)))
    env["PYTHONPATH"] = os.pathsep.join(
        [package_parent] + [p for p in env.get("PYTHONPATH", "").split(
            os.pathsep) if p])
    host = [np.asarray(c) if hasattr(c, "shape") else c for c in constants]
    folder = tempfile.mkdtemp(prefix="brainsp-workers-")
    results = queue.Queue()
    workers, readers = [], []
    try:
        for k in range(n_workers):
            idx = np.arange(k, len(fl), n_workers)
            path = os.path.join(folder, f"job{k}.pkl")
            with open(path, "wb") as handle:
                pickle.dump(dict(constants=host, options=options,
                                 fitter=fitter, indices=idx,
                                 flux=np.asarray(fl[idx]),
                                 isig=np.asarray(isl[idx]),
                                 istat=np.asarray(ist[idx])), handle)
            proc = subprocess.Popen(
                [sys.executable, "-m", "brainsp.cube_worker", path],
                stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env)
            workers.append(proc)

            def read(proc=proc):
                stream = proc.stdout
                while True:
                    head = stream.read(4)
                    if len(head) < 4:
                        break
                    (n,) = struct.unpack("<I", head)
                    results.put(pickle.loads(stream.read(n)))
                results.put(None)

            thread = threading.Thread(target=read, daemon=True)
            thread.start()
            readers.append(thread)
        finished = 0
        received = 0
        while finished < n_workers:
            item = results.get()
            if item is None:
                finished += 1
                continue
            received += 1
            on_result(*item)
        for proc in workers:
            proc.wait()
        if received < len(fl):
            errors = "\n".join(p.stderr.read().decode(errors="replace")[-2000:]
                               for p in workers if p.returncode)
            raise RuntimeError(f"{len(fl) - received} spaxels came back from "
                               f"no worker.\n{errors}")
    finally:
        for proc in workers:
            if proc.poll() is None:
                proc.kill()
        import shutil
        shutil.rmtree(folder, ignore_errors=True)

def estimate_batch_size(mem_gb, n_pix, n_ssp, n_gas, n_params, is_gpu, max_cpu_cores=55, max_ram_gb=60.0):
    n_t = n_ssp + n_gas
    per = (2*n_params*(n_pix*n_ssp + n_pix*n_gas + n_t**2) + n_pix*n_t + n_pix*5 + n_t*5) * 8 * 4
    if is_gpu:
        avail = max(0.5e9, (mem_gb - 2.0) * 1e9)
        b = int(avail / per)
        b = 2 ** max(0, int(np.log2(max(1, b))))
        return min(max(1, b), 256)
    
    avail_cpu = max_ram_gb * 1e9
    b = int(avail_cpu / per)
    return min(max(1, b), max_cpu_cores, CPU_BATCH)

def lsf_decision_pixels(matrix, inv_sigma_batch):
    """
    The pixels the line-spread-function decision is taken over, for a cube.

    The whole cube is fitted with one template matrix, so the verdict on each
    group of templates — convolve it up to the data, or leave it at its native
    resolution — is reached once and binds every spaxel. A single spaxel's mask
    is the wrong unit to reach it with: a convolution justified on one spaxel's
    window would be forced onto spaxels that fit pixels where it is not.

    Taken here as the union of the good pixels of the spaxels that will be
    fitted, intersected with the pixels at least one template covers. The union
    is the conservative member of the family: prepare_templates refuses a group
    when the library is broader than the data anywhere in the mask, so a wider
    mask can only turn a convolution into a refusal, and a refusal changes
    nothing about the fit and is reported. The intersection would license a
    convolution over a window narrower than any spaxel actually fits. The
    coverage mask alone would drag in pixels every spaxel has discarded.

    The coverage term is not optional. Pixels beyond the reach of the library
    are identically zero in every template, and letting them into the decision
    binds it to wavelengths that carry no template flux at all.
    """
    covered = np.any(np.abs(np.asarray(matrix)) > 0, axis=1)
    used = np.any(np.asarray(inv_sigma_batch) > 0, axis=0)
    return covered & used

def to_rest_frame(wl, cfg, logger):
    """
    Put an observed-frame cube onto rest wavelengths.

    The two entry points take different frames, and that asymmetry is the
    kind of thing this code fails silently on, so it is stated here and in
    brain_setup.yaml rather than left implicit. The scalar path expects a
    spectrum you de-redshifted yourself; a datacube arrives straight from the
    telescope in the observed frame, so the cube paths do it before any
    spaxel is fitted.

    Applied to the wavelength axis rather than to the flux, and before the
    log grid is built, so every template, mask and line window downstream is
    in one frame. Nothing is interpolated and no flux is touched.

    cube_redshift and spectrum_redshift do opposite jobs and setting both is
    refused in brain_settings.validate, so reaching here with a non-zero
    value means exactly one correction is wanted.
    """
    z = float(getattr(cfg, 'cube_redshift', 0.0) or 0.0)
    if z == 0.0:
        return wl, 0.0
    logger.info(f"Rest frame: dividing the wavelength axis by 1+z with "
                f"z = {z:g} ({wl[0]:.0f}-{wl[-1]:.0f} A observed becomes "
                f"{wl[0] / (1.0 + z):.0f}-{wl[-1] / (1.0 + z):.0f} A rest)")
    return wl / (1.0 + z), z

def line_table_for(cfg, lam_grid, weight=None):
    """
    The emission-line design the cube path will fit, for this configuration.

    Named and separate so the cube path's answer to "which lines am I
    fitting" can be read without running a fit, and so it has one definition
    rather than being inlined where nothing can check it.

    Gated on cfg.fit_emission_lines, as brainv1 gates it. to_config already
    forces that false whenever nebular_mode is spop, so the single flag
    carries the rule that spop mode is purely stellar and needs no second
    test for the mode here. Without the gate the cube fitted fourteen gas
    templates in off mode, where the scalar path fits none, so the two were
    fitting different models and no map from a cube was of the quantity the
    scalar path reports.

    Returns the precompiled line table, the kinematic group labels in the
    order their fitted parameters are laid out, and the number of groups.
    With lines switched off the table is empty and the design matrix is
    stellar alone.

    The order comes from emission_lines.group_positions, which is the same
    expression the table itself uses to turn a group label into a position
    in the parameter vector, so the bounds built from these labels and the
    kinematics each line reads cannot drift apart.
    """
    reach = (cfg.v0_gas_bounds, cfg.sigma_gas_bounds[1])
    if not cfg.fit_emission_lines:
        return precompile_line_table(lam_grid, {}, {}, {}, *reach), [], 0

    lines, doublets = get_line_catalog(
        (float(lam_grid[0]), float(lam_grid[-1])), cfg.wavelength_scale)
    masked = []
    if weight is not None:
        # The wavelength mask is the same for every spaxel: a line it covers
        # has no data in any of them and is left out, as for a spectrum.
        from brainsp.emission_lines import drop_masked_lines
        lines, doublets, masked = drop_masked_lines(
            lines, doublets, lam_grid, weight, cfg.sigma_gas_init)
    groups, n_groups = assign_kinematic_groups(lines, doublets,
                                               cfg.kinematic_groups)
    labels = list(group_positions(groups))
    table = precompile_line_table(lam_grid, lines, doublets, groups, *reach)
    # What was fitted, for the line products to name each member by.
    table["doublets"] = doublets
    table["masked"] = masked
    return table, labels, n_groups

def quality_mask(data, name, logger=None):
    """
    A quality-mask cube as the integers its bits are tested on.

    BRAIN reads a quality mask as MaNGA's: a pixel is bad when bit 10
    (DONOTUSE, 1024) is set. A mask written as floating point (a JWST cube's
    MASK of 0.0 and 2.0) cannot be bit-tested, and the test raised a numpy
    TypeError mid-run. Whole-number floats are read as those integers, and a
    NaN entry as a bad pixel; anything else is refused with the reason. A mask
    whose non-zero values never carry bit 10 flags nothing, and the log says
    so rather than leave the user believing the mask was applied.
    """
    data = np.asarray(data)
    if not np.issubdtype(data.dtype, np.integer):
        values = data.astype(np.float64)
        finite = np.isfinite(values)
        if np.any(values[finite] != np.round(values[finite])):
            raise SystemExit(
                f"\nThe quality mask {name!r} holds non-integer values, so it "
                f"is not a bit mask. Leave 'Quality mask' empty (bad pixels "
                f"that are NaN in the flux are skipped anyway).\n")
        data = np.where(finite, values, MANGA_DONOTUSE).astype(np.int64)
        if logger:
            logger.info(f"Mask: {name} is stored as floating point; read as "
                        f"integers")
    if logger and np.any(data != 0) and not np.any(data & MANGA_DONOTUSE):
        shown = ", ".join(str(int(v)) for v in np.unique(data[data != 0])[:6])
        logger.warning(
            f"Mask: {name} has non-zero values ({shown}) but none carries "
            f"bit 10 (1024), the only bit BRAIN reads as bad, so it flags no "
            f"pixel. If they mark bad pixels, rewrite them as 1024.")
    return data

def preprocess_datacube(flux_cube, ivar_cube, mask_cube, wl_lin, lam_grid,
                        norm_wl, has_err=True, file_weight_lin=None,
                        spaxel_select=None, missing_error="exclude",
                        logger=None):
    """
    Every valid spaxel's flux and inverse sigma on the log grid, batched.

    This follows brainv1 step for step, because a cube fitted here and a
    spectrum fitted there must give the same answer, and every step where the
    two differ is a place they silently do not.

    The flux is moved onto the log grid by the same flux-conserving integral
    the scalar path uses, not by linear interpolation. The two are not
    approximations of each other: the integral redistributes flux between
    bins of unequal width, and on the reference spectrum the interpolated
    version was wrong by 18.9 per cent at its worst pixel.

    That choice forces the rest. Once the rebin blends neighbouring input
    pixels, an output pixel next to a rejected one holds some of its flux,
    so the quality mask has to be carried by the same operator -- the clean
    fraction -- and not by a nearest-neighbour lookup alone. Both carriers
    are kept, as in brainv1: the nearest-neighbour weight says which input
    pixel an output pixel sits on, the clean fraction says how much of its
    value came from input worth keeping.

    The error is filled only where the inverse variance carries no
    information, never where the mask rejects a pixel. Those are different
    questions and conflating them inflates the error at flagged pixels, which
    then bleeds into their neighbours through the interpolation and reaches
    pixels that were never in doubt.

    The resampling is vmapped over spaxels rather than looped, since the two
    grids are shared and only the flux differs.

    spaxel_select, when given, is a (n_y, n_x) boolean or integer plane and a
    spaxel is fitted only where it is true. That is how a signal-to-noise mask
    reaches the fit: SN_MASKS_<t> is 1 where the spaxel passed the threshold,
    so selection is intersected with the emptiness and quality tests here
    rather than replacing them -- a spaxel above the threshold whose flux is
    entirely flagged is still not fittable, and quietly fitting it would
    produce exactly the plausible-looking answer this code has to avoid.

    The per-pixel error on the log grid is returned alongside the inverse
    sigma, and is None when the cube carries no inverse variance. Outlier
    clipping needs it: its criterion is abs(resid) / err, while inv_sigma is
    sqrt(mask weight) / err, and the two agree only where the mask weight is
    exactly one. load_mask allows any non-negative weight, so a cube fitted
    with a weighted mask would otherwise clip at a different threshold from
    the same spectrum fitted on its own.

    missing_error says what a pixel with a finite flux but no usable error is.
    "exclude" (the default) leaves it out: in MaNGA an inverse variance of
    zero marks a pixel not to be trusted. "interpolate" keeps it, with the
    error interpolated from the spaxel's neighbouring errors, when the flux
    is finite and the quality mask does not flag it. Some cubes give no error
    for real data: the NIRSpec cubes of Riffel et al. carry NaN errors on 15
    per cent of their pixels, and on 45 per cent at 2.20-2.27 um, where the
    flux scatters exactly as it does elsewhere. Leaving those out discards
    half the red end, where hot dust and A_V are told apart.
    """
    n_wave, n_y, n_x = flux_cube.shape
    n_sp = n_y * n_x
    flux_flat = flux_cube.reshape(n_wave, n_sp).T.astype(np.float64)
    ivar_flat = (ivar_cube.reshape(n_wave, n_sp).T.astype(np.float64)
                 if ivar_cube is not None and has_err else None)
    mask_flat = mask_cube.reshape(n_wave, n_sp).T if mask_cube is not None else None

    valid = ((np.nansum(np.abs(flux_flat), axis=1) > 0)
             & (~np.all(np.isnan(flux_flat), axis=1)))
    if mask_flat is not None:
        valid &= ~np.all((mask_flat & MANGA_DONOTUSE) > 0, axis=1)
    if spaxel_select is not None:
        valid &= np.asarray(spaxel_select).reshape(n_sp) != 0
    vi = np.where(valid)[0]
    if len(vi) == 0:
        return None, None, None, vi, None

    fv = flux_flat[vi].copy()
    fv[np.isnan(fv)] = 0.0

    dq_good = np.ones(fv.shape, dtype=bool)
    have_ivar = np.ones(fv.shape, dtype=bool)
    if ivar_flat is not None:
        iv = ivar_flat[vi]
        have_ivar = np.isfinite(iv) & (iv > 0)
        dq_good &= have_ivar
    if mask_flat is not None:
        dq_good &= (mask_flat[vi] & MANGA_DONOTUSE) == 0

    if ivar_flat is not None:
        with np.errstate(divide='ignore', invalid='ignore'):
            err_lin = np.where(have_ivar,
                               1.0 / np.sqrt(np.where(have_ivar, iv, 1.0)), 0.0)
        flagged = ((mask_flat[vi] & MANGA_DONOTUSE) != 0
                   if mask_flat is not None else np.zeros(fv.shape, dtype=bool))
        restored = 0
        for i in range(err_lin.shape[0]):
            miss = ~have_ivar[i]
            if miss.any():
                keep = have_ivar[i]
                if missing_error == "interpolate" and keep.sum() > 1:
                    real = miss & np.isfinite(flux_flat[vi[i]]) & ~flagged[i]
                    err_lin[i, real] = np.interp(wl_lin[real], wl_lin[keep],
                                                 err_lin[i, keep])
                    dq_good[i, real] = True
                    restored += int(real.sum())
                    miss = miss & ~real
                err_lin[i, miss] = (np.median(err_lin[i, keep]) * 10.0
                                    if keep.any() else 1.0)
        if missing_error == "interpolate" and logger is not None:
            logger.info(f"missing_error 'interpolate': {restored} pixels with a "
                        f"finite flux and no error kept, their error "
                        f"interpolated from their neighbours "
                        f"({100 * restored / max(1, fv.size):.1f}% of all)")
    else:
        err_lin = None

    idx_n = np.argmin(np.abs(wl_lin - norm_wl))
    hw = 10
    nf = np.median(fv[:, max(0, idx_n - hw):min(n_wave, idx_n + hw + 1)], axis=1)
    bad_n = nf <= 0
    for i in np.where(bad_n)[0]:
        pos = fv[i][fv[i] > 0]
        nf[i] = np.median(pos) if len(pos) > 0 else 1.0
    fv = fv / nf[:, None]
    if err_lin is not None:
        err_lin = err_lin / nf[:, None]

    if file_weight_lin is None:
        file_weight_lin = np.ones(len(wl_lin), dtype=np.float64)
    file_weight_lin = np.asarray(file_weight_lin, dtype=np.float64)

    wl_j = jnp.asarray(np.asarray(wl_lin, dtype=np.float64))
    lam_j = jnp.asarray(np.asarray(lam_grid, dtype=np.float64))
    resample = jax.vmap(lambda y: flux_conserving_resample(wl_j, y, lam_j))

    fl = np.asarray(resample(jnp.asarray(fv)))

    usable = dq_good & (file_weight_lin > 0)[None, :]
    clean = np.asarray(resample(jnp.asarray(usable.astype(np.float64))))
    uncontaminated = clean >= 1.0 - 1e-8

    weight_lin = file_weight_lin[None, :] * dq_good
    nn = interp1d(np.asarray(wl_lin), np.arange(len(wl_lin), dtype=np.float64),
                  kind='nearest', bounds_error=False,
                  fill_value=-1.0)(np.asarray(lam_grid))
    inside = nn >= 0
    nn_idx = np.clip(nn, 0, len(wl_lin) - 1).astype(int)
    weight_log = np.where(inside[None, :], weight_lin[:, nn_idx], 0.0)
    weight_log = np.clip(weight_log, 0.0, None)
    good_log = (weight_log > 0) & uncontaminated
    sqrt_w = np.sqrt(weight_log)

    if err_lin is not None:
        err_log = np.empty((len(vi), len(lam_grid)), dtype=np.float64)
        for i in range(err_lin.shape[0]):
            err_log[i] = np.interp(lam_grid, wl_lin, err_lin[i])
        with np.errstate(divide='ignore', invalid='ignore'):
            isl = np.where(good_log & (err_log > 0),
                           sqrt_w / np.where(err_log > 0, err_log, 1.0), 0.0)
    else:
        err_log = None
        isl = sqrt_w * good_log

    good = np.any(isl > 0, axis=1)
    if not np.all(good):
        k = np.where(good)[0]
        fl, isl, nf, vi = fl[k], isl[k], nf[k], vi[k]
        if err_log is not None:
            err_log = err_log[k]
    return fl, isl, nf, vi, err_log




@partial(jax.jit, static_argnames=('max_comp', 'n_groups', 'solver',
                                   'simplex', 'max_iter', 'ftol', 'xtol',
                                   'n_clip', 'clip_threshold', 'have_errors',
                                   'clip_scaled'))
def _fit_batch(flux_b, isig_b, istat_b, M_star, lam, velscale, dk, D_full,
               yav_flags, rest_wl, ratios, n_comp, gids, max_comp, n_groups,
               blo, bhi, v0_grid, av0, sig_s0, sig_g0,
               solver="active_set", simplex=False,
               max_iter=200, ftol=1e-7, xtol=1e-7,
               n_clip=0, clip_threshold=3.0, have_errors=True,
               clip_scaled=False):
    """
    One batch of spaxels fitted, clipped and re-fitted, under one set of
    kernel launches.

    Clipping runs the schedule brainv1.clip_outliers_and_refit runs: every
    pass measures each fitted pixel's deviation from the current model in
    units of its own error, zeroes the weight beyond clip_threshold and
    re-optimises from the parameters already reached. It is a weight update,
    jnp.where on isig, so nothing changes shape and nothing recompiles per
    spaxel; a spaxel with no outlier keeps the array it had.

    istat_b is the reciprocal of that error, 1/err_log, zero where the cube
    reports no error. It is passed instead of being recovered from isig
    because isig carries the mask weight as well, and abs(resid) * isig equals
    the deviation in sigma only where that weight is one.

    The control flow is static, so every spaxel runs every pass whether or
    not it has anything to clip. A pass that clips nothing would re-optimise
    from a converged point and return it, but the result is selected on
    whether the pass clipped anything, so a spaxel that stops producing
    outliers is bit-for-bit the spaxel the previous pass left -- the same
    thing the scalar path's break statement achieves.
    """
    n_ssp = M_star.shape[1]

    def _fit_one(flux, isig, istat):
        def eval_v0(v0):
            sga = jnp.full(n_groups, sig_g0)
            return _grid_point_chi2(v0, M_star, lam, flux, isig, velscale,
                                    sig_s0, dk, av0, sga, rest_wl, ratios,
                                    n_comp, gids, max_comp, D_full,
                                    solver=solver, simplex=simplex)
        all_c2 = jax.vmap(eval_v0)(v0_grid)
        bv0 = v0_grid[jnp.argmin(all_c2)]
        x0 = jnp.concatenate([jnp.array([av0, 0.0, sig_s0, bv0]),
                               jnp.tile(jnp.array([sig_g0, bv0]), n_groups)])
        bp, bc2, ni, nf, conv = _lm_optimize(
            x0, blo, bhi, M_star, lam, flux, isig, velscale, dk, D_full,
            yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
            max_iter=max_iter, ftol=ftol, xtol=xtol, diff_step=1e-3,
            solver=solver, simplex=simplex)
        w, fitted, Msd, Mgd, res = _forward_full(
            bp, M_star, lam, flux, isig, velscale, dk, D_full,
            yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
            solver=solver, simplex=simplex)

        for _ in range(n_clip):
            resid = flux - fitted
            if have_errors:
                nsigma = outlier_nsigma(resid, istat, isig, clip_scaled)
            else:
                rms = jnp.std(jnp.where(isig > 0, resid, 0.0))
                nsigma = jnp.abs(resid) / jnp.maximum(rms, 1e-30)
            outliers = (nsigma > clip_threshold) & (isig > 0)
            clipped = jnp.any(outliers)
            isig = jnp.where(outliers, 0.0, isig)
            bp2, bc22, ni2, nf2, conv2 = _lm_optimize(
                bp, blo, bhi, M_star, lam, flux, isig, velscale, dk, D_full,
                yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
                max_iter=max_iter, ftol=ftol, xtol=xtol, diff_step=1e-3,
                solver=solver, simplex=simplex)
            bp = jnp.where(clipped, bp2, bp)
            bc2 = jnp.where(clipped, bc22, bc2)
            ni = jnp.where(clipped, ni2, ni)
            conv = jnp.where(clipped, conv2, conv)
            w, fitted, Msd, Mgd, res = _forward_full(
                bp, M_star, lam, flux, isig, velscale, dk, D_full,
                yav_flags, rest_wl, ratios, n_comp, gids, max_comp,
                solver=solver, simplex=simplex)

        stellar = Msd @ w[:n_ssp]
        gas = Mgd @ w[n_ssp:]
        neff = jnp.sum(isig > 0)
        return bp, w, fitted, stellar, gas, bc2, neff, conv, ni, isig

    return jax.vmap(_fit_one)(flux_b, isig_b, istat_b)

def spaxel_selection(hdu_list, hdu_flag, n_y, n_x, flux_cube=None, logger=None):
    """
    The per-spaxel selection plane a flag extension carries, or None.

    Which value of a two-dimensional flag means "fit this spaxel" is not
    settled in the wild, and getting it backwards is silent and total: the
    fit runs on empty sky, every spaxel converges on noise, and the cube is
    full of numbers. Both conventions are found in published megacubes:
    "anything different from 0 is a bad spaxel" is documented, and
    SN_MASKS_<t> planes are written that way -- measured on a real megacube,
    SN_MASKS_10 == 0 marks 145 spaxels of median S/N 13.6 and != 0 marks
    5331 of median S/N 0.00 -- while readers exist that do the opposite,
    skipping the spaxel when the flag is falsy.

    So the polarity is measured rather than assumed. The convention that
    selects the spaxels carrying signal is the right one, and the decision
    is logged with the evidence that produced it. Without a flux cube to
    measure against, the documented convention is used and the log says it
    was assumed.

    "Carrying signal" is most of the spectrum being positive, not any of it.
    On a sky-subtracted cube an empty spaxel is noise about zero, so its
    maximum is positive nearly always: measured on the raw FLUX extension of
    manga-8626-12704, "max > 0" calls 60.6% of the empty spaxels signal and
    leaves a 39-point margin, where requiring most wavelengths positive
    leaves 84. The weaker test reaches the same verdict here and would not
    survive a noisier cube.

    A three-dimensional flag rejects instead, per wavelength, non-zero being
    bad, and is handled by the mask path rather than here.
    """
    if hdu_flag is None:
        return None, None
    flag = np.asarray(hdu_list[hdu_flag].data)
    if flag.ndim == 3:
        return None, flag
    if flag.shape != (n_y, n_x):
        raise SystemExit(
            f"\nFlag extension {hdu_flag!r} is {flag.shape}, which is not the "
            f"{n_y}x{n_x} spatial grid of the cube.\n")

    zero_keeps = flag == 0
    nonzero_keeps = flag != 0

    def say(msg):
        if logger is not None:
            logger.info(msg)

    if flux_cube is None:
        say(f"Selection: {hdu_flag} polarity assumed, 0 means fit "
            f"(no flux cube available to measure it)")
        return zero_keeps, None

    with np.errstate(invalid='ignore'):
        has_signal = np.nanmean(np.asarray(flux_cube) > 0, axis=0) > 0.6

    def fraction_with_signal(sel):
        n = int(np.sum(sel))
        return (float(np.sum(has_signal & sel)) / n) if n else 0.0

    f_zero = fraction_with_signal(zero_keeps)
    f_nonzero = fraction_with_signal(nonzero_keeps)

    if f_zero >= f_nonzero:
        select, kept, other = zero_keeps, f_zero, f_nonzero
        meaning = "0 means fit"
    else:
        select, kept, other = nonzero_keeps, f_nonzero, f_zero
        meaning = "non-zero means fit"

    say(f"Selection: {hdu_flag} read as {meaning} -- "
        f"{int(np.sum(select))} spaxels selected, {kept:.1%} of them carrying "
        f"signal against {other:.1%} the other way round")

    if kept < 0.5:
        raise SystemExit(
            f"\nFlag extension {hdu_flag!r} does not select spaxels with data "
            f"under either reading: {f_zero:.1%} of the {int(np.sum(zero_keeps))} "
            f"spaxels flagged 0 carry signal, and {f_nonzero:.1%} of the "
            f"{int(np.sum(nonzero_keeps))} flagged non-zero do.\n"
            f"Fitting would run on empty sky and converge on noise without "
            f"complaining. Check that {hdu_flag!r} is the extension you meant.\n")

    return select, None

def pop_bin_planes(cfg, age_bins, fit, base, sfr_bins=None, fc_bins=None,
                   bb_bins=None, mass_factor=None):
    """
    Every plane of PoPBins, in order, as (card value, comment, values).

    The order is the published one: the featureless continuum first, then the
    age-binned light fractions, then the age-binned mass fractions, then the
    fixed tail of scalars ending at SNR. Quantities BRAIN reports and
    the published megacubes do not are appended after that tail, so a reader that indexes
    the stack by position rather than by card name still finds everything it
    knew about where it was.

    Age bins may overlap, so totals such as xy_Tot (xyy + xiy) are bins of
    their own. fc_bins and bb_bins group the featureless components by the
    power-law exponent or blackbody temperature their names give
    (ifu_output.featureless_parameter), lower edge exclusive, upper
    inclusive, as per cent of all the light.

    Mstar, Mpcross and the SFR planes are written only when mass_factor is
    given: 4*pi*d^2/L_sun times the cube's flux unit, which turns BRAIN's
    apparent masses (the observed flux over the library's flux per unit
    mass) into solar masses. Without a distance they would carry a unit
    wrong by that factor, about 3.5 dex for a MaNGA spaxel. The mass
    *fractions*, which carry no such factor, are in PoPVecsM and the
    <pop>_mass planes either way.

    The maps follow the definitions of the megacubes of Riffel et al.
    (2023) exactly, so that the two can be compared map for map:
      SFR_<bin> = Mstar * sum(Mini_j %, bin) / 100 / (max - min), in
        Msun/yr: the initial-mass fractions times the present stellar mass;
      Mage_L, MZ_L: the stellar sum over the light of EVERY component,
        featureless ones included (Mage_M, MZ_M are unaffected: a
        featureless component has no mass).
    """
    pops = fit['pops']
    params = fit['params']
    membership = ifu_output.age_bin_membership(
        base['ages_yr'], age_bins, base['is_ssp'])
    light = ifu_output.binned(pops['x_j'], membership)
    mass = ifu_output.binned(pops['mcor_pct'], membership)

    planes = []
    featureless = [i for i, is_ssp in enumerate(base['is_ssp']) if not is_ssp]
    for index in featureless:
        planes.append((ifu_output.power_law_card(base['short'][index]),
                       'Featureless (AGN) component, per cent of light',
                       pops['x_j'][:, index]))
    if len(featureless) > 1:
        # The AGN templates are nearly collinear in the optical, so only their
        # sum is a measurement; it gets its own card so nobody has to add them.
        planes.append(('AGN_light',
                       'All featureless (AGN) components, per cent of light',
                       pops['x_j'][:, featureless].sum(axis=1)))
    kinds = [ifu_output.featureless_parameter(base['short'][i])
             if not base['is_ssp'][i] else (None, None)
             for i in range(len(base['is_ssp']))]
    for family, bins, unit, text in (("FC", fc_bins, "", "Featureless continuum"
                                       ": {lo:.2f} < exp <= {hi:.2f}"),
                                      ("BB", bb_bins, " K", "Black body: "
                                       "{lo:g} < temp (K) <= {hi:g}")):
        for name, lo, hi in bins or []:
            members = [i for i, (fam, value) in enumerate(kinds)
                       if fam == family and lo < value <= hi]
            planes.append((name, text.format(lo=lo, hi=hi),
                           pops['x_j'][:, members].sum(axis=1)
                           if members else np.zeros(len(pops['x_j']))))

    for (name, lo, hi), values in zip(age_bins, light):
        planes.append((f"{name}_light",
                       f"Light binned pop: {lo:.1E} < age <= {hi:.1E}",
                       values))
    for (name, lo, hi), values in zip(age_bins, mass):
        planes.append((f"{name}_mass",
                       f"Mass binned pop:  {lo:.1E} < age <= {hi:.1E}",
                       values))
    if mass_factor is not None:
        stellar = np.asarray(base['is_ssp'], dtype=bool)
        ages = np.asarray(base['ages_yr'], dtype=np.float64)
        for name, lo, hi in sfr_bins or []:
            member = stellar & (ages > lo) & (ages <= hi)
            fraction = (pops['mini_j'][:, member].sum(axis=1)
                        / np.where(pops['mini_tot'] > 0, pops['mini_tot'], 1.0))
            formed = pops['mcor_tot'] * mass_factor * fraction
            planes.append((name, f"SFR (Msun/yr): Mstar * Mini fraction with "
                                 f"{lo:.1E} < age <= {hi:.1E} yr over the bin",
                           formed / (hi - lo)))

    lam = np.asarray(fit['lam'])
    centre = int(np.argmin(np.abs(lam - cfg.norm_wavelength)))
    window = slice(max(0, centre - 10), min(len(lam), centre + 11))

    planes += [
        ('Av', 'Optical extinction (mag)', params[:, 0]),
        ('Mage_L', 'Mean age, light weighted, log10(yr)',
         ifu_output.mean_log_age(pops['x_j'], base['ages_yr'],
                                 base['is_ssp'], over_all=True)),
        ('Mage_M', 'Mean age, mass weighted, log10(yr)',
         ifu_output.mean_log_age(pops['mcor_pct'], base['ages_yr'],
                                 base['is_ssp'])),
        ('MZ_L', 'Mean metallicity, light weighted (mass frac)',
         ifu_output.mean_metallicity(pops['x_j'], base['metals'],
                                     base['is_ssp'], over_all=True)),
        ('MZ_M', 'Mean metallicity, mass weighted (mass frac)',
         ifu_output.mean_metallicity(pops['mcor_pct'], base['metals'],
                                     base['is_ssp'])),
    ]
    if mass_factor is not None:
        planes += [
            ('Mstar', 'Present mass in stars (Msun)',
             pops['mcor_tot'] * mass_factor),
            ('Mpcross', 'Mass processed in stars, initial (Msun)',
             pops['mini_tot'] * mass_factor),
        ]
    planes += [
        ('F_Norm', 'Normalisation flux in input units', fit['norm']),
        ('Sigma_star', 'Stellar velocity dispersion (km/s), observed',
         params[:, 2]),
        ('vrot_star', 'Stellar velocity (km/s)', params[:, 3]),
        ('Adev', 'Mean |O-M|/O over the fitted pixels (per cent)',
         fit['adev']),
        ('ChiSqrt', 'chi2 / Nl_eff, the pixels that entered the chi2',
         fit['chi2']),
        ('SNR', 'Mean S/N per pixel, normalisation window',
         ifu_output.signal_to_noise(fit['flux'], fit['inv_sigma'], window)),
        ('YAV', 'Extra extinction, young populations (mag)',
         params[:, 1]),
        ('Nl_eff', 'Pixels that entered the chi2', fit['n_eff']),
        ('Niter', 'Levenberg-Marquardt iterations used', fit['n_iter']),
        ('BadSpaxel', '1 where this spaxel is not a converged fit',
         fit['bad']),
    ]

    for index, group in enumerate(fit['groups']):
        planes.append((f'Sigma_gas_g{group}',
                       f'Gas velocity dispersion (km/s), group {group}',
                       params[:, 4 + 2 * index]))
        planes.append((f'vrot_gas_g{group}',
                       f'Gas velocity (km/s), group {group}',
                       params[:, 5 + 2 * index]))

    if fit['lsf'] is not None:
        sigma_corrected, delta, flag = fit['lsf']
        planes += [
            ('Sigma_star_c',
             'Sigma_star, LSF difference removed (km/s)',
             sigma_corrected),
            ('Dsigma_inst', 'Data minus library LSF in quadrature (km/s)',
             delta),
            ('LSF_flag', '0 measurement 1 upper limit 2 not reportable',
             flag),
        ]

    missed = ifu_output.unbinned_percent(pops['x_j'], membership,
                                         base['is_ssp'])
    if np.median(missed) > 0.5:
        logger.warning(
            f"   The age bins claim {100 - np.median(missed):.1f} per cent of "
            f"the light in the median spaxel: {np.median(missed):.1f} per "
            f"cent sits on templates that fall in no bin, and is in no "
            f"<pop>_light map. The bins and this template library do not "
            f"cover the same ages.")
    return planes

def fit_cube_hdus(cfg, hdu_list, hdu_flux='FLUX', hdu_ivar='IVAR',
                  hdu_mask='MASK', hdu_flag=None, batch_size=None,
                  progress=None, age_bins=None,
                  flxobs_step=ifu_output.LINEAR_WAVELENGTH_STEP,
                  spaxels=None, stat_type='inv_variance', sfr_bins=None,
                  fc_bins=None, bb_bins=None, galaxy_distance=None,
                  flux_scale=None):
    """
    Fit every selected spaxel of an open cube and return the fit as HDUs.

    Separated from run_megacube so that one fit has one implementation
    whether it is reached from the command line, from the IFU runner with a
    chain of pre-pass modules in front of it, or from the desktop interface.
    The arrays, the log grid, the template matrix, the batching and the
    extensions are all unchanged by the split: run_megacube is now this
    function plus the file it writes around it.

    hdu_flux, hdu_ivar and hdu_mask name the extensions to read, because a
    pre-pass module renames what it produces -- dereddening writes FLUX_DRD,
    a spectral resampler writes FLUX_BIN -- and the fit has to be told which
    of them it is now looking at. Naming an extension that is not there is an
    error, not a silent fallback to FLUX, since falling back would fit the
    cube the user thought they had stopped fitting.

    hdu_flag names a per-spaxel selection plane, normally the SN_MASKS_<t>
    an S/N mask stage wrote.

    stat_type says which of the four kinds of uncertainty the statistical
    extension holds -- error, inv_error, variance or inv_variance -- and it
    is converted to the inverse variance the weights are built from on the
    way in. MaNGA ships an inverse variance and MUSE ships a variance, and a
    plane read as the wrong one of the two produces weights that are wrong by
    a factor that varies pixel by pixel and a fit that still looks like a
    fit.

    progress, when given, is called with the number of batches finished and
    the total. The total is known before the first batch, so this is a real
    fraction rather than a spinner.

    spaxels, when given, is called with the flat spaxel indices a batch has
    just finished and the spatial shape those indices address, so that an
    interface can fill the field in as the fit works through it. It is called
    once with an empty array before the first batch, which is what tells a
    listener the shape and that nothing is done yet. It is called from the
    fitting thread and must return immediately; anything it does that takes
    time is time the fit is not running.

    age_bins are the (name, min, max) age bins the PoPBins light and mass
    maps are summed into, in years, min exclusive and max inclusive.
    flxobs_step is the step of the linear wavelength grid FLXOBS and FLXSYN
    are written on, which exists because the analysis tools these files are
    read with take their wavelength axis from that extension's CRVAL3 and
    CD3_3 and have no other route to it.

    Returns the products as an HDUList, the cards to add to the primary
    header, and a summary dictionary. Spaxels that were not fitted, and
    spaxels whose fit did not converge, are both recorded: STATUS is 0, 1 and
    2 for unfitted, converged and not converged, and BADSPXL is 1 for
    anything that is not a converged fit, so no map plane can be read as a
    measurement without the reader having been told otherwise.
    """
    if float(getattr(cfg, 'galactic_ebv', 0.0) or 0.0):
        raise SystemExit(
            "\n'galactic_ebv' is set, and it corrects a single spectrum. A cube's "
            "Galactic reddening is removed by the dereddening stage (or the 'ebv' "
            "column of a targets table). Doing it in both places "
            "would correct the cube twice; set galactic_ebv to 0 and use the "
            "stage.\n")

    t0 = time.time()
    age_bins = tuple(ifu_output.DEFAULT_AGE_BINS if age_bins is None
                     else [(str(n), float(lo), float(hi))
                           for n, lo, hi in age_bins])
    backend, mem_gb, dev = get_device_info()
    logger.info(f"Device: {dev} ({backend})")

    for name, what in ((hdu_flux, "'hdu flux'"), (hdu_ivar, "'hdu ivar'"),
                       (hdu_mask, "'hdu mask'"), (hdu_flag, "'hdu flag'")):
        if name is not None and name not in hdu_list:
            if what in ("'hdu ivar'", "'hdu mask'"):
                continue
            raise SystemExit(
                f"\n{what} names extension {name!r}, which this cube does "
                f"not have. It holds: "
                + ", ".join(h.name for h in hdu_list if h.name) + "\n")

    fc = np.asarray(hdu_list[hdu_flux].data, dtype=np.float64)
    ic = (inverse_variance(np.asarray(hdu_list[hdu_ivar].data,
                                      dtype=np.float64), stat_type)
          if hdu_ivar is not None and hdu_ivar in hdu_list else None)
    if ic is not None and stat_type != 'inv_variance':
        logger.info(f"Uncertainty: {hdu_ivar} is a {stat_type}, converted to "
                    f"an inverse variance for the weights")
    mc = (quality_mask(hdu_list[hdu_mask].data, hdu_mask, logger)
          if hdu_mask is not None and hdu_mask in hdu_list else None)
    wl = wavelength_axis(hdu_list, hdu_flux)
    logger.info(f"Cube: {fc.shape[1]}x{fc.shape[2]}, {fc.shape[0]} "
                f"wavelengths [{wl[0]:.0f}-{wl[-1]:.0f} A] from {hdu_flux}")

    n_y0, n_x0 = fc.shape[1], fc.shape[2]
    select, flag_cube = spaxel_selection(hdu_list, hdu_flag, n_y0, n_x0,
                                         flux_cube=fc, logger=logger)
    if flag_cube is not None:
        mc = np.where(flag_cube != 0, MANGA_DONOTUSE, 0).astype(np.int32) \
            if mc is None else (mc | np.where(flag_cube != 0,
                                              MANGA_DONOTUSE, 0)).astype(mc.dtype)
        logger.info(f"Flag: {hdu_flag} is three-dimensional, so it rejects "
                    f"pixels rather than selecting spaxels")

    wl, z_applied = to_rest_frame(wl, cfg, logger)

    if cfg.wavelength_min is not None or cfg.wavelength_max is not None:
        lo = cfg.wavelength_min if cfg.wavelength_min is not None else wl[0]
        hi = cfg.wavelength_max if cfg.wavelength_max is not None else wl[-1]
        keep = (wl >= lo) & (wl <= hi)
        if not np.any(keep):
            raise SystemExit(
                f"\n'wavelength_min'/'wavelength_max' ({lo}-{hi} A) select no "
                f"part of this cube, which spans {wl[0]:.0f}-{wl[-1]:.0f} A.\n")
        wl = wl[keep]; fc = fc[keep]
        ic = ic[keep] if ic is not None else None
        mc = mc[keep] if mc is not None else None
        logger.info(f"Clipped to {lo:.0f}-{hi:.0f} A: {len(wl)} wavelengths")

    nw, ny, nx = fc.shape

    _, ll, vs = log_rebin(wl, fc[:, ny//2, nx//2])
    lg = np.exp(np.asarray(ll)); npx = len(lg)

    bi = bases_info_for(cfg)
    Ml, sn, sa, sm, ss, smst, syf, sfn = load_ssp_templates(cfg.ssp_folder, bi, lg, cfg.norm_wavelength)
    nssp = Ml.shape[1]
    is_ssp = stellar_component_flags(bi, sn)
    logger.info(f"SSPs: {nssp} templates"
                + (f", {int(np.sum(~is_ssp))} of them featureless components "
                   f"carrying light but no mass" if not np.all(is_ssp) else ""))

    # The wavelength mask on the fitting grid, to leave out the lines it
    # covers. Pixels are masked on the input grid below (preprocess_datacube).
    lt, gids, ng = line_table_for(
        cfg, lg, load_mask(cfg.mask_file, lg) if cfg.mask_file else None)
    if lt.get("masked"):
        logger.warning(f"{len(lt['masked'])} emission line(s) lie under the "
                       f"wavelength mask and are NOT fitted or mapped: "
                       f"{', '.join(lt['masked'])}. Unmask them to fit them.")
    mc_ = int(lt['rest_wavelengths'].shape[1]) if lt['names'] else 1
    ngas = len(lt['names'])
    logger.info(f"Lines: {ngas} templates, {ng} groups")

    dk = jnp.array(_get_reddening_curve(lg, cfg.reddening_law, cfg.norm_wavelength))
    nt = nssp + ngas
    D = build_regularization_matrix(nssp, cfg.reg_strength, cfg.reg_order)
    Df = jnp.zeros((D.shape[0], nt)).at[:, :nssp].set(D) if D is not None else jnp.zeros((0, nt))

    blo = [cfg.av_bounds[0], 0.0, cfg.sigma_star_bounds[0], cfg.v0_star_bounds[0]]
    bhi = [cfg.av_bounds[1], 5.0, cfg.sigma_star_bounds[1], cfg.v0_star_bounds[1]]
    for _ in gids: blo += [cfg.sigma_gas_bounds[0], cfg.v0_gas_bounds[0]]; bhi += [cfg.sigma_gas_bounds[1], cfg.v0_gas_bounds[1]]
    blo_d, bhi_d = jnp.array(blo), jnp.array(bhi)
    npar = len(blo)
    v0g = jnp.arange(cfg.v0_star_bounds[0], cfg.v0_star_bounds[1], cfg.v0_grid_step)
    ld = jnp.array(lg)
    rwd = jnp.array(lt['rest_wavelengths']); rad = jnp.array(lt['ratios'])
    ncd = jnp.array(lt['n_components']); gid = jnp.array(lt['group_indices'])

    logger.info("Preprocessing...")
    tp = time.time()
    file_weight = load_mask(cfg.mask_file, wl)
    fl, isl, nf, vi, err_l = preprocess_datacube(fc, ic, mc, wl, lg, cfg.norm_wavelength,
                                         cfg.has_errorbars,
                                         file_weight_lin=file_weight,
                                         spaxel_select=select,
                                         missing_error=getattr(cfg, 'missing_error',
                                                               'exclude'),
                                         logger=logger)
    nv = len(vi)
    if spaxels is not None:
        spaxels(np.empty(0, dtype=np.int64), ny, nx)
    logger.info(f"Valid: {nv}/{ny*nx} in {time.time()-tp:.1f}s")
    if nv == 0:
        raise SystemExit(
            "\nNo spaxel of this cube can be fitted. Every spaxel is either "
            "empty, entirely flagged, or excluded by the selection plane.\n")

    have_err = err_l is not None
    if have_err:
        with np.errstate(divide='ignore', invalid='ignore'):
            ist = np.where(err_l > 0, 1.0 / np.where(err_l > 0, err_l, 1.0),
                           0.0)
    else:
        ist = np.zeros_like(isl)
    n_clip = int(getattr(cfg, 'n_clip_iterations', 0))
    clip_threshold = float(getattr(cfg, 'sigma_clip_threshold', 3.0))
    clip_scaled = getattr(cfg, 'sigma_clip_scale', 'error') == 'typical'
    if n_clip:
        logger.info(f"Outlier clipping: {n_clip} clip-and-refit "
                    f"pass{'es' if n_clip > 1 else ''} at "
                    f"{clip_threshold:g} sigma"
                    + (" of the larger of the error and the typical residual"
                       if clip_scaled else " of each pixel's error")
                    + ", every pass a complete refit")

    fit_px = lsf_decision_pixels(Ml, isl)
    Ml, sfn, lsf = prepare_templates(Ml, lg, cfg, bi, fit_px, sfn)
    if getattr(cfg, 'instrumental_resolution', None) is not None:
        logger.info(f"LSF, cube-wide: {lsf['state_text']} "
                    f"({lsf['n_applied']} pre-convolved, {lsf['n_refused']} at "
                    f"library resolution), decided over {int(np.sum(fit_px))}/"
                    f"{npx} pixels used by at least one of the {nv} fitted "
                    f"spaxels and covered by at least one template")
    Md = jnp.array(Ml); yd = jnp.array(syf)

    if batch_size is None:
        batch_size = estimate_batch_size(mem_gb, npx, nssp, ngas, npar, backend=='gpu', 
                                         max_cpu_cores=getattr(cfg, 'ncpu', 55), 
                                         max_ram_gb=getattr(cfg, 'max_ram', 60.0))
        # Never larger than the spaxels there are: a batch is padded to its
        # full size and the padding is fitted like real data. Nine spaxels
        # used to be fitted as a batch of 55.
        batch_size = max(1, min(batch_size, int(2 ** np.ceil(np.log2(max(nv, 1))))))
    n_workers = 1
    if backend == 'cpu':
        n_workers = cpu_worker_count(getattr(cfg, 'cpu_workers', 'auto'), nv)
    logger.info("JIT warmup..." if n_workers == 1 else
                f"{n_workers} CPU workers will each compile the fit")
    tj = time.time()
    solver = cfg.linear_solver
    simplex = bool(cfg.enforce_simplex)
    max_iter = int(cfg.lm_max_iter)
    ftol = float(cfg.lm_ftol)
    xtol = float(cfg.lm_xtol)
    fitter = str(getattr(cfg, 'cube_fitter', 'fast'))
    if fitter == 'fast' and solver != 'active_set':
        logger.info(f"cube_fitter 'fast' carries the active set between solves "
                    f"and {solver!r} has none; this cube is fitted with the "
                    f"reference fitter.")
        fitter = 'reference'
    fast_opts = {}
    if fitter == 'fast':
        from brainsp import fast_cube
        batch_fn = fast_cube.fit_batch_fast
        # The broadening kernel never reaches further than 8 sigma plus the
        # largest shift the bounds allow; padding by that (and a margin)
        # instead of the full spectrum makes every transform about three
        # times shorter. The model is the same to 1e-13 from a kernel two
        # pixels wide up; a narrower one is not band-limited and rings, and
        # at most 2e-6 of the flux reaches the outermost pixels
        # (tests/test_fast_cube_jacobian.py).
        sig_max = float(max(abs(b) for b in cfg.sigma_star_bounds))
        v_max = float(max(abs(b) for b in cfg.v0_star_bounds))
        fast_opts = dict(pad_pixels=int(np.ceil((8.0 * sig_max + v_max)
                                                / float(vs))) + 16)
    else:
        batch_fn = _fit_batch
    logger.info(f"Solver: {solver} | simplex {simplex} | max_iter {max_iter} "
                f"| ftol {ftol:g} | xtol {xtol:g} | cube fitter: {fitter}")
    # One device, fast fitter: the spaxels go through a work queue with
    # batch_size lanes (fast_cube.fit_queue_fast), so no lane waits for the
    # slowest spaxel of its batch. Same fit, spaxel by spaxel.
    use_queue = n_workers == 1 and fitter == 'fast'
    while n_workers == 1:
        df = jnp.zeros((batch_size, npx)); di = jnp.ones((batch_size, npx))
        dt_ = jnp.ones((batch_size, npx))
        try:
            if use_queue:
                # Warm up every kernel the queue uses, the refill included.
                warm = batch_size + max(1, batch_size // 4)
                wo = fast_cube.fit_queue_fast(
                    np.zeros((warm, npx)), np.ones((warm, npx)),
                    np.ones((warm, npx)), Md, ld, vs, dk, Df, yd, rwd, rad,
                    ncd, gid, mc_, ng, blo_d, bhi_d, v0g, cfg.av_init,
                    cfg.sigma_star_init, cfg.sigma_gas_init, solver=solver,
                    simplex=simplex, max_iter=max_iter, ftol=ftol,
                    xtol=xtol, n_clip=n_clip, clip_threshold=clip_threshold,
                    clip_scaled=clip_scaled,
                    have_errors=have_err, lanes=batch_size, **fast_opts)
                del wo, df, di, dt_
                break
            wo = batch_fn(df, di, dt_, Md, ld, vs, dk, Df, yd, rwd, rad, ncd,
                            gid, mc_, ng, blo_d, bhi_d, v0g, cfg.av_init,
                            cfg.sigma_star_init, cfg.sigma_gas_init,
                            solver=solver, simplex=simplex, max_iter=max_iter,
                            ftol=ftol, xtol=xtol, n_clip=n_clip,
                            clip_threshold=clip_threshold,
                            clip_scaled=clip_scaled,
                            have_errors=have_err, **fast_opts)
            jax.block_until_ready(wo)
        except Exception as exc:
            del df, di, dt_
            if batch_size <= 1 or not out_of_memory(exc):
                raise
            batch_size = max(1, batch_size // 2)
            logger.warning(
                f"The device ran out of memory compiling a batch, so the "
                f"batch size drops to {batch_size} and the warm-up runs "
                f"again. Every spaxel is fitted independently inside a batch, "
                f"so each one gets the same fit; what changes is the order "
                f"XLA sums in, measured at about one part in 1e5 on the "
                f"fitted parameters. Set 'batch size' in the pipeline to "
                f"choose it yourself and skip this.")
            continue
        del wo, df, di, dt_
        break
    jt = time.time() - tj
    logger.info(f"JIT: {jt:.1f}s")
    if n_workers > 1:
        batch_size = 1
    nb = (nv + batch_size - 1) // batch_size
    logger.info(f"Batch: {batch_size} | Batches: {nb}")
    if progress is not None:
        progress(0, nb)

    ap = np.zeros((nv, npar)); aw = np.zeros((nv, nt))
    af = np.zeros((nv, npx)); ast = np.zeros((nv, npx)); ag = np.zeros((nv, npx))
    ac2 = np.zeros(nv); ane = np.zeros(nv); acv = np.zeros(nv, dtype=bool); ani = np.zeros(nv, dtype=int)
    n_pre_clip = np.sum(isl > 0, axis=1)

    logger.info("="*60)
    logger.info(f"Fitting {nv} spaxels on {backend.upper()}"
                + (f", {n_workers} worker processes of one thread each"
                   if n_workers > 1 else "") + "...")
    tf = time.time()
    if n_workers > 1:
        constants = [Md, ld, vs, dk, Df, yd, rwd, rad, ncd, gid, mc_, ng,
                     blo_d, bhi_d, v0g, cfg.av_init, cfg.sigma_star_init,
                     cfg.sigma_gas_init]
        options = dict(solver=solver, simplex=simplex, max_iter=max_iter,
                       ftol=ftol, xtol=xtol, n_clip=n_clip,
                       clip_threshold=clip_threshold, clip_scaled=clip_scaled,
                       have_errors=have_err,
                       **fast_opts)
        done = [0]
        step = max(1, nv // 50)

        def collect(i, out):
            ap[i] = out[0]; aw[i] = out[1]; af[i] = out[2]; ast[i] = out[3]
            ag[i] = out[4]; ac2[i] = out[5]; ane[i] = out[6]
            acv[i] = out[7]; ani[i] = out[8]; isl[i] = out[9]
            done[0] += 1
            if spaxels is not None:
                spaxels(np.asarray([vi[i]], dtype=np.int64), ny, nx)
            if progress is not None:
                progress(done[0], nv)
            if done[0] % step == 0 or done[0] == nv:
                el = time.time() - tf
                logger.info(f"  {done[0]}/{nv} spaxels, {done[0] / el:.1f} "
                            f"spaxels/s, ETA {(nv - done[0]) * el / done[0]:.0f}s")

        _fit_cpu_workers(n_workers, fl, isl.copy(), ist, constants, options,
                         fitter, collect)
        nb = 0
    if use_queue and nv > 0:
        logger.info(f"Work queue: {batch_size} lanes (a spaxel leaves its lane "
                    f"when it converges and the next one takes it)")
        last = [0, time.time()]

        def report(n_done, n_total, finished):
            if spaxels is not None:
                spaxels(np.asarray(vi[finished], dtype=np.int64), ny, nx)
            if progress is not None:
                progress(n_done, n_total)
            if (n_done - last[0] >= max(1, n_total // 20) or n_done == n_total):
                el = time.time() - tf
                logger.info(f"  {n_done}/{n_total} spaxels, "
                            f"{n_done / el:.1f} spaxels/s, "
                            f"ETA {(n_total - n_done) * el / n_done:.0f}s")
                last[0] = n_done

        r = fast_cube.fit_queue_fast(
            fl, isl, ist, Md, ld, vs, dk, Df, yd, rwd, rad, ncd, gid, mc_, ng,
            blo_d, bhi_d, v0g, cfg.av_init, cfg.sigma_star_init,
            cfg.sigma_gas_init, solver=solver, simplex=simplex,
            max_iter=max_iter, ftol=ftol, xtol=xtol, n_clip=n_clip,
            clip_threshold=clip_threshold, clip_scaled=clip_scaled,
            have_errors=have_err,
            lanes=batch_size, progress=report, **fast_opts)
        ap[:] = r[0]; aw[:] = r[1]; af[:] = r[2]; ast[:] = r[3]
        ag[:] = r[4]; ac2[:] = r[5]; ane[:] = r[6]; acv[:] = r[7]
        ani[:] = r[8]; isl[:] = r[9]
        nb = 0
    for b in range(nb):
        i0 = b*batch_size; i1 = min(i0+batch_size, nv); act = i1-i0
        if act < batch_size:
            fb = np.zeros((batch_size, npx)); sb = np.zeros((batch_size, npx))
            eb = np.zeros((batch_size, npx))
            fb[:act] = fl[i0:i1]; sb[:act] = isl[i0:i1]; eb[:act] = ist[i0:i1]
        else:
            fb = fl[i0:i1]; sb = isl[i0:i1]; eb = ist[i0:i1]
        tb = time.time()
        r = batch_fn(jnp.array(fb), jnp.array(sb), jnp.array(eb),
                       Md, ld, vs, dk, Df, yd,
                       rwd, rad, ncd, gid, mc_, ng, blo_d, bhi_d, v0g,
                       cfg.av_init, cfg.sigma_star_init, cfg.sigma_gas_init,
                       solver=solver, simplex=simplex,
                       max_iter=max_iter, ftol=ftol, xtol=xtol,
                       n_clip=n_clip, clip_threshold=clip_threshold,
                       clip_scaled=clip_scaled,
                       have_errors=have_err, **fast_opts)
        ap[i0:i1]=np.array(r[0][:act]); aw[i0:i1]=np.array(r[1][:act])
        af[i0:i1]=np.array(r[2][:act]); ast[i0:i1]=np.array(r[3][:act])
        ag[i0:i1]=np.array(r[4][:act]); ac2[i0:i1]=np.array(r[5][:act])
        ane[i0:i1]=np.array(r[6][:act]); acv[i0:i1]=np.array(r[7][:act])
        ani[i0:i1]=np.array(r[8][:act]); isl[i0:i1]=np.array(r[9][:act])
        dt = time.time()-tb; el = time.time()-tf; rt = i1/el
        eta = (nv-i1)/rt if rt > 0 else 0
        nc = int(np.sum(np.array(r[7][:act])))
        mc2 = np.median(chi2_per_effective_pixel(np.array(r[5][:act]),
                                                 np.array(r[6][:act])))
        logger.info(f"  Batch {b+1}/{nb}: {act} spaxels {dt:.1f}s | conv {nc}/{act} | chi2/Nl_eff {mc2:.3f} | ETA {eta:.0f}s")
        if spaxels is not None:
            spaxels(np.asarray(vi[i0:i1], dtype=np.int64), ny, nx)
        if progress is not None:
            progress(b + 1, nb)

    ft = time.time()-tf
    tc = int(np.sum(acv))
    c2n = chi2_per_effective_pixel(ac2, ane)
    c2r = ac2/np.maximum(ane-npar, 1)
    logger.info("="*60)
    logger.info(f"Done: {ft:.1f}s ({nv/ft:.1f} spaxels/s) | conv {tc}/{nv} | med chi2/Nl_eff {np.median(c2n):.4f} | med chi2/(Nl_eff-n_par) {np.median(c2r):.4f}")
    if n_clip:
        gone = n_pre_clip - np.sum(isl > 0, axis=1)
        logger.info(f"   Clipped {int(np.sum(gone))} pixels over {nv} "
                    f"spaxels, median {int(np.median(gone))} per spaxel, "
                    f"worst {int(np.max(gone))}. Nl_eff, the chi2 above and "
                    f"adev are all over what is left.")
    if tc < nv:
        logger.warning(f"   {nv - tc} of {nv} fitted spaxels did not converge. "
                       f"They carry STATUS 2 and BADSPXL 1, and their map "
                       f"values are the last state of an optimisation that "
                       f"had not finished, not measurements.")

    lsf_planes = None
    if getattr(cfg, 'instrumental_resolution', None) is not None:
        sig_int = np.zeros(nv); dsig = np.zeros(nv)
        lflag = np.zeros(nv); unreported = None
        for i in range(nv):
            rep = dispersion_report(
                lsf, float(ap[i, 2]), aw[i, :nssp], af[i], isl[i] > 0,
                cfg.sigma_star_bounds[0], upper_bound=cfg.sigma_star_bounds[1],
                reference_wavelength=cfg.norm_wavelength)
            if rep['sigma_star'] is None:
                sig_int[i] = np.nan; dsig[i] = np.nan; lflag[i] = 2
                unreported = unreported or rep['reason']
            else:
                sig_int[i] = rep['sigma_star']; dsig[i] = rep['delta_sigma_inst']
                lflag[i] = 1 if rep['upper_limit'] else 0
        ok_lsf = np.isfinite(sig_int)
        if unreported:
            logger.warning(f"   {unreported}")
        if np.any(ok_lsf):
            logger.info(f"   sigma_star corrected for the line-spread function "
                        f"in {int(np.sum(ok_lsf))}/{nv} spaxels: median "
                        f"{np.median(np.abs(sig_int[ok_lsf])):.2f} km/s, median "
                        f"delta_sigma_inst {np.median(dsig[ok_lsf]):+.2f} km/s, "
                        f"{int(np.sum(lflag == 1))} upper limits, "
                        f"{int(np.sum(sig_int[ok_lsf] < 0))} unphysical")
        else:
            logger.warning(f"   sigma_star in KINEMATICS is an OBSERVED "
                           f"dispersion in all {nv} spaxels, not an intrinsic "
                           f"one")
        lsf_planes = (sig_int, dsig, lflag)

    lnp = np.asarray(lg)
    io = np.clip(np.searchsorted(lnp, wl, side='right')-1, 0, npx-2)
    fo = (wl - lnp[io])/(lnp[io+1]-lnp[io])
    def tol(a): return a[:, io]*(1-fo) + a[:, io+1]*fo

    st_lin = tol(ast)*nf[:,None]; gs_lin = tol(ag)*nf[:,None]
    yc = vi // nx; xc = vi % nx

    os_ = np.zeros_like(fc); om = np.zeros_like(fc)
    os_[:, yc, xc] = st_lin.T; om[:, yc, xc] = (st_lin+gs_lin).T

    grid = ifu_output.linear_wavelength_grid(lg, flxobs_step)
    obs_lin = ifu_output.resample_batch(lg, fl*nf[:, None], grid)
    syn_lin = ifu_output.resample_batch(lg, ast*nf[:, None], grid)
    oobs = np.zeros((len(grid), ny, nx)); oobs[:, yc, xc] = obs_lin.T
    osyn = np.zeros((len(grid), ny, nx)); osyn[:, yc, xc] = syn_lin.T
    logger.info(f"FLXOBS/FLXSYN on a linear grid: {len(grid)} points, "
                f"{grid[0]:.1f}-{grid[-1]:.1f} A, step "
                f"{grid[1]-grid[0]:.3f} A")

    k_norm = float(reddening_laws.get_k_lambda(
        jnp.array([cfg.norm_wavelength]), cfg.reddening_law)[0])
    pops = population_statistics(aw[:, :nssp], nf, ap[:, 0], ap[:, 1],
                                 syf, smst, sfn, k_norm, is_ssp)
    adev, _, _ = deviation_percent(fl, af, isl > 0)

    opl = np.zeros((nssp, ny, nx)); opl[:, yc, xc] = pops['x_j'].T
    opm = np.zeros((nssp, ny, nx)); opm[:, yc, xc] = pops['mcor_pct'].T
    opi = np.zeros((nssp, ny, nx)); opi[:, yc, xc] = pops['mini_pct'].T
    ok = np.zeros((npar, ny, nx)); ok[:, yc, xc] = ap.T
    oc2 = np.zeros((ny, nx)); oc2[yc, xc] = c2n
    ocr = np.zeros((ny, nx)); ocr[yc, xc] = c2r
    ost = np.zeros((ny, nx), dtype=np.int32); ost[yc, xc] = np.where(acv, 1, 2)
    obad = np.ones((ny, nx), dtype=np.int32); obad[yc, xc] = np.where(acv, 0, 1)

    ages_yr = np.asarray(sa) * 1e9
    mass_factor = None
    if galaxy_distance not in (None, "", 0):
        scale = (float(flux_scale) if flux_scale not in (None, "") else
                 ifu_output.flux_unit_scale(
                     hdu_list[hdu_flux].header.get("BUNIT", "")))
        mass_factor = ifu_output.distance_factor(float(galaxy_distance)) * scale
        logger.info(f"Masses and SFRs in Msun: distance "
                    f"{float(galaxy_distance):g} Mpc, flux unit x{scale:g}")
    planes = pop_bin_planes(
        cfg, age_bins,
        {'pops': pops, 'adev': adev, 'params': ap, 'norm': nf, 'chi2': c2n,
         'n_eff': ane, 'n_iter': ani, 'bad': obad[yc, xc], 'flux': fl,
         'inv_sigma': isl, 'lam': lg, 'groups': gids, 'lsf': lsf_planes},
        {'ages_yr': ages_yr, 'metals': np.asarray(sm), 'short': ss,
         'is_ssp': is_ssp},
        sfr_bins=sfr_bins, fc_bins=fc_bins, bb_bins=bb_bins,
        mass_factor=mass_factor)

    products = fits.HDUList([
        ifu_output.named(fits.ImageHDU(data=np.column_stack(
            [ages_yr, np.asarray(sm)])), 'BaseAgeMetal',
            'Ages (yr) & metallicities (mass fraction) in the base'),
        ifu_output.pop_bins_hdu(planes, (ny, nx), (yc, xc),
                                unfitted={'BadSpaxel': 1.0}),
        ifu_output.named(fits.ImageHDU(data=opl), 'PoPVecsL',
                         'Population vectors, light fractions, per cent'),
        ifu_output.named(fits.ImageHDU(data=opm), 'PoPVecsM',
                         'Population vectors, present-mass fractions, per cent'),
        ifu_output.named(fits.ImageHDU(data=opi), 'PoPVecsMini',
                         'Population vectors, initial-mass fractions, per cent'),
        ifu_output.wcs_cards(ifu_output.named(
            fits.ImageHDU(data=oobs), 'FLXOBS',

def run_megacube(cfg, cube_path, out_path, batch_size=None, hdu_flux='FLUX',
                 hdu_ivar='IVAR', hdu_mask='MASK', hdu_flag=None,
                 progress=None, age_bins=None,
                 flxobs_step=ifu_output.LINEAR_WAVELENGTH_STEP,
                 spaxels=None):
    """
    Fit one cube from disk and write the megacube.

    The file it writes is the fitted input -- flux, inverse variance, mask and
    wavelength on the clipped, rest-frame grid the fit used -- followed by
    every extension fit_cube_hdus produced.
    """
    t0 = time.time()
    with fits.open(cube_path) as h:
        hdr = h['PRIMARY'].header.copy()
        result = fit_cube_hdus(cfg, h, hdu_flux=hdu_flux, hdu_ivar=hdu_ivar,
                               hdu_mask=hdu_mask, hdu_flag=hdu_flag,
                               batch_size=batch_size, progress=progress,
                               age_bins=age_bins, flxobs_step=flxobs_step,
                               spaxels=spaxels)

    fc, ic, mc, wl = (result['flux'], result['ivar'], result['mask'],
                      result['wave'])
    hdus = [fits.PrimaryHDU(header=hdr)]
    hdus.append(fits.ImageHDU(data=fc.astype(np.float32), name='FLUX'))
    if ic is not None: hdus.append(fits.ImageHDU(data=ic.astype(np.float32), name='IVAR'))
    if mc is not None: hdus.append(fits.ImageHDU(data=mc, name='MASK'))
    hdus.append(fits.ImageHDU(data=wl, name='WAVE'))
    hl = fits.HDUList(hdus + list(result['products']))
    for key, value, comment in result['cards']:
        hl[0].header[key] = (value, comment)
    hl.writeto(out_path, overwrite=True)
    summary = dict(result['summary'])
    summary['total_s'] = time.time() - t0
    logger.info(f"Saved: {out_path} | Total: {summary['total_s']:.1f}s")
    return summary
