"""
The uncertainties of everything a fit reports, in every mode.

Two methods, chosen by error_samples:

  0 (the default)  FORMAL. Each fit's covariance from the data's errors
                   (fast_cube.parameter_covariance): the inverse Fisher
                   matrix of the weights and the non-linear parameters
                   together, at the solution, on the fit's active set. It is
                   scaled by the fit's chi2 per degree of freedom where that
                   exceeds 1, so a fit worse than its errors allow (template
                   mismatch, underestimated errors) gets wider bars; it is
                   never narrowed below what the errors give. Every reported
                   quantity is then propagated from it to first order.
  N > 0            BOOTSTRAP. The whole fit, every stage of it, is repeated
                   on N spectra made from the best model and the fit's own
                   residuals with random signs (the wild bootstrap; Davidson
                   & Flachaire 2008, as Kacharov et al. 2018 for pPXF), and
                   each quantity's error is half the 16-84 per cent range of
                   its N values. It sees what the formal errors cannot: a
                   template that leaves or joins the solution, a parameter
                   that moves to a bound, a spaxel that changes between FC
                   and NC, T_e that is measured in some resamples and not in
                   others, and residuals larger than the errors.

Where each quantity's formal error comes from:

  A_V, Y_V, sigma_*, v_*, the weights and everything built from them (light
  and mass fractions, mean ages and metallicities, masses, SFRs, the age
  bins):  the population fit, the one whose values are reported. In nespop
  that fit has A_V^neb free within its Balmer-decrement error and, in FC,
  the measured H-alpha and H-beta with their errors constraining the
  ionising photons, so y_neb, Q(H) and tau carry both.
  sigma_gas, v_gas:  the joint fit of stars and lines, which measures them.
  Line fluxes:  the linear measurement at the fitted kinematics
  (fast_cube.line_flux_errors), scaled as above.
  A_V^neb, H-alpha/H-beta, T_e, n_e:  from the line-flux errors, through
  the same formulae that compute them.

A parameter on a bound of its range has no error (NaN): the fit did not
find a minimum there, it stopped at the edge. Flags, counts and the fit's
own statistics (STATUS, MODE, Niter, chi2 ...) have none either.
"""

# Planes of PoPBins and NEBMAPS that are flags, counts or the fit's own
# statistics: no error.
NO_ERROR = frozenset((
    'F_Norm', 'Adev', 'ChiSqrt', 'SNR', 'Nl_eff', 'Niter', 'BadSpaxel',
    'LSF_flag', 'Dsigma_inst', 'AV_src', 'MODE', 'BPT', 'cond_src'))

# The quantities whose formal errors the calibration bears out (noise draws
# of mock spectra with known truth, every mode, 2026-10-14: the scatter of
# the fits over the median error within a factor of ~1.3, or conservative).
# Every other product is a sum or ratio over the weights that a template
# entering or leaving the fit moves -- the age bins, x_j, mass-weighted
# means, masses, SFRs, y_neb, tau -- where a linear covariance at one
# solution cannot see the change (an empty bin gets an error of ~0, y_neb's
# was 5-8 times too small): no formal error is written for them (NaN), and
# the bootstrap (error_samples) gives theirs.
FORMAL_POPBINS = frozenset(('Av', 'YAV', 'Sigma_star', 'vrot_star', 'Mage_L',
                            'MZ_L', 'Sigma_star_c'))
FORMAL_NEBMAPS = frozenset(('AV_neb', 'logQH', 'Hb_pred', 'Hb_obs', 'HaHb',
                            'Te', 'ne'))


# chi2 per degree of freedom above which a spaxel's formal errors are not
# written. On MaNGA 8626-12704 (60 spaxels, nespop, formal against a
# 50-resample bootstrap) the two agreed within 0.9-1.5 where chi2/dof ~ 1.4
# (S/N 3-8); at chi2/dof 9 and 45 (S/N 8-20, 20-90) the misfit is template
# mismatch, correlated over pixels, which no chi2 factor turns into an error:
# the formal errors were 1.5-6.5 times too small for most quantities and
# too large for some lines.
FORMAL_CHI2_MAX = 2.0

def withhold_misfit(errs, chi2_dof, limit=FORMAL_CHI2_MAX):
    """NaN for every formal error of the spaxels whose chi2/dof exceeds
    limit; returns how many."""
    bad = ~(np.asarray(chi2_dof, float) <= limit)
    if not bad.any():
        return 0
    for key, v in list(errs.items()):
        if isinstance(v, dict):
            for k2 in v:
                v[k2] = np.where(bad, np.nan, np.asarray(v[k2], float))
        elif v is not None:
            a = np.array(v, dtype=float)
            if a.shape[:1] == bad.shape:
                a[bad] = np.nan
            elif a.ndim == 2 and a.shape[1] == bad.shape[0]:
                a[:, bad] = np.nan
            errs[key] = a
    return int(bad.sum())

def formal_validated(ext, name):
    """Whether a formal error of this plane is written (see above)."""
    if ext == "PoPBins":
        return (name in FORMAL_POPBINS or str(name).startswith("Sigma_gas_g")
                or str(name).startswith("vrot_gas_g"))
    if ext == "NEBMAPS":
        return name in FORMAL_NEBMAPS
    return ext in ("KINEMATICS", "LSFCORR", "LINES")




# Relative step of the first-order propagation along each direction of the
# covariance (q(z + h L_k) - q(z - h L_k)) / 2h: small enough to be the
# derivative, large enough to be far above rounding in float64.
FD_STEP = 1e-4

def chi2_scale(chi2, n_eff, n_free):
    """max(1, chi2 / (N_eff - n_free)): the formal covariance's factor."""
    dof = np.maximum(np.asarray(n_eff, float) - np.asarray(n_free, float), 1.0)
    with np.errstate(divide="ignore", invalid="ignore"):
        s = np.asarray(chi2, float) / dof
    return np.where(np.isfinite(s), np.maximum(s, 1.0), 1.0)

def stage_covariance(ap, aw, fl, isig, lo, hi, T, ld, vs, dk, Df, yd, rwd,
                     rad, ncd, gid, mc_, batch_size, nssp, neb=None, nl=None,
                     lines=None, i0=0, pad_pixels=None):
    """
    One fit's formal covariance, for every spaxel, in batches.

    ap, aw, isig: the fit's parameters, weights and the pixel weights it
    ended on (clipped, masked); lo, hi its per-spaxel bounds. The rest are
    that fit's constants as fast_cube.fit_queue_fast took them. It returns::

      sig    (n, n_par + 1 + n_cols)  1-sigma of every quantity, scaled,
             NaN where it was not free (on a bound, zero weight)
      L      (n, n_par + 1 + nssp, K) the population block's square root
             (A_V .. gas, A_V^neb, stellar weights), scaled: its covariance
             is L L^T. K is the largest number of free quantities in it.
      g      (n, 3, n_par + 1 + nssp) or None: gradients of S, the nebular
             continuum and the model at pixel i0 (nebular fits)
      scale  (n,) the chi2 factor applied
    """
    from brainsp import fast_cube
    if neb is not None:
        # as fit_queue_fast prepares it: without its transform the model
        # has no nebular continuum
        neb = fast_cube.prepared_neb(neb, T, pad_pixels)
    n, npar = ap.shape
    lo = np.broadcast_to(np.asarray(lo, float), (n, npar))
    hi = np.broadcast_to(np.asarray(hi, float), (n, npar))
    nl = np.zeros((n, 1)) if nl is None else np.asarray(nl, float)
    n_r = 0 if neb is None else int(neb.line_idx.shape[0])
    # (F_L, E_L, dA_V^neb/dF_L) of nespop's row lines, per spaxel
    lines = (np.zeros((n, 3, max(n_r, 1))) if lines is None
             else np.nan_to_num(np.asarray(lines, float)))
    nb = npar + 1 + nssp
    d = npar + 1 + aw.shape[1]
    sig = np.full((n, d), np.nan)
    Ls, gs, scale = [], [], np.ones(n)
    step = max(1, min(int(batch_size or 32), 32))
    if fast_cube.POOL is not None:
        # on CPU workers each computes a few rows of its own, so a batch
        # can keep them all busy (the cap of 32 is a GPU's memory)
        step = max(step, 4 * fast_cube.POOL.n)
    for i0b in range(0, n, step):
        sl = slice(i0b, min(n, i0b + step))
        # one batch at a time (a cube's full covariances would not fit in
        # memory), padded to a shared shape (fast_cube.run_rows)
        Cd, Sl, chi2, nfree, g = fast_cube.run_rows(
            fast_cube.parameter_covariance,
            tuple(np.asarray(a)[sl] for a in (ap, aw, fl, isig, lo, hi, nl,
                                              lines)),
            sl.stop - sl.start, step, T[0], T[1], int(T.n_left),
            int(T.n_fft), ld, vs, dk, Df, yd, rwd, rad, ncd, gid, int(mc_),
            int(i0), neb=neb)
        f = chi2_scale(chi2, np.sum(np.asarray(isig[sl]) > 0, 1), nfree)
        scale[sl] = f
        # the data's part widened by the fit's chi2 factor; the measured
        # lines' part by their own errors (already widened by the fit that
        # measured them)
        SE = Sl * lines[sl, 1][:, None, :]
        Cb_ = Cd * f[:, None, None] + np.einsum("nik,njk->nij", SE, SE)
        dg = np.einsum('nii->ni', Cb_)
        sig[sl] = np.where(dg > 0, np.sqrt(np.clip(dg, 0, None)), np.nan)
        vals, vecs = np.linalg.eigh(Cb_[:, :nb, :nb])
        Ls.append((vals, vecs))
        if neb is not None:
            gs.append(np.asarray(g)[:, :, :nb])
    # the rank: eigenvalues above rounding relative to the largest
    K = max(1, max(int(np.max(np.sum(
        v > 1e-13 * np.max(np.abs(v), axis=1, keepdims=True), axis=1)))
        for v, _ in Ls))
    L = np.zeros((n, nb, K))
    i = 0
    for vals, vecs in Ls:
        m = len(vals)
        L[i:i + m] = vecs[:, :, -K:] * np.sqrt(np.clip(vals[:, -K:], 0, None))[:, None, :]
        i += m
    return dict(sig_p=sig[:, :npar], sig_a=sig[:, npar],
                sig_w=sig[:, npar + 1:], L=L,
                g=(np.concatenate(gs) if gs else None), scale=scale,
                npar=npar)

def pad_params(cov, npar):
    """A stage_covariance result of a fit with fewer non-linear
    parameters (the population fits carry A_V .. v_* only) in the layout of
    one with npar: the missing ones (the gas kinematics) not free."""
    if cov is None or cov["npar"] == npar:
        return cov
    k = cov["npar"]
    ins = lambda a, axis: np.insert(a, [k] * (npar - k), 0.0, axis=axis)
    out = dict(cov)
    out["sig_p"] = np.insert(cov["sig_p"], [k] * (npar - k), np.nan, axis=1)
    out["L"] = ins(cov["L"], 1)
    if cov["g"] is not None:
        out["g"] = ins(cov["g"], 2)
    out["npar"] = npar
    return out

def merge(parts, n):
    """stage_covariance results of disjoint spaxel subsets, in one."""
    parts = [(idx, c) for idx, c in parts if c is not None]
    if not parts:
        return None
    c0 = parts[0][1]
    K = max(c["L"].shape[2] for _, c in parts)
    out = dict(sig_p=np.full((n,) + c0["sig_p"].shape[1:], np.nan),
               sig_a=np.full(n, np.nan),
               sig_w=np.full((n,) + c0["sig_w"].shape[1:], np.nan),
               L=np.zeros((n, c0["L"].shape[1], K)),
               g=(None if c0["g"] is None else np.zeros((n,) + c0["g"].shape[1:])),
               scale=np.ones(n), npar=c0["npar"])
    for idx, c in parts:
        for k in ("sig_p", "sig_a", "sig_w", "scale"):
            out[k][idx] = c[k]
        out["L"][idx, :, :c["L"].shape[2]] = c["L"]
        if out["g"] is not None:
            out["g"][idx] = c["g"]
    return out

def bound_free(sig, ap, lo, hi):
    """NaN for a parameter on a bound (or held), whatever its column gave."""
    ap = np.asarray(ap, float)
    lo = np.broadcast_to(np.asarray(lo, float), ap.shape)
    hi = np.broadcast_to(np.asarray(hi, float), ap.shape)
    tol = 1e-6 * np.maximum(hi - lo, 1e-30)
    free = (hi - lo > 0) & (ap - lo > tol) & (hi - ap > tol)
    return np.where(free, sig, np.nan)

def propagate(fn, z, L, step=FD_STEP, chunk=128):
    """
    First-order propagation of z's covariance L L^T through fn, for every
    spaxel: sigma_q^2 = sum_k (dq/dz . L_k)^2, the derivative along each
    column of L by central differences of size step x L_k.

    fn(zs, idx) maps a (m, nz) array of z's, each a perturbed copy of
    spaxel idx[i], to a dict of (m,) or (m, k) arrays; it is called on all
    2 K copies of `chunk` spaxels at a time.
    """
    n, nz, K = L.shape
    out = {}
    for c0 in range(0, n, chunk):
        sl = slice(c0, min(n, c0 + chunk))
        m = sl.stop - sl.start
        Lt = np.transpose(L[sl], (0, 2, 1))                  # (m, K, nz)
        zp = (z[sl, None, :] + step * Lt).reshape(m * K, nz)
        zm = (z[sl, None, :] - step * Lt).reshape(m * K, nz)
        idx = np.repeat(np.arange(sl.start, sl.stop), K)
        q = fn(np.concatenate([zp, zm]), np.concatenate([idx, idx]))
        for name, v in q.items():
            v = np.asarray(v, float)
            dv = (v[:m * K] - v[m * K:]).reshape((m, K) + v.shape[1:]) / (2.0 * step)
            e = np.sqrt(np.sum(np.where(np.isfinite(dv), dv, 0.0) ** 2, axis=1))
            if name not in out:
                out[name] = np.zeros((n,) + e.shape[1:])
            out[name][sl] = e
    return out

def robust_sigma(x, axis=-1):
    """Half the 16-84 per cent range: the 1-sigma of a Gaussian, and not
    pulled by the odd resample that ran to a bound."""
    lo, hi = np.nanpercentile(x, [15.865525, 84.134475], axis=axis)
    return 0.5 * (hi - lo)

def wild_resamples(fl, af, isl, n_samples, seed=0):
    """
    N spectra per spaxel, (n, N, n_pix): the model plus the residual with a
    random sign on every pixel the fit used, the data as they are on the
    rest (masked or clipped pixels stay what they were, so they are masked
    or clipped again).
    """
    rng = np.random.default_rng(seed)
    n, npx = fl.shape
    used = isl > 0
    sign = rng.choice((-1.0, 1.0), size=(n, n_samples, npx))
    r = (fl - af)[:, None, :]
    star = af[:, None, :] + sign * r
    return np.where(used[:, None, :], star, fl[:, None, :])

def err_hdu(hdu, sigma, method, extra=None):
    """<EXT>_ERR: hdu's layout and cards, sigma in place of the values."""
    from astropy.io import fits
    h = fits.ImageHDU(data=np.asarray(sigma, dtype=np.float64),
                      header=hdu.header.copy(), name=f"{hdu.name}_ERR")
    h.header["EXTNAME"] = f"{hdu.name}_ERR"
    h.header["ERRMETH"] = (method, "formal covariance or wild bootstrap")
    if method == "formal":
        h.header["ERRNOTE"] = ("NaN planes: bootstrap only (error_samples)",
                               "formal errors kept where calibrated")
    h.header["ERRWHAT"] = ("1 sigma of the same plane of " + hdu.name,
                           "NaN: no error defined")
    for k, v in (extra or {}).items():
        h.header[k] = v
    return h

def formal_errors(cov_pop, gas_sig, ap, aw, nssp, pop_fn, av_neb=None,
                  lsf_planes=None, gerr=None):
    """
    Every reported quantity's formal 1-sigma, from the population fit's
    covariance (cov_pop, in the layout of the final parameters) and the
    gas kinematics' (gas_sig, (n, n_par - 4), NaN where not available).

    pop_fn(z, idx) -> {key: values}: the products of a (m, n_par + 1 +
    nssp) array of [parameters, A_V^neb, stellar weights], keys
    ('PoPBins', card) for the PoPBins planes and 'PoPVecsL', 'PoPVecsM',
    'PoPVecsMini' for the vectors. Returns the errors as a dict of the
    same keys plus 'KINEMATICS' (n, n_par), 'LSFCORR' {plane: values} and
    'LINES' (the line weights' errors, gerr as given).
    """
    n, npar = ap.shape
    a = np.zeros(n) if av_neb is None else np.nan_to_num(np.asarray(av_neb, float))
    z = np.concatenate([ap, a[:, None], aw[:, :nssp]], axis=1)
    out = propagate(pop_fn, z, cov_pop["L"])
    p_sig = np.array(cov_pop["sig_p"], dtype=float)
    if npar > 4:
        p_sig[:, 4:] = (np.full((n, npar - 4), np.nan) if gas_sig is None
                        else gas_sig)
    errs = {"KINEMATICS": p_sig, "LINES": gerr, "PoPBins": {}}
    for key, v in out.items():
        if isinstance(key, tuple):
            errs.setdefault(key[0], {})[key[1]] = v
        else:
            errs[key] = v
    pb = errs["PoPBins"]
    # parameters read straight off: their own errors, NaN on a bound
    for name, k in (("Av", 0), ("YAV", 1), ("Sigma_star", 2),
                    ("vrot_star", 3)):
        if name in pb:
            pb[name] = p_sig[:, k]
    for name in list(pb):
        if name in NO_ERROR:
            pb[name] = np.full(n, np.nan)
    # only what the calibration bears out (FORMAL_POPBINS)
    for name in list(pb):
        if not formal_validated("PoPBins", name) and not (
                str(name).startswith(("Sigma_gas_g", "vrot_gas_g"))):
            pb[name] = np.full(n, np.nan)
    for key in ("PoPVecsL", "PoPVecsM", "PoPVecsMini"):
        if key in errs:
            errs[key] = np.full_like(np.asarray(errs[key], float), np.nan)
    errs.pop("SUM", None)
    if lsf_planes is not None:
        sig_c = np.asarray(lsf_planes[0], float)
        with np.errstate(divide="ignore", invalid="ignore"):
            e_c = p_sig[:, 2] * np.abs(ap[:, 2] / sig_c)
        e_c = np.where(np.isfinite(sig_c) & (sig_c != 0), e_c, np.nan)
        pb["Sigma_star_c"] = e_c
        errs["LSFCORR"] = {0: e_c}
    return errs

def gas_planes(errs, groups):
    """The Sigma_gas / vrot_gas planes of PoPBins from KINEMATICS' errors."""
    p_sig = errs["KINEMATICS"]
    for index, group in enumerate(groups):
        errs["PoPBins"][f"Sigma_gas_g{group}"] = p_sig[:, 4 + 2 * index]
        errs["PoPBins"][f"vrot_gas_g{group}"] = p_sig[:, 5 + 2 * index]

def nebular_errors(neb_out, cov_pop, p_sig, af, nf, i0, k_hb):
    """
    NEBMAPS' formal errors: y_neb, Q(H) and the predicted H-beta from the
    population fit's covariance (their gradients, cov_pop["g"]); A_V^neb
    its posterior there (the decrement's error as the prior, the photons
    in FC) or A_V's where tied; the measured lines', the decrement's, T_e's
    and n_e's from the lines' errors. Flags have none.
    """
    si, S = neb_out["inputs"], np.asarray(neb_out["S"], float)
    n = len(S)
    L, g = cov_pop["L"], cov_pop["g"]
    npar = cov_pop["npar"]
    nan = np.full(n, np.nan)

    def sd(grad):
        return np.sqrt(np.sum(np.einsum("nb,nbk->nk", grad, L) ** 2, axis=1))

    tied = np.asarray(si["av_source"]) != 1
    with np.errstate(divide="ignore", invalid="ignore"):
        N0 = np.asarray(neb_out["neb_spec"])[:, i0]
        T0 = np.asarray(af)[:, i0]
        gy = 100.0 * (g[:, 1] / T0[:, None] - (N0 / T0 ** 2)[:, None] * g[:, 2])
        e_a = np.zeros_like(g[:, 0])
        e_a[np.arange(n), np.where(tied, 0, npar)] = 1.0
        g_lnpred = g[:, 0] / S[:, None] - 0.4 * np.log(10.0) * k_hb * e_a
        s_lnpred = sd(g_lnpred)
        F2, E2 = neb_out["lines"]
        hb, ehb = F2.get("Hbeta", nan), E2.get("Hbeta", nan)
        c_hb = np.asarray(si.get("C_Hb", nan), float)
        av_neb = np.asarray(si["av_neb"], float)
        hb_pred = c_hb * S * 10.0 ** (-0.4 * av_neb * k_hb)
        rel_obs = ehb / hb
        out = {
            "AV_neb": np.where(tied, p_sig[:, 0], cov_pop["sig_a"]),
            "y_neb": sd(gy),
            "logQH": sd(g[:, 0]) / (S * np.log(10.0)),
            "tau_Hb": (hb_pred / hb) * np.hypot(s_lnpred, rel_obs),
            "Hb_pred": hb_pred * nf * s_lnpred,
            "Hb_obs": ehb * nf,
            "HaHb": np.asarray(si.get("decrement_err", nan), float),
            "Te": np.asarray(si.get("Te_err", nan), float),
            "ne": np.asarray(si.get("ne_err", nan), float),
        }
    return {k: (np.where(np.isfinite(v), v, np.nan)
                if formal_validated("NEBMAPS", k) else np.full(n, np.nan))
            for k, v in out.items()}

def error_hdus(products, errs, method, yc, xc, npar):
    """<EXT>_ERR for every product extension errs covers, aligned with
    the extension's own planes (by card for PoPBins and NEBMAPS)."""
    hdus = []
    ny, nx = products[1].data.shape[1:]

    def place(values_by_plane, n_planes):
        cube = np.full((n_planes, ny, nx), np.nan)
        for k, v in values_by_plane.items():
            if v is not None and k < n_planes:
                cube[k, yc, xc] = v
        return cube

    if "PoPBins" in errs and "PoPBins" in products:
        h = products["PoPBins"]
        n_planes = h.data.shape[0]
        by = {}
        for k in range(n_planes):
            name = h.header.get(f"DATA{k}")
            if name in errs["PoPBins"] and name not in NO_ERROR:
                by[k] = errs["PoPBins"][name]
        hdus.append(err_hdu(h, place(by, n_planes), method))
    for ext in ("PoPVecsL", "PoPVecsM", "PoPVecsMini"):
        if ext in errs and ext in products:
            v = np.asarray(errs[ext])
            hdus.append(err_hdu(products[ext], place(
                {k: v[:, k] for k in range(v.shape[1])},
                products[ext].data.shape[0]), method))
    if "KINEMATICS" in errs and "KINEMATICS" in products:
        v = np.asarray(errs["KINEMATICS"])
        hdus.append(err_hdu(products["KINEMATICS"], place(
            {k: v[:, k] for k in range(min(npar, v.shape[1]))},
            products["KINEMATICS"].data.shape[0]), method))
    if "NEBMAPS" in errs and "NEBMAPS" in products:
        h = products["NEBMAPS"]
        n_planes = h.data.shape[0]
        by = {k: errs["NEBMAPS"][h.header[f"PLANE{k}"]]
              for k in range(n_planes)
              if h.header.get(f"PLANE{k}") in errs["NEBMAPS"]
              and h.header.get(f"PLANE{k}") not in NO_ERROR}
        hdus.append(err_hdu(h, place(by, n_planes), method))
    if "LSFCORR" in errs and "LSFCORR" in products:
        h = products["LSFCORR"]
        hdus.append(err_hdu(h, place(errs["LSFCORR"], h.data.shape[0]),
                            method))
    return hdus
