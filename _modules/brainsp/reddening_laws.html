def _red_fitzpatrick(wavelength_angstroms, al0, ga, c1, c2, c3, c4):
    """
    Helper function for HZ2 (Seaton) and HZ3 (Fitzpatrick) laws.
    Ported from SUBROUTINE red (lines 7384-7395).
    """
    ala = wavelength_angstroms * 1.0e-4
    p = 1.0 / ala
    
    _c4 = jnp.where(p < 5.9, 0.0, c4)
        
    t1 = c3 / (((p - (al0**2 / p))**2) + ga*ga)
    t2 = _c4 * (0.539 * ((p - 5.9)**2) + 0.0564 * ((p - 5.9)**3))
    
    ff = c1 + c2*p + t1 + t2
    return ff

def _linear_interp(ws, vals, w_query):
    """Simple linear interpolation for table lookups."""
    ws_j = jnp.asarray(ws)
    vals_j = jnp.asarray(vals)
    w_q = jnp.atleast_1d(jnp.asarray(w_query))
    return jnp.interp(w_q, ws_j, vals_j, left=vals_j[0], right=vals_j[-1])

def ccm_89_law(wavelength, rv=3.1):
    """
    Cardelli, Clayton & Mathis (1989), q(lambda) = A_lambda / A_V = a + b/R_V.

    CCM is a one-parameter family, and R_V sets its SHAPE through b/R_V, not
    only its normalisation. R_V used to be fixed at 3.1 here, so a caller that
    passed another R_V got the 3.1 curve scaled by R_V * E(B-V): against
    the correct a + b/R_V, that was wrong by up to 11 per cent
    in the correction at R_V = 5. The default is still 3.1, so every in-fit use
    -- all of which pass no R_V -- is unchanged.
    Ported from FUNCTION Cardelli_RedLaw.
    """
    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    x = 10000.0 / wl
    
    y_fuv = x - 8.0
    a_fuv = -1.073 - 0.628*y_fuv + 0.137*(y_fuv**2) - 0.070*(y_fuv**3)
    b_fuv = 13.670 + 4.257*y_fuv - 0.420*(y_fuv**2) + 0.374*(y_fuv**3)
    q_fuv = a_fuv + b_fuv/rv

    y_bump = x - 5.9
    fa = jnp.where(x >= 5.9, -0.04473*(y_bump**2) - 0.009779*(y_bump**3), 0.0)
    fb = jnp.where(x >= 5.9, 0.2130*(y_bump**2) + 0.1207*(y_bump**3), 0.0)
    a_uv = 1.752 - 0.316*x - 0.104/((x - 4.67)**2 + 0.341) + fa
    b_uv = -3.090 + 1.825*x + 1.206/((x - 4.62)**2 + 0.263) + fb
    q_uv = a_uv + b_uv/rv

    y_opt = x - 1.82
    a_opt = 1.0 + 0.17699*y_opt - 0.50447*(y_opt**2) - 0.02427*(y_opt**3) + 0.72085*(y_opt**4) + 0.01979*(y_opt**5) - 0.77530*(y_opt**6) + 0.32999*(y_opt**7)
    b_opt = 1.41338*y_opt + 2.28305*(y_opt**2) + 1.07233*(y_opt**3) - 5.38434*(y_opt**4) - 0.62251*(y_opt**5) + 5.30260*(y_opt**6) - 2.09002*(y_opt**7)
    q_opt = a_opt + b_opt/rv

    a_ir = 0.574 * x**1.61
    b_ir = -0.527 * x**1.61
    q_ir = a_ir + b_ir/rv

    q = jnp.where((x >= 8.0) & (x <= 10.0), q_fuv,
        jnp.where((x >= 3.3) & (x < 8.0), q_uv,
        jnp.where((x >= 1.1) & (x < 3.3), q_opt,
        jnp.where((x >= 0.3) & (x < 1.1), q_ir,
        0.0))))

    return q

def cal_00_law(wavelength):
    """
    Calzetti et al. (2000) - Starburst.
    Ported from FUNCTION Calzetti_RedLaw.
    """
    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    x = 10000.0 / wl
    rv = 4.05

    q1 = (2.659/rv)*(-2.156 + 1.509*x - 0.198*(x**2) + 0.011*(x**3)) + 1.0
    q2 = (2.659/rv)*(-1.857 + 1.040*x) + 1.0

    q = jnp.where((wl >= 1200.0) & (wl <= 6300.0), q1,
        jnp.where((wl > 6300.0) & (wl <= 22000.0), q2,
        0.0))
    return q

def ccc_law(wavelength):
    """
    Calzetti + Claus + Cid (Hybrid)
    Ported from FUNCTION CCC_RedLaw.
    Leitherer (2002) for FUV (<1846), Calzetti (2000) for Optical/IR.
    """
    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    x = 10000.0 / wl
    rv = 4.05
    
    q1 = (5.472 + 0.671*x - 9.218e-3*(x**2) + 2.620e-3*(x**3)) / rv
    q2 = (2.659/rv)*(-2.156 + 1.509*x - 0.198*(x**2) + 0.011*(x**3)) + 1.0
    q3 = (2.659/rv)*(-1.857 + 1.040*x) + 1.0

    q = jnp.where((wl >= 912.0) & (wl <= 1846.0), q1,
        jnp.where((wl > 1846.0) & (wl <= 6300.0), q2,
        jnp.where((wl > 6300.0) & (wl <= 31149.0), q3,
        0.0)))
    return q

def hz1_allen_mw(wavelength):
    """
    HZ1: Allen (1976) - Milky Way.
    Table lookup + linear interpolation.
    """
    wab = jnp.array([1000., 1110., 1250., 1430., 1670., 2000., 2220., 2500., 2850., 3330., 
                    3650., 4000., 4400., 5000., 5530., 6700., 9000., 10000., 20000., 100000.])

    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    return jnp.interp(wl, wab, amab, left=amab[0], right=amab[-1])

def hz2_seaton_mw(wavelength):
    """
    HZ2: Seaton (1979) fit by Fitzpatrick (1986) - MW.
    Hybrid: Formula for < 1200, Table for > 3650.
    """
    wab = jnp.array([3650., 4000., 4400., 5000., 5530., 6700., 9000., 10000., 20000., 100000.])
    amab = jnp.array([1.58, 1.45, 1.32, 1.13, 1.00, 0.74, 0.46, 0.38, 0.11, 0.0])
    
    rv = 3.1
    al0, ga = 4.595, 1.051
    c1, c2, c3, c4 = -0.38, 0.74, 3.96, 0.26
    
    ff11 = _red_fitzpatrick(1100., al0, ga, c1, c2, c3, c4)
    ff12 = _red_fitzpatrick(1200., al0, ga, c1, c2, c3, c4)
    slope = (ff12 - ff11) / 100.0

    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    
    ff_below1200 = ff11 + (wl - 1100.)*slope
    ff_formula = _red_fitzpatrick(wl, al0, ga, c1, c2, c3, c4)
    
    q_below1200 = ff_below1200 / rv + 1.0
    q_formula = ff_formula / rv + 1.0
    q_table = jnp.interp(wl, wab, amab, left=amab[0], right=amab[-1])
    
    q = jnp.where(wl < 1200., q_below1200,
        jnp.where(wl <= 3650., q_formula,
        q_table))
    
    return q

def hz4_prevot_smc(wavelength):
    """
    HZ4: Prevot (1984) & Bouchet (1985) - SMC.
    Table lookup.
    """
    wab = jnp.array([1275., 1330., 1385., 1435., 1490., 1545., 1595., 1647., 1700., 1755., 
                    1810., 1860., 1910., 2000., 2115., 2220., 2335., 2445., 2550., 2665., 
                    2778., 2890., 2995., 3105., 3704., 4255., 5291., 12500., 16500., 22000.])
    
    amab = jnp.array([13.54, 12.52, 11.51, 10.80, 9.84, 9.28, 9.06, 8.49, 8.01, 7.71,
                     7.17, 6.90, 6.76, 6.38, 5.85, 5.30, 4.53, 4.24, 3.91, 3.49,
                     3.15, 3.00, 2.65, 2.29, 1.81, 1.00, 0.00, -2.02, -2.36, -2.47])
    
    rv = 2.72
    
    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    val = jnp.interp(wl, wab, amab, left=amab[0], right=amab[-1])
    q = (val / rv) + 1.0
    
    q = jnp.maximum(q, 0.0)
    return q

def gd03_law(wavelength, variant='GD1'):
    """
    Gordon et al. (2003).
    GD1: SMC Bar
    GD2: LMC2 Supershell
    GD3: LMC Average
    """
    x_gordon = np.array([0.455, 0.606, 0.800, 1.235, 1.538, 1.818, 
                         2.273, 2.703, 3.375, 3.625, 3.875, 4.125, 4.375,
                         4.625, 4.875, 5.125, 5.375, 5.625, 5.875, 6.125,
                         6.375, 6.625, 6.875, 7.125, 7.375, 7.625, 7.875, 
                         8.125, 8.375])
    
    wl_table = 10000.0 / x_gordon
    
    if variant == 'GD1':
        q_vals = np.array([0.016, 0.169, 0.131, 0.567, 0.801, 1.000, 
                           1.374, 1.672, 2.000, 2.220, 2.428, 2.661, 2.947, 
                           3.161, 3.293, 3.489, 3.637, 3.866, 4.013, 4.243, 
                           4.472, 4.776, 5.000, 5.272, 5.575, 5.795, 6.074, 
                           6.297, 6.436])
    elif variant == 'GD2':
        q_vals = np.array([0.101, 0.097, 0.299, 0.5004, 0.7600, 1.000, 
                           1.349, 1.665, 1.899, 2.067, 2.249, 2.447, 2.777, 
                           2.922, 2.921, 2.812, 2.805, 2.863, 2.932, 3.060, 
                           3.110, 3.299, 3.408, 3.515, 3.670, 3.862, 3.937, 
                           4.055, 3.969])
    elif variant == 'GD3':
        q_vals = np.array([0.030, 0.186, 0.257, 0.4705, 0.7457, 1.000,
                           1.293, 1.518, 1.786, 1.969, 2.149, 2.391, 2.771,
                           2.967, 2.846, 2.646, 2.565, 2.566, 2.598, 2.607,
                           2.668, 2.787, 2.874, 2.983, 3.118, 3.231, 3.374,
                           3.366, 3.467])
    else:
        q_vals = np.zeros_like(x_gordon)
        
    idx = np.argsort(wl_table)
    wl_sorted = jnp.array(wl_table[idx])
    q_sorted = jnp.array(q_vals[idx])
    
    wl = jnp.atleast_1d(jnp.asarray(wavelength))
    return jnp.interp(wl, wl_sorted, q_sorted, left=q_sorted[0], right=q_sorted[-1])

def get_k_lambda(wavelength, law_name="GD24", rv=3.1):
    """
    Returns the extinction curve value k(lambda) [or q(lambda) = A_lam/Av]
    for the specified law.
    """
    key = law_name.upper().strip()
    
    if key == "CCM": return ccm_89_law(wavelength, rv)
    if key == "CAL": return cal_00_law(wavelength)
    if key == "CCC": return ccc_law(wavelength)
    
    if key == "HZ1": return hz1_allen_mw(wavelength)
    if key == "HZ2": return hz2_seaton_mw(wavelength)
    if key == "HZ3": return hz3_fitzpatrick_lmc(wavelength)
    if key == "HZ4": return hz4_prevot_smc(wavelength)
    if key == "HZ5": return hz5_calzetti_sb(wavelength)
    
    if key == "GD1": return gd03_law(wavelength, 'GD1')
    if key == "GD2": return gd03_law(wavelength, 'GD2')
    if key == "GD3": return gd03_law(wavelength, 'GD3')
    
    if key == "GD24": return gd24_law(wavelength)
    
    raise ValueError(
        f"Unknown reddening law {law_name!r}. Available: "
        + ", ".join(LAW_NAMES))
