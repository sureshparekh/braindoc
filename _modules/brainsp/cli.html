"""
The ``brainsp`` command.

    brainsp gui                              the desktop interface
    brainsp fit galaxy.spec [more ...]       fit spectra
    brainsp cube cube.fits --redshift 0.02   fit a datacube
    brainsp selftest [--quick]               check the installation
    brainsp examples [FOLDER]                write the mock spectra and cube
    brainsp init run.py                      a run as one Python file
    brainsp init settings.yaml               the documented settings file
    brainsp check settings.yaml              validate settings, fit nothing
    brainsp settings [NAME]                  what a setting does
    brainsp libraries [info NAME | install FOLDER]
    brainsp plot RESULT [--map NAME] -o FILE a figure from a result
    brainsp desktop [--remove]               add BRAIN to the applications menu
    brainsp clear-cache                      forget compiled kernels and templates
    brainsp version

``brainsp COMMAND --help`` describes each one. Any setting can be given on
the command line as ``--set name=value`` (repeatable); values are read as
YAML, so ``--set dust_range=[0,3]`` and ``--set nebular_mode=espop`` work.
"""

import argparse
import os
import sys


def _settings(args):
    import yaml
    import brainsp
    s = brainsp.Settings(args.settings) if getattr(args, "settings", None) \
        else brainsp.Settings()
    for item in getattr(args, "set", None) or []:
        if "=" not in item:
            raise SystemExit(f"--set needs name=value, got {item!r}")
        key, value = item.split("=", 1)
        s[key.strip()] = yaml.safe_load(value)
    return s


def _fit(args):
    import glob
    import brainsp
    s = _settings(args)
    spectra = []
    for pattern in args.spectra:
        found = sorted(glob.glob(pattern))
        spectra += found or [pattern]
    if not spectra and not args.corrections:
        if not s.get("spectrum"):
            raise SystemExit("Name the spectra to fit, or set 'spectrum' in "
                             "the settings file.")
        spectra = sorted(glob.glob(str(s["spectrum"]))) or [s["spectrum"]]
    results = brainsp.fit_spectra(spectra or None, s,
                                  corrections=args.corrections,
                                  output_dir=args.output, device=args.device,
                                  quiet=args.quiet)
    print()
    for r in results:
        print(f"  {r!r}")
    return 1 if any(r.error_message for r in results) else 0


def _cube(args):
    import brainsp
    if not args.cubes and not args.table:
        raise SystemExit("Name the cubes, or give a --table whose first "
                         "column names them.")
    s = _settings(args)
    pipeline = brainsp.CubePipeline(
        args.cubes, redshift=args.redshift, settings=s,
        output_dir=args.output, flux_hdu=args.flux_hdu,
        error_hdu=args.error_hdu, error_type=args.error_type,
        mask_hdu=None if args.mask_hdu in ("", "none", "None") else
        args.mask_hdu, wavelength_scale=args.wavelength_scale,
        galactic_ebv=args.ebv, galactic_rv=args.rv,
        sn_mask=args.sn or None, fit_sn_threshold=args.fit_sn,
        wavelength_mask=args.mask, batch_size=args.batch_size,
        per_cube=args.table, spatial_binning=args.bin)
    if args.export:
        pipeline.export_script(args.export, device=args.device or "auto")
        print(f"wrote {args.export}; run it with: python {args.export}")
        return 0
    if args.check:
        pipeline.validate()
        print("valid")
        return 0
    written = pipeline.run(device=args.device)
    print("\nMegacubes written:\n  " + "\n  ".join(written))
    expected = len(pipeline._per_cube_rows()[1])
    return 0 if len(written) == expected else 1


def _selftest(args):
    from brainsp.selftest import run
    report = run(quick=args.quick, device=args.device, folder=args.keep,
                 keep=True if args.keep else None)
    return 0 if report.passed else 1


def _examples(args):
    from brainsp.mock import write_examples
    folder = write_examples(args.folder, cube=not args.no_cube,
                            overwrite=args.overwrite)
    print(f"Mock data written to {folder}:")
    for root, _, files in sorted(os.walk(folder)):
        for name in sorted(files):
            print("  " + os.path.relpath(os.path.join(root, name), folder))
    return 0


def _init(args):
    import shutil
    from brainsp._paths import SETTINGS_TEMPLATE
    path = args.path
    if os.path.exists(path) and not args.overwrite:
        raise SystemExit(f"{path} exists; pass --overwrite to replace it.")
    if path.endswith((".yaml", ".yml")):
        shutil.copyfile(SETTINGS_TEMPLATE, path)
        print(f"wrote {path}: every setting, documented. Edit it, then "
              f"run: brainsp fit --settings {path}")
        return 0
    import brainsp
    if args.kind == "cube":
        brainsp.write_script(path, kind="cube",
                             cubes=args.inputs or ["my_cube.fits"],
                             redshift=None)
    else:
        brainsp.write_script(path, spectra=args.inputs or [])
    print(f"wrote {path}: one Python file with every setting. Edit it, "
          f"then run: python {path}")
    return 0


def _check(args):
    import brainsp
    s = _settings(args)
    s.validate(for_cube=args.cube)
    print(f"{args.settings or 'the default settings'}: valid")
    return 0


def _settings_cmd(args):
    import brainsp
    print(brainsp.Settings.describe(args.name))
    return 0


def _libraries(args):
    from brainsp import libraries
    if args.action == "install":
        print(libraries.install(args.target, overwrite=args.overwrite))
        return 0
    if args.action == "info":
        info = libraries.info(args.target or "BasesGM")
        for key, value in info.items():
            if isinstance(value, list) and len(value) > 8:
                value = f"{len(value)}: {value[0]:g} ... {value[-1]:g}"
            print(f"  {key:<16} {value}")
        return 0
    print("Template libraries (first match wins):")
    for name in libraries.available():
        print(f"  {name:<20} {libraries.path(name)}")
    print("\nSearched: " + ", ".join(libraries.search_path()))
    return 0


def _plot(args):
    import brainsp
    target = args.result
    out = args.output
    if target.endswith(".fits"):
        cube = brainsp.read_megacube(target)
        if args.diagnostic:
            brainsp.plots.diagnostic(cube, args.diagnostic, out)
        elif args.spaxel:
            y, x = (int(v) for v in args.spaxel.split(","))
            brainsp.plots.spaxel(cube, y, x, out)
        elif args.export:
            written = brainsp.plots.export_maps(
                cube, args.export, args.map or None,
                formats=tuple(args.formats.split(",")))
            print(f"{len(written)} files in {args.export}")
            return 0
        else:
            if not args.map:
                print("Maps in this megacube:\n  "
                      + "\n  ".join(cube.names()))
                return 0
            if len(args.map) == 1:
                brainsp.plots.map(cube, args.map[0], out, cmap=args.cmap)
            else:
                brainsp.plots.maps(cube, args.map, out, cmap=args.cmap)
    else:
        r = brainsp.read_result(target)
        if args.sfh:
            brainsp.plots.sfh(r, out)
        else:
            brainsp.plots.spectrum(r, out, xlim=tuple(
                float(v) for v in args.xlim.split(",")) if args.xlim
                else None)
    print(f"wrote {out}")
    return 0

def main(argv=None):
    argv = sys.argv[1:] if argv is None else argv
    parser = argparse.ArgumentParser(
        prog="brainsp",
        description="BRAIN: stellar population synthesis of spectra and "
                    "datacubes.",
        epilog="Documentation: https://sureshparekh.github.io/braindoc/")
    sub = parser.add_subparsers(dest="command")

    def common(p, device=True):
        p.add_argument("--settings", help="a settings file (YAML)")
        p.add_argument("--set", action="append", metavar="NAME=VALUE",
                       help="change one setting (repeatable)")
        if device:
            p.add_argument("--device", choices=("auto", "cpu", "gpu"),
                           help="processor to fit on")

    p = sub.add_parser("gui", help="open the desktop interface")
    p.set_defaults(run=lambda a: __import__(
        "brainsp.brain_gui", fromlist=["main"]).main() or 0)

    p = sub.add_parser("fit", help="fit one or more spectra")
    p.add_argument("spectra", nargs="*", help="files or glob patterns")
    common(p)
    p.add_argument("-o", "--output", help="output folder (default brain_out)")
    p.add_argument("--corrections", help="CSV: filename, redshift, ebv, rv, "
                                         "law, rest_frame")
    p.add_argument("-q", "--quiet", action="store_true")
    p.set_defaults(run=_fit)

    p = sub.add_parser("cube", help="fit datacubes")
    p.add_argument("cubes", nargs="*",
                   help="FITS cubes (or give them in --table)")
    common(p)
    p.add_argument("--redshift", type=float, help="the galaxy's z (required "
                   "unless --table gives one per cube; 0 = rest frame)")
    p.add_argument("-o", "--output", default="brain_ifu_out")
    p.add_argument("--flux-hdu", default="FLUX")
    p.add_argument("--error-hdu", default="IVAR")
    p.add_argument("--error-type", default="inv_variance",
                   choices=("inv_variance", "variance", "error",
                            "inv_error"))
    p.add_argument("--mask-hdu", default="MASK")
    p.add_argument("--wavelength-scale", choices=("air", "vacuum"))
    p.add_argument("--ebv", type=float, default=0.0,
                   help="Galactic E(B-V) to remove")
    p.add_argument("--rv", type=float, default=3.1)
    p.add_argument("--sn", type=float, nargs="*",
                   help="S/N thresholds to make masks for, e.g. 3 5 10")
    p.add_argument("--fit-sn", type=float,
                   help="fit only spaxels with S/N above this")
    p.add_argument("--mask", help="wavelength mask file (.sm/.gm/list)")
    p.add_argument("--bin", type=int, help="bin spaxels n x n")
    p.add_argument("--batch-size", type=int)
    p.add_argument("--table", help="CSV of per-cube settings")
    p.add_argument("--check", action="store_true", help="validate only")
    p.add_argument("--export", metavar="FILE.py",
                   help="write the run as one Python file instead")
    p.set_defaults(run=_cube)

    p = sub.add_parser("selftest", help="check the installation")
    p.add_argument("--quick", action="store_true")
    p.add_argument("--device", default="auto",
                   choices=("auto", "cpu", "gpu"))
    p.add_argument("--keep", metavar="FOLDER",
                   help="write the mocks and fits here and keep them")
    p.set_defaults(run=_selftest)

    p = sub.add_parser("examples", help="write the mock spectra and cube")
    p.add_argument("folder", nargs="?", help="default ~/.brainsp/examples")
    p.add_argument("--no-cube", action="store_true")
    p.add_argument("--overwrite", action="store_true")
    p.set_defaults(run=_examples)

    p = sub.add_parser("init", help="write a run script (.py) or a settings "
                                    "file (.yaml)")
    p.add_argument("path")
    p.add_argument("inputs", nargs="*", help="spectra or cubes to put in it")
    p.add_argument("--kind", choices=("spectrum", "cube"), default="spectrum")
    p.add_argument("--overwrite", action="store_true")
    p.set_defaults(run=_init)

    p = sub.add_parser("check", help="validate settings without fitting")
    p.add_argument("settings", nargs="?")
    p.add_argument("--set", action="append", metavar="NAME=VALUE")
    p.add_argument("--cube", action="store_true",
                   help="settings for a datacube (no spectrum needed)")
    p.set_defaults(run=_check)

    p = sub.add_parser("settings", help="describe the settings")
    p.add_argument("name", nargs="?")
    p.set_defaults(run=_settings_cmd)

    p = sub.add_parser("libraries", help="list, describe or install "
                                         "template libraries")
    p.add_argument("action", nargs="?", choices=("list", "info", "install"),
                   default="list")
    p.add_argument("target", nargs="?")
    p.add_argument("--overwrite", action="store_true")
    p.set_defaults(run=_libraries)

    p = sub.add_parser("plot", help="draw a figure from a result")
    p.add_argument("result", help="a *_summary.txt or a megacube")
    p.add_argument("-o", "--output", default="brainsp_plot.png")
    p.add_argument("--map", nargs="*", help="map name(s) of a megacube")
    p.add_argument("--cmap", default="viridis")
    p.add_argument("--diagnostic", choices=("BPT [N II]", "BPT [S II]",
                                            "WHAN"))
    p.add_argument("--spaxel", metavar="Y,X")
    p.add_argument("--export", metavar="FOLDER",
                   help="write maps to files (all, or --map ones)")
    p.add_argument("--formats", default="png,fits")
    p.add_argument("--sfh", action="store_true",
                   help="star-formation history of a spectrum fit")
    p.add_argument("--xlim", metavar="LO,HI")
    p.set_defaults(run=_plot)

    p = sub.add_parser("desktop", help="add BRAIN to the applications menu")
    p.add_argument("--remove", action="store_true")
    p.set_defaults(run=lambda a: __import__(
        "brainsp.install_desktop", fromlist=["main"]).main(
            ["--remove"] if a.remove else []))

    p = sub.add_parser("clear-cache", help="forget compiled kernels and "
                                           "prepared templates")
    p.set_defaults(run=lambda a: (__import__(
        "brainsp.brain_cache", fromlist=["clear"]).clear("all"),
        print("caches cleared"), 0)[-1])

    p = sub.add_parser("version", help="print the version")
    p.set_defaults(run=lambda a: (print(__import__("brainsp").__version__),
                                  0)[-1])

    args = parser.parse_args(argv)
    if not getattr(args, "command", None):
        parser.print_help()
        return 0
    try:
        return int(args.run(args) or 0)
    except SystemExit as exc:
        if exc.code in (None, 0) or isinstance(exc.code, int):
            raise
        print(str(exc.code), file=sys.stderr)
        return 2
    except (KeyError, ValueError) as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 2




if __name__ == "__main__":
    sys.exit(main())
