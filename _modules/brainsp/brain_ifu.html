#!/usr/bin/env python3
"""
BRAIN — the IFU runner.

    python -m brainsp.brain_ifu run  my_pipeline.json
    python -m brainsp.brain_ifu check my_pipeline.json
    python -m brainsp.brain_ifu defaults > my_pipeline.json

A pipeline is an ordered chain of pre-pass modules followed by the fit, run
over a list of targets. Each target is opened once, every stage appends the
HDUs it produced to that open list, and the list that accumulates is the
megacube, in the format of the MaNGA megacubes of Riffel et al. (2023, MNRAS
524, 5640), so a megacube BRAIN writes opens in tooling written for those.

Targets run one after another rather than in parallel. A single cube's batch
already saturates the machine through XLA's own thread pool, so a second cube
alongside it contends for the same cores and doubles peak memory for no
throughput.

The chaining of extension names between stages lives here rather than in the
desktop interface, so the command line gets it too: a resampler run on the
flux and on the inverse variance produces FLUX_RSP and IVAR_RSP, dereddening
produces FLUX_DRD, and the fit is handed whatever the last stage called its
output. Getting one of those names wrong by hand fails late, inside the fit.
"""

def _optional_number(value):
    """A number from a setting or a table cell; None when blank."""
    if value is None or (isinstance(value, str) and not value.strip()):
        return None
    return float(value)

def extra_bin_errors(rows, label, positive=True):
    """What is wrong with a list of [name, min, max] bins (SFR, FC, BB)."""
    if not rows:
        return []
    try:
        bins = age_bins(rows)
    except (TypeError, ValueError, IndexError):
        return [f"A {label} bin needs three fields: a name, a minimum and a "
                f"maximum."]
    errors = []
    names = [n for n, _, _ in bins]
    if len(set(names)) != len(names):
        errors.append(f"Two {label} bins have the same name.")
    for name, low, high in bins:
        if not name:
            errors.append(f"A {label} bin has no name.")
        elif low >= high:
            errors.append(f"{label} bin {name!r} runs from {low:g} to "
                          f"{high:g}; the first must be the smaller.")
        elif positive and high <= 0:
            errors.append(f"{label} bin {name!r} ends at {high:g}.")
    return errors

def age_bins(rows):
    """
    Population age bins as (name, min, max) in years, from whatever they
    arrived as.

    A pipeline file, a CSV cell and the desktop interface all describe a bin
    as three fields in a row, and the numbers reach here as strings often
    enough that converting them at one point is the only way the wrong type
    does not surface as a header card that reads '1e7' inside a map nobody
    looks at twice.
    """
    out = []
    for row in rows or []:
        if isinstance(row, str):
            row = [part.strip() for part in row.split(",")]
        name, low, high = row[0], float(row[1]), float(row[2])
        out.append((str(name).strip(), low, high))
    return out

def age_bin_errors(rows):
    """
    What is wrong with a set of age bins, before a cube is opened.

    Bins may overlap: a total such as xy_Tot (all of xyy and xiy) is a bin
    of its own, and each bin is its own map, so nothing is counted twice
    within a map (the bins' maps are not meant to be added). Bins may also
    leave gaps -- the published sets do -- and the fit reports how much
    light falls into no bin.
    """
    errors = []
    try:
        bins = age_bins(rows)
    except (TypeError, ValueError, IndexError):
        return ["A population bin needs three fields: a name, the youngest "
                "age in years and the oldest."]
    if not bins:
        return ["No population age bin is defined, so PoPBins would carry no "
                "light or mass map."]
    names = [name for name, _, _ in bins]
    if len(set(names)) != len(names):
        errors.append("Two population bins have the same name.")
    for name, low, high in bins:
        if not name:
            errors.append("A population bin has no name.")
        if low <= 0 or high <= 0:
            errors.append(f"Population bin {name!r} has an age of zero or "
                          f"less; ages are in years.")
        elif low >= high:
            errors.append(f"Population bin {name!r} runs from {low:g} to "
                          f"{high:g} yr; the first must be the smaller.")
    return errors

class BrainFit(IFUModule):
    """
    The batched fit, as a stage of the chain. Writes the fit extensions.

    Every physics setting comes from the settings file named in `settings`,
    exactly as for a single spectrum: the template library, the fitting range,
    the extinction law, the nebular mode, the solver and the bounds all have
    one place they are declared. What belongs here is only what is about the
    cube -- which extension holds the flux, which the uncertainty and of
    which kind, which plane selects the spaxels to fit, and how large a batch
    to compile for.

    Three keys override the settings file per target, because they are the
    three that legitimately differ between cubes in one run: `redshift` sets
    `cube_redshift`, `mask file` sets `mask`, and `wavelength scale` sets
    `wavelength_scale`. Left at None, the settings file's own values stand.

    `wavelength scale` is here rather than in the settings file's exclusive
    keeping for the same reason `redshift` is: it is a fact about the
    instrument that produced this cube, not about the physics being fitted,
    and a run that sweeps cubes from two surveys has two answers to it. It
    takes 'air' or 'vacuum' and nothing else; the validator refuses every
    other spelling by name, so the control that sets it emits the exact
    token rather than anything it has had to case-fold.

    There is no timeout: a vmapped batch has a fixed iteration count and
    cannot hang on one spaxel.
    """

    name = "BRAIN fit"

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.primary_cards = []
        self.summary = {}

    def _set_default_parameters(self):
        self.default_parameters["hdu flux"] = "FLUX"
        self.default_parameters["hdu ivar"] = "IVAR"
        self.default_parameters["stat type"] = "inv_variance"
        self.default_parameters["hdu mask"] = "MASK"
        self.default_parameters["hdu flag"] = None
        self.default_parameters["settings"] = "brain_setup.yaml"
        self.default_parameters["batch size"] = None
        self.default_parameters["redshift"] = None
        self.default_parameters["mask file"] = None
        self.default_parameters["wavelength scale"] = None
        self.default_parameters["population bins"] = [
            list(b) for b in DEFAULT_AGE_BINS]
        self.default_parameters["flxobs step"] = LINEAR_WAVELENGTH_STEP
        # Optional maps: SFR in age bins, featureless continua grouped by
        # power-law exponent or blackbody temperature, each [name, min, max]
        # with min exclusive and max inclusive; and the distance (Mpc) and
        # flux-unit factor that turn apparent masses into Msun (Mstar,
        # Mpcross and the SFRs are written only with a distance).
        self.default_parameters["sfr bins"] = []
        self.default_parameters["fc bins"] = []
        self.default_parameters["bb bins"] = []
        self.default_parameters["galaxy distance"] = None
        self.default_parameters["flux scale"] = None

    def configuration(self):
        """The validated settings this stage would fit with."""
        from brainsp import brain_settings

        path = str(self["settings"])
        settings = brain_settings.load(path)
        if self["redshift"] is not None:
            settings["cube_redshift"] = float(self["redshift"])
            settings["spectrum_redshift"] = 0.0
        if self["mask file"]:
            settings["mask"] = str(self["mask file"])
        if self["wavelength scale"]:
            settings["wavelength_scale"] = str(self["wavelength scale"])
        settings = brain_settings.validate(settings, path, for_cube=True)
        return brain_settings.to_config(settings)

    def execute(self, hdu_list):
        from brainsp.fit_megacube_jax import fit_cube_hdus

        cfg = self.configuration()
        batch = self["batch size"]

        def report(done, total):
            self.progress_hook(done / max(total, 1),
                               f"batch {done} of {total}")

        result = fit_cube_hdus(
            cfg, hdu_list,
            hdu_flux=self["hdu flux"], hdu_ivar=self["hdu ivar"],
            hdu_mask=self["hdu mask"], hdu_flag=self["hdu flag"],
            stat_type=str(self["stat type"]),
            batch_size=None if batch in (None, "", 0) else int(batch),
            progress=report, age_bins=age_bins(self["population bins"]),
            flxobs_step=float(self["flxobs step"]),
            spaxels=self.spaxel_hook,
            sfr_bins=age_bins(self["sfr bins"]),
            fc_bins=age_bins(self["fc bins"]),
            bb_bins=age_bins(self["bb bins"]),
            galaxy_distance=_optional_number(self["galaxy distance"]),
            flux_scale=_optional_number(self["flux scale"]))
        self.primary_cards = result["cards"]
        self.summary = result["summary"]
        return result["products"]

def build_module_chain(cfg):
    """
    The enabled stages, in order, with the extension names already threaded.

    Returned as (class name, configuration) pairs rather than as constructed
    objects so that the same function serves the live runner and the script
    exporter, and the exported script therefore cannot describe a pipeline
    different from the one the runner would have executed.

    The threading is the part that is easy to get wrong by hand. A resampler
    or a degrader runs twice, once on the flux and once on the statistical
    plane with the data type set so the two combine correctly, and each pass
    renames its output; dereddening always writes the fixed name FLUX_DRD;
    the S/N mask writes SN_MASKS_<threshold>, which the fit then consumes as
    its selection plane. Each stage is handed the name the stage before it
    produced.

    A stage that changes the shape of the cube drops the data-quality mask
    rather than carrying one that no longer lines up with the data: a flag is
    not an additive quantity, so there is no honest way to block-sum it or to
    interpolate it onto a new wavelength grid.
    """
    chain = []
    flux = cfg["input"]["hdu flux"]
    stat = cfg["input"]["hdu stat"]
    stat_type = cfg["input"]["stat type"]
    mask = cfg["input"].get("hdu mask") or None

    spatial = cfg["spatial"]
    if spatial["enabled"]:
        size = spatial["resample size"]
        chain.append(("SpatialResampler", {
            "hdu target": flux, "data type": "flux", "resample size": size}))
        chain.append(("SpatialResampler", {
            "hdu target": stat, "data type": stat_type,
            "resample size": size}))
        flux, stat = f"{flux}_RSP", f"{stat}_RSP"
        mask = None

    resolution = cfg["resolution"]
    if resolution["enabled"]:
        def degrade(target, data_type):
            return {"hdu target": target, "data type": data_type,
                    "input type": resolution["input type"],
                    "input value": resolution["input value"],
                    "input ref wave": resolution["input ref wave"],
                    "output type": resolution["output type"],
                    "output value": resolution["output value"],
                    "output ref wave": resolution["output ref wave"]}
        chain.append(("DegradeResolution", degrade(flux, "flux")))
        chain.append(("DegradeResolution", degrade(stat, stat_type)))
        flux, stat = f"{flux}_DEGR", f"{stat}_DEGR"

    spectral = cfg["spectral_bin"]
    if spectral["enabled"]:
        size = spectral["resample size"]
        chain.append(("SpectralResampler", {
            "hdu target": flux, "data type": "flux", "resample size": size}))
        chain.append(("SpectralResampler", {
            "hdu target": stat, "data type": stat_type,
            "resample size": size}))
        flux, stat = f"{flux}_BIN", f"{stat}_BIN"
        mask = None

    filtering = cfg["filter"]
    if filtering["enabled"]:
        chain.append(("ButterworthFilter", {
            "hdu flux": flux, "order": filtering["order"],
            "range": filtering["range"]}))
        flux = "FLUX_BW"

    dereddening = cfg["dereddening"]
    if dereddening["enabled"]:
        chain.append(("Dereddening", {
            "hdu flux": flux, "law": dereddening["law"],
            "ebv": dereddening["ebv"], "rv": dereddening["rv"]}))
        flux = "FLUX_DRD"

    sn_mask = cfg["sn_mask"]
    if sn_mask["enabled"]:
        mask_cfg = {"hdu flux": flux, "method": sn_mask["method"],
                    "sn window": list(sn_mask["sn window"]),
                    "thresholds": list(sn_mask["thresholds"]),
                    "redshift": sn_mask["redshift"]}
        stat_key = SN_METHOD_STAT_KEY.get(sn_mask["method"])
        if stat_key is not None:
            mask_cfg[stat_key] = stat
        chain.append(("SNMask", mask_cfg))

    fit = cfg["fit"]
    if fit["enabled"]:
        fit_cfg = {"hdu flux": flux, "hdu ivar": stat, "hdu mask": mask,
                   "stat type": stat_type,
                   "settings": fit["settings"],
                   "batch size": fit["batch size"],
                   "redshift": fit["redshift"],
                   "mask file": fit["mask file"],
                   "wavelength scale": fit.get("wavelength scale"),
                   "population bins": [list(b) for b in
                                       fit.get("population bins")
                                       or DEFAULT_AGE_BINS],
                   "flxobs step": fit.get("flxobs step",
                                          LINEAR_WAVELENGTH_STEP),
                   "sfr bins": [list(b) for b in fit.get("sfr bins") or []],
                   "fc bins": [list(b) for b in fit.get("fc bins") or []],
                   "bb bins": [list(b) for b in fit.get("bb bins") or []],
                   "galaxy distance": fit.get("galaxy distance"),
                   "flux scale": fit.get("flux scale"),
                   "hdu flag": None}
        if sn_mask["enabled"] and fit["flag threshold"] is not None:
            fit_cfg["hdu flag"] = f"SN_MASKS_{int(fit['flag threshold'])}"
        chain.append(("BrainFit", fit_cfg))

    return chain

def accepted_columns(cfg):
    """
    Every per-target override key the enabled stages of this chain declare.

    Which keys those are depends on which stages are switched on, so the set
    is derived from the chain rather than listed somewhere: an interface
    offering a column the chain has no reader for is offering a column that
    does nothing.
    """
    accepted = set()
    for class_name, _ in build_module_chain(cfg):
        accepted |= set(STAGE_CLASSES[class_name]().config)
    return accepted

def override_column_errors(cfg):
    """
    Refuse a per-target column no stage in the chain would read.

    A module ignores a key it never declared, which is what lets one row
    carry columns for several stages -- and is also how a column named
    'reddening' instead of 'ebv' runs a whole campaign at the wrong
    extinction without a word. The keys are known before the run, so the
    mismatch is caught here instead of being discovered in the output.
    """
    columns = [str(c).strip() for c in cfg["targets"]["columns"]]
    if not columns:
        return []
    accepted = accepted_columns(cfg)
    unknown = [c for c in columns[1:] if c and c not in accepted]
    if not unknown or not accepted:
        return []
    return [f"The targets table has a column {c!r}, which no enabled stage "
            f"reads. The columns the chain accepts are: "
            + ", ".join(sorted(accepted)) + "." for c in unknown]

SN_METHOD_FOR_STAT = {"inv_variance": ("ivar", "ivar_median"),
                      "variance": ("variance",), "error": ("error",)}

def sn_method_problem(cfg):
    """
    Why the S/N method cannot read the uncertainty the cube has, or None.

    The S/N stage is handed the uncertainty extension under the name its
    method reads (ivar, var, error), whatever that extension holds. An ivar
    formula given a 1-sigma error computes mean(flux * sqrt(error)), which
    for a JWST cube in erg/s/cm^2/A is of order 1e-28: every spaxel falls
    below every threshold and the fit is handed nothing.
    """
    method = cfg["sn_mask"]["method"]
    stat_type = cfg["input"].get("stat type")
    allowed = SN_METHOD_FOR_STAT.get(stat_type)
    if method == "mean_std" or allowed is None or method in allowed:
        return None
    return (f"The S/N is measured as {method!r}, which reads "
            f"{dict(ivar='an inverse variance', ivar_median='an inverse variance', variance='a variance', error='an error').get(method, method)}, "
            f"but the uncertainty extension holds {stat_type!r}: use "
            f"{allowed[0]!r} (or 'mean_std', which needs no uncertainty).")

def missing_extensions(cfg, path):
    """
    The extensions the run reads that the cube at `path` does not have.

    Checked before anything runs: a name the file lacks used to surface as a
    KeyError from inside whichever stage first needed it, after the run had
    started, with nothing pointing at the field that named it.
    """
    want = [("Flux", cfg["input"]["hdu flux"])]
    if cfg["input"].get("stat type") != "flux" or cfg["fit"]["enabled"]:
        want.append(("Uncertainty", cfg["input"]["hdu stat"]))
    if cfg["input"].get("hdu mask"):
        want.append(("Quality mask", cfg["input"]["hdu mask"]))
    try:
        with fits.open(path, memmap=True) as hdul:
            names = [h.name for h in hdul]
    except Exception as exc:
        return [f"{os.path.basename(path)} cannot be opened: {exc}"]
    return [f"{field} extension {name!r} is not in "
            f"{os.path.basename(path)}, which holds {', '.join(names)}."
            for field, name in want if name and name not in names]

def validate_detailed(cfg, check_paths=True):
    """
    Everything that can be refused before a single spaxel is read, each
    message paired with the section of the pipeline it is about.

    The pairing is what lets an interface put a message next to the control
    that is wrong instead of only in a dialog, and it belongs here rather than
    in the interface because the code that decides a setting is wrong is the
    only code that knows which setting it was.

    check_paths=False skips the existence tests, so a pipeline can be
    configured against a directory that only exists on the machine it will
    eventually run on. Everything else is checked either way, because a
    threshold that does not match, or a stage chain that produces nothing,
    is wrong on every machine.
    """
    errors = []

    targets = cfg["targets"]
    rows = [r for r in targets["rows"] if r and str(r[0]).strip()]
    if not rows:
        errors.append(("targets",
                       "The targets table has no row naming a cube to fit."))
    errors.extend(("targets", m) for m in override_column_errors(cfg))
    if check_paths and targets["dir"] and not os.path.isdir(targets["dir"]):
        errors.append(("targets",
                       f"Targets directory {targets['dir']!r} does not "
                       f"exist."))
    if check_paths:
        for row in rows:
            path = os.path.join(targets["dir"], str(row[0]).strip())
            if not os.path.isfile(path):
                errors.append(("targets", f"Target {path!r} does not exist."))
            else:
                errors.extend(("input", m) for m in
                              missing_extensions(cfg, path))

    stages = ("spatial", "resolution", "spectral_bin", "filter",
              "dereddening", "sn_mask", "fit")
    if not any(cfg[s]["enabled"] for s in stages):
        errors.append(("pipeline",
                       "No stage is enabled, so the run would do nothing."))

    if cfg["sn_mask"]["enabled"]:
        problem = sn_method_problem(cfg)
        if problem:
            errors.append(("sn_mask", problem))
        if cfg["sn_mask"]["method"] not in SN_METHODS:
            errors.append((
                "sn_mask",
                f"The S/N method is {cfg['sn_mask']['method']!r}; it must be "
                f"one of " + ", ".join(SN_METHODS) + "."))
        if not cfg["sn_mask"]["thresholds"]:
            errors.append(("sn_mask",
                           "The S/N mask needs at least one threshold."))
        low, high = cfg["sn_mask"]["sn window"]
        if float(low) >= float(high):
            errors.append((
                "sn_mask",
                f"The S/N window is {low}-{high} A; the first value must be "
                f"the smaller one."))

    if cfg["fit"]["enabled"]:
        if cfg["input"]["stat type"] in ("flux", "inv_flux"):
            errors.append((
                "input",
                f"The statistical plane is declared as "
                f"{cfg['input']['stat type']!r}, which is not an uncertainty, "
                f"so the fit has nothing to weight with. Point 'hdu stat' at "
                f"an error, a variance or an inverse variance and say which, "
                f"or switch the fit off and run the pre-pass stages alone."))
        settings_path = cfg["fit"]["settings"]
        threshold = cfg["fit"]["flag threshold"]
        if threshold is not None and not cfg["sn_mask"]["enabled"]:
            errors.append((
                "fit",
                "The fit is told to fit only the spaxels a S/N mask selects, "
                "but the S/N mask stage is switched off, so no such plane "
                "will exist."))
        elif threshold is not None:
            available = [int(t) for t in cfg["sn_mask"]["thresholds"]]
            if int(threshold) not in available:
                errors.append((
                    "fit",
                    f"The fit selects spaxels with SN_MASKS_{int(threshold)}, "
                    f"which the S/N mask stage does not write. It writes "
                    + ", ".join(f"SN_MASKS_{t}" for t in available) + "."))
        if check_paths and not os.path.isfile(str(settings_path)):
            errors.append(("fit",
                           f"Settings file {settings_path!r} does not "
                           f"exist."))
        elif os.path.isfile(str(settings_path)):
            errors.extend(_settings_errors(cfg, settings_path))

    if cfg["fit"]["enabled"]:
        errors.extend(("fit", m) for m in
                      age_bin_errors(cfg["fit"].get("population bins")))
        for key, label in (("sfr bins", "SFR"), ("fc bins", "FC"),
                           ("bb bins", "BB")):
            errors.extend(("fit", m) for m in
                          extra_bin_errors(cfg["fit"].get(key), label))
        distance = cfg["fit"].get("galaxy distance")
        if distance not in (None, ""):
            try:
                if float(distance) <= 0:
                    raise ValueError
            except (TypeError, ValueError):
                errors.append(("fit", f"The galaxy distance must be a "
                                      f"positive number of Mpc, not "
                                      f"{distance!r}."))
        if cfg["fit"].get("sfr bins") and distance in (None, "") \
                and "galaxy distance" not in [str(c).strip() for c in
                                              cfg["targets"]["columns"][1:]]:
            errors.append(("fit", "SFR maps need the galaxy distance (Mpc), "
                                  "for the whole run or as a 'galaxy "
                                  "distance' column of the targets table."))
        try:
            step = float(cfg["fit"].get("flxobs step",
                                        LINEAR_WAVELENGTH_STEP))
        except (TypeError, ValueError):
            step = 0.0
        if step <= 0:
            errors.append((
                "fit",
                "The FLXOBS wavelength step must be a positive number of "
                "Angstrom; it is the linear grid the observed and synthetic "
                "spectra are written on."))

    if cfg["fit"]["mask file"] and check_paths \
            and not os.path.isfile(str(cfg["fit"]["mask file"])):
        errors.append((
            "fit",
            f"Emission-line mask file {cfg['fit']['mask file']!r} does not "
            f"exist."))

    return errors




def _with_fit_overrides(settings, cfg):
    """
    A copy of the settings file with what the page overrides already in it.

    Check validates the settings file, and the fit does not run the settings
    file: it runs the settings file with the page's overrides applied. Those
    are two different things whenever the page answers something the file
    leaves out, and `wavelength_scale` is exactly that case -- required in a
    line-fitting mode, absent from plenty of existing setup files, and
    answerable on the page. Validating the raw file would report it missing
    after the user had supplied it, which teaches people that Check is wrong
    and to press Run anyway.

    Only the keys the chain genuinely overrides are applied, and the caller
    keeps the raw file for the checks that are about the file disagreeing
    with the page rather than about the fit being runnable.
    """
    merged = dict(settings)
    if cfg["fit"].get("wavelength scale"):
        merged["wavelength_scale"] = str(cfg["fit"]["wavelength scale"])
    if cfg["fit"].get("redshift") is not None:
        merged["cube_redshift"] = float(cfg["fit"]["redshift"])
        merged["spectrum_redshift"] = 0.0
    if cfg["fit"].get("mask file"):
        merged["mask"] = str(cfg["fit"]["mask file"])
    return merged


def _settings_errors(cfg, settings_path):
    """
    What the settings file and the chain disagree about, by section.

    The one that matters is asking for both routes to the line-spread
    function at once. Degrading the data smooths it down to a target
    resolution; `instrumental_resolution` convolves the templates up to the
    data. Doing both counts the instrument twice, and the second correction
    is applied to a resolution the first one already moved. It is the same
    class of error as setting `cube_redshift` and `spectrum_redshift`
    together, and it is refused in the same way, before the run rather than
    in the numbers afterwards.
    """
    from brainsp import brain_settings

    errors = []
    try:
        settings = brain_settings.load(str(settings_path))
        brain_settings.validate(_with_fit_overrides(settings, cfg),
                                str(settings_path), for_cube=True)
    except SystemExit as exc:
        return [("fit", str(exc).strip())]

    if cfg["resolution"]["enabled"] \
            and settings.get("instrumental_resolution") is not None:
        errors.append((
            "resolution",
            f"The resolution stage degrades the data, and "
            f"{os.path.basename(str(settings_path))} also sets "
            f"'instrumental_resolution', which convolves the templates up to "
            f"the data. Together they count the instrument twice. Drop one: "
            f"leave 'instrumental_resolution' set and switch the resolution "
            f"stage off, which is BRAIN's route, or clear "
            f"'instrumental_resolution' and keep the stage, which reproduces "
            f"an existing analysis made that way."))

    if cfg["fit"]["redshift"] is not None \
            and float(settings.get("spectrum_redshift") or 0.0) != 0.0:
        errors.append((
            "fit",
            "The fit is given a redshift to remove from the cube, and "
            "'spectrum_redshift' in the settings file says a shift has "
            "already been applied. Set whichever is true and leave the other "
            "at 0."))

    return errors

def read_targets_csv(path):
    """
    A targets CSV.

    The `target` column names the file; every other column is a per-target
    override of a configuration key, matched to module keys by exact name, so
    one file carries a whole campaign. A module ignores a key it never
    declared, which is what lets a single row carry columns meant for several
    different stages.
    """
    with open(path, newline="") as handle:
        rows = list(csv.reader(handle))
    if not rows:
        raise SystemExit(f"\n{path} is empty.\n")
    columns = [c.strip() for c in rows[0]]
    body = [[cell.strip() for cell in row] for row in rows[1:]
            if any(cell.strip() for cell in row)]
    return columns, body

def _coerce(text):
    """
    A CSV cell as the type a module expects.

    A configuration value that arrives as the string '0.0296' where a number
    belongs is the sort of thing that survives all the way to a header card
    and only shows up as a wrong number much later, so the conversion happens
    once, here. A bracketed cell is a list, which is how a CSV carries a
    window or a set of thresholds.
    """
    if text is None:
        return None
    text = str(text).strip()
    if text == "":
        return None
    if text.startswith("[") and text.endswith("]"):
        return [_coerce(part) for part in text[1:-1].split(",") if part.strip()]
    if text.lower() in ("true", "false"):
        return text.lower() == "true"
    try:
        return int(text)
    except ValueError:
        pass
    try:
        return float(text)
    except ValueError:
        return text

def target_overrides(columns, row):
    """The per-target configuration one CSV row carries, and its filename."""
    values = dict(zip(columns, row))
    name_column = "target" if "target" in columns else columns[0]
    name = values.pop(name_column, "").strip()
    overrides = {}
    for key, text in values.items():
        value = _coerce(text)
        if value is not None:
            overrides[key] = value
    return name, overrides

def _write_atomically(opened, out):
    """
    Write a megacube under a temporary name and move it into place.

    The interface tells the reader that stopping a run leaves the megacubes
    already written in place and does not write the one in progress. Writing
    straight to the final name makes that true everywhere except during the
    write itself, where a stop, a crash or an exhausted disk would leave a
    truncated file under the name a finished result has. A truncated FITS
    still opens, and the planes that made it to disk still read as numbers,
    which is this project's characteristic way of being wrong.

    The temporary name is in the destination directory so the move is a
    rename within one filesystem, which is atomic. A reader therefore sees
    either the previous file or the complete new one, never a partial one.

    mkstemp creates private to its owner, and a rename carries those
    permissions to the result. A megacube on a shared machine is read by
    people who did not run the fit, so the mode is put back to what an
    ordinary create under the same umask would have given.
    """
    folder = os.path.dirname(os.path.abspath(out))
    os.makedirs(folder, exist_ok=True)
    handle, staged = tempfile.mkstemp(
        dir=folder, prefix=os.path.basename(out) + ".", suffix=".partial")
    os.close(handle)
    try:
        opened.writeto(staged, overwrite=True)
        umask = os.umask(0)
        os.umask(umask)
        os.chmod(staged, 0o666 & ~umask)
        os.replace(staged, out)
    except BaseException:
        if os.path.exists(staged):
            os.remove(staged)
        raise

def config_record_hdu(chain, records):
    """
    The pipeline that produced this file, as a table inside it.

    A table rather than header cards: a configuration value routinely runs
    past the 68 characters a card's value can hold, and the megacubes this
    format is being matched to already carry a two-column table of exactly
    this shape.
    """
    stages, keys, values = [], [], []
    for index, (class_name, module_config) in enumerate(chain):
        for key, value in records[index].items():
            stages.append(f"{index}:{class_name}")
            keys.append(str(key))
            values.append(str(value))
    columns = [
        fits.Column(name="stage", format="64A", array=stages),
        fits.Column(name="key", format="64A", array=keys),
        fits.Column(name="value", format="128A", array=values),
    ]
    hdu = fits.BinTableHDU.from_columns(columns)
    hdu.header["EXTNAME"] = CONFIG_EXTNAME
    hdu.header["SUMMARY"] = "Every stage of the BRAIN IFU run, with its settings"
    return hdu

class BrainIFU:
    """
    A chain of stages run over a list of targets.

    add_module, add_target and execute. Targets run sequentially, the
    progress callback reports real fractions of the work done, and
    every stage is constructed once per target so a per-target override
    reaches it.

    `spaxels` is the finer of the two progress callbacks: the fit reports the
    spaxels each batch finished as it finishes them, so an interface can show
    the field filling in. It runs on the fitting thread, so a listener that
    does real work in it slows the fit down.
    """

    def __init__(self, progress=None, log=None, spaxels=None):
        self._chain = []
        self._targets = []
        self.progress = progress or (lambda fraction, message="": None)
        self.log = log or (lambda message: logger.info(message))
        self.spaxels = spaxels or (lambda indices, n_y, n_x: None)

    def execute(self, save_path_root="brain_ifu_out", save_config=True,
                overwrite=True):
        """
        Run every stage over every target and write one megacube each.

        With overwrite False a target whose megacube already exists is
        skipped before its first stage runs, so nothing is recomputed.
        """
        os.makedirs(save_path_root, exist_ok=True)
        if not self._chain:
            raise SystemExit("\nNo stage is enabled; there is nothing to run.\n")
        if not self._targets:
            raise SystemExit("\nNo target to run on.\n")

        summaries = []
        n_targets = len(self._targets)
        for index, (target, overrides) in enumerate(self._targets):
            out = output_path(save_path_root, target)
            base = index / n_targets

            def stage_progress(fraction, message="", _base=base):
                self.progress(_base + fraction / n_targets, message)

            if not os.path.isfile(target):
                self.log(f"MISSING {target}")
                summaries.append({"target": target, "state": "missing"})
                continue
            if not overwrite and os.path.isfile(out):
                self.log(f"SKIPPING {os.path.basename(target)}: "
                         f"{out} already exists")
                summaries.append({"target": target, "state": "skipped",
                                  "output": out})
                continue

            self.log(f"--- {os.path.basename(target)}  "
                     f"({index + 1} of {n_targets})")
            started = time.time()
            summary = self._run_one(target, overrides, out, save_config,
                                    stage_progress)
            summary["seconds"] = time.time() - started
            summaries.append(summary)
            self.progress((index + 1) / n_targets, "")
        return summaries



    @staticmethod
    def _copy_settings(records, out):
        """
        The settings file the fit ran against, written beside the megacube.

        A megacube and the settings that produced it are only reproducible
        together, and the settings file is the one input that can be edited
        between two runs without leaving any trace in the output.
        """
        import shutil
        for record in records:
            path = record.get("settings")
            if path and os.path.isfile(str(path)):
                beside = os.path.splitext(out)[0] + ".settings.yaml"
                shutil.copyfile(str(path), beside)
                return beside
        return None

    def _run_one(self, target, overrides, out, save_config, progress):
        records = []
        cards = []
        summary = {"target": target, "state": "done", "output": out}
        with fits.open(target) as opened:
            n_stages = len(self._chain)
            for index, (class_name, config) in enumerate(self._chain):
                merged = dict(config)
                merged.update(overrides)
                module = STAGE_CLASSES[class_name](**merged)

                def hook(fraction, message="", _i=index):
                    progress((_i + fraction) / n_stages, message)

                module.progress_hook = hook
                module.spaxel_hook = self.spaxels
                self.log(f"    [{index + 1}/{n_stages}] {module.name}")
                progress(index / n_stages, module.name)
                for produced in module.execute(opened):
                    opened.append(produced)
                records.append(module.config)
                cards.extend(getattr(module, "primary_cards", []))
                summary.update(getattr(module, "summary", {}))
                progress((index + 1) / n_stages, module.name)

            if save_config:
                opened.append(config_record_hdu(self._chain, records))
                self._copy_settings(records, out)
            for key, value, comment in cards:
                opened[0].header[key] = (value, comment)
            _write_atomically(opened, out)
        self.log(f"    wrote {out}")
        return summary

def build_runner(cfg, progress=None, log=None, spaxels=None):
    """A BrainIFU loaded from a configuration dictionary, ready to execute."""
    runner = BrainIFU(progress=progress, log=log, spaxels=spaxels)
    for class_name, module_config in build_module_chain(cfg):
        runner.add_module(class_name, module_config)

    targets = cfg["targets"]
    columns = targets["columns"]
    for row in targets["rows"]:
        name, overrides = target_overrides(columns, row)
        if name:
            runner.add_target(os.path.join(targets["dir"], name), overrides)
    return runner

def run(cfg, progress=None, log=None, spaxels=None):
    """Validate a configuration and run it."""
    errors = validate(cfg)
    if errors:
        raise SystemExit("\nThis pipeline will not run:\n\n"
                         + "\n".join(f"  * {e}" for e in errors) + "\n")
    runner = build_runner(cfg, progress=progress, log=log, spaxels=spaxels)
    out = cfg["output"]
    return runner.execute(save_path_root=out["save_path_root"],
                          save_config=out["save_config"],
                          overwrite=out["overwrite"])

def load_config(path):
    """
    Read a pipeline file, in whichever of the two formats it is written in.

    The extension decides: .json is read as JSON, .yaml and .yml as YAML.
    Both are supported because the settings file BRAIN already ships is YAML
    and the interface this one is modelled on round-trips JSON, and choosing
    between them is not this code's decision to make.
    """
    with open(path) as handle:
        if os.path.splitext(path)[1].lower() in (".yaml", ".yml"):
            import yaml
            loaded = yaml.safe_load(handle)
        else:
            loaded = json.load(handle)
    return merge_config(loaded)

def merge_config(loaded):
    """A loaded pipeline filled out with the defaults it did not mention."""
    cfg = default_config()
    for section, values in (loaded or {}).items():
        if section in cfg and isinstance(cfg[section], dict):
            cfg[section].update(values or {})
        else:
            cfg[section] = values
    return cfg

def describe(cfg):
    """The chain a configuration would run, one stage per line."""
    lines = []
    for index, (class_name, module_config) in enumerate(build_module_chain(cfg), 1):
        detail = ", ".join(f"{k}={v!r}" for k, v in module_config.items())
        lines.append(f"  {index}. {class_name}({detail})")
    return "\n".join(lines) or "  (nothing enabled)"

def main(argv):
    logging.basicConfig(level=logging.INFO, format="%(message)s",
                        stream=sys.stdout)
    command = argv[1] if len(argv) > 1 else "help"

    if command == "defaults":
        json.dump(default_config(), sys.stdout, indent=2)
        sys.stdout.write("\n")
        return 0

    if command in ("run", "check", "script") and len(argv) > 2:
        cfg = load_config(argv[2])
        if command == "check":
            errors = validate(cfg)
            print("Pipeline:")
            print(describe(cfg))
            if errors:
                print("\nThis pipeline will not run:")
                for error in errors:
                    print(f"  * {error}")
                return 1
            print("\nValid.")
            return 0
        if command == "script":
            from brainsp.ifu_script_export import export
            out = argv[3] if len(argv) > 3 else "run_brain_ifu.py"
            written, outside = export(cfg, out)
            print("Wrote " + ", ".join(written))
            if outside:
                print("Outside the script's folder (copy them along):\n  "
                      + "\n  ".join(outside))
            return 0
        run(cfg)
        return 0

    print(__doc__)
    return 0




if __name__ == "__main__":
