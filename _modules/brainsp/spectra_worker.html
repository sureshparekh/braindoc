Source code for brainsp.spectra_worker

"""
One CPU worker of a batch of spectra: ``python -m brainsp.spectra_worker JOB``.

:func:`brainsp.fit_spectra` writes a job file for each worker: the shared
settings, the output folder and this worker's share of the spectra, each
with its own settings. The worker fits them one after another in this one
process, so the fit is compiled once and stays warm, and prints one line per
spectrum, ``DONE <index> <summary file>`` or ``FAILED <index> <message>``.
"""

import pickle
import sys

def main(path):
    with open(path, "rb") as handle:
        job = pickle.load(handle)
    if "configs" in job:
        return _configs(job)
    import brainsp
    for index, spectrum, extra in job["spectra"]:
        try:
            r = brainsp.fit_spectrum(spectrum, job["settings"],
                                     output_dir=job["output_dir"],
                                     quiet=True, **extra)
            print(f"DONE {index} {r.summary_file}", flush=True)
        except (Exception, SystemExit) as exc:
            text = str(exc).replace("\n", " ")[:2000]
            print(f"FAILED {index} {text}", flush=True)
    return 0




def _configs(job):
    """The interface's form: ready-made fit configurations."""
    from brainsp import brain_cache, brainv1, io_utils
    from brainsp.brain_child import config_from_dict
    brain_cache.enable_jax_cache(verbose=False)
    raw = io_utils.load_ssp_templates
    memo = {}

    def cached(folder, bases, target_wl, norm):
        key = brain_cache.ssp_cache_key(folder, bases, target_wl, norm)
        if key not in memo:
            memo[key] = brain_cache.load_ssp_templates_cached(
                folder, bases, target_wl, norm, raw)
        return memo[key]

    io_utils.load_ssp_templates = cached
    brainv1.load_ssp_templates = cached
    import logging
    logging.getLogger("NEWBRAIN").setLevel(logging.WARNING)
    for index, values in job["configs"]:
        cfg = config_from_dict(values)
        try:
            brainv1.run(cfg)
            print(f"DONE {index} {cfg.output_summary}", flush=True)
        except (Exception, SystemExit) as exc:
            text = str(exc).replace("\n", " ")[:2000]
            print(f"FAILED {index} {text}", flush=True)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
