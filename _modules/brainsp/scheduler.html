Source code for brainsp.scheduler

"""
Several cubes at once, on every GPU there is.

A datacube campaign (brain_ifu.run) fits its cubes one after another in one
process. With more than one GPU, or a GPU large enough to hold more than one
fit, the cubes are instead dealt out to *slots*: each slot is a process of
its own (brainsp.target_worker) on one GPU, given its share of that card's
memory and of the CPU cores (never more, together, than every core but two),
and running exactly the fit a campaign of that one cube would run. A slot
that finishes takes the next cube. Every cube is fitted by the same code
with the same settings, so its megacube is the one the sequential campaign
writes.

    slots("auto")   one slot per GPU, two on a card of 40 GB or more
    slots(3)        three slots, dealt over the GPUs in turn

On one GPU of under 40 GB, "auto" is one slot, and the campaign runs as it
always has. With no GPU the cubes run one after another on the CPU worker
pool, which already uses every core but two (cpu_pool).
"""

import json
import os
import queue
import subprocess
import sys
import tempfile
import threading
import time

# A card with at least this much memory holds two fits at once by default.
# The second slot overlaps one fit's host-side stages (reading, the
# population statistics, writing the megacube) with the other's GPU work;
# its gain is not yet measured on such a card.
TWO_SLOT_GB = 40.0

def slots(requested="auto"):
    """
    The GPU slots to run cubes on: a list of dicts with the card's index,
    its name and the fraction of its memory the slot may take (None for a
    whole card). Empty without a GPU.
    """
    from brainsp import machine
    gpus = machine.gpus()
    if not gpus:
        return []
    if str(requested).lower() == "auto":
        per = [2 if g["memory_gb"] >= TWO_SLOT_GB else 1 for g in gpus]
    else:
        n = max(1, int(requested))
        per = [n // len(gpus) + (1 if i < n % len(gpus) else 0)
               for i in range(len(gpus))]
    out = []
    for g, k in zip(gpus, per):
        for _ in range(k):
            out.append(dict(gpu=g["index"], name=g["name"],
                            fraction=None if k == 1 else round(0.9 / k, 3)))
    return out




def _cores_for(n_slots):
    """The CPU cores of each slot: the budget (every core but two) dealt
    out, physical cores first."""
    from brainsp.fit_megacube_jax import worker_cores
    cores = worker_cores()
    return [cores[i::n_slots] or cores for i in range(n_slots)]

def run(cfg, slot_list, log=print, progress=None):
    """
    brain_ifu.run(cfg), its targets dealt out over `slot_list` (slots()).
    Returns the campaign's summaries, in the order of its targets.
    """
    rows = list(cfg["targets"]["rows"])
    jobs = queue.Queue()
    for i, row in enumerate(rows):
        jobs.put((i, row))
    results = [None] * len(rows)
    folder = tempfile.mkdtemp(prefix="brainsp-slots-")
    lock = threading.Lock()
    done = [0]
    cores = _cores_for(len(slot_list))
    package_parent = os.path.dirname(os.path.dirname(os.path.abspath(
        __file__)))

    def name_of(row):
        from brainsp.brain_ifu import target_overrides
        name, _ = target_overrides(cfg["targets"]["columns"], row)
        return os.path.basename(str(name))

    def serve(k, slot):
        env = dict(os.environ, CUDA_VISIBLE_DEVICES=str(slot["gpu"]),
                   JAX_PLATFORMS="cuda")
        if slot["fraction"] is not None:
            env["XLA_PYTHON_CLIENT_MEM_FRACTION"] = str(slot["fraction"])
        env["PYTHONPATH"] = os.pathsep.join(
            [package_parent] + [p for p in env.get("PYTHONPATH", "").split(
                os.pathsep) if p])
        mine = set(cores[k])
        while True:
            try:
                i, row = jobs.get_nowait()
            except queue.Empty:
                return
            one = json.loads(json.dumps(cfg))
            one["targets"]["rows"] = [row]
            path = os.path.join(folder, f"target{i}.json")
            with open(path, "w") as handle:
                json.dump(one, handle)
            label = f"[GPU {slot['gpu']}.{k}] {name_of(row)}"
            log(f"{label}: started")
            started = time.time()
            proc = subprocess.Popen(
                [sys.executable, "-m", "brainsp.target_worker", path,
                 path + ".out"], stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT, env=env, text=True,
                preexec_fn=lambda: os.sched_setaffinity(0, mine))
            for line in proc.stdout:
                line = line.rstrip()
                if line:
                    log(f"{label}: {line}")
            proc.wait()
            try:
                with open(path + ".out") as handle:
                    summary = json.load(handle)[0]
            except (OSError, ValueError, IndexError):
                summary = {"target": name_of(row), "state": "failed",
                           "error": f"the slot's process ended with code "
                                    f"{proc.returncode}"}
            summary["seconds"] = time.time() - started
            with lock:
                results[i] = summary
                done[0] += 1
                if progress is not None:
                    progress(done[0] / len(rows), name_of(row))
            log(f"{label}: {summary.get('state')} in "
                f"{summary['seconds']:.0f} s")

    log(f"{len(rows)} cubes on {len(slot_list)} GPU slots: "
        + ", ".join(f"GPU {s['gpu']} ({s['name']})"
                    + (f" {s['fraction']:.0%} of its memory"
                       if s["fraction"] else "") for s in slot_list))
    threads = [threading.Thread(target=serve, args=(k, s), daemon=True)
               for k, s in enumerate(slot_list)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    import shutil
    shutil.rmtree(folder, ignore_errors=True)
    return results
