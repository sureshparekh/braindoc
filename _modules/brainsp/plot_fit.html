"""
Plot NewBrain spectral fit results.
Creates a multi-panel figure adapting automatically to the fitting mode:
  "spop"     — 2-panel: fit overview + residuals
  "nepop"    — 3-panel: fit overview + nebular detail + residuals
  "photopop" — 3-panel: fit overview + nebular detail + residuals
                         (with predicted vs observed emission line annotations)
"""

import numpy as np
from brainsp import emission_lines
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import sys
import os

def detect_mode(spectrum_file, summary_file=None):
    """
    Detect the photoionization mode from the output files.

    Returns
    -------
    mode : str
        One of "spop", "espop", "nepop", "photopop"
    """
    mode = "spop"

    if summary_file and os.path.isfile(summary_file):
        with open(summary_file) as f:
            for line in f:
                if "[photoionization_mode]" in line:
                    tag = line.split()[0].strip()
                    if tag in ("spop", "espop", "nepop", "photopop"):
                        mode = tag
                    return mode

    with open(spectrum_file) as f:
        header = f.readline().strip().lstrip('#').split()
    if 'Nebular_Fit' in header:
        mode = "nepop"

    return mode

def parse_summary(summary_file):
    """
    Extract key diagnostics from the summary file.

    Returns
    -------
    info : dict
        Dictionary with stellar and nebular parameters.
    predicted_lines : dict or None
        {name: (wavelength, predicted_flux, observed_flux)}
    """
    info = {}
    predicted_lines = {}

    if not summary_file or not os.path.isfile(summary_file):
        return info, None

    in_predicted_block = False

    with open(summary_file) as f:
        for line in f:
            if "[AV_min" in line:
                info['Av'] = line.split()[0]
            elif "[vd_min" in line:
                info['sigma_star'] = line.split()[0]
            elif "[v0_min" in line:
                info['v0_star'] = line.split()[0]
            elif "[AV_neb" in line:
                info['Av_neb'] = line.split()[0]
            elif "[tau" in line and "tolerance" not in line:
                info['tau'] = line.split()[0]
            elif "[Q_eff" in line:
                info['Q_eff'] = line.split()[0]
            elif "[f_neb" in line:
                info['f_neb'] = line.split()[0]
            elif "[T_e" in line:
                info['T_e'] = line.split()[0]
            elif "[n_e" in line:
                info['n_e'] = line.split()[0]
            elif "[photoionization_mode]" in line:
                info['mode'] = line.split()[0]
            elif "[chi2/Nl_eff]" in line:
                info['chi2'] = line.split()[0]
            elif "[adev" in line:
                info['adev'] = line.split()[0]

            if "## Predicted Emission Lines" in line:
                in_predicted_block = True
                continue
            if in_predicted_block:
                if line.startswith("##") or line.strip() == "":
                    in_predicted_block = False
                    continue
                if "|" in line and "Line" not in line:
                    parts = line.split("|")
                    if len(parts) >= 4:
                        try:
                            name = parts[0].strip()
                            wl = float(parts[1].strip())
                            pred = float(parts[2].strip())
                            obs = float(parts[3].strip())
                            predicted_lines[name] = (wl, pred, obs)
                        except ValueError:
                            pass

    return info, predicted_lines if predicted_lines else None

def load_spectrum_data(spectrum_file):
    """
    Load the fitted spectrum file and detect columns from header.

    Returns
    -------
    wl, flux_obs, total_fit, stellar_fit, emission_fit : arrays
    nebular_fit : array or None
    err : array or None
    resid : array
    """
    data = np.loadtxt(spectrum_file)
    wl = data[:, 0]
    flux_obs = data[:, 1]
    total_fit = data[:, 2]
    stellar_fit = data[:, 3]
    emission_fit = data[:, 4]

    nebular_fit = None
    err = None
    resid = None

    with open(spectrum_file) as f:
        header = f.readline().strip().lstrip('#').split()

    if 'Nebular_Fit' in header:
        nebular_fit = data[:, 5]
        if data.shape[1] == 8:
            err = data[:, 6]
            resid = data[:, 7]
        elif data.shape[1] == 7:
            err = None
            resid = data[:, 6]
        else:
            resid = flux_obs - total_fit
    else:
        if data.shape[1] == 7:
            err = data[:, 5]
            resid = data[:, 6]
        elif data.shape[1] == 6:
            err = None
            resid = data[:, 5]
        else:
            resid = flux_obs - total_fit

    return wl, flux_obs, total_fit, stellar_fit, emission_fit, nebular_fit, err, resid

def plot_fit(spectrum_file, summary_file=None, output_png="newbrain_fit.pdf"):
    """
    Plot the fitted spectrum from NewBrain output.
    Automatically adapts layout to the detected photoionization mode.

    Parameters
    ----------
    spectrum_file : str
        Path to fitted spectrum file.
    summary_file : str or None
        Path to fit summary file for title information.
    output_png : str
        Output filename for the plot.
    """
    wl, flux_obs, total_fit, stellar_fit, emission_fit, nebular_fit, err, resid = \
        load_spectrum_data(spectrum_file)

    mode = detect_mode(spectrum_file, summary_file)
    info, predicted_lines = parse_summary(summary_file)

    if 'mode' in info:
        mode = info['mode']

    has_nebular = (nebular_fit is not None and np.max(np.abs(nebular_fit)) > 1e-10)
    has_emission = np.max(np.abs(emission_fit)) > 1e-10

    if mode != "spop" and has_nebular:
        fig, axes = plt.subplots(3, 1, figsize=(16, 11),
                                  gridspec_kw={'height_ratios': [4, 2, 1.5]},
                                  sharex=True)
        ax_main, ax_neb, ax_res = axes
    else:
        fig, axes = plt.subplots(2, 1, figsize=(16, 8),
                                  gridspec_kw={'height_ratios': [4, 1.5]},
                                  sharex=True)
        ax_main, ax_res = axes
        ax_neb = None

    mode_labels = {"spop": "Stellar-Only", "espop": "Stars + Emission Lines",
                   "nepop": "Nebular Continuum",
                   "photopop": "Full Self-Consistent"}
    title_parts = [f"Mode: {mode_labels.get(mode, mode)}"]
    for key, label in [('Av', 'Av'), ('sigma_star', 'σ★'),
                        ('v0_star', 'v0★'), ('chi2', 'χ²/N'),
                        ('adev', 'ADEV%')]:
        if key in info:
            title_parts.append(f"{label}={info[key]}")
    if mode != "spop":
        for key, label in [('Av_neb', 'Av_neb'), ('tau', 'τ'),
                            ('f_neb', 'f_neb'), ('T_e', 'Te'), ('n_e', 'ne')]:
            if key in info:
                title_parts.append(f"{label}={info[key]}")

    offset = np.max(flux_obs) * 0.6
    ax_main.plot(wl, flux_obs + offset, color='gray', lw=0.5, alpha=0.6,
                 label=f'Original (offset +{offset:.2f})')
    ax_main.axhline(y=offset, color='gray', ls=':', lw=0.4, alpha=0.3)

    ax_main.plot(wl, flux_obs, 'k-', lw=0.5, alpha=0.7, label='Observed')
    ax_main.plot(wl, total_fit, 'r-', lw=1.2, label='Total Fit')
    ax_main.plot(wl, stellar_fit, 'b-', lw=0.8, alpha=0.6, label='Stellar')

    if has_emission:
        ax_main.plot(wl, emission_fit, 'g-', lw=0.8, alpha=0.6, label='Emission (empirical)')

    if has_nebular:
        ax_main.plot(wl, nebular_fit, color='darkorange', lw=1.0, alpha=0.7,
                     label='Nebular Continuum')

    ax_main.set_ylabel('Normalized Flux')
    ax_main.legend(loc='upper right', fontsize=9)
    ax_main.set_title('  |  '.join(title_parts), fontsize=11)
    ax_main.grid(True, alpha=0.3)

    if ax_neb is not None and has_nebular:
        ax_neb.fill_between(wl, stellar_fit, stellar_fit + nebular_fit,
                            color='darkorange', alpha=0.3, label='Nebular Continuum')
        ax_neb.plot(wl, stellar_fit + nebular_fit, color='darkorange',
                    lw=0.8, alpha=0.7)
        ax_neb.plot(wl, stellar_fit, 'b-', lw=0.8, alpha=0.5, label='Stellar Only')
        ax_neb.plot(wl, flux_obs, 'k-', lw=0.4, alpha=0.4, label='Observed')

        for jump_wl, jump_name in [(3646, 'Balmer jump'), (8207, 'Paschen jump')]:
            if wl[0] <= jump_wl <= wl[-1]:
                ax_neb.axvline(x=jump_wl, color='darkorange', ls='--', lw=0.8, alpha=0.5)
                ax_neb.text(jump_wl + 15, ax_neb.get_ylim()[1] * 0.85 if ax_neb.get_ylim()[1] > 0
                            else 0.02, jump_name,
                            fontsize=7, color='darkorange', alpha=0.8, rotation=90, va='top')

        if mode == "photopop" and predicted_lines:
            y_top = ax_neb.get_ylim()[1]
            for name, (line_wl, pred, obs) in predicted_lines.items():
                if wl[0] <= line_wl <= wl[-1]:
                    ax_neb.axvline(x=line_wl, color='green', ls=':', lw=0.5, alpha=0.4)
                    short_name = emission_lines.species_of(name) or name
                    ax_neb.text(line_wl, y_top * 0.75, short_name,
                                fontsize=6, color='green', alpha=0.7,
                                rotation=90, va='top', ha='right')

        ax_neb.set_ylabel('Flux')
        ax_neb.legend(loc='upper right', fontsize=8)
        neb_title = 'Nebular Continuum Contribution'
        if mode == "photopop":
            neb_title += '  +  Predicted Line Positions'
        ax_neb.set_title(neb_title, fontsize=10)
        ax_neb.grid(True, alpha=0.3)

    ax_res.plot(wl, resid, 'k-', lw=0.5, alpha=0.7)
    ax_res.axhline(y=0, color='r', ls='--', lw=0.8, alpha=0.5)

    if err is not None:
        ax_res.fill_between(wl, -err, err, alpha=0.15, color='blue', label='±1σ')
        ax_res.legend(fontsize=9)

    rms = np.std(resid)
    ax_res.set_ylabel('Residual')
    ax_res.set_xlabel('Wavelength (Å)')
    ax_res.set_title(f'Residuals (RMS = {rms:.4e})', fontsize=10)
    ax_res.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig(output_png, dpi=1200, bbox_inches='tight')
    print(f"Plot saved to {output_png}")
    plt.close()



if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python plot_fit.py <spectrum_filename>")
        sys.exit(1)
        
    spec_input = sys.argv[1]
    spec_name = os.path.splitext(os.path.basename(spec_input))[0]
    
    spec_file = f"brain_out/{spec_name}_fitted.txt"
    summ_file = f"brain_out/{spec_name}_summary.txt"
    out_png   = f"brain_out/{spec_name}_fit.pdf"

    plot_fit(spec_file, summ_file, out_png)
