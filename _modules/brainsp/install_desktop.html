#!/usr/bin/env python3
"""
Register BRAIN with the desktop so it appears in the applications menu.

    brainsp desktop              install
    brainsp desktop --remove     take it back out

Linux gets a .desktop entry, macOS a small .app bundle. On Windows the
launcher is a .lnk in the Start Menu, which needs pywin32; if that is not
present the script says so rather than half-installing.
"""

import os
import sys
import shutil
import subprocess

from brainsp._paths import LOGO_DIR

# The interface starts in the user's home folder, which is where it writes
# its results by default, and is run as a module of the installed package.
HERE = os.path.expanduser("~")
ICON = os.path.join(LOGO_DIR, "withbg.png")
ENTRY = "-m brainsp.brain_gui"
NAME = "BRAIN"
COMMENT = "Stellar population fitting"

def linux(remove):
    apps = os.path.expanduser("~/.local/share/applications")
    icons = os.path.expanduser("~/.local/share/icons/hicolor/256x256/apps")
    desktop = os.path.join(apps, "brain.desktop")
    icon = os.path.join(icons, "brain.png")

    if remove:
        for p in (desktop, icon):
            if os.path.exists(p):
                os.remove(p)
        print("removed")
    else:
        os.makedirs(apps, exist_ok=True)
        os.makedirs(icons, exist_ok=True)
        shutil.copyfile(ICON, icon)
        with open(desktop, "w") as f:
            f.write(f"""[Desktop Entry]
Type=Application
Name={NAME}
Comment={COMMENT}
Exec={sys.executable} {ENTRY}
Path={HERE}
Icon=brain
Terminal=false
Categories=Science;Astronomy;Education;
StartupWMClass={NAME}
""")
        os.chmod(desktop, 0o755)
        print(f"installed  {desktop}")

    for cmd in (["update-desktop-database", apps],
                ["gtk-update-icon-cache", "-f", "-t",
                 os.path.expanduser("~/.local/share/icons/hicolor")]):
        try:
            subprocess.run(cmd, check=False, capture_output=True)
        except FileNotFoundError:
            pass

def macos(remove):
    app = os.path.expanduser(f"~/Applications/{NAME}.app")
    if remove:
        shutil.rmtree(app, ignore_errors=True)
        print("removed")
        return
    macos_dir = os.path.join(app, "Contents", "MacOS")
    res_dir = os.path.join(app, "Contents", "Resources")
    os.makedirs(macos_dir, exist_ok=True)
    os.makedirs(res_dir, exist_ok=True)

    icns = os.path.join(res_dir, "brain.icns")
    try:
        iconset = os.path.join(res_dir, "brain.iconset")
        os.makedirs(iconset, exist_ok=True)
        for s in (16, 32, 64, 128, 256, 512):
            subprocess.run(["sips", "-z", str(s), str(s), ICON, "--out",
                            os.path.join(iconset, f"icon_{s}x{s}.png")],
                           check=True, capture_output=True)
        subprocess.run(["iconutil", "-c", "icns", iconset, "-o", icns],
                       check=True, capture_output=True)
        shutil.rmtree(iconset)
    except Exception:
        icns = None

    launcher = os.path.join(macos_dir, NAME)
    with open(launcher, "w") as f:
        f.write(f'#!/bin/sh\ncd "{HERE}"\nexec "{sys.executable}" {ENTRY}\n')
    os.chmod(launcher, 0o755)

    with open(os.path.join(app, "Contents", "Info.plist"), "w") as f:
        f.write(f"""<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN"
 "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0"><dict>
  <key>CFBundleName</key><string>{NAME}</string>
  <key>CFBundleExecutable</key><string>{NAME}</string>
  <key>CFBundleIdentifier</key><string>org.brain.gui</string>
  <key>CFBundlePackageType</key><string>APPL</string>
  <key>CFBundleShortVersionString</key><string>1.0</string>
  {'<key>CFBundleIconFile</key><string>brain.icns</string>' if icns else ''}
  <key>NSHighResolutionCapable</key><true/>
</dict></plist>
""")
    print(f"installed  {app}")

def windows(remove):
    import winreg
    start = os.path.join(os.environ["APPDATA"], "Microsoft", "Windows",
                         "Start Menu", "Programs", f"{NAME}.lnk")
    if remove:
        if os.path.exists(start):
            os.remove(start)
        print("removed")
        return
    try:
        from win32com.client import Dispatch
    except ImportError:
        sys.exit("Needs pywin32 for the Start Menu shortcut:\n"
                 "    pip install pywin32\n"
                 "Or just run:  brainsp gui")
    shell = Dispatch("WScript.Shell")
    lnk = shell.CreateShortCut(start)
    pythonw = sys.executable.replace("python.exe", "pythonw.exe")
    lnk.Targetpath = pythonw if os.path.exists(pythonw) else sys.executable
    lnk.Arguments = ENTRY
    lnk.WorkingDirectory = HERE
    lnk.IconLocation = ICON
    lnk.Description = COMMENT
    lnk.save()
    print(f"installed  {start}")

def main(argv=None):
    argv = sys.argv[1:] if argv is None else argv
    remove = "--remove" in argv
    if not os.path.isfile(ICON):
        sys.exit(f"Missing icon: {ICON}")
    {"linux": linux, "darwin": macos, "win32": windows}.get(
        sys.platform, linux)(remove)
    return 0




if __name__ == "__main__":
    sys.exit(main())
