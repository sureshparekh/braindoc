Source code for brainsp.megacube

"""
Reading a BRAIN megacube, and the emission-line diagnostics drawn from it.

A *megacube* is the FITS file BRAIN writes for a fitted datacube
(``<cube>-BRAIN_MEGACUBE.fits``): the input cube's own extensions, then the
observed and synthetic spectra on a linear grid (``FLXOBS``, ``FLXSYN``), the
population vectors (``PoPVecsL``, ``PoPVecsM``), the population maps and fit
statistics (``PoPBins``), and, in a line-fitting mode, the emission-line
products in the layout of Riffel et al. 2023 (``SOLUTION``, ``FLUX_M``,
``EQW_M`` ...). :class:`Megacube` opens one and hands out any of its maps by
name; :func:`brainsp.read_megacube` is the same thing.

Nothing here fits anything or imports JAX or Qt, so it works on a machine
with no display.

The diagnostic curves, with their sources:

    Kewley et al. 2001     [N II] BPT maximum starburst
    Kauffmann et al. 2003  [N II] BPT pure star formation, to its asymptote
                           at log [N II]/Halpha = 0.05
    Cid Fernandes et al. 2010   LINER / Seyfert in the [N II] BPT
    Kewley et al. 2006     [S II] BPT, star forming and Seyfert / LINER
    Cid Fernandes et al. 2011   WHAN

A diagnostic is drawn only where every line it needs has a positive model
flux. BRAIN does not yet write line-flux uncertainties, so there is no S/N
cut on the lines: a weak line the fit placed at a small positive flux is
classified like a strong one.
"""

import os

import numpy as np
from astropy.io import fits

C_KMS = 299792.458

def bpt_nii_class(n2ha, o3hb):
    """1 star forming, 2 composite, 3 LINER, 4 Seyfert, NaN undefined."""
    out = np.full(n2ha.shape, np.nan)
    ok = np.isfinite(n2ha) & np.isfinite(o3hb)
    with np.errstate(invalid="ignore", divide="ignore"):
        sf = ok & (n2ha < 0.05) & (o3hb < kauffmann03(n2ha))
        agn = ok & ~sf & ((n2ha >= 0.47) | (o3hb > kewley01(n2ha)))
        comp = ok & ~sf & ~agn
        sey = agn & (o3hb > cid10(n2ha))
        lin = agn & ~sey
    out[sf], out[comp], out[lin], out[sey] = 1, 2, 3, 4
    return out

def bpt_sii_class(s2ha, o3hb):
    """1 star forming, 2 LINER, 3 Seyfert, NaN undefined."""
    out = np.full(s2ha.shape, np.nan)
    ok = np.isfinite(s2ha) & np.isfinite(o3hb)
    with np.errstate(invalid="ignore", divide="ignore"):
        sf = ok & (s2ha < 0.32) & (o3hb < 0.72 / (s2ha - 0.32) + 1.30)
        agn = ok & ~sf
        sey = agn & (o3hb > 1.89 * s2ha + 0.76)
    out[sf], out[agn & ~sey], out[sey] = 1, 2, 3
    return out

def whan_class(n2ha, ew_ha):
    """1 star forming, 2 retired, 3 weak AGN, 4 strong AGN (Cid Fernandes 2011)."""
    out = np.full(n2ha.shape, np.nan)
    ok = np.isfinite(n2ha) & np.isfinite(ew_ha)
    with np.errstate(invalid="ignore"):
        rg = ok & (ew_ha < 3)
        sf = ok & ~rg & (n2ha < -0.4)
        sagn = ok & ~rg & (n2ha >= -0.4) & (ew_ha > 6)
        wagn = ok & ~rg & (n2ha >= -0.4) & (ew_ha <= 6)
    out[sf], out[rg], out[wagn], out[sagn] = 1, 2, 3, 4
    return out

class Megacube:
    """
    A BRAIN megacube, read once, with its maps named and grouped.

    Parameters
    ----------
    path : str
        A ``*-BRAIN_MEGACUBE.fits`` file.

    Attributes
    ----------
    path : str
    shape : tuple of int
        ``(n_y, n_x)``, the spatial shape.
    wave : numpy.ndarray
        Rest-frame wavelength of ``FLXOBS`` and ``FLXSYN``, in Angstrom.
    groups : dict
        Map names by group: ``"Populations"``, ``"Kinematics"``,
        ``"Fit quality"``, ``"Emission lines"``, ``"Diagnostics"``.
    maps : dict
        Every map name, with the extension and plane it is read from.
    lines : list of str
        The emission lines fitted (empty in ``spop``).
    ages : numpy.ndarray
        Age of every template, in years, in population-vector order.
    status : numpy.ndarray
        1 where a spaxel was fitted, 0 where it was not.

    Examples
    --------
    >>> import brainsp
    >>> cube = brainsp.read_megacube("mock_cube-BRAIN_MEGACUBE.fits")
    >>> cube.groups["Kinematics"]
    ['Sigma_star', 'vrot_star', 'Sigma_gas_g0', 'vrot_gas_g0', ...]
    >>> velocity = cube.plane("vrot_star")        # 2-D array, NaN where unfitted
    >>> obs, model, lines = cube.spectrum(8, 8)
    """

    def __init__(self, path):
        self.path = path
        self.h = fits.open(path, memmap=True)
        names = [x.name for x in self.h]
        if "PoPBins" not in self.h or "FLXOBS" not in self.h:
            raise ValueError(f"{os.path.basename(path)} is not a megacube: it "
                             f"has no PoPBins or FLXOBS extension")
        self.groups = {}
        self.maps = {}
        pop = self.h["PoPBins"]
        for i in range(pop.data.shape[0]):
            name = str(pop.header.get(f"DATA{i}", f"plane {i}"))
            group = ("Populations" if name.endswith(("_light", "_mass"))
                     or name.startswith(("FC", "AGN"))
                     else "Fit quality" if name in ("Adev", "ChiSqrt", "SNR",
                                                    "Nl_eff", "Niter",
                                                    "BadSpaxel")
                     else "Kinematics" if "Sigma" in name or "vrot" in name
                     else "Populations")
            self._add(group, name, ("PoPBins", i))
        if "CHI2RED" in names:
            self._add("Fit quality", "chi2 reduced", ("CHI2RED", None))
        if "STATUS" in names:
            self._add("Fit quality", "status", ("STATUS", None))
        self.lines = []
        if "PARNAMES" in names and "FLUX_M" in names:
            pn = self.h["PARNAMES"].data
            self.lines = [str(c).strip() for c, p in pn if str(p).strip() == "A"]
            cfg = {str(k).strip(): str(v).strip()
                   for k, v in self.h["FITCONFIG"].data}
            self.rest = {n: float(cfg.get(f"{n}.rest_wavelength", "nan"))
                         for n in self.lines}
            for k, n in enumerate(self.lines):
                self._add("Emission lines", f"{n}  flux", ("FLUX_M", k))
                self._add("Emission lines", f"{n}  EW", ("EQW_M", k))
                self._add("Emission lines", f"{n}  velocity", ("SOLUTION", 3 * k + 1))
                self._add("Emission lines", f"{n}  sigma", ("SOLUTION", 3 * k + 2))
            for name in ("BPT [N II]", "BPT [S II]", "WHAN"):
                if self._diagnostic(name) is not None:
                    self._add("Diagnostics", name, ("DIAG", name))
        self.shape = self.h["FLXOBS"].data.shape[1:]
        hd = self.h["FLXOBS"].header
        n = self.h["FLXOBS"].data.shape[0]
        self.wave = (hd["CRVAL3"] + (np.arange(n) + 1 - hd.get("CRPIX3", 1))
                     * hd.get("CD3_3", hd.get("CDELT3", 1.0)))
        self.ages = np.asarray(self.h["BaseAgeMetal"].data[:, 0], dtype=float)
        self.status = self._fitted(names)
        self.units = str(self.h["FLUX"].header.get("BUNIT", "")) \
            if "FLUX" in names else ""

    def _fitted(self, names):
        """
        Which spaxels hold a fit: BRAIN's STATUS when the file has it.

        A megacube written by another pipeline has no STATUS. Its unfitted
        spaxels are zeros in every map, and counting them as fitted put
        hundreds of zeros into every map's colour scale (an age map running
        from 0 to 10 shows the whole galaxy in one colour). BADSPXL (0 for a
        good fit) and a non-empty FLXOBS say which spaxels were fitted.
        """
        if "STATUS" in names:
            return np.asarray(self.h["STATUS"].data)
        fitted = np.nansum(np.abs(self.h["FLXOBS"].data), axis=0) > 0
        if "BADSPXL" in names and self.h["BADSPXL"].data is not None:
            fitted &= np.asarray(self.h["BADSPXL"].data) == 0
        return fitted.astype(np.int32)

    def _add(self, group, name, where):
        self.groups.setdefault(group, []).append(name)
        self.maps[name] = where

    def line_flux(self, name):
        if name not in self.lines:
            return None
        return np.asarray(self.h["FLUX_M"].data[self.lines.index(name)], dtype=float)



    def _ratio(self, num, den):
        a, b = self.line_flux(num), self.line_flux(den)
        if a is None or b is None:
            return None
        with np.errstate(invalid="ignore", divide="ignore"):
            r = np.log10(a / b)
        r[~((a > 0) & (b > 0))] = np.nan
        return r

    def diagnostic_inputs(self):
        n2 = self._ratio("[N II]_6583", "Halpha_6563")
        o3 = self._ratio("[O III]_5007", "Hbeta_4861")
        s2a = self.line_flux("[S II]_6716")
        s2b = self.line_flux("[S II]_6731")
        ha = self.line_flux("Halpha_6563")
        s2 = None
        if s2a is not None and s2b is not None and ha is not None:
            with np.errstate(invalid="ignore", divide="ignore"):
                s2 = np.log10((s2a + s2b) / ha)
            s2[~((s2a + s2b > 0) & (ha > 0))] = np.nan
        ew = None
        if "Halpha_6563" in self.lines:
            ew = -np.asarray(self.h["EQW_M"].data[self.lines.index("Halpha_6563")],
                             dtype=float)
        return n2, o3, s2, ew



    def _diagnostic(self, name):
        n2, o3, s2, ew = self.diagnostic_inputs()
        if name == "BPT [N II]" and n2 is not None and o3 is not None:
            return bpt_nii_class(n2, o3)
        if name == "BPT [S II]" and s2 is not None and o3 is not None:
            return bpt_sii_class(s2, o3)
        if name == "WHAN" and n2 is not None and ew is not None:
            return whan_class(n2, ew)
        return None

    def names(self):

        return [n for g in self.groups.values() for n in g]

    def plane(self, name):
        """
        One map as a 2-D array, NaN where the spaxel was not fitted.

        Parameters
        ----------
        name : str
            A name from :meth:`names`, e.g. ``"vrot_star"``, ``"xyy_light"``,
            ``"Halpha_6563  flux"`` or ``"BPT [N II]"``.
        """
        ext, index = self.maps[name]
        if ext == "DIAG":
            return self._diagnostic(index)
        data = self.h[ext].data
        out = np.asarray(data if index is None else data[index], dtype=float)
        out = out.copy()
        out[self.status == 0] = np.nan
        return out

    def spectrum(self, y, x):
        """
        One spaxel's observed and model spectra on :attr:`wave`.

        Returns
        -------
        observed, model : numpy.ndarray
        lines : list of (str, numpy.ndarray)
            Each fitted emission line's Gaussian profile, for lines with a
            positive amplitude.
        """
        obs = np.asarray(self.h["FLXOBS"].data[:, y, x], dtype=float)
        syn = np.asarray(self.h["FLXSYN"].data[:, y, x], dtype=float)
        # Wavelengths the cube does not cover are stored as exact zeros by
        # some pipelines; drawn, they are a cliff down to zero.
        obs = np.where(obs == 0.0, np.nan, obs)
        lines = []
        if self.lines:
            sol = self.h["SOLUTION"].data[:, y, x]
            for k, n in enumerate(self.lines):
                A, v, s = (float(q) for q in sol[3 * k:3 * k + 3])
                rest = self.rest.get(n, np.nan)
                if not (np.isfinite(A) and A > 0 and s > 0 and np.isfinite(rest)):
                    continue
                lam_c = rest * (1 + v / C_KMS)
                sig = s * lam_c / C_KMS
                lines.append((n, A * np.exp(-0.5 * ((self.wave - lam_c) / sig) ** 2)))
        return obs, syn, lines

    def populations(self, y, x):
        """One spaxel's population vector: light fractions, and mass
        fractions when the file has them, in :attr:`ages` order."""
        light = np.asarray(self.h["PoPVecsL"].data[:, y, x], dtype=float)
        mass = np.asarray(self.h["PoPVecsM"].data[:, y, x], dtype=float) \
            if "PoPVecsM" in self.h else None
        return light, mass

    def numbers(self, y, x):
        """Every PoPBins value at one spaxel, as {name: value}."""
        pop = self.h["PoPBins"]
        out = {}
        for i in range(pop.data.shape[0]):
            out[str(pop.header.get(f"DATA{i}"))] = float(pop.data[i, y, x])
        return out
