"""
BRAIN — the datacube viewer.

A map of the cube, and the spectrum of whichever spaxel is clicked on it.

Everything that touches the file happens on a worker thread and nothing is
read that is not about to be drawn: a collapsed map is built from a few
hundred wavelength planes rather than all of them, and a spectrum is one
strided read. A 74x74x6732 cube opens and draws without the interface
stopping, which is the whole point of the arrangement.

The map offers whatever the file actually holds. A plain datacube offers its
flux collapsed over wavelength or over a band, and any two-dimensional
extension of the right shape, which for MaNGA includes the broadband images.
A megacube offers all of that plus every plane of PoPBins, addressed by its
DATA<n> card exactly as the analysis tools address it, and the fit's own
maps. Nothing here is a hard-coded list of quantities, so a plane added to
the output appears here with no change.

While a fit runs, the spaxels each batch finishes light up on a progress map
as they complete. The animation is driven by a timer on the interface thread
and the fitting thread only ever appends indices to an array, so drawing
never holds the fit up.

Whatever `SN_MASKS_<t>` planes a file carries are offered as a cut, and the
spaxels a chosen threshold throws away are drawn yellow over the map with
the count of what survives next to the selector. Two formats write these
planes with opposite polarity and neither says which it used, so the viewer
works it out from the file and states what it decided.

Colour maps are perceptually uniform and the state of a spaxel is never
carried by colour alone: what is not a converged fit is hatched as well as
greyed, the yellow cut is cross-hatched, and the counts are written under
the map.
"""

def orientation_text(header):
    """
    Which way round the map is drawn, read from the file rather than assumed.

    The map is drawn with the array's first row at the bottom, so a cube whose
    world coordinate system has right ascension decreasing with column and
    declination increasing with row is north-up and east-left as displayed.
    Any other arrangement is named instead of being quietly mislabelled, and a
    cube with no celestial coordinates is reported as having none.
    """
    try:
        cd1 = float(header.get("CD1_1", header.get("CDELT1", 0.0)))
        cd2 = float(header.get("CD2_2", header.get("CDELT2", 0.0)))
    except (TypeError, ValueError):
        return "no celestial WCS; rows increase upwards, columns rightwards"
    if cd1 == 0.0 or cd2 == 0.0:
        return "no celestial WCS; rows increase upwards, columns rightwards"
    vertical = "north up" if cd2 > 0 else "north down"
    horizontal = "east left" if cd1 < 0 else "east right"
    return f"{vertical}, {horizontal}; origin at the lower left"

def plane_cards(header, prefix):
    """The PREFIX<n> cards of a stacked extension, in order, as a list."""
    names = []
    index = 0
    while f"{prefix}{index}" in header:
        names.append(str(header[f"{prefix}{index}"]).strip())
        index += 1
    return names

def map_entries(hdu_list, flux_ext, shape):
    """
    Every map this file can draw, derived from the file and not from a list.

    A three-dimensional extension carrying DATA<n>, PLANE<n> or PARAM<n> cards
    is a stack of named maps and is offered one plane at a time under the name
    its own card gives it, which is how the analysis tools address PoPBins. A
    two-dimensional extension of the cube's spatial shape is one map. The
    flux cube itself is offered collapsed, since that is the image a cube is
    usually recognised by.
    """
    entries = []
    if flux_ext and flux_ext in hdu_list:
        units = str(hdu_list[flux_ext].header.get("BUNIT", "")).strip()
        entries.append(dict(key="collapse:median", kind="collapse",
                            hdu=flux_ext, stat="median", units=units,
                            label="Flux, median over wavelength"))
        entries.append(dict(key="collapse:sum", kind="collapse",
                            hdu=flux_ext, stat="sum", units=units,
                            label="Flux, summed over wavelength"))
        entries.append(dict(key="collapse:band", kind="collapse",
                            hdu=flux_ext, stat="band", units=units,
                            label="Flux, median over a band"))

    for hdu in hdu_list:
        name = (hdu.name or "").strip()
        if not name or name == flux_ext:
            continue
        data = hdu.data
        if data is None or not hasattr(data, "ndim"):
            continue
        if data.ndim == 2 and data.shape == shape:
            entries.append(dict(key=f"plane:{name}", kind="plane", hdu=name,
                                index=None, label=name,
                                units=str(hdu.header.get("BUNIT", "")).strip(),
                                comment=str(hdu.header.get("SUMMARY", ""))))
        elif data.ndim == 3 and data.shape[1:] == shape:
            for prefix in ("DATA", "PLANE", "PARAM"):
                names = plane_cards(hdu.header, prefix)
                if not names:
                    continue
                for index, plane in enumerate(names):
                    if index >= data.shape[0]:
                        break
                    comment = hdu.header.comments[f"{prefix}{index}"]
                    entries.append(dict(
                        key=f"stack:{name}:{index}", kind="stack", hdu=name,
                        index=index, label=f"{name}    {plane}",
                        units="", comment=str(comment)))
                break
    return entries

def sn_mask_planes(hdu_list, shape):
    """
    The `SN_MASKS_<t>` extensions a file actually carries, lowest threshold
    first, each as a boolean plane of where the value is non-zero.

    Discovered from the file rather than from a list, because the ladder of
    thresholds is a choice whoever wrote the cube made: the published
    megacubes carry 1, 5, 10 and 20, a BRAIN chain carries whatever its
    `thresholds` setting said, and most cubes carry none at all. A name that
    does not end in a number, or a plane that is not on the cube's spatial
    grid, is not one of these and is skipped.
    """
    found = []
    for hdu in hdu_list:
        name = (hdu.name or "").strip()
        if not name.upper().startswith(SN_MASK_PREFIX):
            continue
        try:
            threshold = float(name[len(SN_MASK_PREFIX):])
        except ValueError:
            continue
        data = hdu.data
        if data is None or getattr(data, "ndim", 0) != 2:
            continue
        if tuple(data.shape) != tuple(shape):
            continue
        found.append(dict(name=name, threshold=threshold,
                          header=hdu.header,
                          nonzero=(np.asarray(data) != 0)))
    found.sort(key=lambda entry: entry["threshold"])
    return found

def sn_mask_convention(planes, hdu_list=None, flux_ext=None):
    """
    Which value in an `SN_MASKS_<t>` plane means the spaxel was kept.

    Two codes write these masks with opposite polarity and neither stamps a
    card that says so, so the viewer works it out from the file instead of
    assuming. Getting it backwards paints exactly the wrong spaxels, and it
    does so plausibly, which is why this is decided rather than guessed.

    Three rules, in the order of how little they have to trust:

    The kept set can only shrink as the threshold rises -- a spaxel above
    S/N 20 is above S/N 10 -- so with two or more thresholds present the
    value whose count falls monotonically is the kept one. This needs no
    header and no flux and it decides both conventions.

    Failing that, a `THRESH` or `SN_METH` card is stamped by BRAIN's own S/N
    stage, whose planes are 1 where the threshold was exceeded.

    Failing that, the two groups are compared against a signal-to-noise
    measure: `SNMAP` when the file carries it, otherwise the median flux per
    spaxel sampled from the cube. The group with the higher median is the
    kept one.

    Returns the keep value, whether anything actually decided it, and the
    sentence the interface shows. When nothing decides it the smaller group
    is called the kept one, which is the reading that overstates neither how
    much data survives nor how little, and the caller is told it is a guess.
    """
    if not planes:
        return None
    total = int(planes[0]["nonzero"].size)
    kept = [int(np.count_nonzero(plane["nonzero"])) for plane in planes]
    thresholds = [plane["threshold"] for plane in planes]

    ladder = ", ".join(f"{t:g}" for t in thresholds)
    for value, series in ((1, kept), (0, [total - n for n in kept])):
        if not _monotonically_shrinking(series):
            continue
        counts = ", ".join(str(n) for n in series)
        return dict(
            keep=value, decided=True,
            reason=(f"{value} means keep: the kept set shrinks as the "
                    f"threshold rises ({counts} spaxels at S/N {ladder}), "
                    f"which only the kept set can do"))

    for plane in planes:
        header = plane["header"]
        if "THRESH" in header or "SN_METH" in header:
            return dict(
                keep=1, decided=True,
                reason=("1 means keep: the planes carry the THRESH card this "
                        "pipeline's signal-to-noise stage stamps, which is 1 "
                        "where the spaxel is above the threshold"))

    signal = _sn_proxy(hdu_list, flux_ext, planes[0]["nonzero"].shape)
    if signal is not None:
        flag = planes[-1]["nonzero"]
        high = _median_of(signal[flag])
        low = _median_of(signal[~flag])
        if np.isfinite(high) and np.isfinite(low) and high != low:
            keep = 1 if high > low else 0
            better, worse = (high, low) if keep else (low, high)
            return dict(
                keep=keep, decided=True,
                reason=(f"{keep} means keep: at S/N "
                        f"{planes[-1]['threshold']:g} the spaxels marked "
                        f"{keep} have a median {signal_name(hdu_list)} of "
                        f"{better:.3g} against {worse:.3g} for the rest"))

    keep = 1 if kept[-1] <= total - kept[-1] else 0
    return dict(
        keep=keep, decided=False,
        reason=(f"nothing in this file says which value means keep -- no "
                f"second threshold to compare, no THRESH card and no S/N map "
                f"-- so the smaller group, {keep}, is taken as the kept one "
                f"and the yellow may be inverted"))

def signal_name(hdu_list):
    """What the polarity was decided against, named rather than implied."""
    if hdu_list is not None and "SNMAP" in hdu_list:
        return "SNMAP"
    return "flux"




def _monotonically_shrinking(counts):
    """
    True when a sequence never rises and falls at least once.

    A flat sequence -- every spaxel of a small example cube passing every
    threshold -- is not shrinking and says nothing about polarity, so it
    fails here and the decision falls through to the next rule rather than
    being made on the strength of no evidence.
    """
    if len(counts) < 2:
        return False
    return (all(b <= a for a, b in zip(counts, counts[1:]))
            and any(b < a for a, b in zip(counts, counts[1:])))


def _median_of(values):
    finite = np.asarray(values, dtype=np.float64)
    finite = finite[np.isfinite(finite)]
    return float(np.median(finite)) if finite.size else float("nan")


def _sn_proxy(hdu_list, flux_ext, shape):
    """
    A per-spaxel measure that rises with signal, for deciding polarity.

    `SNMAP` is used when it is there, because it is the map the thresholds
    were cut from and nothing else in the file is closer to the question. A
    file without one falls back to the median flux over a strided sample of
    the cube, which is not a signal-to-noise but does separate the sky from
    the galaxy, and separating them is all this is being asked to do.
    """
    if hdu_list is None:
        return None
    if "SNMAP" in hdu_list:
        data = hdu_list["SNMAP"].data
        if data is not None and tuple(getattr(data, "shape", ())) == tuple(shape):
            return np.asarray(data, dtype=np.float64)
    if not flux_ext or flux_ext not in hdu_list:
        return None
    cube = hdu_list[flux_ext].data
    if cube is None or getattr(cube, "ndim", 0) != 3:
        return None
    step = max(1, cube.shape[0] // COLLAPSE_PLANES)
    block = np.asarray(cube[::step], dtype=np.float64)
    with np.errstate(invalid="ignore"):
        return np.nanmedian(np.where(np.isfinite(block), block, np.nan),
                            axis=0)

def safe_name(text):
    """
    A plane's name turned into one a file system will take, unambiguously.

    Spaces and punctuation become single underscores and the result keeps its
    case, so `PoPBins    Mage_L` becomes `PoPBins_Mage_L` and is still the
    name the file calls it. A name that collapses to nothing is given one,
    because an empty file name silently becomes a dot file.
    """
    kept = [c if (c.isalnum() or c in "-+.") else "_" for c in str(text)]
    out = "".join(kept)
    while "__" in out:
        out = out.replace("__", "_")
    return out.strip("_") or "map"

def scaled(values, scale):
    """
    The limits and the array a map is drawn with, for one of the three scales.

    The clipped linear scale is the default because a single bright spaxel
    otherwise takes the whole colour bar and the field goes flat. The full
    range is offered next to it so that what the clip hides is one click
    away, and neither is applied without the colour bar saying which.
    """
    finite = np.asarray(values, dtype=np.float64)
    good = np.isfinite(finite)
    if not np.any(good):
        return finite, 0.0, 1.0
    if scale == "log":
        positive = good & (finite > 0)
        if not np.any(positive):
            return finite, 0.0, 1.0
        out = np.where(positive, np.log10(np.where(positive, finite, 1.0)),
                       np.nan)
        return out, float(np.nanmin(out)), float(np.nanmax(out))
    if scale == "clip":
        lo, hi = np.percentile(finite[good], [1.0, 99.0])
        if hi <= lo:
            lo, hi = float(np.min(finite[good])), float(np.max(finite[good]))
        return finite, float(lo), float(hi)
    return finite, float(np.min(finite[good])), float(np.max(finite[good]))

class CubeService(QThread):
    """
    Every read of a FITS file the viewer makes, on one thread of its own.

    The interface posts a request and forgets about it; the answer arrives as
    a signal carrying the path it was computed from, so a result for a cube
    that has since been closed is recognised and dropped instead of being
    drawn over the new one. Only the newest request of each kind is kept:
    dragging across the map queues one spectrum, not two hundred.
    """

    opened = Signal(dict)
    map_ready = Signal(dict)
    spectrum_ready = Signal(dict)
    exported = Signal(dict)
    sn_previewed = Signal(dict)
    failed = Signal(str)

    def __init__(self, parent=None):
        super().__init__(parent)
        self._lock = threading.Lock()
        self._pending = {}
        self._wake = threading.Event()
        self._alive = True
        self._hdus = None
        self._path = ""
        self._flux = ""
        self._wave = None
        self._shape = (0, 0)

    def run(self):
        while self._alive:
            self._wake.wait(0.25)
            self._wake.clear()
            while self._alive:
                with self._lock:
                    kind = next((k for k in ("open", "map", "spectrum",
                                             "export", "sn_preview")
                                 if k in self._pending), None)
                    payload = self._pending.pop(kind) if kind else None
                if kind is None:
                    break
                try:
                    getattr(self, "_" + kind)(**payload)
                except Exception as exc:
                    self.failed.emit(f"{type(exc).__name__}: {exc}")
        self._close()



    def _close(self):
        if self._hdus is not None:
            try:
                self._hdus.close()
            except Exception:
                pass
        self._hdus = None

    def _open(self, path, flux):
        self._close()
        self._hdus = fits.open(path, memmap=True)
        self._path = path
        self._flux = self._flux_extension(flux)
        cube = self._hdus[self._flux]
        self._shape = tuple(cube.data.shape[1:])
        self._wave = np.asarray(wavelength_axis(self._hdus, self._flux),
                                dtype=np.float64)
        entries = map_entries(self._hdus, self._flux, self._shape)
        sn_masks, sn_convention = self._sn_masks()
        self.opened.emit(dict(
            path=path, flux=self._flux, shape=self._shape,
            entries=entries,
            wavelength=(float(self._wave[0]), float(self._wave[-1])),
            n_wave=int(len(self._wave)),
            orientation=orientation_text(cube.header),
            units=str(cube.header.get("BUNIT", "")).strip(),
            megacube="FLXOBS" in self._hdus,
            bad=self._bad_plane(),
            sn_masks=sn_masks, sn_convention=sn_convention))

    def _sn_masks(self):
        """
        Every signal-to-noise threshold this file offers, as a kept mask each.

        The polarity is settled once, here, from all the planes together
        rather than from each on its own, and the planes are turned into
        `keep` maps before they leave this thread so that nothing downstream
        has to remember which value meant what.
        """
        planes = sn_mask_planes(self._hdus, self._shape)
        convention = sn_mask_convention(planes, self._hdus, self._flux)
        if convention is None:
            return [], None
        masks = []
        for plane in planes:
            keep = (plane["nonzero"] if convention["keep"] == 1
                    else ~plane["nonzero"])
            masks.append(dict(name=plane["name"],
                              threshold=float(plane["threshold"]),
                              keep=keep, n_keep=int(np.count_nonzero(keep)),
                              n_total=int(keep.size)))
        return masks, convention

    def _sn_preview(self, settings, threshold):
        """
        The S/N masks the pipeline's own S/N stage would write for the open
        file, computed here so the interface never reads the cube itself.
        """
        from brainsp.ifu_modules import SNMask
        if self._hdus is None:
            return
        planes = SNMask(**settings).execute(self._hdus)
        masks = []
        for hdu in planes:
            if hdu.name.startswith("SN_MASKS_"):
                keep = np.asarray(hdu.data) == 1
                masks.append(dict(name=hdu.name,
                                  threshold=float(hdu.header["THRESH"]),
                                  keep=keep, n_keep=int(keep.sum()),
                                  n_total=int(keep.size)))
        self.sn_previewed.emit(dict(path=self._path, masks=masks,
                                    threshold=float(threshold)))

    def _flux_extension(self, wanted):
        """
        The extension the map and the spectra come from.

        A megacube says in BRAINHDU which extension it was fitted from, and
        that one wins, because a chain with a spatial resampling stage in it
        leaves the input FLUX at one shape and every fitted map at another:
        taking FLUX there would size the viewer to a grid none of the maps are
        on, and silently offer none of them. Failing that, the name the
        pipeline is configured with is used, then FLUX, then the first
        three-dimensional image -- so a raw cube opened while the pipeline is
        set up for a dereddened one still opens.
        """
        fitted = str(self._hdus[0].header.get("BRAINHDU", "")).strip()
        for name in (fitted, wanted, "FLUX"):
            if name and name in self._hdus:
                data = self._hdus[name].data
                if data is not None and data.ndim == 3:
                    return name
        for hdu in self._hdus:
            if hdu.data is not None and getattr(hdu.data, "ndim", 0) == 3:
                return hdu.name
        raise ValueError(f"{os.path.basename(self._path)} holds no datacube")

    def _bad_plane(self):
        """BADSPXL as a boolean map, or None when the file does not carry it."""
        if self._hdus is None or "BADSPXL" not in self._hdus:
            return None
        data = np.asarray(self._hdus["BADSPXL"].data)
        if data.shape != self._shape:
            return None
        return (data != 0)

    def _plane(self, key, band=None):
        """One map, as values with the name and unit that belong to it."""
        parts = key.split(":")
        if parts[0] == "collapse":
            values, title, units = self._collapsed(parts[1], band)
        elif parts[0] == "plane":
            hdu = self._hdus[parts[1]]
            values = np.asarray(hdu.data, dtype=np.float64)
            title = parts[1]
            units = str(hdu.header.get("BUNIT", "")).strip()
        else:
            hdu = self._hdus[parts[1]]
            index = int(parts[2])
            values = np.asarray(hdu.data[index], dtype=np.float64)
            name = plane_cards(hdu.header, "DATA") or \
                plane_cards(hdu.header, "PLANE") or \
                plane_cards(hdu.header, "PARAM")
            title = f"{parts[1]}    {name[index]}"
            units = ""
        return values, title, units

    def _map(self, key, band=None):
        if self._hdus is None:
            return
        values, title, units = self._plane(key, band)
        self.map_ready.emit(dict(path=self._path, key=key, values=values,
                                 title=title, units=units))

    def _export(self, keys, folder, formats, band=None, scale="clip",
                colour_map="viridis", sn_index=None, hide_unfitted=True):
        """
        Write one map, or every map, to the folder, on this thread.

        Reading and writing a hundred planes takes long enough to freeze an
        interface that does it on the drawing thread, and the only reason the
        viewer has a thread at all is that it must not. Progress is reported
        as it goes, so the dialog that started it can say which file it is on
        rather than only that it has not come back yet.

        Every file is written through the same trust mask the map is drawn
        with, so an exported plane cannot disagree with the plane on screen
        about which spaxels are measurements.
        """
        if self._hdus is None:
            return
        masks, convention = self._sn_masks()
        keep = None
        cut_text = "no signal-to-noise cut"
        if sn_index is not None and 0 <= int(sn_index) < len(masks):
            chosen = masks[int(sn_index)]
            keep = chosen["keep"]
            cut_text = (f"S/N > {chosen['threshold']:g} kept, "
                        f"{chosen['n_keep']} of {chosen['n_total']} spaxels")
        bad = self._bad_plane()
        converged = None
        if bad is not None and hide_unfitted:
            converged = ~bad
        wcs = ifu_output.spatial_wcs(self._hdus[self._flux].header)
        orientation = orientation_text(self._hdus[self._flux].header)
        scale_text = dict(SCALES).get(scale, scale)

        written, failures = [], []
        total = max(len(keys) * len(formats), 1)
        step = 0
        for key, name in keys:
            try:
                values, title, units = self._plane(key, band)
            except Exception as exc:
                failures.append(f"{name}: {type(exc).__name__}: {exc}")
                step += len(formats)
                continue
            mask = ifu_output.trust_mask(values, sn_keep=keep,
                                         converged=converged)
            shown, low, high = scaled(values, scale)
            settings = {
                "SNCUT": (cut_text, "signal-to-noise cut applied"),
                "FITCUT": ("spaxels that are not converged fits are blanked"
                           if converged is not None else
                           "no convergence cut applied",
                           "convergence cut applied"),
                "CSCALE": (scale_text,
                           "colour scaling of the image; data here are raw"),
                "CMAP": (str(colour_map), "colour map of the image"),
                "ORIENT": (orientation, "how the map reads on the sky")}
            for fmt in formats:
                path = os.path.join(folder, f"{name}.{fmt}")
                try:
                    if fmt == "fits":
                        ifu_output.write_plane_fits(
                            path, values, mask, plane=title,
                            cube=self._path, units=units, wcs=wcs,
                            settings=settings)
                    else:
                        ifu_output.write_map_image(
                            path, shown, mask, plane=title, cube=self._path,
                            units=units, colour_map=colour_map, vmin=low,
                            vmax=high, scale=scale_text, cut=cut_text,
                            orientation=orientation)
                    written.append(path)
                except Exception as exc:
                    failures.append(f"{os.path.basename(path)}: "
                                    f"{type(exc).__name__}: {exc}")
                step += 1
                self.exported.emit(dict(path=self._path, state="working",
                                        done=step, total=total,
                                        file=os.path.basename(path)))
        self.exported.emit(dict(path=self._path, state="done", folder=folder,
                                written=written, failures=failures,
                                done=total, total=total))

    def _collapsed(self, stat, band):
        """
        The flux cube as one image, from a few hundred planes and not all of
        them.

        A collapsed image is a picture of where the light is, and it is the
        same picture whether it is built from six thousand planes or from two
        hundred spread evenly through them. Reading two hundred is what makes
        a large cube open at once instead of after several seconds, and the
        caption says how many were used so nobody mistakes it for a
        measurement.
        """
        cube = self._hdus[self._flux].data
        n_wave = cube.shape[0]
        inside = np.ones(n_wave, dtype=bool)
        window = ""
        if stat == "band" and band:
            low, high = float(band[0]), float(band[1])
            inside = (self._wave >= low) & (self._wave <= high)
            if not np.any(inside):
                raise ValueError(
                    f"the band {low:g}-{high:g} A lies outside this cube, "
                    f"which covers {self._wave[0]:.0f}-{self._wave[-1]:.0f} A")
            window = f", {low:g} to {high:g} Å"
        indices = np.flatnonzero(inside)
        total = len(indices)
        step = max(1, total // COLLAPSE_PLANES)
        indices = indices[::step]
        block = np.asarray(cube[indices], dtype=np.float32)
        block = np.where(np.isfinite(block), block, np.nan)
        sampled = f"{len(indices)} of {total} planes{window}"
        with np.errstate(invalid="ignore"):
            if stat == "sum":
                values = np.nansum(block, axis=0) * float(step)
                title = f"Flux, summed over wavelength, from {sampled}"
            else:
                values = np.nanmedian(block, axis=0)
                title = f"Flux, median over wavelength, from {sampled}"
        empty = np.all(~np.isfinite(block), axis=0)
        values = np.where(empty, np.nan, values)
        units = str(self._hdus[self._flux].header.get("BUNIT", "")).strip()
        return np.asarray(values, dtype=np.float64), title, units

    def _spectrum(self, y, x, redshift, known=False):
        """
        One spaxel, with the fit over it when the file carries one.

        A megacube's FLXOBS is the observed flux the fit actually used, on the
        linear grid and in the rest frame, and FLXSYN is the model on the same
        grid, so the two are plotted against each other with nothing
        interpolated here. A plain cube has only its own flux, which is in the
        frame the file is in, and the redshift the pipeline is configured with
        moves it to the rest frame when the viewer is asked for that.
        """
        if self._hdus is None:
            return
        n_y, n_x = self._shape
        if not (0 <= y < n_y and 0 <= x < n_x):
            return
        curves = []
        if "FLXOBS" in self._hdus:
            header = self._hdus["FLXOBS"].header
            n = self._hdus["FLXOBS"].data.shape[0]
            start = float(header.get("CRVAL3", 0.0))
            step = float(header.get("CD3_3", header.get("CDELT3", 1.0)))
            grid = start + step * np.arange(n, dtype=np.float64)
            curves.append(dict(
                wave=grid, role="observed",
                flux=np.asarray(self._hdus["FLXOBS"].data[:, y, x],
                                dtype=np.float64)))
            if "FLXSYN" in self._hdus:
                curves.append(dict(
                    wave=grid, role="model",
                    flux=np.asarray(self._hdus["FLXSYN"].data[:, y, x],
                                    dtype=np.float64)))
            frame = "rest frame, as fitted"
            at_rest = True
        else:
            shift = float(redshift or 0.0)
            wave = self._wave / (1.0 + shift) if shift else self._wave
            curves.append(dict(
                wave=wave, role="observed",
                flux=np.asarray(self._hdus[self._flux].data[:, y, x],
                                dtype=np.float64)))
            # A redshift that was given -- 0 included, for a cube already at
            # rest -- puts the axis in the rest frame; a blank one does not.
            frame = (f"rest frame, z = {shift:g} removed" if shift
                     else "rest frame, z = 0" if known
                     else "the frame the file is in")
            at_rest = bool(shift) or bool(known)
        self.spectrum_ready.emit(dict(
            path=self._path, y=int(y), x=int(x), curves=curves, frame=frame,
            at_rest=at_rest, units=str(self._hdus[self._flux].header.get(
                "BUNIT", "")).strip(),
            values=self._spaxel_values(y, x)))

    def _spaxel_values(self, y, x):
        """
        What the fit says about this spaxel, as name and number pairs.

        Taken from PoPBins by its own DATA<n> cards, so a quantity added to
        the output is reported here without this code being told about it.
        The handful that are worth reading next to a spectrum are named, and
        the rest of the stack stays in the map list where it belongs.
        """
        wanted = ("Av", "Sigma_star", "vrot_star", "ChiSqrt", "Adev", "SNR",
                  "Mage_L", "MZ_L", "BadSpaxel")
        if self._hdus is None or "PoPBins" not in self._hdus:
            return []
        header = self._hdus["PoPBins"].header
        names = plane_cards(header, "DATA")
        data = self._hdus["PoPBins"].data
        out = []
        for name in wanted:
            if name not in names:
                continue
            index = names.index(name)
            out.append((name, float(data[index, y, x])))
        return out

    def _fill_sn_masks(self, masks, convention):
        """
        Offer whichever signal-to-noise thresholds the opened file carries.

        The list is rebuilt from the file every time, and a threshold that
        was being shown is re-selected if the new file has it, so switching
        between the cube and the megacube does not silently drop the cut. A
        file with no `SN_MASKS_<t>` at all -- which is most files -- leaves
        the row present but disabled and saying so, rather than offering a
        control that does nothing.
        """
        previous = self.cb_sn.currentData()
        self.sn_masks = list(masks)
        self.sn_convention = convention
        self.cb_sn.blockSignals(True)
        self.cb_sn.clear()
        if self.sn_masks:
            self.cb_sn.addItem("every spaxel, no cut", None)
        else:
            self.cb_sn.addItem("every spaxel; this file offers no cut", None)
        for index, mask in enumerate(self.sn_masks):
            share = 100.0 * mask["n_keep"] / max(mask["n_total"], 1)
            self.cb_sn.addItem(
                f"S/N > {mask['threshold']:g}   {mask['n_keep']} of "
                f"{mask['n_total']} spaxels   {share:.0f} %", index)
        self.cb_sn.setEnabled(bool(self.sn_masks))
        self.lbl_sn.setEnabled(bool(self.sn_masks))
        restored = -1
        if previous is not None and previous < len(self.sn_masks):
            restored = self.cb_sn.findData(previous)
        self.cb_sn.setCurrentIndex(max(restored, 0))
        self.cb_sn.blockSignals(False)
        if self.sn_masks:
            self.cb_sn.setToolTip(
                "Spaxels the threshold throws away stay on the map in "
                "yellow, cross-hatched, so what a cut costs is visible "
                "rather than merely absent.\n"
                + self.sn_convention["reason"])
        else:
            self.cb_sn.setToolTip(
                "A signal-to-noise cut is a set of SN_MASKS_<t> planes "
                "written beside the cube. This file carries none, so there "
                "is nothing to choose between.")
        self._sn_cost()

    def _export_dialog(self):
        """
        Ask what to export, then hand the work to the reading thread.

        The dialog is opened from the strip the plane is chosen in, not from
        a menu, because the question it answers is about the plane now on
        screen and that is where a user is looking when they want it.

            return
        entries = [entry for entry in self.entries
                   if not entry["key"].startswith("collapse:")]
        current = self.cb_plane.currentData()
        chosen = next((entry for entry in self.entries
                       if entry["key"] == current), None)
        if chosen is None:
            return

                           os.path.dirname(self.path) or os.getcwd(), self)
        if dialog.exec() != QDialog.Accepted:
            return

        if not os.path.isdir(folder):
            try:
                os.makedirs(folder, exist_ok=True)
            except OSError as exc:
                self.status.emit(f"cannot write to {folder}: {exc}")

        wanted = entries if dialog.every_plane() else [chosen]
        if not wanted:
            wanted = [chosen]
        stem = os.path.splitext(os.path.basename(self.path))[0]
        keys = [(entry["key"], f"{stem}-{safe_name(entry['label'])}")
                for entry in wanted]
        self._exporting = True

        self.btn_export.setText("Writing\u2026")

            "export", keys=keys, folder=folder,

    def _draw_map(self):
        """
        The selected map, with the spaxels that are not measurements marked.

        A spaxel that was not fitted, or whose fit did not converge, is
        greyed and hatched rather than coloured, so that a map cannot be read
        as a field of measurements it is not: the published megacubes write 0
        into an unfitted spaxel, and 0 is a plausible value for most of these
        quantities. The count is written under the map as well, because a
        hatch is a texture and a number is a number.

        The plot carries no title and no axis names. Which map this is, is
        said once in the chooser above it; the two axes are spaxel indices,
        which is what the numbers along them already say. A rotated name down
        the left margin cost sixty pixels and told nobody anything.
        """
        if self._map_values is None or self._fit_level is not None:
            return
        C = self.C
        values, low, high = scaled(self._map_values,
                                   self.cb_scale.currentData())
        untrusted = np.zeros(values.shape, dtype=bool)
        if self.bad is not None and self.ck_converged.isChecked():
            untrusted = np.asarray(self.bad, dtype=bool)
            values = np.where(untrusted, np.nan, values)
        masked = np.ma.masked_invalid(values)

        self.map_ax.clear()
        self.map_cax.clear()
        self.map_fig.texts.clear()
        self.map_cax.set_visible(True)
        n_y, n_x = self.shape
        extent = (-0.5, n_x - 0.5, -0.5, n_y - 0.5)
        self._image = self.map_ax.imshow(
            masked, origin="lower", cmap=self._colour_map(), vmin=low,
            vmax=high, extent=extent, interpolation="nearest", aspect="equal")
        if np.any(untrusted):
            contours = self.map_ax.contourf(
                np.arange(n_x), np.arange(n_y), untrusted.astype(np.float64),
                levels=[0.5, 1.5], colors="none", hatches=["///"])
            contours.set_edgecolor(C["text_mute"])
            contours.set_linewidth(0.0)
        cut = self._draw_sn_cut(extent)
        self._draw_colour_bar()
        self._style_map_axes()
        self._zoom.set_full(extent)
        self._zoom.apply()
        self._draw_selection()

        note = []
        if high <= low:
            finite = np.asarray(self._map_values, dtype=np.float64)
            finite = finite[np.isfinite(finite)]
            level = float(finite[0]) if finite.size else float("nan")
            note.append(f"constant at {level:g} across the field")
        hidden = int(np.sum(untrusted))
        if hidden:
            plural = "" if hidden == 1 else "s"
            note.append(f"{hidden} spaxel{plural} hatched, not a converged fit")
        if cut:
            note.append(cut)
        if self.geometry_text:
            note.append(self.geometry_text)
        note.append("scroll to zoom, drag to pan, double-click for the "
                    "whole field")
        self.lbl_foot.setText(".  ".join(note))
        self.map_canvas.draw_idle()

    def _draw_colour_bar(self):
        """
        The scale the map is read against, in the palette and in mono.

        The bar used to arrive with matplotlib's own white axes under it,
        which put the brightest rectangle on a dark page next to the data and
        beat the data for attention. It is drawn here from the same palette

    def _bar_caption(self):

        What the colour bar measures, in as few words as say it.

        The unit when the file carries one, because a number with no unit is

    def _draw_sn_cut(self, extent):
        """
        The spaxels the chosen threshold throws away, in yellow over the map.

        Drawn a cell at a time so that the edge of the yellow is the edge of
        the mask and not a contour interpolated between spaxel centres: the
        point of the overlay is to say exactly which spaxels a threshold
        costs, and a boundary drawn half a spaxel out would be a picture of
        something else.

        All four colour maps offered here end in yellow, so the overlay
        carries a cross-hatch as well as a colour and the count is written
        under the map. Nothing about the cut is readable from the hue alone.
        """
        mask = self._sn_selected()
        if mask is None:
            return ""
        cut = ~np.asarray(mask["keep"], dtype=bool)
        if not np.any(cut):
            return f"S/N > {mask['threshold']:g} costs no spaxel"
        n_y, n_x = self.shape
        colour = SN_CUT_COLOUR.get(self.C.get("name", "light"),
                                   SN_CUT_COLOUR["light"])
        rgba = np.zeros((n_y, n_x, 4), dtype=np.float64)
        rgba[..., :3] = matplotlib.colors.to_rgb(colour)
        rgba[..., 3] = np.where(cut, 1.0, 0.0)
        self.map_ax.imshow(rgba, origin="lower", extent=extent,
                           interpolation="nearest", aspect="equal", zorder=3)
        patches = self.map_ax.contourf(
            np.arange(n_x), np.arange(n_y), cut.astype(np.float64),
            levels=[0.5, 1.5], colors="none", hatches=[SN_CUT_HATCH],
            zorder=4)
        patches.set_edgecolor(self.C["text"])
        patches.set_linewidth(0.0)
        dropped = int(np.count_nonzero(cut))
        plural = "" if dropped == 1 else "s"
        return (f"S/N > {mask['threshold']:g} drops {dropped} spaxel{plural} "
                f"in yellow, keeping {mask['n_keep']} of {mask['n_total']}")

    def step(self, down, across):
        """
        Move the selection to a neighbouring spaxel, and stop at the edge.

        Stopping rather than wrapping: a cursor that reappears on the far
        side of the field after one key press has moved 73 spaxels, and the
        spectrum that arrives is of somewhere else entirely.
        """
        if self.shape == (0, 0):
            return
        n_y, n_x = self.shape
        row, column = self._picked if self._picked is not None else (0, 0)
        if self._picked is None:
            self.pick(n_y // 2, n_x // 2)
            return
        self.pick(max(0, min(n_y - 1, row + int(down))),

        text = f"wavelength [\u00c5, {frame}]"
        if units or factor:
            text = f"flux [{factor}{units}]".replace(" ]", "]") + "      " + text
        self.spec_ax.text(0.0, 1.0, text, transform=self.spec_ax.transAxes,

    def begin_run(self, n_y, n_x, background=None):
        """
        Start the progress map: a field of the fit's own shape, all unfitted.

        The shape is the fit's, not the file's, because a spatial resampling
        stage in front of the fit changes it, and showing progress on the
        wrong grid would put a spaxel in the wrong place. The collapsed flux
        map is dimmed underneath it when the two shapes agree, and left out
        when they do not.

        The denominator starts as the whole field, which is the only number
        known at this point, and is replaced by the count of fittable spaxels
        as soon as the fit reports one. Saying 'of 5476' while the fit will
        only ever attempt 1452 of them is a progress bar that stops at a
        quarter and looks broken.
        """
        self._fit_shape = (int(n_y), int(n_x))
        self._fit_level = np.zeros(self._fit_shape, dtype=np.float64)
        self._fit_target = np.zeros(self._fit_shape, dtype=np.float64)
        self._fit_flash = np.zeros(self._fit_shape, dtype=np.float64)
        self._fit_running = True
        self._background = None
        self._fit_image = None
        self._fit_started = time.monotonic()
        self._fit_valid = None
        self._fit_converged = None
        self._fit_accounted = 0
        self._fit_batches = 0
        self._fit_reason = "finished"
        self.lbl_run.setVisible(True)
        if background is None and self._map_values is not None \
                and self._map_values.shape == self._fit_shape:
            background = self._map_values
        if background is not None:
            # A log stretch: a galaxy's flux spans decades, and a linear one
            # shows the nucleus on a black field.
            good = np.isfinite(background) & (background > 0)
            if np.any(good):
                logged = np.where(good, np.log10(np.where(good, background, 1.0)),
                                  np.nan)
                low, high = np.nanpercentile(logged[good], [2.0, 99.5])
                span = max(high - low, 1e-30)
                self._background = np.clip(
                    (np.where(good, logged, low) - low) / span, 0.0, 1.0)
        self._draw_progress()
        if not self._rise.isActive():
            self._rise.start()

    def _advance_rise(self):
        """
        One frame of the progress map.

        An exponential approach rather than a jump, so a batch appears over
        about a fifth of a second instead of snapping on, and the timer stops
        as soon as nothing is moving, so an idle viewer costs nothing.
        """
        if self._fit_level is None:
            self._rise.stop()
            return
        rate = 1.0 - np.exp(-RISE_INTERVAL_MS / (1000.0 * RISE_SECONDS / 3.0))
        gap = self._fit_target - self._fit_level
        self._fit_level += gap * rate
        moving = float(np.max(np.abs(gap))) if gap.size else 0.0
        flash = getattr(self, "_fit_flash", None)
        flashing = 0.0
        if flash is not None and flash.size:
            flash *= np.exp(-RISE_INTERVAL_MS / (1000.0 * FLASH_SECONDS / 3.0))
            flash[flash < 0.01] = 0.0
            flashing = float(flash.max())
        if moving < 0.004 and flashing == 0.0:
            self._fit_level[...] = self._fit_target
            self._rise.stop()
        self._draw_progress()

    def _progress_rgb(self):
        """
        The frame as one RGB array: the flux map in grey, fitted spaxels in
        the accent's hue at their own brightness, so the galaxy shows through
        what is done.
        """
        C = self.C
        n_y, n_x = self._fit_shape
        if self._background is not None:
            grey = PROGRESS_FIELD + PROGRESS_LIT * self._background
        else:
            grey = np.full((n_y, n_x), PROGRESS_FIELD + 0.5 * PROGRESS_LIT)
        base = np.repeat(grey[:, :, None], 3, axis=2)
        accent = np.array([int(C["accent"][i:i + 2], 16) / 255.0
                           for i in (1, 3, 5)])
        # The accent scaled to the spaxel's brightness: a bright knot stays
        # bright, a faint outskirt stays faint, both turn blue once fitted.
        shade = 0.45 + 0.75 * (grey - PROGRESS_FIELD) / PROGRESS_LIT
        tinted = np.clip(shade[:, :, None] * accent[None, None, :], 0.0, 1.0)
        alpha = np.clip(self._fit_level, 0.0, 1.0)[:, :, None] * PROGRESS_TINT
        frame = base * (1.0 - alpha) + tinted * alpha
        flash = getattr(self, "_fit_flash", None)
        if flash is not None and flash.shape == frame.shape[:2]:
            # A spaxel just fitted burns near-white in the accent's hue and
            # cools into the tint: the pop that shows where the fit is now.
            hot = accent * 0.25 + 0.75
            f = np.clip(flash, 0.0, 1.0)[:, :, None] * 0.9
            frame = frame * (1.0 - f) + hot[None, None, :] * f
        return frame

    def _draw_progress(self):
        """
        The fit filling the field in, twenty-five frames a second.

        The image is built once and its data replaced on every frame after
        that. Clearing the axes and rebuilding the image each time, which is
        what this did, threw the artist away and re-ran the layout twenty-five
        times a second; on a 74 by 74 cube that is visible as a flicker and it
        is also most of the cost.

        What rises is a spaxel the fit has ATTEMPTED. Whether it converged is
        not in the message the fitter sends, so it is never implied here: the
        heading says attempted, and the converged count underneath it is
        whatever the run has reported, or nothing at all when it has reported
        none.
        """
        if self._fit_level is None:
            return
        C = self.C
        n_y, n_x = self._fit_shape
        rgb = self._progress_rgb()
        if self._fit_image is None or self._fit_image.get_array().shape[:2] \
                != (n_y, n_x):
            self.map_ax.clear()
            self.map_cax.clear()
            self.map_fig.texts.clear()
            self.map_cax.set_visible(False)
            self._fit_image = self.map_ax.imshow(

                extent=(-0.5, n_x - 0.5, -0.5, n_y - 0.5))
            self._style_map_axes()
            self._zoom.set_full((-0.5, n_x - 0.5, -0.5, n_y - 0.5))
            self._zoom.apply()

    def _progress_heading(self):
        """What the field is showing, in words, with attempted said plainly."""
        attempted = int(np.sum(self._fit_target > 0.5))
        n_y, n_x = self._fit_shape
        total = self._fit_valid if self._fit_valid is not None else n_y * n_x
        of = ("of the {0} spaxels this fit will attempt".format(total)
              if self._fit_valid is not None
              else f"of the {total} spaxels in the field")
        if self._fit_running:
            return (f"Fitting. {attempted} {of} have been attempted; "
                    f"those are tinted, the rest are still grey.")
        if self._fit_reason == "stopped":
            return (f"Fit stopped. {attempted} {of} had been attempted when "
                    f"it was ended.")
        if self._fit_reason == "failed":
            return (f"Fit failed. {attempted} {of} had been attempted when "
                    f"it stopped.")
        return f"Fit finished. {attempted} {of} were attempted."

    def _progress_readout(self):
        """
        Elapsed, remaining and converged, in mono, under the field.

        Remaining is extrapolated from the spaxels finished so far and is
        called an estimate, because it is one. It is left out entirely until
        a batch has finished, since a rate measured over no work is not a
        rate.
        """
        attempted = int(np.sum(self._fit_target > 0.5))
        n_y, n_x = self._fit_shape
        total = self._fit_valid if self._fit_valid is not None else n_y * n_x
        elapsed = max(time.monotonic() - self._fit_started, 0.0)
        parts = [f"elapsed {_clock(elapsed)}"]
        if self._fit_running and attempted > 0 and attempted < total:
            rate = attempted / max(elapsed, 1e-6)
            parts.append(f"about {_clock((total - attempted) / rate)} left")
        if self._fit_converged is not None and self._fit_accounted > 0:
            parts.append(f"converged {self._fit_converged} of "
                         f"{self._fit_accounted} reported so far")
        if self._fit_batches:
            parts.append(f"{self._fit_batches} batches done")
        return "   ".join(parts)

def _clock(seconds):
    """Seconds as mm:ss, or h:mm:ss once there is an hour of it."""
    seconds = int(max(float(seconds), 0.0))
    hours, rest = divmod(seconds, 3600)
    minutes, second = divmod(rest, 60)
    if hours:
        return f"{hours:d}:{minutes:02d}:{second:02d}"
    return f"{minutes:02d}:{second:02d}"
