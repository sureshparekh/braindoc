Source code for brainsp.plots

"""
Figures from code: every plot the desktop interface draws, written to a file.

Each function draws on a matplotlib Figure (no display needed) and, when
given a ``path``, writes it there; the suffix picks the format (``.png``,
``.pdf``, ``.svg``, ``.jpg``). Each returns the Figure, so it can be changed
further before it is saved, and a function given ``ax=`` draws into an
existing axes instead, for a figure of your own.

For a single spectrum (from :func:`brainsp.fit_spectrum`):

* :func:`spectrum_fit` -- the standard BRAIN figure: data, model, stellar and
  emission components, residuals and the headline numbers.
* :func:`spectrum` -- the data and model on one axes, optionally zoomed.
* :func:`sfh` -- the star-formation history, light- or mass-weighted.

For a datacube (from :func:`brainsp.fit_cube` or
:func:`brainsp.read_megacube`):

* :func:`map` -- one map, with the unfitted spaxels grey and hatched.
* :func:`maps` -- several maps on one page.
* :func:`export_maps` -- maps to files: images for a paper and FITS with the
  cube's sky coordinates and a mask saying which spaxels are measurements.
* :func:`spaxel` -- one spaxel's spectrum, model and emission lines.
* :func:`sfh` -- one spaxel's star-formation history (``y=``, ``x=``).
* :func:`diagnostic` -- the BPT [N II], BPT [S II] or WHAN diagram, the
  spaxels coloured by class, with the dividing curves.

Figures are drawn for print: white ground, black text, a stated unit on
every axis, perceptually uniform colour maps.
"""

import os

import numpy as np

__all__ = ["spectrum_fit", "spectrum", "sfh", "map", "maps", "export_maps",
           "spaxel", "diagnostic"]

INK = "#111111"
DIM = "#555555"
FIT = "#1D5BBF"
STARS = "#D9822B"
GAS = "#1C7C54"
MASKED = "#C0362C"


def _figure(figsize):
    from matplotlib.figure import Figure
    fig = Figure(figsize=figsize, dpi=150)
    fig.set_facecolor("white")
    return fig


def _style(ax):
    ax.set_facecolor("white")
    for side in ("top", "right"):
        ax.spines[side].set_visible(False)
    for side in ("left", "bottom"):
        ax.spines[side].set_color(DIM)
    ax.tick_params(colors=DIM, labelsize=8)
    ax.xaxis.label.set_color(INK)
    ax.yaxis.label.set_color(INK)


def _save(fig, path, dpi=300):
    if path:
        folder = os.path.dirname(os.path.abspath(str(path)))
        os.makedirs(folder, exist_ok=True)
        fig.savefig(str(path), dpi=dpi, facecolor="white",
                    bbox_inches="tight")
    return fig


def _result(result):
    from brainsp.api import FitResult
    if isinstance(result, FitResult):
        return result
    return FitResult(str(result))

def spectrum_fit(result, path=None):
    """
    The standard BRAIN figure of one fit: what ``make_plot`` writes.

    Parameters
    ----------
    result : FitResult or str
        A fit, or the path of its ``*_summary.txt``.
    path : str, optional
        Output file; beside the summary as ``<name>_fit.pdf`` by default.

    Returns
    -------
    str
        The file written.
    """
    return _result(result).plot(path)

def spectrum(result, path=None, *, ax=None, xlim=None, components=True,
             residuals=True, title=None):
    """
    Observed spectrum and model, with the stellar and emission parts.

    Parameters
    ----------
    result : FitResult or str
    path : str, optional
        Write the figure here.
    ax : matplotlib.axes.Axes, optional
        Draw into these axes instead (no residual panel).
    xlim : (float, float), optional
        Wavelength range to show, in Angstrom, e.g. ``(6450, 6650)`` for
        H-alpha and [N II].
    components : bool
        Draw the stellar and emission-line models separately as well.
    residuals : bool
        Add a residual panel underneath.
    title : str, optional

    Returns
    -------
    matplotlib.figure.Figure
    """
    r = _result(result)
    wave, obs, model = r.wavelength, r.observed, r.model
    if ax is None:
        fig = _figure((9, 4.6 if residuals else 3.6))
        if residuals:
            grid = fig.add_gridspec(2, 1, height_ratios=(3, 1), hspace=0.05)
            ax = fig.add_subplot(grid[0])
            rax = fig.add_subplot(grid[1], sharex=ax)
        else:
            ax = fig.add_subplot(111)
            rax = None
    else:
        fig, rax = ax.figure, None
    _style(ax)
    ax.plot(wave, obs, color=INK, lw=0.7, label="observed")
    ax.plot(wave, model, color=FIT, lw=1.0, label="model")
    if components:
        if r.emission is not None and np.any(r.emission != 0):
            ax.plot(wave, r.stellar, color=STARS, lw=0.8, alpha=0.9,
                    label="stars")
            # The lines on their own sit on zero; in a zoom that baseline is
            # far below the data, and the model already shows them on top of
            # the stars, so they are drawn separately only in the full view.
            if not xlim:
                ax.plot(wave, r.emission, color=GAS, lw=0.8,
                        label="emission")
    ax.set_ylabel("flux (normalised)")
    ax.legend(frameon=False, fontsize=8, ncol=4, loc="upper right")
    if title or r.spectrum:
        ax.set_title(title or os.path.basename(str(r.spectrum)), fontsize=9,
                     color=INK, loc="left")
    if xlim:
        ax.set_xlim(*xlim)
        inside = (wave >= xlim[0]) & (wave <= xlim[1])
        if np.any(inside):
            lo = np.nanmin([obs[inside].min(), model[inside].min()])
            hi = np.nanmax([obs[inside].max(), model[inside].max()])
            pad = 0.08 * (hi - lo or 1.0)
            ax.set_ylim(lo - pad, hi + pad)
    if rax is not None:
        _style(rax)
        with np.errstate(invalid="ignore", divide="ignore"):
            res = obs - model
        rax.plot(wave, res, color=DIM, lw=0.6)
        rax.axhline(0, color=FIT, lw=0.6)
        rax.set_ylabel("data - model", fontsize=8)
        rax.set_xlabel("rest wavelength [Å]")
        for t in ax.get_xticklabels():
            t.set_visible(False)
    else:
        ax.set_xlabel("rest wavelength [Å]")
    return _save(fig, path)




def _sfh_vector(source, y=None, x=None, kind="light"):
    """Ages and fractions, summed over metallicity, from a fit or a cube."""
    from brainsp.megacube import Megacube
    if isinstance(source, Megacube):
        light, mass = source.populations(int(y), int(x))
        frac = light if kind == "light" or mass is None else mass
        ages = np.asarray(source.ages, dtype=float)
        n = min(len(ages), len(frac))
        ages, frac = ages[:n], frac[:n]
        # Power-law (AGN) components are catalogued with an age of 1 yr;
        # they are not a population and have no place in a history.
        stellar = ages > 10.0
        ages, frac = ages[stellar], frac[stellar]
    else:
        r = _result(source)
        p = r.populations[r.populations["stellar"]]
        ages = p["age"]
        frac = p["x"] if kind == "light" else p["mcor"]
    uniq = np.unique(ages)
    total = np.array([frac[ages == a].sum() for a in uniq])
    if total.sum() > 0:
        total = 100.0 * total / total.sum()
    return uniq, total

def sfh(source, path=None, *, y=None, x=None, kind="both", ax=None,
        cumulative=False):
    """
    The star-formation history: the fraction of light or mass in each age.

    Fractions at the same age (different metallicities) are summed; light
    fractions are at the normalisation wavelength, mass fractions are of the
    present stellar mass.

    Parameters
    ----------
    source : FitResult, str or Megacube
        A spectrum fit, or a megacube with ``y`` and ``x`` naming a spaxel.
    path : str, optional
    y, x : int, optional
        The spaxel, for a megacube.
    kind : {"light", "mass", "both"}
    ax : matplotlib.axes.Axes, optional
        Draw into these axes (``kind`` must then be ``"light"`` or
        ``"mass"``).
    cumulative : bool
        Show the cumulative fraction formed, oldest first, instead.

    Returns
    -------
    matplotlib.figure.Figure
    """
    kinds = ["light", "mass"] if kind == "both" else [kind]
    if ax is None:
        fig = _figure((4.6 * len(kinds), 3.2))
        axes = [fig.add_subplot(1, len(kinds), i + 1)
                for i in range(len(kinds))]
    else:
        fig, axes = ax.figure, [ax]
    for a, k in zip(axes, kinds):
        _style(a)
        ages, frac = _sfh_vector(source, y, x, k)
        logt = np.log10(ages)
        colour = FIT if k == "light" else STARS
        if cumulative:
            order = np.argsort(logt)[::-1]
            a.step(logt[order], np.cumsum(frac[order]), where="post",
                   color=colour)
            a.set_ylabel(f"cumulative {k} fraction [%]")
        else:
            # Template ages are unevenly spaced in log t; each bar spans
            # halfway to its neighbours, so none is drawn as a hairline.
            if len(logt) > 1:
                mids = 0.5 * (logt[1:] + logt[:-1])
                edges = np.concatenate([[logt[0] - (mids[0] - logt[0])],
                                        mids,
                                        [logt[-1] + (logt[-1] - mids[-1])]])
                width = np.minimum(np.diff(edges), 0.3) * 0.9
            else:
                width = 0.1
            a.bar(logt, frac, width=width, color=colour, alpha=0.85,
                  edgecolor="white", linewidth=0.3)
            a.set_ylabel(f"{k} fraction [%]")
        a.set_xlabel("log age [yr]")
    return _save(fig, path)

def _cube(source):
    from brainsp.megacube import Megacube
    return source if isinstance(source, Megacube) else Megacube(str(source))


def _limits(values, scale):
    good = np.isfinite(values)
    if not np.any(good):
        return values, 0.0, 1.0
    if scale == "log":
        pos = good & (values > 0)
        out = np.where(pos, np.log10(np.where(pos, values, 1.0)), np.nan)
        return out, float(np.nanmin(out)), float(np.nanmax(out))
    if scale == "clip":
        lo, hi = np.percentile(values[good], [1.0, 99.0])
        if hi > lo:
            return values, float(lo), float(hi)
    return values, float(np.nanmin(values)), float(np.nanmax(values))

def map(cube, name, path=None, *, ax=None, cmap="viridis", scale="clip",
        vmin=None, vmax=None, contours=False, title=None, units=None):
    """
    One map of a megacube.

    Parameters
    ----------
    cube : Megacube or str
    name : str
        A map name from ``cube.names()``: ``"vrot_star"``, ``"Sigma_star"``,
        ``"Av"``, ``"Mage_L"``, ``"xyy_light"``, ``"Halpha_6563  flux"``,
        ``"BPT [N II]"`` ...
    path : str, optional
    ax : matplotlib.axes.Axes, optional
    cmap : str
        Any matplotlib colour map.
    scale : {"clip", "full", "log"}
        Linear between the 1st and 99th percentile, linear over the full
        range, or log10.
    vmin, vmax : float, optional
        Fixed colour limits (override `scale`'s).
    contours : bool
        Overlay contours of the map itself.
    title, units : str, optional

    Returns
    -------
    matplotlib.figure.Figure
    """
    import matplotlib
    from matplotlib import colors as mcolors
    from brainsp.megacube import CLASS_STYLES, SHORT
    c = _cube(cube)
    data = c.plane(name)
    if ax is None:
        fig = _figure((4.8, 4.2))
        ax = fig.add_axes([0.12, 0.11, 0.70, 0.80])
        cax = fig.add_axes([0.85, 0.11, 0.035, 0.80])
    else:
        fig, cax = ax.figure, None
    ax.set_facecolor("#d9d9d9")
    n_y, n_x = data.shape
    extent = (-0.5, n_x - 0.5, -0.5, n_y - 0.5)
    if name in CLASS_STYLES:
        names, colours = CLASS_STYLES[name]
        cm = mcolors.ListedColormap(colours)
        norm = mcolors.BoundaryNorm(np.arange(0.5, len(colours) + 1), cm.N)
        im = ax.imshow(data, origin="lower", cmap=cm, norm=norm,
                       extent=extent, interpolation="nearest")
        if cax is not None:
            bar = fig.colorbar(im, cax=cax, ticks=range(1, len(colours) + 1))
            bar.ax.set_yticklabels([SHORT.get(n, n) for n in names])
    else:
        shown, lo, hi = _limits(data, scale)
        cm = matplotlib.colormaps[cmap].copy()
        cm.set_bad("#d9d9d9")
        im = ax.imshow(shown, origin="lower", cmap=cm,
                       vmin=lo if vmin is None else vmin,
                       vmax=hi if vmax is None else vmax,
                       extent=extent, interpolation="nearest")
        if contours and np.isfinite(shown).any():
            ax.contour(shown, levels=np.linspace(lo, hi, 8)[1:-1],
                       colors=INK, linewidths=0.5, alpha=0.7, extent=extent)
        if cax is not None:
            bar = fig.colorbar(im, cax=cax)
            label = units if units is not None else (
                "log10" if scale == "log" else "")
            bar.set_label(label, fontsize=8)
    unfitted = ~np.isfinite(data)
    if np.any(unfitted):
        ax.contourf(np.arange(n_x), np.arange(n_y), unfitted.astype(float),
                    levels=[0.5, 1.5], colors="none", hatches=["///"])
    ax.set_xlabel("x [spaxel]")
    ax.set_ylabel("y [spaxel]")
    ax.set_title(title or name, fontsize=9, loc="left")
    ax.tick_params(labelsize=8)
    return _save(fig, path)

def maps(cube, names, path=None, *, ncols=3, cmap="viridis", scale="clip"):
    """
    Several maps of one megacube on a single page.

    Parameters
    ----------
    cube : Megacube or str
    names : list of str
    path : str, optional
    ncols : int
    cmap, scale : as for :func:`map`

    Returns
    -------
    matplotlib.figure.Figure
    """
    c = _cube(cube)
    nrows = int(np.ceil(len(names) / ncols))
    fig = _figure((3.6 * ncols, 3.3 * nrows))
    for i, name in enumerate(names):
        ax = fig.add_subplot(nrows, ncols, i + 1)
        map(c, name, ax=ax, cmap=cmap, scale=scale)
        im = ax.images[-1]
        fig.colorbar(im, ax=ax, fraction=0.046, pad=0.03)
    fig.tight_layout()
    return _save(fig, path)

def export_maps(cube, folder, names=None, *, formats=("png", "fits"),
                cmap="viridis", scale="clip"):
    """
    Write maps to files, one per map and format.

    Images (``png``, ``pdf``) are drawn for print. ``fits`` writes the map
    with the cube's sky coordinates and a ``TRUST`` extension whose bits say
    why a spaxel is not a measurement (not fitted, no value), so the file
    can be read without the megacube.

    Parameters
    ----------
    cube : Megacube or str
    folder : str
        Created if needed.
    names : list of str, optional
        Maps to write; every map in the megacube by default.
    formats : tuple of str
        Any of ``"png"``, ``"pdf"``, ``"svg"``, ``"fits"``.
    cmap, scale : as for :func:`map`

    Returns
    -------
    list of str
        The files written.
    """
    from brainsp import ifu_output
    c = _cube(cube)
    os.makedirs(folder, exist_ok=True)
    names = list(names or c.names())
    header = c.h["FLUX"].header if "FLUX" in c.h else c.h["FLXOBS"].header
    wcs = ifu_output.spatial_wcs(header)
    written = []
    for name in names:
        values = c.plane(name)
        mask = ifu_output.trust_mask(values, converged=c.status != 0)
        safe = "".join(ch if ch.isalnum() or ch in "._-" else "_"
                       for ch in name).strip("_")
        for fmt in formats:
            path = os.path.join(folder, f"{safe}.{fmt}")
            if fmt == "fits":
                ifu_output.write_plane_fits(
                    path, values, mask, plane=name, cube=c.path, wcs=wcs,
                    settings={"CMAP": (cmap, "colour map of the images"),
                              "CSCALE": (scale, "image colour scaling")})
            else:
                fig = map(c, name, cmap=cmap, scale=scale)
                _save(fig, path)
            written.append(path)
    return written

def spaxel(cube, y, x, path=None, *, xlim=None, lines=True):
    """
    One spaxel of a megacube: the spectrum, the model and each line.

    Parameters
    ----------
    cube : Megacube or str
    y, x : int
    path : str, optional
    xlim : (float, float), optional
    lines : bool
        Draw each fitted emission line's profile.

    Returns
    -------
    matplotlib.figure.Figure
    """
    c = _cube(cube)
    obs, syn, profiles = c.spectrum(int(y), int(x))
    fig = _figure((9, 3.6))
    ax = fig.add_subplot(111)
    _style(ax)
    ax.plot(c.wave, obs, color=INK, lw=0.7, label="observed")
    ax.plot(c.wave, syn, color=FIT, lw=1.0, label="model")
    if lines:
        for k, (name, profile) in enumerate(profiles):
            ax.plot(c.wave, profile, color=GAS, lw=0.7,
                    label="emission lines" if k == 0 else None)
    if xlim:
        ax.set_xlim(*xlim)
    ax.set_xlabel("rest wavelength [Å]")
    ax.set_ylabel(f"flux {c.units}".strip())
    ax.set_title(f"spaxel y = {y}, x = {x}", fontsize=9, loc="left")
    ax.legend(frameon=False, fontsize=8)
    return _save(fig, path)

def diagnostic(cube, kind="BPT [N II]", path=None, *, ax=None):
    """
    An emission-line diagnostic diagram, every spaxel coloured by class.

    Parameters
    ----------
    cube : Megacube or str
        Fitted in a line mode (``espop``, ``nepop`` or ``photopop``).
    kind : {"BPT [N II]", "BPT [S II]", "WHAN"}
        Baldwin, Phillips & Terlevich (1981) with the Kewley et al. (2001),
        Kauffmann et al. (2003) and Cid Fernandes et al. (2010) lines; the
        [S II] diagram with Kewley et al. (2006); or the WHAN diagram of Cid
        Fernandes et al. (2011).
    path : str, optional
    ax : matplotlib.axes.Axes, optional

    Returns
    -------
    matplotlib.figure.Figure
    """
    from brainsp.megacube import (CLASS_STYLES, kewley01, kauffmann03,
                                  cid10)
    c = _cube(cube)
    n2, o3, s2, ew = c.diagnostic_inputs()
    classes = c._diagnostic(kind)
    if classes is None:
        raise ValueError(f"{kind} needs emission lines this megacube does "
                         f"not have: fit with nebular_mode='espop', and do "
                         f"not mask the lines it uses (H-beta, [O III], "
                         f"H-alpha, [N II], [S II])")
    if ax is None:
        fig = _figure((4.6, 4.2))
        ax = fig.add_subplot(111)
    else:
        fig = ax.figure
    _style(ax)
    names, colours = CLASS_STYLES[kind]
    if kind == "BPT [N II]":
        xs, ys = n2, o3
        grid = np.linspace(-2.0, 0.04, 200)
        ax.plot(grid, kauffmann03(grid), color=DIM, lw=0.8, ls="--")
        grid = np.linspace(-2.0, 0.46, 200)
        ax.plot(grid, kewley01(grid), color=INK, lw=0.9)
        grid = np.linspace(-0.2, 1.0, 50)
        ax.plot(grid, cid10(grid), color=DIM, lw=0.8, ls=":")
        ax.set_xlabel("log [N II]6583 / Hα")
        ax.set_ylabel("log [O III]5007 / Hβ")
    elif kind == "BPT [S II]":
        xs, ys = s2, o3
        grid = np.linspace(-2.0, 0.31, 200)
        ax.plot(grid, 0.72 / (grid - 0.32) + 1.30, color=INK, lw=0.9)
        grid = np.linspace(-0.31, 1.0, 50)
        ax.plot(grid, 1.89 * grid + 0.76, color=DIM, lw=0.8, ls=":")
        ax.set_xlabel("log [S II]6716+6731 / Hα")
        ax.set_ylabel("log [O III]5007 / Hβ")
    else:
        xs, ys = n2, np.log10(np.where(ew > 0, ew, np.nan))
        ax.axvline(-0.4, color=INK, lw=0.8)
        ax.axhline(np.log10(3), color=DIM, lw=0.8, ls="--")
        ax.axhline(np.log10(6), color=DIM, lw=0.8, ls=":")
        ax.set_xlabel("log [N II]6583 / Hα")
        ax.set_ylabel("log EW(Hα) [Å]")
    for k, (label, colour) in enumerate(zip(names, colours), start=1):
        sel = classes == k
        if np.any(sel):
            ax.scatter(xs[sel], ys[sel], s=6, color=colour, label=label,
                       alpha=0.8, linewidths=0)
    # The usual window of each diagram, widened only to take in the data;
    # the dividing curves run to infinity at their asymptotes and would
    # otherwise set the scale.
    xr, yr = ((-1.5, 0.8), (-1.2, 1.5)) if kind != "WHAN" \
        else ((-1.5, 0.8), (-1.0, 2.5))
    good = np.isfinite(xs) & np.isfinite(ys) & np.isfinite(classes)
    if np.any(good):
        xq = np.percentile(xs[good], [1, 99])
        yq = np.percentile(ys[good], [1, 99])
        xr = (min(xr[0], xq[0] - 0.1), max(xr[1], xq[1] + 0.1))
        yr = (min(yr[0], yq[0] - 0.1), max(yr[1], yq[1] + 0.1))
    ax.set_xlim(*xr)
    ax.set_ylim(*yr)
    ax.legend(frameon=False, fontsize=7, markerscale=2)
    ax.set_title(kind, fontsize=9, loc="left")
    return _save(fig, path)
