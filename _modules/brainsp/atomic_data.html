"""
The Balmer wavelengths this module needs as scalars, in VACUUM Angstrom.

Read from line_catalogue rather than written out again here. There were
three tables of hydrogen wavelengths in this tree: air values to two
decimals here, whole-Angstrom labels in the emission-line catalogue, and
whole-Angstrom values again in METAL_LINE_CALIBRATIONS. The predicted Case
B lines were therefore placed on a different scale from the lines they are
printed beside and compared against. A second copy of a wavelength is a
second thing to keep in step; BRAIN has already been bitten by four copies
of the [N II] branching ratio.

These five are used for exactly two things: evaluating the extinction curve
at H alpha and H beta, and computing NU_HALPHA and NU_HBETA for the Case B
line energies. Neither is sensitive to the scale. k(lambda) for CCM changes
by 2.8e-4 between the air and vacuum H alpha, which multiplies A_V in a
magnitude exponent, and the photon energy changes by 2.8e-4 relative. The
tables that DO care which scale they are on -- the ones whose wavelengths
are compared against fitted lines, or printed beside them -- take the scale
as an argument and do not read these constants.

LAMBDA_LYMAN_ALPHA stays a literal because it is below the 2000 A floor
where an air wavelength does not exist, so vacuum is the only scale it has.
"""

def alpha_B_powerlaw(Te):
    """Power-law approximation for alpha_B (Osterbrock & Ferland 2006)."""
    t4 = Te / 1e4
    exponent = -0.833 - 0.034 * np.log(t4)
    return 2.59e-13 * t4**exponent

def hydrogen_case_b_ratios(Te):
    """
    Case B hydrogen intensities relative to H beta, with no wavelengths.

    {species: ratio}, Balmer and Paschen together, interpolated in log Te
    and clipped to the ends of the tabulated grid.

    Separate from balmer_ratios and all_hydrogen_ratios because a caller
    that wants only the intrinsic Balmer decrement -- which is what the
    nebular extinction is derived from -- has no wavelength to place and
    therefore no wavelength scale to declare. Making it ask for one would be
    demanding an answer that cannot change its result.
    """
    log_Te = np.log10(np.clip(Te, _balmer_ratio_Te[0], _balmer_ratio_Te[-1]))
    log_Te_grid = np.log10(_balmer_ratio_Te)
    result = {}
    for table in (_balmer_ratios, _paschen_ratios):
        for name, ratios in table.items():
            result[name] = np.interp(log_Te, log_Te_grid, ratios)
    return result

def balmer_ratios(Te, wavelength_scale):
    """
    Case B Balmer line ratios relative to H beta at a given Te.

    Parameters
    ----------
    Te : float
        Electron temperature [K].
    wavelength_scale : {'air', 'vacuum'}
        The scale the returned wavelengths are on. REQUIRED. These
        wavelengths are printed beside the fitted line fluxes they are meant
        to be compared with, so a predicted line on the other scale from the
        fitted one sits 83 km/s away from it with nothing in the output to
        say why.

    Returns
    -------
    ratios : dict {species: (wavelength, ratio)}
    """
    wavelengths = hydrogen_lines_on_scale(wavelength_scale)
    ratios = hydrogen_case_b_ratios(Te)
    return {name: (wavelengths[name], ratios[name]) for name in _balmer_ratios}

def qh_from_spectrum(wavelength, flux, ssp_flux_unit='Lsun'):
    """
    Ionising photon rates of an SSP, integrated from its own spectrum.

        Q = integral_0^lambda_edge  L_lambda * lambda / (h c)  d lambda

    evaluated at the three ionisation edges, which is what sets how many
    photons are available to ionise H, He and He+ respectively.

    This replaces the age/metallicity lookup BRAIN used previously. A lookup
    cannot know what is actually in the library it is being applied to: it
    silently returns a number for an SSP whose real ionising output is
    different, and there is no way for the fit to notice.

    Parameters
    ----------
    wavelength : 1D array
        SSP wavelength grid in Angstrom. Must extend below 911.76 A, or there
        is no ionising continuum to integrate and the result is meaningless.
    flux : 1D array
        L_lambda per unit initial mass, on the same grid.
    ssp_flux_unit : {'Lsun', 'cgs'}
        Units of `flux`: L_sun/A/Msun or erg/s/A/Msun.

    Returns
    -------
    dict
        {'QHc', 'QHe1', 'QHe2'} in photons/s per Msun formed. Entries whose
        edge lies outside the library's coverage are absent, not zero — a
        caller must not be able to mistake "not measured" for "no photons".

    Notes
    -----
    The edge bin is closed by interpolating the flux onto the edge exactly
    rather than truncating at the last grid point, which otherwise biases the
    result by the width of one bin. The integrand is interpolated
    logarithmically, because the Lyman continuum falls by orders of magnitude
    across a few Angstrom and a linear rule over the native grid overestimates
    it — by 2% at the He II edge on the BC03 grid.
    """
    wavelength = np.asarray(wavelength, dtype=np.float64)
    flux = np.asarray(flux, dtype=np.float64)

    if ssp_flux_unit == 'Lsun':
        L = flux * L_SUN_ERG
    elif ssp_flux_unit == 'cgs':
        L = flux
    else:
        raise ValueError(f"Unknown ssp_flux_unit: {ssp_flux_unit!r}")

    order = np.argsort(wavelength)
    w, L = wavelength[order], L[order]

    out = {}
    for name, edge in IONISATION_EDGES.items():
        if w[0] >= edge:
            continue
        inside = w < edge
        if inside.sum() < 2:
            continue

        wi, Li = w[inside], L[inside]
        integrand = Li * wi

        j = np.searchsorted(w, edge)
        if 0 < j < len(w):
            L_edge = np.interp(edge, w[j - 1:j + 1], L[j - 1:j + 1])
            wi = np.append(wi, edge)
            integrand = np.append(integrand, L_edge * edge)

        pos = integrand > 0
        if pos.sum() >= 2:
            lw, li = np.log(wi[pos]), np.log(integrand[pos])
            fine = np.linspace(lw[0], np.log(edge), 4096)
            total = np.trapezoid(np.exp(np.interp(fine, lw, li)), np.exp(fine))
        else:
            total = np.trapezoid(integrand, wi)

        out[name] = float(total / HC_ERG_ANGSTROM)

    return out

def Zgas_to_logOH(Z_gas, Z_solar=0.0142):
    """Convert gas metallicity to 12+log(O/H)."""
    return 8.69 + np.log10(Z_gas / Z_solar)




def _oiii_5007_ratio(logOH):
    """[O III] 5007 / H-beta vs 12+log(O/H).
    Peaks at low metallicity, decreases at high Z due to cooling."""
    x = logOH - 8.0
    return 10.0**(0.7 - 0.5*x - 0.15*x**2)

def _branching(name):
    """
    The weak-over-strong branching ratio of a tied doublet, from the catalogue.

    Read from emission_lines.TIED_DOUBLETS rather than written out again,
    because it is the same atomic constant the design matrix is built with
    and a second copy is a second thing to keep in step. There were four
    values of the [N II] ratio in this tree before this: 0.32739 in the
    catalogue, 1/2.95 here, 1/3.06 in SPECTRAL_COMPLEXES, and 0.33 in the
    catalogue before that. Imported inside the function because these
    calibrations are read at import time by a module-level table and
    emission_lines is the heavier import of the two.
    """
    from brainsp.emission_lines import TIED_DOUBLETS

    ratios = [ratio for _, ratio in TIED_DOUBLETS[name]]
    return min(ratios) / max(ratios)


def _oiii_4959_ratio(logOH):
    """[O III] 4959 / H-beta, tied to 5007 by its branching ratio."""
    return _oiii_5007_ratio(logOH) * _branching('[O III]_Doublet')

def _nii_6583_ratio(logOH):
    """[N II] 6583 / H-beta vs 12+log(O/H).
    Increases with metallicity (nitrogen is secondary at high Z)."""
    x = logOH - 8.69
    return 10.0**(-0.3 + 1.1*x + 0.2*x**2)

def _nii_6548_ratio(logOH):
    """[N II] 6548 / H-beta, tied to 6583 by its branching ratio."""
    return _nii_6583_ratio(logOH) * _branching('[N II]_Doublet')

def _sii_6716_ratio(logOH):
    """[S II] 6716 / H-beta. Moderate metallicity dependence."""
    x = logOH - 8.69
    return 10.0**(-0.5 + 0.4*x)

def _sii_6731_ratio(logOH):
    """[S II] 6731 / H-beta. Similar to 6716, density-dependent ratio."""
    return _sii_6716_ratio(logOH) * 0.75

def _oii_3727_ratio(logOH):
    """[O II] 3726+3729 / H-beta. Strong at moderate Z."""
    x = logOH - 8.69
    return 10.0**(0.1 + 0.3*x - 0.2*x**2)

def _oi_6300_ratio(logOH):
    """[O I] 6300 / H-beta. Weak, increases with Z."""
    x = logOH - 8.69
    return 10.0**(-1.5 + 0.5*x)

def _neiii_3869_ratio(logOH):
    """[Ne III] 3869 / H-beta. Decreases with Z."""
    x = logOH - 8.69
    return 10.0**(-0.5 - 0.3*x)

"""
Semi-empirical metal line strengths relative to H beta, keyed by catalogue
name.

The whole-Angstrom wavelength that used to travel with each calibration is
gone. It was a third copy of the emission-line wavelengths, rounded harder
than either of the other two, and it placed the predicted lines on no scale
at all. metal_calibration_wavelengths supplies the wavelength from
line_catalogue on a declared scale instead.

'[O II]_3727' is not a catalogue entry. It is the historic name of the
UNRESOLVED 3726 + 3729 blend, and _oii_3727_ratio predicts the SUM of the
two, so the entry is kept under that name and the comparison against the
fit sums the two fitted components. Splitting the prediction by some fixed
3729/3726 ratio would be wrong: that ratio is the density diagnostic, which
is the reason the doublet is fitted as two untied lines in the first place.
"""

OII_BLEND_COMPONENTS = ('[O II]_3726', '[O II]_3729')
"""
The two catalogue entries whose fitted fluxes sum to the '[O II]_3727'
prediction.

Named here, where the prediction is defined, so that the consumer doing the
summing does not have to know which components the blend is made of. The
[O II] doublet used to be a single catalogue entry at 3727 and the
prediction matched it by name; once the catalogue split it, the predicted
name matched nothing fitted and [O II] fell silently out of the
predicted-versus-observed table.
"""

def metal_calibration_wavelengths(wavelength_scale):
    """
    {catalogue name: wavelength} for every metal line calibration, on a
    declared scale.

    Every key but '[O II]_3727' is a catalogue entry and takes that entry's
    wavelength. '[O II]_3727' names the unresolved blend and is placed at
    the mean of its two components, which is a label for a prediction that
    is itself a sum, not a claim about where either component sits.
    """
    catalogue = emission_lines_on_scale(wavelength_scale)
    wavelengths = {}
    for name in METAL_LINE_CALIBRATIONS:
        if name in catalogue:
            wavelengths[name] = catalogue[name]
        elif name == '[O II]_3727':
            wavelengths[name] = float(np.mean(
                [catalogue[c] for c in OII_BLEND_COMPONENTS]))
        else:
            raise KeyError(
                f"metal line calibration {name!r} names no catalogue entry "
                f"and has no wavelength. Add it to line_catalogue, or give "
                f"it a rule here saying what it is a blend of.")
    return wavelengths

def predicted_metal_ratios(Z_gas, wavelength_scale, Z_solar=0.0142):
    """
    Return semi-empirical metal line ratios relative to H-beta.

    Parameters
    ----------
    Z_gas : float
        Gas-phase metallicity (mass fraction).
    wavelength_scale : {'air', 'vacuum'}
        The scale the returned wavelengths are on. REQUIRED; see
        balmer_ratios for why there is no default.

    Returns
    -------
    ratios : dict {name: (wavelength, ratio_to_Hbeta)}
    """
    logOH = Zgas_to_logOH(Z_gas, Z_solar)
    wavelengths = metal_calibration_wavelengths(wavelength_scale)
    return {name: (wavelengths[name], func(logOH))
            for name, func in METAL_LINE_CALIBRATIONS.items()}

def two_photon_shape_HI(x):
    """
    Two-photon spectral shape function A(x) for H I.
    x = nu / nu_Lya (0 to 1).

    Returns A(x) in [s^-1], unnormalized.
    The total rate is integral_0^1 A(x) dx = A_2q = 8.2249 s^-1.
    """
    x = np.asarray(x, dtype=np.float64)
    y = x * (1.0 - x)

    y = np.clip(y, 0.0, 0.25)

    term1 = y * (1.0 - (4.0 * y)**_2q_gamma)
    term2 = _2q_delta * y**_2q_beta * (4.0 * y)**_2q_gamma

    A = _2q_C * (term1 + term2)
    A = np.where((x > 0) & (x < 1), A, 0.0)
    return A

def two_photon_shape_HeI(x):
    """
    Two-photon spectral shape for He I (1s2s ^1S_0 -> 1s^2 ^1S_0).
    x = nu / nu_2q_HeI (0 to 1), where nu_2q_HeI corresponds to 601.4 A.
    """
    x = np.asarray(x, dtype=np.float64)
    C_HeI = 145.29
    beta_HeI = -2.259
    gamma_HeI = -8.77

    A = C_HeI * (1.0 + beta_HeI * (x - 0.5)**2 + gamma_HeI * (x - 0.5)**4)
    A = np.where((x > 0) & (x < 1), A, 0.0)
    return np.maximum(A, 0.0)
