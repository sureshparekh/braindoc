"""
NewBrain Solver Module — JAX Backend
Bounded Variable Least Squares (BVLS) with Tikhonov regularization.
Handles the linear sub-problem: given fixed nonlinear params, solve for weights.

Uses a JIT-compiled FISTA (Fast Iterative Shrinkage-Thresholding) NNLS solver
for GPU-accelerated non-negative least squares.
"""

import jax
import jax.numpy as jnp
import numpy as np
from functools import partial

def build_regularization_matrix(n_stellar, reg_strength, order=1):
    """
    Build a Tikhonov regularization matrix (difference operator).

    Returns D of shape (n_stellar - order, n_stellar), or None.
    """
    if reg_strength <= 0 or n_stellar <= order:
        return None

    rows = jnp.arange(n_stellar - order)
    if order == 1:
        D = jnp.zeros((n_stellar - 1, n_stellar))
        D = D.at[rows, rows].set(-reg_strength)
        D = D.at[rows, rows + 1].set(reg_strength)
    elif order == 2:
        D = jnp.zeros((n_stellar - 2, n_stellar))
        D = D.at[rows, rows].set(reg_strength)
        D = D.at[rows, rows + 1].set(-2.0 * reg_strength)
        D = D.at[rows, rows + 2].set(reg_strength)
    else:
        raise ValueError(f"Unsupported regularization order: {order}")
    return D




@partial(jax.jit, static_argnames=('max_iter',))
def _fista_nnls(ATA, ATb, max_iter=300, tol=1e-8):
    """
    FISTA-accelerated projected gradient descent for NNLS.

    Solves: min_{x >= 0} 0.5 * x^T ATA x - ATb^T x
    (equivalent to min ||Ax - b||^2 when ATA = A^T A, ATb = A^T b)

    JIT-compiled with early convergence termination via jax.lax.while_loop.

    Parameters
    ----------
    ATA : (n, n)  Gram matrix A^T A (precomputed by caller).
    ATb : (n,)    A^T b (precomputed by caller).
    max_iter : int  Static — controls compilation shape only.
    tol : float     Relative convergence tolerance.

    Returns
    -------
    x : (n,) non-negative solution.
    """
    n = ATb.shape[0]
    x0 = jnp.zeros(n)
    y0 = x0
    t0 = 1.0

    L = jnp.linalg.norm(ATA, ord='fro')
    step = 1.0 / jnp.maximum(L, 1e-30)

    init = (x0, y0, t0, 0, False)

    def cond(state):
        _, _, _, i, converged = state
        return (~converged) & (i < max_iter)

    def body(state):
        x, y, t, i, _ = state

        grad = ATA @ y - ATb
        x_new = jnp.maximum(y - step * grad, 0.0)

        t_new = 0.5 * (1.0 + jnp.sqrt(1.0 + 4.0 * t * t))
        beta = (t - 1.0) / t_new
        y_new = jnp.maximum(x_new + beta * (x_new - x), 0.0)

        dx = jnp.sum((x_new - x) ** 2)
        xnorm = jnp.sum(x_new ** 2) + 1e-30
        converged = (dx / xnorm < tol * tol) & (i > 5)

        return (x_new, y_new, t_new, i + 1, converged)

    x, _, _, _, _ = jax.lax.while_loop(cond, body, init)
    return x


@jax.jit
def _solve_core(M_full, flux, inv_sigma, D_full, extra_rows, extra_rhs):
    """
    JIT-compiled core: assemble augmented system, solve NNLS, compute chi2.

    All arguments are concrete JAX arrays (no None — use dummy + flag instead).
    D_full and extra_rows may be (0, n) when unused; vstack is a no-op
    for 0-row arrays, so no conditional branching is needed.
    """
    W = inv_sigma[:, jnp.newaxis]
    A = M_full * W
    b = flux * inv_sigma

    A = jnp.vstack([A, D_full, extra_rows])
    b = jnp.concatenate([b, jnp.zeros(D_full.shape[0]), extra_rhs])

    ATA = A.T @ A
    ATb = A.T @ b
    weights = _fista_nnls(ATA, ATb)

    fitted = M_full @ weights
    residuals = (flux - fitted) * inv_sigma
    chi2 = jnp.sum(residuals ** 2)

    return weights, fitted, chi2

def solve_nnls_regularized(M_stellar, M_gas, flux, inv_sigma,
                           reg_strength=1.0, reg_order=1,
                           extra_rows=None, extra_rhs=None):
    """
    NNLS with Tikhonov regularization and optional SC penalty rows.

    This is a thin wrapper that prepares the inputs for the JIT-compiled
    _solve_core.  The JIT boundary is inside _solve_core so that this
    function can handle Python-level None checks and shape logic.

    Parameters / Returns — same as before.
    """
    M_stellar = jnp.asarray(M_stellar)
    flux = jnp.asarray(flux)
    inv_sigma = jnp.asarray(inv_sigma)

    n_star = M_stellar.shape[1]
    n_gas = 0
    if M_gas is not None:
        M_gas = jnp.asarray(M_gas)
        if M_gas.shape[1] > 0:
            n_gas = M_gas.shape[1]

    if n_gas > 0:
        M_full = jnp.hstack([M_stellar, M_gas])
    else:
        M_full = M_stellar

    n_total = M_full.shape[1]

    D = build_regularization_matrix(n_star, reg_strength, reg_order)
    has_D = D is not None
    if has_D:
        D_full = jnp.zeros((D.shape[0], n_total))
        D_full = D_full.at[:, :n_star].set(D)
    else:
        D_full = jnp.zeros((0, n_total))

    has_extra = extra_rows is not None and extra_rhs is not None
    if has_extra:
        extra_rows = jnp.asarray(extra_rows)
        extra_rhs = jnp.asarray(extra_rhs)
    else:
        extra_rows = jnp.zeros((0, n_total))
        extra_rhs = jnp.zeros(0)

    weights, fitted, chi2 = _solve_core(
        M_full, flux, inv_sigma, D_full, extra_rows, extra_rhs,
    )

    return weights, fitted, chi2

@partial(jax.jit, static_argnames=('max_outer','max_inner','simplex'))
def solve_active_set(ATA, ATb, cvec, P_init=None, max_outer=120, max_inner=15,
                      tol=1e-10, rank_tol=1e-12, simplex=True):
    """
    Active-set least squares with x >= 0 and optionally  c.x = 1.

    KKT on the passive set P:
        [ A_P^T A_P   c_P ] [ x_P ]   [ A_P^T b ]
        [   c_P^T      0  ] [ mu  ] = [    1    ]

    The multiplier matters for more than the solve: the rule for admitting a
    new column is  w_j > mu, not  w_j > 0. Using the unconstrained test makes
    the passive set collapse to a single column and the fit is ruined.

    `P_init` warm-starts the passive set. A set carried over from a
    neighbouring evaluation is only a guess: the KKT solve on it can return
    negative components, and such a set is not a legal starting state. It is
    therefore purged first — columns whose KKT component is non-positive are
    dropped and the system re-solved, until every passive component is
    strictly positive. That restores the invariant the outer admission loop
    depends on: x is the exact least-squares solution on its own support and
    positive there. Only then is warm starting exact — it changes the path and
    not the answer, measured max|x_warm - x_cold| = 0 exactly over 80 random
    feasible P_init, with and without the simplex row.

    Without the purge the clipped vector is not a KKT point at all, no
    improving column exists, and the outer loop returns it immediately:
    passive max|gradient| 3.6e+04 against 1.5e-11 for the cold solve.

    The warm start earns its keep — perturbed photopop-mode solves drop from 20
    outer iterations to 8 — but it earns nothing unless the purge runs.

    Note the contrast with warm-starting an *iterative* solver, which changes
    the map being evaluated and silently breaks the Jacobian. Warm-starting is
    safe here only because this solver reaches the same vertex wherever it
    starts.

    A column of the design matrix that is identically — or numerically —
    zero over the fitted pixels carries no information, and until the
    `usable` mask below it was handled only by accident. Three accidents,
    in fact: the off-passive diagonal `scl*eye` kept H invertible, the
    admission test happened to be a strict inequality against a weight of
    exactly 0, and under the simplex the gas block's `cvec` entry is 0 so
    the multiplier never multiplied it. Each of the three breaks on its own
    input, and all three break silently:

    - Cold start. `argmax(ATb)` lands on the dead column whenever every
      live column has ATb_j < 0, because the dead column's ATb_j is exactly
      0 and therefore the largest. The first KKT system is then singular
      and `jnp.linalg.solve` returns NaN without raising — NumPy raises
      LinAlgError on the same matrix, JAX does not.
    - Warm start. A `P_init` containing the dead column cannot be purged:
      the purge drops columns with `sp <= 0`, `NaN <= 0` is False, so the
      purge declares the set legal on its first pass and the NaN survives
      into the answer with an empty passive set.
    - Simplex. In the constrained block `cvec_j` is 1, the bordered row
      rescues K from singularity, and the solve returns finite garbage: on
      a 5-column test system with one dead column, scaling ATb by 0.1 puts
      50% of the light on the dead column, by 0.01 puts 95% on it, and mu
      is pinned to ~0, which degrades the admission rule to the
      unconstrained `w_j > 0` that ruins the fit as described above. Which
      scalings break is not predictable from the matrix.

    `usable` closes all three. A column is usable when its diagonal of ATA
    — the squared norm of that column, since ATA = A^T A — is finite and
    above `rank_tol` times the mean diagonal. Both sides of that test are
    quadratic in A, so unlike the admission test below it is invariant
    under rescaling the design matrix. An unusable column is excluded from
    the cold start, from any `P_init`, and from admission, so it can never
    enter the passive set, can never make the KKT system singular, and
    comes back with weight exactly 0. On the reference fit the smallest
    diagonal is 0.657 of the mean and `rank_tol` is 1e-12, so nothing is
    anywhere near the test and the answer is bit-for-bit what it was. A
    narrow emission-line column is the weakest thing the design matrix
    legitimately holds, and in an espop fit of the same spectrum the
    weakest of the 14 of them is 4.8e-06 of the mean — still six orders of
    magnitude clear of the test.

    Weight 0 is the right answer for a dead column in the unconstrained
    problem — it contributes nothing, so it earns nothing — and no warning
    is raised for it there. Under the simplex it is not: the weights are
    then fractions of a whole, and a dead column inside the constrained
    block means the whole itself is not what the caller thinks it is. A
    zero-filled stellar template is the case on record, and every fraction
    reported would be a fraction of a library that is partly not there.
    So the solver refuses instead of solving around it: if `simplex` and
    any unusable column carries a non-zero `cvec`, x comes back all-NaN.
    It also refuses when any diagonal is non-finite (in either mode, which
    is what already happened, now by decision rather than by propagation),
    and when no column at all is usable. NaN reaches chi2 and the summary;
    it is not a plausible fit and cannot be mistaken for one. Fixing it
    means fixing the design matrix, which is where the bug is.

    The tolerances are scale-dependent and are left that way deliberately,
    since changing them would move the reference fit:

    - Admission, `wf[j] > tol*scl`. The left side w = ATb - ATA x is
      linear in b; the right side tol*trace(ATA)/n is quadratic in A. The
      test is therefore not invariant under rescaling b alone. Measured on
      the reference system: scaling ATb by 1e-2 and 1e-4 changes the answer
      by 8.8e-12 and 6.6e-13 relative, 1e-6 returns 3 non-zero populations
      instead of 4 with weights wrong by 36%, and 1e-8 returns 2, wrong by
      127%. The symptom of crossing that line is fewer and older
      components — the exact signature of an unconverged iterative solver,
      arriving from an exact one. BRAIN normalises the flux at
      `norm_wavelength`, which leaves four to five orders of headroom, so
      nothing in the pipeline is near it today. Anything that feeds this
      solver an un-normalised system is.
    - The drop test `xi > tol*1e-3` is a bare absolute 1e-13 with no scale
      reference at all.

    Returns (x, P) so the caller can pass P back in.
    """
    n = ATb.shape[0]; eye = jnp.eye(n)
    scl = jnp.maximum(jnp.trace(ATA)/n, 1e-30)

    diagonal = jnp.diag(ATA)
    usable = jnp.isfinite(diagonal) & (diagonal > rank_tol*scl)
    dead = ~usable
    refuse = (~jnp.all(jnp.isfinite(diagonal))) | (~jnp.any(usable))
    if simplex:
        refuse = refuse | jnp.any(dead & (cvec != 0.0))

    def kkt(P):
        H = jnp.where(P[:, None] & P[None, :], ATA, scl*eye)
        rhs = jnp.where(P, ATb, 0.0)
        if not simplex:
            return jnp.linalg.solve(H, rhs), jnp.zeros(())
        c = jnp.where(P, cvec, 0.0)
        K = jnp.concatenate([
            jnp.concatenate([H, c[:, None]], axis=1),
            jnp.concatenate([c, jnp.zeros(1)])[None, :]], axis=0)
        K = K.at[n, n].set(-1e-12*scl)
        sol = jnp.linalg.solve(K, jnp.concatenate([rhs, jnp.ones(1)]))
        return sol[:n], sol[n]

    def outer(st):
        x, P, mu, i, _ = st
        w = ATb - ATA @ x
        thr = mu*cvec if simplex else jnp.zeros(n)
        wf = jnp.where(P | dead, -jnp.inf, w - thr)
        j = jnp.argmax(wf)
        can = (~P[j]) & (wf[j] > tol*scl)
        P_new = jnp.where(can, P.at[j].set(True), P)

        def inner(ist):
            _, xx, PP, mm, k, _ = ist
            s_full, m_new = kkt(PP)
            s = jnp.where(PP, s_full, 0.0)
            neg = PP & (s <= 0.0); ok = ~jnp.any(neg)
            ratio = jnp.where(neg, xx/jnp.maximum(xx - s, 1e-30), jnp.inf)
            alpha = jnp.minimum(jnp.min(ratio), 1.0)
            xi = jnp.maximum(jnp.where(ok, s, xx + alpha*(s - xx)), 0.0)
            PPn = jnp.where(ok, PP, PP & (xi > tol*1e-3))
            return (s, xi, PPn, m_new, k+1, ok)

        _, xn, Po, mo, _, _ = jax.lax.while_loop(
            lambda t: (~t[5]) & (t[4] < max_inner), inner,
            (jnp.zeros(n), x, P_new, mu, 0, False))
        done = ~can
        return (jnp.where(done, x, xn), jnp.where(done, P, Po),
                jnp.where(done, mu, mo), i+1, done)

    cold = jnp.zeros(n, dtype=bool).at[
        jnp.argmax(jnp.where(usable, ATb, -jnp.inf))].set(True)
    if P_init is None:
        P0 = cold
    else:
        P_init = P_init & usable
        def purge(st):
            P, k, _ = st
            sp, _m = kkt(P)
            sp = jnp.where(P, sp, 0.0)
            ok = ~jnp.any(P & (sp <= 0.0))
            return (jnp.where(ok, P, P & (sp > 0.0)), k + 1, ok)

        P0 = jax.lax.while_loop(
            lambda st: (~st[2]) & (st[1] < n) & jnp.any(st[0]), purge,
            (jnp.where(jnp.any(P_init), P_init, cold), 0, False))[0]
        P0 = jnp.where(jnp.any(P0), P0, cold)

    s0, m0 = kkt(P0)
    x0 = jnp.maximum(jnp.where(P0, s0, 0.0), 0.0)
    out = jax.lax.while_loop(lambda st: (~st[4]) & (st[3] < max_outer),
                             outer, (x0, P0, m0, 0, False))
    x = jnp.where(usable, out[0], 0.0)
    return jnp.where(refuse, jnp.nan, x), out[1]

def compute_residual_vector(weights, M_stellar, M_gas, flux, inv_sigma,
                            reg_strength=1.0, reg_order=1,
                            extra_rows=None, extra_rhs=None):
    """
    Compute full residual vector (data + regularization + SC penalty).
    """
    M_stellar = jnp.asarray(M_stellar)
    weights = jnp.asarray(weights)
    flux = jnp.asarray(flux)
    inv_sigma = jnp.asarray(inv_sigma)

    n_star = M_stellar.shape[1]
    n_gas = 0
    if M_gas is not None:
        M_gas = jnp.asarray(M_gas)
        if M_gas.shape[1] > 0:
            n_gas = M_gas.shape[1]

    if n_gas > 0:
        M_full = jnp.hstack([M_stellar, M_gas])
    else:
        M_full = M_stellar

    model = M_full @ weights
    data_resid = (flux - model) * inv_sigma

    parts = [data_resid]

    D = build_regularization_matrix(n_star, reg_strength, reg_order)
    if D is not None:
        parts.append(D @ weights[:n_star])

    if extra_rows is not None and extra_rhs is not None:
        extra_rows = jnp.asarray(extra_rows)
        extra_rhs = jnp.asarray(extra_rhs)
        parts.append(extra_rows @ weights - extra_rhs)

    return jnp.concatenate(parts)
