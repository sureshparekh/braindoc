"""
NewBrain — Next-Generation Stellar Population Fitting Code

Direct Pixel-Fitting spectral synthesis engine.
Pure physical model: No multiplicative or additive polynomials.
All core operations in Fourier/matrix space → GPU-ready.

Supports three photoionization modes:
  "spop"       — Standard stellar-only fitting with empirical Gaussian emission lines.
  "nepop" — Nebular continuum from Q(H) included; emission lines still empirical.
  "photopop"      — Self-consistent: nebular continuum + predicted H lines from Q(H).

Main pipeline driver.

Author: Suresh Parekh
"""

def _gas_kinematics(cfg, fitter, result, n_ssp, lsf_info):
    """
    The per-group gas kinematics, with each group's claim to be quotable.

    Assembled here rather than in the optimiser because the judgement needs
    the fitted line fluxes, which the optimiser returns alongside the
    kinematics but does not interpret, and the instrumental line-spread
    function, which it never sees at all.

    Each gas column's wavelength is the ratio-weighted mean of its components,
    which for a single line is the line and for a tied doublet is a point
    between its two members. It is used only to look the instrument up, and
    the instrumental dispersion varies by a few km/s across a doublet's
    separation, so the distinction between this and either component is far
    below anything it is compared against.

    Returns [] when the fit has no emission lines in it, which is what makes
    the caller's summary omit the block entirely rather than write an empty
    one.
    """
    if not cfg.fit_emission_lines or not fitter._line_table['names']:
        return []

    from brainsp.gas_diagnostics import gas_kinematics_report

    table = fitter._line_table
    rest = np.asarray(table['rest_wavelengths'], dtype=np.float64)
    ratios = np.asarray(table['ratios'], dtype=np.float64)
    present = (rest > 0) & (ratios > 0)
    weight = np.where(present, ratios, 0.0)
    total = np.sum(weight, axis=1)
    wavelengths = np.where(
        total > 0, np.sum(weight * rest, axis=1) / np.maximum(total, 1e-30),
        np.max(rest, axis=1))

    names = list(table['names'])
    fluxes = np.asarray(result['weights'])[n_ssp:n_ssp + len(names)]

    return gas_kinematics_report(
        fitter._group_labels, result.get('group_kinematics', {}),
        names, np.asarray(table['group_indices'], dtype=int), fluxes,
        wavelengths, cfg.sigma_gas_bounds[0], cfg.sigma_gas_bounds[1],
        lsf_info=lsf_info)


def _log_gas_kinematics(gas_report, indent):
    """
    The same refusal in the log as in the summary file, and the same words.

    A bound printed in the log as a plain number is quoted as readily as one
    printed in the summary, and the log is what a user watching a fit run
    actually reads.

    A group that is quotable but built on fewer lines than it has carries a
    note rather than a warning: it is a description of the fit and not a
    reason to distrust the number, and a warning that fires on a healthy fit
    teaches the reader to skip warnings.
    """
    from brainsp.gas_diagnostics import gas_dispersion_text, gas_velocity_text

    for entry in gas_report:
        label = entry['label']
        note = "" if entry['quotable'] else f" ({entry['status']})"
        logger.info(f"{indent}σ_gas[{label}] = "
                    f"{gas_dispersion_text(entry)} km/s, OBSERVED{note}")
        logger.info(f"{indent}v0_gas[{label}] = "
                    f"{gas_velocity_text(entry)} km/s")
        if entry['reason']:
            say = logger.info if entry['quotable'] else logger.warning
            say(f"{indent}{entry['reason']}")


def _observed_balmer(gas_names, gas_weights, av, k_norm, reddening_law):
    """
    Halpha and Hbeta as they reach the detector, not as the solver stores them.

    The gas block of the design matrix already carries the stellar
    attenuation, so a fitted gas weight is a de-reddened amplitude and the
    flux actually observed is

        F_L = w_L * 10^(-0.4 * A_V * (k_L - k_norm)).

    A Balmer decrement is a statement about observed fluxes and has to be
    formed from F, not from w. Forming it from w subtracts A_V from every
    nebular extinction derived from it, exactly and every time, which is what
    made the outer self-consistency iteration walk by A_V per pass.
    """
    from brainsp.nebular import extract_line_flux
    from brainsp.atomic_data import LAMBDA_HALPHA, LAMBDA_HBETA
    from brainsp import reddening_laws
    w_Ha = extract_line_flux(gas_names, gas_weights, 'Halpha')
    w_Hb = extract_line_flux(gas_names, gas_weights, 'Hbeta')
    k = reddening_laws.get_k_lambda(
        np.array([LAMBDA_HALPHA, LAMBDA_HBETA]), reddening_law)
    return (w_Ha * 10.0 ** (-0.4 * av * (float(k[0]) - k_norm)),
            w_Hb * 10.0 ** (-0.4 * av * (float(k[1]) - k_norm)))


def _seek_av_neb(fitter, cfg, p0, budget):
    """
    Find the nebular extinction the data prefers, and say how well it is fixed.

    What this replaces was a fixed-point iteration: refit, read the Balmer
    decrement back out of the fitted hydrogen amplitudes, feed it in again.
    That could not converge, for a structural reason rather than a numerical
    one. In full mode the Case B penalty rows drive every hydrogen amplitude
    to the value implied by the Av_neb that was put in, so reading the
    decrement out returns the same number: the map is the identity and its
    residual carries no information about Av_neb at all. Measured on
    10517-9102_10_21 the local gain was 0.998 +- 0.004 across the whole of
    [0, 6] mag, so no damping, secant step or Anderson mixing could have
    rescued it either. Where it appeared to converge it had reached the
    Av_neb = 0 clip and stopped moving because it was pinned.

    chi2 is the only thing that carries information about Av_neb, so chi2 is
    what is minimised, by golden section over av_nebular_bounds. Every trial
    is refitted from the same warm start, so the objective is a function of
    Av_neb alone and the search is reproducible. The augmented chi2 is used
    rather than the data rows alone because the penalty block contributes
    under 0.03 of ~1600 here and does not move the minimum.

    The two bounds and `nebular_dust_start` are always evaluated, so a user's
    stated prior is among the candidates however the bracket happens to walk,
    and the width of chi2 across the whole range is always available to
    report.

    A minimum found sitting on a bound is not a measurement and is never
    reported as converged; `status` says which bound. `dchi2_range` and
    `one_sigma` are there so a reader can see how flat the objective is: on
    the example spectra chi2 varies by under 5 over the full 6 mag, so
    Av_neb is barely constrained even where the search succeeds.
    """
    lo = float(cfg.av_nebular_bounds[0])
    hi = float(cfg.av_nebular_bounds[1])
    tol = max(float(cfg.av_neb_tolerance), 1e-4)
    seen = {}

    def chi2_at(x):
        key = round(float(np.clip(x, lo, hi)), 6)
        if key not in seen:
            fitter.update_Av_neb(key)
            seen[key] = fitter.optimize(initial_guess=list(np.asarray(p0)))
        return float(seen[key]['chi2'])

    chi2_at(lo)
    chi2_at(hi)
    chi2_at(getattr(cfg, 'av_nebular_init', 0.5 * (lo + hi)))
    phi = 0.5 * (np.sqrt(5.0) - 1.0)
    a, b = lo, hi
    c, d = b - phi * (b - a), a + phi * (b - a)
    fc, fd = chi2_at(c), chi2_at(d)
    guard = 0
    while len(seen) < budget and (b - a) > tol and guard < 4 * budget:
        guard += 1
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - phi * (b - a)
            fc = chi2_at(c)
        else:
            a, c, fc = c, d, fd
            d = a + phi * (b - a)
            fd = chi2_at(d)

    xs = sorted(seen)
    chi2s = [float(seen[x]['chi2']) for x in xs]
    i_best = int(np.argmin(chi2s))
    x_best, chi2_best = xs[i_best], chi2s[i_best]

    one_sigma = np.nan
    for j in range(i_best + 1, len(xs)):
        if chi2s[j] - chi2_best >= 1.0:
            f = (1.0 - (chi2s[j - 1] - chi2_best)) / max(
                chi2s[j] - chi2s[j - 1], 1e-30)
            one_sigma = xs[j - 1] + f * (xs[j] - xs[j - 1])
            break

    if x_best <= lo + tol:
        status = 'LOWER_BOUND'
    elif x_best >= hi - tol:
        status = 'UPPER_BOUND'
    elif (b - a) <= tol:
        status = 'converged'
    else:
        status = 'BUDGET_EXHAUSTED'

    return {
        'av_neb': x_best,
        'status': status,
        'result': seen[x_best],
        'n_evals': len(seen),
        'bracket': b - a,
        'dchi2_range': max(chi2s) - chi2_best,
        'one_sigma': one_sigma,
    }


def _model_spectrum(result, flux_log, n_ssp):
    """
    The spectrum a fit result describes: stars, emission lines, nebular.

    Assembled in one place because the clipping loop needs the residual after
    every re-fit and the summary needs it once at the end, and two copies of
    this sum have already drifted apart once.
    """
    M_gas_dust = result['M_gas_dust']
    w = result['weights']
    n_gas = len(result['gas_names'])
    total = result['M_star_dust'] @ w[:n_ssp]
    if M_gas_dust is not None and n_gas > 0:
        total = total + M_gas_dust @ w[n_ssp:n_ssp + n_gas]
    return total + result.get('nebular_spectrum', jnp.zeros_like(flux_log))

def clip_outliers_and_refit(fitter, result, inv_sigma, flux_log, err_log,
                            n_ssp, cfg):
    """
    Zero the weight of pixels the model cannot explain, and fit again.

    A pass measures the deviation of every fitted pixel from the current
    model in units of its own error, zeroes the weight of everything beyond
    sigma_clip_threshold, and re-optimises from the parameters already
    reached. The re-fit is inside the loop on purpose: the criterion is a
    property of the residual, so a second pass that reuses the first pass's
    residual can only find the pixels it has already removed and is a no-op.
    That is what the loop used to be. cfg.n_clip_iterations is therefore a
    count of clip-and-refit passes, and one pass -- the default -- is one
    clip and one re-fit.

    The deviation is abs(resid) / err_log, the error of the pixel as the file
    reports it, not abs(resid) * inv_sigma. The two differ by the square root of
    the mask weight, and a mask weight is an instruction about how much a
    pixel should count, not a statement about how noisy it is; folding it in
    would make the threshold mean something different wherever a mask is
    loaded with a weight other than zero or one. Without errorbars there is
    no per-pixel scale and the residual's own scatter stands in for one.

    Returns the last result, the surviving weights, how many pixels went, and
    the time the re-fits took. Assigning to fitter.inv_sigma is what makes
    the re-fit see the clip; the property does the cache invalidation.
    """
    total_clipped = 0
    t_refit_total = 0.0
    total_fit = _model_spectrum(result, flux_log, n_ssp)
    resid = flux_log - total_fit

    for clip_iter in range(cfg.n_clip_iterations):
        if err_log is not None:
            nsigma = jnp.abs(resid) / jnp.where(err_log > 0, err_log, 1.0)
            # outlier_scale 'typical': in units of the larger of the error
            # and the typical residual (1.4826 MAD of residual/error), so a
            # systematic misfit at high S/N is not taken for outliers in
            # every pixel. 'error' (the default): the error alone, as
            # STARLIGHT's NSIGMA clipping.
            typical = 1.4826 * float(jnp.nanmedian(
                jnp.where(inv_sigma > 0, nsigma, jnp.nan)))
            if (getattr(cfg, "sigma_clip_scale", "error") == "typical"
                    and typical > 1.0):
                logger.info(f"   Residuals are {typical:.2f} times the errors; "
                            f"clipping against that scale.")
                nsigma = nsigma / typical
        else:
            good_mask = inv_sigma > 0
            rms = (float(jnp.std(jnp.where(good_mask, resid, 0.0)))
                   if jnp.any(good_mask) else 1.0)
            nsigma = jnp.abs(resid) / max(rms, 1e-30)

        outliers = (nsigma > cfg.sigma_clip_threshold) & (inv_sigma > 0)
        n_new = int(jnp.sum(outliers))
        if n_new == 0:
            if clip_iter == 0:
                logger.info("   No outliers found.")
            break

        total_clipped += n_new
        inv_sigma = jnp.where(outliers, 0.0, inv_sigma)
        n_left = int(jnp.sum(inv_sigma > 0))
        logger.info(f"   Pass {clip_iter + 1}: clipped {n_new} pixels, "
                     f"re-optimizing on the remaining {n_left}...")

        fitter.inv_sigma = inv_sigma
        t1 = time.time()
        result = fitter.optimize(initial_guess=result['params'])
        t_refit = time.time() - t1
        t_refit_total += t_refit

        resid = flux_log - _model_spectrum(result, flux_log, n_ssp)
        logger.info(f"   Refined χ² = {result['chi2']:.4f} ({t_refit:.2f}s)")

    return result, inv_sigma, total_clipped, t_refit_total

def run(cfg):
    """
    Execute the full NewBrain spectral fitting pipeline.

    Parameters
    ----------
    cfg : NewBrainConfig
        Complete configuration.
    """
    t0 = time.time()

    logger.info("=" * 70)
    logger.info("  NewBrain — Direct Pixel-Fitting Spectral Synthesis")
    if cfg.photoionization_mode != "spop":
        logger.info(f"  Photoionization mode: {cfg.photoionization_mode}")
    logger.info("=" * 70)


    wl_lin, flux_lin, err_lin, flag_lin = load_spectrum(
        cfg.obs_spectrum_file, cfg.has_errorbars
    )
    logger.info(f"   Raw: {len(wl_lin)} pixels, "
                 f"{wl_lin[0]:.1f}-{wl_lin[-1]:.1f} A")

    # Pre-fit corrections, before the fitting range is cut, so that range is
    # read in the rest frame whichever frame the file was in.
    from brainsp.corrections import apply_pre_fit_corrections
    wl_lin, flux_lin, err_lin, notes = apply_pre_fit_corrections(
        wl_lin, flux_lin, err_lin, cfg)
    for note in notes:
        logger.info(f"   {note}")

    if cfg.wavelength_min is not None or cfg.wavelength_max is not None:
        wl_min = cfg.wavelength_min or wl_lin[0]
        wl_max = cfg.wavelength_max or wl_lin[-1]
        clip = (wl_lin >= wl_min) & (wl_lin <= wl_max)
        wl_lin, flux_lin = wl_lin[clip], flux_lin[clip]
        if err_lin is not None:
            err_lin = err_lin[clip]
        if flag_lin is not None:
            flag_lin = flag_lin[clip]
        logger.info(f"   Clipped to {wl_min:.0f}-{wl_max:.0f} A: "
                     f"{len(wl_lin)} pixels")

    flux_lin, norm_fac = normalize_at_wavelength(
        flux_lin, wl_lin, cfg.norm_wavelength
    )
    if err_lin is not None:
        err_lin /= norm_fac

    flux_log, log_lam, velscale = log_rebin(wl_lin, flux_lin)
    lam_grid = jnp.exp(log_lam)
    n_pix = len(flux_log)
    logger.info(f"   Log-rebinned: {n_pix} pixels, "
                 f"velscale = {velscale:.2f} km/s/pix")


    file_weight_lin = load_mask(cfg.mask_file, wl_lin)
    dq_good_lin = np.ones(len(wl_lin), dtype=bool)
    if err_lin is not None:
        dq_good_lin &= (err_lin > 0)
    if flag_lin is not None:
        dq_good_lin &= (flag_lin == 0)
    weight_lin = file_weight_lin * dq_good_lin

    usable_lin = dq_good_lin & (np.asarray(file_weight_lin) > 0)
    clean_frac = np.asarray(clean_fraction(wl_lin, usable_lin, lam_grid))
    uncontaminated = clean_frac >= 1.0 - 1e-8

    if err_lin is not None:
        err_log = jnp.interp(jnp.asarray(lam_grid), jnp.asarray(wl_lin), jnp.asarray(err_lin))
    else:
        err_log = None

    from scipy.interpolate import interp1d
    w_interp = interp1d(np.asarray(wl_lin), weight_lin,
                        kind='nearest', bounds_error=False,
                        fill_value=0.0)
    weight_log = np.clip(w_interp(np.asarray(lam_grid)), 0.0, None)
    good_log = (weight_log > 0) & uncontaminated
    sqrt_w = jnp.asarray(np.sqrt(weight_log))

    inv_sigma = jnp.zeros(n_pix)
    if err_log is not None:
        valid = good_log & (np.asarray(err_log) > 0)
        inv_sigma = jnp.where(jnp.asarray(valid), sqrt_w / err_log, 0.0)
    else:
        inv_sigma = jnp.asarray(sqrt_w)

    # The mask is applied as given, in every mode. Pixels near the emission
    # lines used to be put back in the line modes, which also undid every
    # deliberate mask within thousands of km/s of a line (a detector gap, a
    # bad column, a sky residual) and made a spectrum disagree with the same
    # mask on a datacube, where nothing was put back. Not masking the lines
    # one wants fitted is the user's to do; a line left with no data is
    # dropped by the fitter and named (emission_lines.drop_masked_lines).

    would_have_fitted = ~uncontaminated & (weight_log > 0)
    n_contaminated = int(np.sum(would_have_fitted))
    if n_contaminated > 0:
        logger.info(f"   Rejected {n_contaminated} log-rebinned pixels whose "
                    f"flux is partly assembled from rejected input pixels "
                    f"(worst clean fraction "
                    f"{clean_frac[would_have_fitted].min():.3f})")

    n_good = int(jnp.sum(inv_sigma > 0))
    logger.info(f"   Fitting {n_good}/{n_pix} pixels "
                 f"({100 * n_good / n_pix:.1f}%)")


    if getattr(cfg, 'bases_entries', None):
        bases_info = cfg.bases_entries
        logger.info(f"  Base list: {len(bases_info)} templates selected")
    elif cfg.bases_dir_file:
        bases_info = load_bases_dir(cfg.bases_dir_file)
    else:
        bases_info = scan_ssp_folder(cfg.ssp_folder)

    M_log, ssp_names, ssp_ages, ssp_metals, ssp_short, ssp_mstars, ssp_yav_flags, ssp_fbase_norms = load_ssp_templates(
        cfg.ssp_folder, bases_info, lam_grid, cfg.norm_wavelength
    )
    n_ssp = M_log.shape[1]
    ssp_is_stellar = stellar_component_flags(bases_info, ssp_names)

    logger.info(f"   {n_ssp} templates loaded and directly mapped to log grid")
    n_featureless = int(np.sum(~ssp_is_stellar))
    if n_featureless:
        logger.info(f"   {n_featureless} of them are not stellar populations "
                    f"and carry light but no mass")

    covered = np.any(np.abs(np.asarray(M_log)) > 0, axis=1)
    n_bare = int(np.sum(~covered & (np.asarray(inv_sigma) > 0)))
    if n_bare:
        bare_wl = np.asarray(lam_grid)[~covered]
        logger.warning(
            f"   {n_bare} pixels lie outside the template coverage "
            f"({bare_wl.min():.0f}-{bare_wl.max():.0f} A) and were dropped. "
            f"Narrow the fitting range or use a base library that reaches "
            f"further.")
        inv_sigma = jnp.where(jnp.asarray(covered), inv_sigma, 0.0)

    fitted_pixels = covered & (np.asarray(inv_sigma) > 0)
    M_log, ssp_fbase_norms, lsf_info = prepare_templates(
        M_log, lam_grid, cfg, bases_info, fitted_pixels, ssp_fbase_norms)


    nebular_model = None
    qh_ratios = None

    if cfg.photoionization_mode in NEBULAR_MODES:
        from brainsp.nebular import NebularModel, compute_qh_ratios

        nebular_model = NebularModel(
            lam_grid, wavelength_scale=cfg.wavelength_scale,
            Te=cfg.electron_temperature,
            ne=cfg.electron_density,
            Y_He=cfg.helium_abundance,
            Z_gas=cfg.gas_metallicity,
            f_esc=cfg.lyman_escape_fraction,
        )

        qh_ratios, qh_absolute = compute_qh_ratios(
            ssp_ages, ssp_metals, ssp_fbase_norms,
            ssp_flux_unit=cfg.ssp_flux_unit,
            ssp_folder=cfg.ssp_folder,
            bases_info=bases_info,
            return_absolute=True,
            allow_partial=cfg.allow_partial_ionising_coverage,
        )

        n_with_qh = np.sum(qh_ratios > 0)
        logger.info(f"   Q(H) integrated from the SSP spectra below "
                     f"{911.76:.2f} A: {n_with_qh}/{n_ssp} SSPs emit "
                     f"ionizing photons")


    if cfg.photoionization_mode in NEBULAR_MODES:
        from brainsp import gas_diagnostics as _gd
        _gd.nebular_constrained(lam_grid, inv_sigma, flux_log)

    fitter = SpectralFitter(
        M_log, lam_grid, flux_log, inv_sigma, velscale, cfg,
        ssp_yav_flags=ssp_yav_flags,
        nebular_model=nebular_model,
        qh_ratios=qh_ratios,
    )

    if cfg.photoionization_mode == "photopop":
        fitter._photoionization = 'nepop'
        logger.info("   Full mode bootstraps through one continuum-mode fit: "
                    "the Case B penalty rows need line fluxes and a Q(H) to "
                    "tie together, and neither exists before the first fit.")

    t1 = time.time()
    result = fitter.optimize()
    t_fit = time.time() - t1

    best_params = result['params']
    av = best_params[0]
    yav = best_params[1]
    sig_s = best_params[2]
    v0_s = best_params[3]

    logger.info(f"   Completed in {t_fit:.2f}s "
                 f"({result['nfev']} evaluations)")
    logger.info(f"   Av         = {av:.4f}")
    logger.info(f"   YAV        = {yav:.4f}")
    logger.info(f"   σ_star     = {sig_s:.2f} km/s, OBSERVED")
    logger.info(f"   v0_star    = {v0_s:.2f} km/s")
    _log_gas_kinematics(
        _gas_kinematics(cfg, fitter, result, n_ssp, lsf_info), "   ")
    logger.info(f"   χ²         = {result['chi2']:.4f}")
    logger.info(f"   Converged  = {result['success']}")


    nebular_stats = None
    Av_neb = 0.0

    if cfg.photoionization_mode in NEBULAR_MODES:
        from brainsp.nebular import (
            Av_neb_from_balmer_decrement, extract_line_flux, compute_Q_eff,
        )
        from brainsp import reddening_laws

        gas_names = result['gas_names']
        w = result['weights']
        n_gas = len(gas_names)

        k_norm = reddening_laws.get_k_lambda(
            np.array([cfg.norm_wavelength]), cfg.reddening_law
        )[0]

        F_Ha_obs, F_Hb_obs = _observed_balmer(
            gas_names, w[n_ssp:], av, k_norm, cfg.reddening_law)

        has_balmer = (F_Ha_obs > 0) and (F_Hb_obs > 0)

        if has_balmer:
            Av_neb = Av_neb_from_balmer_decrement(
                F_Ha_obs, F_Hb_obs, cfg.reddening_law, cfg.electron_temperature
            )
            Av_neb = np.clip(Av_neb, cfg.av_nebular_bounds[0], cfg.av_nebular_bounds[1])

            logger.info(f"   Av_neb     = {Av_neb:.4f} (from Balmer decrement: "
                         f"Ha/Hb = {F_Ha_obs / F_Hb_obs:.3f})")

            fitter.update_Av_neb(Av_neb)

            t1 = time.time()
            result = fitter.optimize(initial_guess=result['params'])
            t_refit_neb = time.time() - t1
            t_fit += t_refit_neb

            best_params = result['params']
            av, yav, sig_s, v0_s = best_params[0], best_params[1], best_params[2], best_params[3]
            logger.info(f"   Re-fit with Av_neb: χ² = {result['chi2']:.4f} "
                         f"({t_refit_neb:.2f}s)")
        else:
            _missing = ("Hbeta" if F_Hb_obs <= 0 else "Halpha") + (
                " and Halpha" if (F_Hb_obs <= 0 and F_Ha_obs <= 0) else "")
            logger.info(
                f"   {_missing} came back at zero in the bootstrap fit "
                f"(Halpha = {F_Ha_obs:.4g}, Hbeta = {F_Hb_obs:.4g}), so there "
                f"is no Balmer decrement to derive Av_neb from. Nothing else "
                f"depends on it.")

        from brainsp import gas_diagnostics as gd

        _obs_lines = {n: float(w[n_ssp + i])
                      for i, n in enumerate(result['gas_names'])}

        _cond = gd.measure_conditions(_obs_lines)
        _redo = False
        if _cond['ne_measured'] or _cond['Te_measured']:
            if (abs(_cond['Te'] - nebular_model.Te) / max(nebular_model.Te, 1.0) > 0.02
                    or abs(_cond['ne'] - nebular_model.ne) / max(nebular_model.ne, 1.0) > 0.05):
                nebular_model = NebularModel(
                    lam_grid, wavelength_scale=cfg.wavelength_scale,
                    Te=_cond['Te'], ne=_cond['ne'],
                    Y_He=cfg.helium_abundance, Z_gas=cfg.gas_metallicity,
                    f_esc=cfg.lyman_escape_fraction)
                fitter.nebular_model = nebular_model
                fitter._fast_key = None
                _redo = True

        _bpt, _bpt_info = gd.classify_bpt(_obs_lines)
        _bpt_flag = gd.BPT_FLAG.get(_bpt, -999)
        logger.info(f"   BPT class: {_bpt} (flag {_bpt_flag}) — "
                    f"{_bpt_info.get('reason', 'not measurable')}")

        if _redo:
            logger.info("   Re-fitting with updated gas conditions / mode...")
            result = fitter.optimize(initial_guess=result['params'])
            best_params = result['params']
            av, yav, sig_s, v0_s = (best_params[0], best_params[1],
                                    best_params[2], best_params[3])
            w = result['weights']
            gas_names = result['gas_names']

        nebular_stats_extra = {
            'ne': _cond['ne'], 'Te': _cond['Te'],
            'ne_measured': _cond['ne_measured'],
            'Te_measured': _cond['Te_measured'],
            'bpt_class': _bpt, 'bpt_info': _bpt_info,
            'bpt_flag': _bpt_flag,
            'mode_used': cfg.photoionization_mode,
        }

        Q_eff = result['Q_eff']

        if has_balmer and F_Ha_obs > 0:
            tau = nebular_model.tau_ratio(
                Q_eff, F_Ha_obs, Av_neb,
                cfg.reddening_law, cfg.norm_wavelength
            )
        else:
            tau = np.nan

        logger.info(f"   Q_eff      = {Q_eff:.4e}")
        if np.isfinite(tau):
            logger.info(f"   τ ratio    = {tau:.4f}")
        else:
            logger.info(
                f"   τ ratio    = N/A (needs both Balmer lines; "
                f"Halpha = {F_Ha_obs:.4g}, Hbeta = {F_Hb_obs:.4g})")

        sc_converged = False
        sc_status = 'not applicable'
        sc_iterations = 0
        sc_dchi2_range = np.nan
        sc_one_sigma = np.nan

        d_av_neb = np.nan

        if cfg.photoionization_mode == "photopop":
            fitter._photoionization = 'photopop'
            fitter._fast_key = None
            logger.info("   Promoting to full mode: hydrogen line amplitudes "
                        "tied to the Case B prediction from Q(H).")
            result = fitter.optimize(initial_guess=result['params'])
            best_params = result['params']
            av, yav, sig_s, v0_s = (best_params[0], best_params[1],
                                    best_params[2], best_params[3])
            Q_eff = result['Q_eff']
            w = result['weights']
            gas_names = result['gas_names']
            _F_Ha = extract_line_flux(gas_names, w[n_ssp:n_ssp + len(gas_names)],
                                      'Halpha')
            tau = (nebular_model.tau_ratio(Q_eff, _F_Ha, Av_neb,
                                           cfg.reddening_law,
                                           cfg.norm_wavelength)
                   if _F_Ha > 0 else np.nan)
            logger.info(f"   Full-mode fit: χ² = {result['chi2']:.4f}, "
                        f"Q_eff = {Q_eff:.4e}, τ = {tau:.4f}"
                        if np.isfinite(tau) else
                        f"   Full-mode fit: χ² = {result['chi2']:.4f}, "
                        f"Q_eff = {Q_eff:.4e}, τ = N/A")

        if cfg.photoionization_mode == "photopop":
            logger.info(
                f"   Seeking Av_neb by chi2 over "
                f"[{cfg.av_nebular_bounds[0]:.2f}, "
                f"{cfg.av_nebular_bounds[1]:.2f}] mag "
                f"(budget {max(6, int(cfg.max_sc_iterations))} refits)...")
            _search = _seek_av_neb(fitter, cfg, result['params'],
                                   max(6, int(cfg.max_sc_iterations)))

            Av_neb = _search['av_neb']
            sc_status = _search['status']
            sc_converged = (sc_status == 'converged')
            sc_iterations = _search['n_evals']
            d_av_neb = _search['bracket']
            sc_dchi2_range = _search['dchi2_range']
            sc_one_sigma = _search['one_sigma']

            result = _search['result']
            best_params = result['params']
            av, yav, sig_s, v0_s = (best_params[0], best_params[1],
                                    best_params[2], best_params[3])
            Q_eff = result['Q_eff']
            w = result['weights']
            gas_names = result['gas_names']

            _F_Ha, _F_Hb = _observed_balmer(
                gas_names, w[n_ssp:n_ssp + len(gas_names)], av, k_norm,
                cfg.reddening_law)
            tau = (nebular_model.tau_ratio(Q_eff, _F_Ha, Av_neb,
                                           cfg.reddening_law,
                                           cfg.norm_wavelength)
                   if _F_Ha > 0 else np.nan)

            if np.isfinite(tau) and abs(tau - 1.0) > cfg.tau_tolerance:
                logger.warning(
                    f"   τ = {tau:.4f} is further than {cfg.tau_tolerance} "
                    f"from 1, so the Case B penalty rows did not pull the "
                    f"hydrogen amplitudes onto the Q(H) prediction. Reported, "
                    f"not tested: with those rows satisfied τ reduces to a "
                    f"function of A_V alone and says nothing about Av_neb.")

            _sig = (f", Δχ² = 1 at Av_neb = {sc_one_sigma:.3f}"
                    if np.isfinite(sc_one_sigma) else
                    ", Δχ² never reaches 1 inside the bounds")
            if sc_converged:
                logger.info(
                    f"   Av_neb = {Av_neb:.4f} mag, χ² = "
                    f"{result['chi2']:.4f}, bracketed to "
                    f"{d_av_neb:.2e} mag in {sc_iterations} refits. "
                    f"Δχ² over the full range = {sc_dchi2_range:.2f}{_sig}.")
            elif sc_status in ('LOWER_BOUND', 'UPPER_BOUND'):
                logger.warning(
                    f"   Av_neb = {Av_neb:.4f} mag is AT THE "
                    f"{sc_status.replace('_', ' ')}, not determined by the "
                    f"data. Reported as a bound, not as a fit. Δχ² over "
                    f"[{cfg.av_nebular_bounds[0]:.2f}, "
                    f"{cfg.av_nebular_bounds[1]:.2f}] = "
                    f"{sc_dchi2_range:.2f}{_sig}.")
            else:
                logger.warning(
                    f"   Av_neb = {Av_neb:.4f} mag NOT CONVERGED: the search "
                    f"ran out of its {sc_iterations}-refit budget with the "
                    f"bracket still {d_av_neb:.2e} mag wide (want < "
                    f"{cfg.av_neb_tolerance}). Raise 'sc_max_iterations'.")

        neb_spec = result['nebular_spectrum']
        neb_at_norm = np.interp(cfg.norm_wavelength, lam_grid, neb_spec)
        total_at_norm = np.interp(cfg.norm_wavelength, lam_grid, flux_log)
        f_neb_norm = neb_at_norm / max(total_at_norm, 1e-30)

        predicted_lines = nebular_model.predict_all_lines(
            Q_eff, wavelength_range=(lam_grid[0], lam_grid[-1])
        )

        observed_lines = {}
        w = result['weights']
        gas_names = result['gas_names']
        from brainsp.emission_lines import species_of
        from brainsp.atomic_data import OII_BLEND_COMPONENTS
        _h_short = {species_of(k) for k in fitter._hydrogen_line_keys}
        for i, gname in enumerate(gas_names):
            observed_lines[gname] = w[n_ssp + i]
            short = species_of(gname)
            if short in _h_short and short not in observed_lines:
                observed_lines[short] = w[n_ssp + i]

        _oii_parts = [observed_lines[c] for c in OII_BLEND_COMPONENTS
                      if c in observed_lines]
        if _oii_parts:
            observed_lines['[O II]_3727'] = sum(_oii_parts)

        Q_H_total = 0.0
        w_star = w[:n_ssp]
        for j in range(n_ssp):
            if w_star[j] <= 0:
                continue
            av_tot_j = av + (yav if ssp_yav_flags[j] else 0.0)
            unredden = 10.0**(0.4 * av_tot_j * k_norm)
            mini_j_val = w_star[j] * norm_fac * unredden / max(ssp_fbase_norms[j], 1e-30)
            Q_H_total += mini_j_val * qh_absolute['QHc'][j]

        nebular_stats = {
            **nebular_stats_extra,
            'Q_eff': Q_eff,
            'Q_H_total': Q_H_total,
            'tau_ratio': tau,
            'Av_neb': Av_neb,
            'f_neb_norm': f_neb_norm,
            'sc_converged': sc_converged,
            'sc_status': sc_status,
            'sc_iterations': sc_iterations,
            'sc_d_av_neb': d_av_neb,
            'sc_dchi2_range': sc_dchi2_range,
            'sc_one_sigma': sc_one_sigma,
            'predicted_lines': predicted_lines,
            'observed_lines': observed_lines,
        }


    result, inv_sigma, total_clipped, t_clip = clip_outliers_and_refit(
        fitter, result, inv_sigma, flux_log, err_log, n_ssp, cfg)
    t_fit += t_clip

    best_params = result['params']
    av, yav, sig_s, v0_s = (best_params[0], best_params[1],
                            best_params[2], best_params[3])
    M_star_dust = result['M_star_dust']
    M_gas_dust = result['M_gas_dust']
    w = result['weights']
    gas_names = result['gas_names']
    nebular_fit = result.get('nebular_spectrum', jnp.zeros_like(flux_log))


    logger.info(f"   Monte Carlo error estimation "
                 f"({cfg.n_mc_resamples} resamples)...")

    flux_for_mc = (flux_log - nebular_fit
                   if cfg.photoionization_mode in NEBULAR_MODES else flux_log)

    t_mc = time.time()
    w_med, w_err = mc_weight_errors(
        M_star_dust, M_gas_dust, flux_for_mc, inv_sigma, err_log,
        cfg.reg_strength, cfg.reg_order, cfg.n_mc_resamples,
        solver=cfg.linear_solver, simplex=cfg.enforce_simplex
    )
    t_mc = time.time() - t_mc
    logger.info(f"   Completed in {t_mc:.2f}s")


    total_runtime = time.time() - t0

    stellar_fit = M_star_dust @ w[:n_ssp]
    n_gas = len(gas_names)
    if M_gas_dust is not None and n_gas > 0:
        emission_fit = M_gas_dust @ w[n_ssp:n_ssp + n_gas]
    else:
        emission_fit = jnp.zeros_like(flux_log)
    total_fit = stellar_fit + emission_fit + nebular_fit
    total_resid = flux_log - total_fit

    from brainsp import reddening_laws

    n_eff = int(effective_pixel_count(np.asarray(inv_sigma)))
    chi2_red = float(chi2_per_effective_pixel(float(result['chi2']), n_eff))

    chi2_data = float(jnp.sum(((flux_log - total_fit) * inv_sigma) ** 2))
    chi2_data_red = float(chi2_per_effective_pixel(chi2_data, n_eff))

    adev, n_adev_pix, n_good_pix = deviation_percent(
        np.asarray(flux_log), np.asarray(total_fit),
        np.asarray(inv_sigma) > 0)
    adev = float(adev)
    n_good_pix = int(n_good_pix)
    n_adev_skipped = int(n_good_pix - n_adev_pix)
    if n_adev_skipped:
        logger.warning(
            f"   {n_adev_skipped} of {n_good_pix} fitted pixels have a "
            f"non-positive observed flux and are left out of adev, which is a "
            f"deviation relative to that flux and is undefined there. They "
            f"are still fitted and still counted in chi2.")

    k_norm = float(reddening_laws.get_k_lambda(jnp.array([cfg.norm_wavelength]), cfg.reddening_law)[0])

    populations = population_statistics(
        np.asarray(w[:n_ssp]), norm_fac, float(av), float(yav),
        ssp_yav_flags, ssp_mstars, ssp_fbase_norms, k_norm, ssp_is_stellar)

    fit_stats = {
        'chi2_red': chi2_red,
        'chi2_data_red': chi2_data_red,
        'adev': adev,
        'flux_tot': float(populations['flux_tot']),
        'fobs_norm': float(norm_fac),
        'mini_tot': float(populations['mini_tot']),
        'mcor_tot': float(populations['mcor_tot']),
        'x_j': populations['x_j'],
        'mini_pct': populations['mini_pct'],
        'mcor_pct': populations['mcor_pct'],
    }

    lsf_report = dispersion_report(
        lsf_info, sig_s, np.asarray(w[:n_ssp]), np.asarray(total_fit),
        np.asarray(inv_sigma) > 0, cfg.sigma_star_bounds[0],
        upper_bound=cfg.sigma_star_bounds[1],
        reference_wavelength=cfg.norm_wavelength)

    gas_report = _gas_kinematics(cfg, fitter, result, n_ssp, lsf_info)

    if lsf_report['reason']:
        logger.warning(f"   {lsf_report['reason']}")
    if lsf_report['delta_sigma_inst'] is not None:
        logger.info(f"   sigma_obs        = {lsf_report['sigma_obs']:.2f} km/s "
                    f"(kernel applied to the library)")
        logger.info(f"   delta_sigma_inst = "
                    f"{lsf_report['delta_sigma_inst']:+.2f} km/s "
                    f"(data minus library line-spread function)")
        logger.info(f"   sigma_star       = "
                    f"{'< ' if lsf_report['upper_limit'] else ''}"
                    f"{abs(lsf_report['sigma_star']):.2f} km/s"
                    f"{' (upper limit)' if lsf_report['upper_limit'] else ''}")

    write_summary(
        cfg.output_summary, cfg, result['params'],
        w, w_err, n_ssp,
        ssp_short, ssp_ages, ssp_metals, ssp_names, ssp_yav_flags, ssp_mstars, ssp_fbase_norms, fit_stats,
        gas_names, result['chi2'], n_pix,
        (lam_grid[0], lam_grid[-1]),
        total_runtime, result['nfev'],
        nebular_stats=nebular_stats,
        lsf_report=lsf_report,
        gas_report=gas_report,
    )

    write_fitted_spectrum(
        cfg.output_spectrum, lam_grid, flux_log,
        total_fit, stellar_fit, emission_fit,
        err_log, total_resid,
        nebular_fit=(nebular_fit
                     if cfg.photoionization_mode in NEBULAR_MODES else None),
    )


    logger.info("=" * 70)
    logger.info(f"  DONE in {total_runtime:.1f}s")
    logger.info(f"  Av = {av:.4f}, YAV = {yav:.4f}, σ★ = {sig_s:.2f}, "
                f"v0★ = {v0_s:.2f}")
    _log_gas_kinematics(gas_report, "  ")
    logger.info(f"  Mini_tot = {fit_stats['mini_tot']:.4e}, "
                f"Mcor_tot = {fit_stats['mcor_tot']:.4e} "
                f"(apparent; no distance applied, see the summary file)")
    logger.info(f"  χ²/Neff = {chi2_red:.4f} (data + penalty rows), "
                 f"data only = {chi2_data_red:.4f}, ADEV = {adev:.3f}%")
    logger.info(f"  Both over the same {n_eff} pixels"
                + (f", after clipping {total_clipped}" if total_clipped
                   else ", none clipped"))

    if nebular_stats:
        tau_str = f"{nebular_stats['tau_ratio']:.4f}" if np.isfinite(nebular_stats['tau_ratio']) else "N/A"
        logger.info(f"  Q_eff = {nebular_stats['Q_eff']:.4e}, "
                     f"τ = {tau_str}, "
                     f"Av_neb = {nebular_stats['Av_neb']:.4f}")
        f_neb = nebular_stats['f_neb_norm']
        if f_neb >= 0.001:
            logger.info(f"  Nebular fraction at λ₀ = {f_neb:.3f}")
        else:
            logger.info(f"  Nebular fraction at λ₀ = {f_neb:.2e}")

    logger.info(f"  Output: {cfg.output_summary}, {cfg.output_spectrum}")
    logger.info("=" * 70)

    return result




if __name__ == "__main__":
    """
    brainv1 is a module, not an entry point.

    It used to carry a hand-written NewBrainConfig here, which meant that
    running this file fitted whatever that literal said rather than what
    brain_setup.yaml says. The two disagreed -- the literal forced
    photoionization_mode="photopop" and a base library that is not the shipped
    default -- so a `spop` run launched this way tried to compute Q(H) and
    died on the Lyman limit. There is one configuration and it lives in the
    YAML, so this hands straight over to the documented entry point and
    passes any spectra named on the command line through unchanged.
    """
    import sys as _sys
    _sys.dont_write_bytecode = True
    from brainsp.run_brain import main as _main
    _sys.exit(_main(_sys.argv))
