"""
Mouse zoom and pan for a map drawn with imshow.

    wheel           zoom in or out about the pointer
    drag            pan, once zoomed in
    click           whatever the map does with a click (pick a spaxel)
    double-click    back to the whole field

A press that moves less than a few pixels before it is released is a click,
so picking a spaxel works exactly as it did; anything further is a pan. The
view is kept across redraws -- a new plane, a colour map, a run filling in --
and dropped only when the field changes shape, because a zoom on one cube
means nothing on another.
"""

import numpy as np

ZOOM_STEP = 1.25
DRAG_PIXELS = 4

class ImageZoom:

    def __init__(self, canvas, ax, on_click=None):
        self.canvas = canvas
        self.ax = ax
        self.on_click = on_click
        self.full = None
        self.view = None
        self._press = None
        self._panned = False
        canvas.mpl_connect("scroll_event", self._scroll)
        canvas.mpl_connect("button_press_event", self._pressed)
        canvas.mpl_connect("motion_notify_event", self._moved)
        canvas.mpl_connect("button_release_event", self._released)

    def set_full(self, extent):
        """The whole field as (x0, x1, y0, y1); a new shape drops the zoom."""
        extent = tuple(float(v) for v in extent)
        if extent != self.full:
            self.view = None
        self.full = extent

    def apply(self):
        """Put the kept view back after the axes were redrawn."""
        if self.view is not None:
            self.ax.set_xlim(*self.view[0])
            self.ax.set_ylim(*self.view[1])

    def reset(self):
        self.view = None
        if self.full is not None:
            x0, x1, y0, y1 = self.full
            self.ax.set_xlim(x0, x1)
            self.ax.set_ylim(y0, y1)
        self.canvas.draw_idle()

    def zoomed(self):
        return self.view is not None



    def _keep(self):
        x = self.ax.get_xlim()
        y = self.ax.get_ylim()
        if self.full is not None:
            x0, x1, y0, y1 = self.full
            if (abs(x[1] - x[0]) >= abs(x1 - x0) - 1e-9
                    and abs(y[1] - y[0]) >= abs(y1 - y0) - 1e-9):
                self.view = None
                return
        self.view = (tuple(x), tuple(y))

    def _clamp(self, low, high, lo_full, hi_full):
        """An interval no wider than the field, slid back inside it."""
        span = min(high - low, hi_full - lo_full)
        low = max(lo_full, min(low, hi_full - span))
        return low, low + span

    def _scroll(self, event):
        if event.inaxes is not self.ax or event.xdata is None \
                or self.full is None:
            return
        step = 1 / ZOOM_STEP if event.button == "up" else ZOOM_STEP
        x0, x1, y0, y1 = self.full
        lx, hx = self.ax.get_xlim()
        ly, hy = self.ax.get_ylim()
        cx, cy = event.xdata, event.ydata
        lx, hx = cx - (cx - lx) * step, cx + (hx - cx) * step
        ly, hy = cy - (cy - ly) * step, cy + (hy - cy) * step
        # Never narrower than about two spaxels, never wider than the field.
        if hx - lx < 2.0 or hy - ly < 2.0:
            return
        self.ax.set_xlim(*self._clamp(lx, hx, x0, x1))
        self.ax.set_ylim(*self._clamp(ly, hy, y0, y1))
        self._keep()
        self.canvas.draw_idle()

    def _pressed(self, event):
        if event.inaxes is not self.ax or event.button != 1:
            return
        if event.dblclick:
            self._press = None
            self.reset()
            return
        self._press = (event.x, event.y, self.ax.get_xlim(),
                       self.ax.get_ylim())
        self._panned = False

    def _moved(self, event):
        if self._press is None or self.full is None:
            return
        px, py, (lx, hx), (ly, hy) = self._press
        if not self._panned and (abs(event.x - px) < DRAG_PIXELS
                                 and abs(event.y - py) < DRAG_PIXELS):
            return
        if self.view is None:
            return  # nothing to pan across at the whole field
        self._panned = True
        # Pixels to data with the view as it was at the press, so the map
        # follows the pointer rather than accelerating away from it.
        box = self.ax.bbox
        dx = (event.x - px) * (hx - lx) / max(box.width, 1.0)
        dy = (event.y - py) * (hy - ly) / max(box.height, 1.0)
        x0, x1, y0, y1 = self.full
        self.ax.set_xlim(*self._clamp(lx - dx, hx - dx, x0, x1))
        self.ax.set_ylim(*self._clamp(ly - dy, hy - dy, y0, y1))
        self._keep()
        self.canvas.draw_idle()

    def _released(self, event):
        press, self._press = self._press, None
        if press is None or self._panned:
            return
        if self.on_click is not None and event.inaxes is self.ax:
            self.on_click(event)

class AxesZoom:
    """
    Mouse zoom and pan for plots drawn on ordinary axes: spectra, histories,
    diagnostic diagrams.

        wheel           zoom about the pointer: the wavelength axis of a
                        spectrum ("x" axes), both axes of a diagram ("xy")
        shift + wheel   zoom the vertical axis alone
        drag            pan
        double-click    back to the whole plot

    Each axes keeps the limits it was drawn with as its home; call
    `remember(ax)` after drawing it. A logarithmic axis zooms and pans in its
    logarithm, so the pointer stays over the same value.
    """

    def __init__(self, canvas, modes):
        self.canvas = canvas
        self.modes = dict(modes)
        self.home = {}
        self._press = None
        canvas.mpl_connect("scroll_event", self._scroll)
        canvas.mpl_connect("button_press_event", self._pressed)
        canvas.mpl_connect("motion_notify_event", self._moved)
        canvas.mpl_connect("button_release_event", self._released)

    def remember(self, ax):
        """The limits `ax` has now are its whole view."""
        self.home[ax] = (ax.get_xlim(), ax.get_ylim())



    @staticmethod
    def _scale(ax, which):
        log = (ax.get_xscale() if which == "x" else ax.get_yscale()) == "log"
        if log:
            return (lambda v: np.log10(v)), (lambda v: 10.0 ** v)
        return (lambda v: v), (lambda v: v)

    def _zoom_axis(self, ax, which, at, step):
        log = (ax.get_xscale() if which == "x" else ax.get_yscale()) == "log"
        if at is None or (log and at <= 0):
            return
        fwd, back = self._scale(ax, which)
        lo, hi = ax.get_xlim() if which == "x" else ax.get_ylim()
        a, l, h = fwd(at), fwd(lo), fwd(hi)
        new = (back(a - (a - l) * step), back(a + (h - a) * step))
        (ax.set_xlim if which == "x" else ax.set_ylim)(*new)

    def _scroll(self, event):
        ax = event.inaxes
        if ax not in self.modes:
            return
        step = 1 / ZOOM_STEP if event.button == "up" else ZOOM_STEP
        if "shift" in (event.modifiers or ()):
            self._zoom_axis(ax, "y", event.ydata, step)
        else:
            self._zoom_axis(ax, "x", event.xdata, step)
            if self.modes[ax] == "xy":
                self._zoom_axis(ax, "y", event.ydata, step)
        self.canvas.draw_idle()

    def _pressed(self, event):
        ax = event.inaxes
        if ax not in self.modes or event.button != 1:
            return
        if event.dblclick:
            if ax in self.home:
                xlim, ylim = self.home[ax]
                ax.set_xlim(*xlim)
                ax.set_ylim(*ylim)
                self.canvas.draw_idle()
            self._press = None
            return
        self._press = (ax, event.x, event.y, ax.get_xlim(), ax.get_ylim())

    def _moved(self, event):
        if self._press is None:
            return
        ax, x0, y0, xlim, ylim = self._press
        box = ax.bbox
        for which, lim, moved, size in (("x", xlim, event.x - x0, box.width),
                                        ("y", ylim, event.y - y0, box.height)):
            if size <= 0:
                continue
            fwd, back = self._scale(ax, which)
            l, h = fwd(lim[0]), fwd(lim[1])
            shift = (h - l) * moved / size
            (ax.set_xlim if which == "x" else ax.set_ylim)(back(l - shift),
                                                           back(h - shift))
        self.canvas.draw_idle()

    def _released(self, event):
        self._press = None
