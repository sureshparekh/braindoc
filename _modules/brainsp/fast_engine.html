Source code for brainsp.fast_engine

"""
NewBrain Fast Engine — unified 100% JAX forward model for all three
photoionization modes.

Replaces two things at once:

  1. The scipy fallback that `nepop` and `photopop` modes used. The nebular
     fixed point is now a `lax.fori_loop` with a constant trip count, so the
     whole residual is traceable and the pure-JAX LM drives every mode.

  2. The redundant work in the `spop`-mode forward model. `rfft(M_star)` is
     hoisted out of the fit (M_star never changes), and the Jacobian only
     re-broadens for the two parameters that can actually alter the stellar
     broadening.

The parameter vector is unchanged:
    [Av, YAV, sigma_star, v0_star, sig_g0, v0_g0, sig_g1, v0_g1, ...]

Modes:
  "spop"     - stellar populations alone; emission lines are not fitted.
  "nepop"    - nebular continuum carried as a free NNLS column, its amplitude
                read back as Q_eff. Emission lines stay free and empirical.
                The nebular light enters the model ONCE, through that column;
                it is not also subtracted from the data.
  "photopop" - as nepop, plus hydrogen line amplitudes tied to the Case B
                prediction from the same Q(H) via augmented-NNLS penalty rows.
"""

import logging
from functools import partial
from typing import NamedTuple

import jax
import jax.numpy as jnp
import numpy as np
from jax.numpy.fft import rfft, irfft, rfftfreq

from brainsp.emission_lines import _build_emission_jit
from brainsp.solver import _fista_nnls, solve_active_set

logger = logging.getLogger("NEWBRAIN")

def _smooth_length(n, factors=(2, 3, 5, 7)):
    """Smallest m >= n whose only prime factors are in `factors`."""
    while True:
        m = n
        for p in factors:
            while m % p == 0:
                m //= p
        if m == 1:
            return n
        n += 1


def _kinematic_reach(n_pix, velscale, v0_max, sigma_max):
    """
    How many pixels the broadening kernel reaches past either end.

    An FFT convolution is periodic. Without padding the model at the blue end
    of the window is built partly out of the template at the red end: measured
    at sigma = 150 km/s and v0 = 0 on the reference grid, 18 pixels were wrong
    by more than 0.1%, the last one by 315% and the first by 77%.

    The reach is the velocity shift plus the tail of the Gaussian, both in
    pixels, and it must be STATIC -- the transform of the padded template is
    precomputed once at build time and its shape cannot depend on a traced
    sigma. So it is derived from the bounds the fit is allowed to explore,
    which is why `v0_max` and `sigma_max` are arguments here. They default to
    the same numbers `brain_setup.yaml` defaults `velocity_range` and
    `dispersion_range` to. A CALLER THAT WIDENS EITHER BOUND MUST WIDEN THESE
    TOO, or the kernel will reach past the padding again at the far end of the
    widened range.

    Six dispersions rather than five: at five the Gaussian is 3.7e-6 of its
    peak, at six 1.5e-8, and the extra pixels cost nothing next to the length
    the transform is rounded up to anyway.
    """
    reach = (int(abs(v0_max) / velscale) + int(6.0 * sigma_max / velscale) + 2)
    return int(min(reach, n_pix))


def _edge_pad(M_star, n_left, n_right):
    """
    Extend a template matrix at both ends by holding its end values.

    Zero-filling instead is the trap. A template padded with zeros is dragged
    toward zero by the convolution, so the model darkens at both ends of the
    fitted window -- an error that looks exactly like a continuum slope and is
    fitted as one. Holding the first and last value says the template is flat
    beyond the window, which is wrong in detail and right in magnitude, and it
    leaves a flat template exactly flat end to end.
    """
    if n_left <= 0 and n_right <= 0:
        return M_star
    left = jnp.repeat(M_star[:1], n_left, axis=0)
    right = jnp.repeat(M_star[-1:], n_right, axis=0)
    return jnp.concatenate([left, M_star, right], axis=0)

def build_constants(M_star, lam, flux, inv_sigma, velscale, dk, D_full,
                    yav_flags, line_table, mode="spop",
                    qh_ratios=None, neb_template=None, k_norm=0.0,
                    av_neb=0.0, f_capture=1.0,
                    H_gas_idx=None, H_ratios=None, H_dk=None, C_Hbeta=0.0,
                    sc_weight=1000.0, sc_damping=1.0, n_neb_iter=12,
                    v0_max=5000.0, sigma_max=500.0,
                    solver="active_set", simplex=False, fit_lines=None):
    """
    Precompute every constant the forward model needs.

    On `simplex`: the constraint is sum(x_j) + y = 1, but the normalisation flux
    F_lambda0 is itself a fitted parameter bounded to [0.5, 1.5] times the
    measured value (fnormlow / fnormupp), so the equality pins the *fractions*
    while F_lambda0 absorbs the overall scale. That is equivalent to leaving
    the weights free with sum(w) inside [0.5, 1.5] — which is what BRAIN does,
    and its sum lands at ~1.07. Imposing a hard sum(x)=1 instead is a
    different, stricter condition: measured on the reference spectrum it costs
    chi2 (1843 -> 1992), pushes Av from 0.24 to 0.73 and pins sigma_star at its
    lower bound. So it defaults off. The reported light fractions are
    normalised to 100% regardless, as they always were.

    The stellar templates are edge-padded at both ends before their transform
    is taken, so the broadening is a linear convolution and not a periodic one.
    `v0_max` and `sigma_max` set how far it has to reach; see
    _kinematic_reach, which also says what happens if a caller widens the
    fitted bounds past them.
    """
    M_star = jnp.asarray(M_star)
    n_pix, n_star = M_star.shape

    n_left = _kinematic_reach(n_pix, velscale, v0_max, sigma_max)
    n_pad = _smooth_length(n_pix + 2 * n_left)
    M_pad = _edge_pad(M_star, n_left, n_pad - n_pix - n_left)

    if fit_lines is None:
        fit_lines = (mode != "spop")
    n_gas = len(line_table['names']) if fit_lines else 0
    n_neb = 1 if mode == "nepop" else 0

    n_groups_max = 16
    live = [0]
    if bool(jnp.any(jnp.asarray(yav_flags) > 0)):
        live.append(1)
    live += [2, 3]
    if fit_lines and n_gas > 0:
        live += list(range(4, 4 + 2 * n_groups_max))
    live = tuple(live)
    max_comp = (int(line_table['rest_wavelengths'].shape[1])
                if line_table['names'] else 1)

    if mode == "spop":
        neb_shape = jnp.zeros(n_pix)
        neb_col = jnp.zeros(n_pix)
        neb_at_l0 = 0.0
        qh = jnp.zeros(n_star)
    else:
        neb_atten = (10.0 ** (-0.4 * av_neb * dk)) if av_neb > 0 else 1.0
        neb_shape = jnp.asarray(neb_template) * f_capture * neb_atten
        i0 = int(jnp.argmin(jnp.abs(jnp.asarray(dk))))
        neb_at_l0 = float(neb_shape[i0])
        neb_col = neb_shape / jnp.maximum(jnp.abs(neb_at_l0), 1e-300)
        qh = jnp.asarray(qh_ratios)

    if mode == "photopop" and H_gas_idx is not None and len(H_gas_idx) > 0:
        Hg = jnp.asarray(H_gas_idx, dtype=jnp.int32)
        Hr = jnp.asarray(H_ratios)
        Hd = jnp.asarray(H_dk)
    else:
        Hg = jnp.zeros(0, dtype=jnp.int32)
        Hr = jnp.zeros(0)
        Hd = jnp.zeros(0)

    D_full = jnp.asarray(D_full)
    n_total = n_star + n_gas + n_neb
    if D_full.shape[1] < n_total:
        D_full = jnp.pad(D_full, ((0, 0), (0, n_total - D_full.shape[1])))
    elif D_full.shape[1] > n_total:
        D_full = D_full[:, :n_total]

    consts = ForwardConstants(
        M_fft=rfft(M_pad, axis=0),
        omega=2.0 * jnp.pi * rfftfreq(n_pad),
        velscale=float(velscale),
        yav_flags=jnp.asarray(yav_flags),
        lam=jnp.asarray(lam),
        flux=jnp.asarray(flux),
        inv_sigma=jnp.asarray(inv_sigma),
        dk=jnp.asarray(dk),
        D_full=jnp.asarray(D_full),
        rest_wl=line_table['rest_wavelengths'],
        ratios=line_table['ratios'],
        n_comp=line_table['n_components'],
        group_indices=line_table['group_indices'],
        neb_shape=neb_shape,
        neb_col=neb_col,
        neb_at_l0=float(neb_at_l0),
        qh_ratios=qh,
        k_norm=float(k_norm),
        H_gas_idx=Hg, H_ratios=Hr, H_dk=Hd,
        C_Hbeta=float(C_Hbeta),
        av_neb=float(av_neb),
        sc_weight=float(sc_weight),
        sc_damping=float(sc_damping),
    )
    shapes = ForwardShapes(
        n_pix=n_pix, n_pad=n_pad, n_left=n_left, n_star=n_star, n_gas=n_gas,
        n_par=0, max_comp=max_comp, mode=mode, n_neb_iter=int(n_neb_iter),
        live=live, n_neb=int(n_neb), solver=str(solver), simplex=bool(simplex), fit_lines=bool(fit_lines),
    )
    return consts, shapes

@partial(jax.jit, static_argnames=('n_pad', 'n_pix', 'n_left'))
def broaden(C, sigma_v, v0, n_pad, n_pix, n_left):
    """
    Broaden + shift all stellar templates.

    rfft(M_star) is supplied precomputed in C.M_fft, so this is a kernel
    multiply plus one inverse transform — the single biggest saving in the
    whole forward model.

    C.M_fft is the transform of the EDGE-PADDED matrix, `n_left` pixels of it
    before the first fitted pixel, so the window comes back out of the middle
    of the result and not off its front. Slicing from 0 instead would return a
    spectrum shifted by n_left pixels, which at 129 pixels and 58 km/s is
    7500 km/s and would be anything but subtle; slicing from n_left + 1 would
    be one pixel, which would not.
    """
    sigma_pix = sigma_v / C.velscale
    shift_pix = v0 / C.velscale
    kern = jnp.exp(-0.5 * (C.omega * sigma_pix) ** 2 - 1j * C.omega * shift_pix)
    out = irfft(C.M_fft * kern[:, jnp.newaxis], n=n_pad, axis=0)
    return out[n_left:n_left + n_pix]




@partial(jax.jit, static_argnames=('n_star',))
def _q_vector(C, av, yav, n_star):
    """
    Per-SSP ionizing yield per unit NNLS weight, un-reddened.

    Vectorised replacement for nebular.compute_Q_eff's 85-iteration Python
    loop (which branched on device values, forcing a host sync per SSP).
    """
    av_tot = av + yav * (C.yav_flags > 0)
    unredden = 10.0 ** (0.4 * av_tot * C.k_norm)
    return jnp.where(C.qh_ratios > 0, unredden * C.qh_ratios, 0.0)


@partial(jax.jit, static_argnames=('n_star', 'n_total'))
def _sc_rows(C, Q_eff, av, q_vec, n_star, n_total):
    """
    Augmented-NNLS penalty rows tying hydrogen line amplitudes to the Case B
    prediction from Q(H).

    For each hydrogen line L, the model's *observed* flux is
        w_L * 10^(-0.4 * Av * dk_L)                       (gas block wears stellar Av)
    and the Case B prediction, reddened by the nebular extinction, is
        C_Hbeta * ratio_L * Q_eff * 10^(-0.4 * Av_neb * dk_L).
    Equating them,
        w_L = C_Hbeta * ratio_L * Q_eff * 10^(-0.4 * (Av_neb - Av) * dk_L)
    so the residual row is  C_L * (q_vec . w_star) - w_L = 0, scaled by
    sqrt(sc_weight) / expected_L to make the penalty relative.

    Built as one outer product plus a scatter — no lax.map, no Python loop.
    """
    n_H = C.H_gas_idx.shape[0]
    C_L = C.C_Hbeta * C.H_ratios * 10.0 ** (-0.4 * (C.av_neb - av) * C.H_dk)
    expected = C_L * Q_eff
    active = (Q_eff > 1e-10) & (expected > 1e-30)
    scale = jnp.where(active, jnp.sqrt(C.sc_weight) / jnp.maximum(expected, 1e-30), 0.0)

    q_pad = jnp.concatenate([q_vec, jnp.zeros(n_total - n_star)])
    rows = (C_L * scale)[:, jnp.newaxis] * q_pad[jnp.newaxis, :]
    rows = rows.at[jnp.arange(n_H), n_star + C.H_gas_idx].add(-scale)
    return rows, jnp.zeros(n_H)


@partial(jax.jit, static_argnames=('S',))
def _design(C, params, M_broad, S):
    """Assemble the dust-attenuated design matrix from an already-broadened M."""
    av, yav = params[0], params[1]
    gas_p = params[4:]

    atten = 10.0 ** (-0.4 * av * C.dk)
    atten_yav = 10.0 ** (-0.4 * (av + yav) * C.dk)
    M_star_dust = jnp.where(
        C.yav_flags[jnp.newaxis, :] > 0,
        M_broad * atten_yav[:, jnp.newaxis],
        M_broad * atten[:, jnp.newaxis],
    )
    blocks = [M_star_dust]
    if S.fit_lines:
        M_gas = _build_emission_jit(
            C.lam, C.velscale, gas_p[::2], gas_p[1::2],
            C.rest_wl, C.ratios, C.n_comp, C.group_indices, S.max_comp,
        )
        blocks.append(M_gas * atten[:, jnp.newaxis])
    if S.n_neb:
        blocks.append(C.neb_col[:, jnp.newaxis])
    return jnp.hstack(blocks) if len(blocks) > 1 else blocks[0]


@partial(jax.jit, static_argnames=('S',))
def _linear_solve(C, ATA, ATb, S, P_init=None):
    """
    Solve the linear sub-problem for the light fractions.

    "active_set" is exact — it reaches the same optimum as scipy's
    Lawson-Hanson to ~1e-12 — and with simplex=True it additionally enforces
    sum(x) = 1 through the KKT system, so every weight is a light fraction in
    [0, 1]. "fista" is the previous iterative solver, kept
    only for reproducing older results; it does not converge on this problem.
    """
    if S.solver == "active_set":
        n = ATb.shape[0]
        if S.simplex:
            cvec = jnp.where(jnp.arange(n) < S.n_star, 1.0, 0.0)
        else:
            cvec = jnp.ones(n)
        return solve_active_set(ATA, ATb, cvec, P_init=P_init,
                                simplex=S.simplex)
    return _fista_nnls(ATA, ATb), jnp.zeros(ATb.shape[0], dtype=bool)


@partial(jax.jit, static_argnames=('S',))
def _solve(C, M_full, flux_eff, S, P_init=None):
    """
    Weighted, regularised NNLS.

    Deliberately does NOT accept empty (0, n) augmentation blocks — stacking a
    zero-row array measurably blocks XLA fusion (10.2 ms vs 4.5 ms per
    evaluation on CPU). The SC-augmented path folds its rows in as a rank-n_H
    Gram update instead, in forward_from_broad.
    """
    A = jnp.vstack([M_full * C.inv_sigma[:, jnp.newaxis], C.D_full])
    b = jnp.concatenate([flux_eff * C.inv_sigma, jnp.zeros(C.D_full.shape[0])])
    return _linear_solve(C, A.T @ A, A.T @ b, S, P_init)

@partial(jax.jit, static_argnames=('S',))
def forward_from_broad(C, params, M_broad, S, Q_init=0.0, P_init=None):
    """
    Full forward model given an already-broadened stellar matrix.

    Returns (residual, weights, Q_eff, M_full, P_out).

    The nebular continuum enters exactly once, and the two modes put it in
    different places. In "nepop" it is a column of M_full (built in
    _design when S.n_neb is 1) and the NNLS fits its amplitude, so the
    residual is formed against the untouched data. In "photopop" there is no such
    column -- S.n_neb is 0 -- and the nebular is instead subtracted from the
    data at the Q(H) the fixed point has reached. Subtracting in nepop mode
    as well minimised flux - stellar - gas - 2*(nebular), which is chi2 plus a
    shrinkage penalty ||nebular amplitude||^2 that the inner NNLS never sees;
    the outer LM could only satisfy it by driving A_V, sigma and v0 to where
    the NNLS wanted less nebular.

    This is the function the Jacobian calls for the six parameters that cannot
    change the stellar broadening (Av, YAV, and every gas kinematic), which is
    where the speedup comes from: no FFT at all on this path.

    `Q_init` warm-starts the nebular fixed point. With Q_init = 0 the loop
    reproduces the original cold-start behaviour; the Jacobian passes the
    converged Q* from the centre point so a single extra iteration suffices.

    In photopop mode the loop runs a FIXED S.n_neb_iter passes, undamped, and
    returns the Q it actually built the residual with -- not the Q the last
    pass would have gone on to use. Those differ by a factor of about two when
    the loop is truncated, and everything downstream (tau, the nebular
    spectrum, f_neb, the Nebular_Fit column) wants the one that is physically
    in the model.

    On the trip count: the first pass starts from Q = 0, which deactivates the
    SC penalty rows entirely, so it overshoots the fixed point by ~144x.
    Averaging that iterate into every subsequent step, which is what
    sc_damping < 1 does, is what made the sequence non-monotone and slow --
    abs(dQ)/Q < 1e-10 took a median of 44 iterations at damping 0.5 against 7
    undamped, over 25 parameter draws spanning the whole bounds box. Undamped
    with 12 passes the fixed point is reached everywhere tested, and the trip
    count is fixed, so the Jacobian runs the identical schedule to the residual
    by construction rather than by argument.
    """
    n_total = S.n_star + S.n_gas + S.n_neb
    M_full = _design(C, params, M_broad, S)

    if S.mode == "spop":
        w, P_out = _solve(C, M_full, C.flux, S, P_init)
        Q_eff = jnp.array(0.0)
        sc_resid = jnp.zeros(0)
    elif S.mode == "nepop":
        w, P_out = _solve(C, M_full, C.flux, S, P_init)
        y = w[-1]
        Q_eff = y / jnp.maximum(jnp.abs(C.neb_at_l0), 1e-300)
        sc_resid = jnp.zeros(0)
    else:
        av, yav = params[0], params[1]
        q_vec = _q_vector(C, av, yav, S.n_star)
        n_H = C.H_gas_idx.shape[0]
        use_sc = n_H > 0

        A_base = jnp.vstack([M_full * C.inv_sigma[:, jnp.newaxis], C.D_full])
        n_reg = C.D_full.shape[0]
        ATA_base = A_base.T @ A_base
        ATb_flux = A_base.T @ jnp.concatenate(
            [C.flux * C.inv_sigma, jnp.zeros(n_reg)])
        ATb_neb = A_base.T @ jnp.concatenate(
            [C.neb_shape * C.inv_sigma, jnp.zeros(n_reg)])

        def step(i, carry):
            Q_eff, _, _, P_cur = carry
            ATb = ATb_flux - Q_eff * ATb_neb
            if use_sc:
                rows, rhs = _sc_rows(C, Q_eff, av, q_vec, S.n_star, n_total)
                ATA = ATA_base + rows.T @ rows
                ATb = ATb + rows.T @ rhs
            else:
                ATA = ATA_base
            w, P_new = _linear_solve(C, ATA, ATb, S, P_cur)
            Q_new = jnp.sum(w[:S.n_star] * q_vec)
            damped = C.sc_damping * Q_new + (1.0 - C.sc_damping) * Q_eff
            Q_out = jnp.where((i > 0) & (Q_eff > 0), damped, Q_new)
            return (Q_out, w, Q_eff, P_new)

        P_seed = (jnp.zeros(n_total, dtype=bool) if P_init is None
                  else P_init.astype(bool))
        _, w, Q_used, P_out = jax.lax.fori_loop(
            0, S.n_neb_iter, step, (jnp.asarray(Q_init, dtype=float),
                                    jnp.zeros(n_total),
                                    jnp.asarray(Q_init, dtype=float),
                                    P_seed))
        Q_eff = Q_used

        if use_sc:
            rows, rhs = _sc_rows(C, Q_used, av, q_vec, S.n_star, n_total)
            sc_resid = rows @ w - rhs
        else:
            sc_resid = jnp.zeros(0)

        flux_eff = C.flux - C.neb_shape * Q_used
        fitted = M_full @ w
        return (jnp.concatenate([(flux_eff - fitted) * C.inv_sigma,
                                 C.D_full @ w, sc_resid]),
                w, Q_eff, M_full, P_out)

    fitted = M_full @ w
    residual = jnp.concatenate([
        (C.flux - fitted) * C.inv_sigma,
        C.D_full @ w,
        sc_resid,
    ])
    return residual, w, Q_eff, M_full, P_out

@partial(jax.jit, static_argnames=('S', 'n_par', 'central'))
def jacobian(C, params, S, n_par, central=True, diff_step=1e-3):
    """
    Central (or forward) differences, exploiting two pieces of structure.

    1. Only params 2 and 3 (sigma_star, v0_star) alter M_broad. The remaining
       n_par - 2 parameters reuse the centre point's broadened matrix, so of
       the 2*n_par+1 evaluations only 4 pay for an inverse FFT.

    2. The nebular fixed point runs the SAME number of iterations from the
       SAME cold start (Q = 0) as the residual the LM is minimising.

       This matters more than it looks. An earlier version converged Q* once
       at the centre and warm-started each perturbation with a single extra
       iteration — 1.5x faster, and wrong. The Jacobian then described a
       different map than `residual_only`, so the predicted reduction never
       matched the actual one, the gain ratio went to noise, and every LM step
       was rejected: `photopop` mode terminated at its initial guess. Keep the two
       maps identical; the fixed point is cheap now that the Gram matrix is
       hoisted out of the loop.

    3. The centre point's passive set P0 is reused as the warm start for every
       perturbed linear solve. That is exact — solve_active_set purges an
       illegal inherited set before it starts, so it reaches the same vertex
       wherever it starts. Measured max|J_warm - J_cold| = 0 exactly in all
       three modes, and it is cheaper in two of them (nepop Jacobian
       91 ms against 125 ms cold). Before the purge existed this warm start
       was NOT exact and inflated the photopop-mode A_V column by 19x.
    """
    eps = diff_step * jnp.maximum(jnp.abs(params), diff_step)
    E = jnp.diag(eps)
    M_broad = broaden(C, params[2], params[3], S.n_pad, S.n_pix, S.n_left)

    P0 = forward_from_broad(C, params, M_broad, S)[4]

    live = [i for i in S.live if i < n_par]
    no_fft = jnp.array([i for i in live if i not in (2, 3)])
    fft_p = jnp.array([i for i in live if i in (2, 3)])

    rp_n = jax.vmap(lambda i: forward_from_broad(C, params + E[i], M_broad, S, 0.0, P0)[0])(no_fft)
    rp_f = jax.vmap(lambda i: forward(C, params + E[i], S, 0.0, P0)[0])(fft_p)

    if central:
        rm_n = jax.vmap(lambda i: forward_from_broad(C, params - E[i], M_broad, S, 0.0, P0)[0])(no_fft)
        rm_f = jax.vmap(lambda i: forward(C, params - E[i], S, 0.0, P0)[0])(fft_p)
        d_n = (rp_n - rm_n) / (2.0 * eps[no_fft][:, jnp.newaxis])
        d_f = (rp_f - rm_f) / (2.0 * eps[fft_p][:, jnp.newaxis])
    else:
        r0 = forward_from_broad(C, params, M_broad, S, 0.0, P0)[0]
        d_n = (rp_n - r0) / eps[no_fft][:, jnp.newaxis]
        d_f = (rp_f - r0) / eps[fft_p][:, jnp.newaxis]

    J = jnp.zeros((n_par, rp_n.shape[1]))
    if no_fft.size:
        J = J.at[no_fft].set(d_n)
    if fft_p.size:
        J = J.at[fft_p].set(d_f)
    return J.T

@partial(jax.jit, static_argnames=('S', 'n_par', 'central'))
def lm_optimize(C, x0, bounds_lo, bounds_hi, S, n_par,
                max_iter=200, ftol=1e-5, xtol=1e-5, diff_step=1e-3,
                central=True):
    """
    Projected Levenberg-Marquardt, identical in structure and convergence
    logic to optimizer._lm_optimize, but driving the unified forward model so
    it serves all three photoionization modes.

    `max_iter` is a TRACED runtime value, not a static one — it only bounds a
    lax.while_loop, so it need not be baked into the compiled kernel. Together
    with the bounds, tolerances, starting point, reddening curve and
    regularisation strength (all already runtime arrays), this means the
    entire tunable surface of a fit can change without recompiling. Verified:
    max_iter 200 -> 150 -> 300 reuses the cached kernel and returns identical
    results. Only the problem's *shape* forces a rebuild — pixel count,
    template count, regularisation order, photoionization mode.

    On the stagnation rule. `n_stag` counts consecutive iterations whose chi2
    moved by less than ftol*100, and five of them end the fit as converged.
    It advances only on ACCEPTED steps. A rejected trial leaves chi2 exactly
    where it was, so counting rejections as stagnation turned the rule into a
    five-rejection kill switch wearing a convergence label — and a rejection is
    precisely the event LM answers by raising lambda, which is the recovery
    mechanism, not a sign of convergence. Measured: a nepop fit took ten
    accepted steps still improving at 2.2e-3 per iteration, hit five
    rejections while lambda climbed from 5.1e-8 to 4.1e-6, and returned
    "converged" having thrown away 0.12 in chi2 and 18% of A_V. Rejections are
    already bounded by the n_rej < 10 test in `cond`, which is the honest place
    for that limit and which now also reports convergence rather than
    "max iter".

    `step_norm` is measured against the width of the bounds box, not against
    abs(x). Against abs(x) it is scale-broken: sigma_star converges onto its lower
    bound of zero and yav is pinned at zero, so the denominator collapses to
    1e-10 and step_norm reads 1e7 or more, which made the `classical` test
    unreachable and left xtol with no effect at all.
    """
    max_iter = jnp.asarray(max_iter)
    def resid(p):
        return residual_only(C, p, S)

    r0 = resid(x0)
    chi2_0 = jnp.sum(r0 ** 2)
    init = (x0, jnp.array(1e-3), chi2_0, r0, 0, 1, False,
            jnp.ones(n_par), 0, 0, chi2_0)

    def cond(state):
        _, _, _, _, niter, _, conv, _, n_rej, _, _ = state
        return (~conv) & (niter < max_iter) & (n_rej < 10)

    def body(state):
        x, lam_lm, chi2, r, niter, nfev, _, scale, n_rej, n_stag, chi2_ref = state

        J = jacobian(C, x, S, n_par, central, diff_step)
        col_norms = jnp.sqrt(jnp.sum(J ** 2, axis=0))
        scale_new = jnp.maximum(scale, col_norms)
        scale_safe = jnp.maximum(scale_new, 1e-10)

        H = J.T @ J + lam_lm * jnp.diag(scale_safe ** 2)
        dp = jnp.linalg.solve(H, -(J.T @ r))
        x_cand = jnp.clip(x + dp, bounds_lo, bounds_hi)

        r_cand = resid(x_cand)
        chi2_cand = jnp.sum(r_cand ** 2)

        dp_actual = x_cand - x
        pred_red = chi2 - jnp.sum((r + J @ dp_actual) ** 2)
        actual_red = chi2 - chi2_cand
        gain = jnp.where(jnp.abs(pred_red) > 1e-30, actual_red / pred_red, 0.0)

        accept = chi2_cand < chi2
        x_out = jnp.where(accept, x_cand, x)
        chi2_out = jnp.where(accept, chi2_cand, chi2)
        r_out = jnp.where(accept, r_cand, r)
        n_rej_out = jnp.where(accept, 0, n_rej + 1)

        lam_new = jnp.where(
            accept & (gain > 0.75), jnp.maximum(lam_lm / 3.0, 1e-12),
            jnp.where(accept & (gain > 0.25), lam_lm,
                      jnp.minimum(lam_lm * 3.0, 1e10)))

        rel_chi2 = jnp.where(accept,
                             jnp.abs(actual_red) / jnp.maximum(chi2, 1e-30), 1.0)
        step_norm = jnp.max(jnp.abs(dp_actual)
                            / jnp.maximum(bounds_hi - bounds_lo, 1e-30))
        classical = accept & (rel_chi2 < ftol) & (step_norm < xtol) & (niter > 3)

        window = jnp.abs(chi2_ref - chi2_out) / jnp.maximum(chi2_ref, 1e-30)
        big = window > ftol * 100
        n_stag_out = jnp.where(accept, jnp.where(big, 0, n_stag + 1), n_stag)
        chi2_ref_out = jnp.where(big, chi2_out, chi2_ref)
        stagnant = (n_stag_out >= 5) & (niter > 3)

        n_live = len([i for i in S.live if i < n_par])
        return (x_out, lam_new, chi2_out, r_out, niter + 1,
                nfev + 2 * n_live + 1,
                classical | stagnant | (n_rej_out >= 10),
                scale_new, n_rej_out, n_stag_out, chi2_ref_out)

    final = jax.lax.while_loop(cond, body, init)
    return final[0], final[2], final[4], final[5], final[6]

@partial(jax.jit, static_argnames=('S', 'n_groups'))
def grid_chi2(C, v0_grid, sigma_star_init, av_init, sigma_gas_init,
              S, n_groups):
    """
    chi2 across a velocity grid, vmapped.

    Deliberately evaluated in "spop" mode regardless of the requested mode:
    this only picks the velocity basin for the LM, the weights are discarded,
    and adding the nebular fixed point here would cost 4x for no benefit.
    This matches the existing behaviour of SpectralFitter.grid_search_v0.
    """
    S_off = S._replace(mode="spop")

    def one(v0):
        p = jnp.concatenate([
            jnp.array([av_init, 0.0, sigma_star_init, v0]),
            jnp.tile(jnp.array([sigma_gas_init, v0]), n_groups),
        ])
        M_broad = broaden(C, sigma_star_init, v0, S.n_pad, S.n_pix, S.n_left)
        r = forward_from_broad(C, p, M_broad, S_off)[0]
        return jnp.sum(r ** 2)

    return jax.vmap(one)(v0_grid)
