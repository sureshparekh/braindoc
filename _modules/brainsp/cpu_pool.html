Source code for brainsp.cpu_pool

"""
Worker processes for every stage of a datacube fit: on the CPU cores, or on
several GPUs.

One process cannot use a many-core CPU for this problem (XLA's threads speed
up each spaxel's linear algebra a little, and a batch waits for its slowest
spaxel), so on a CPU the spaxels go to worker processes, each pinned to a
core of its own (fit_megacube_jax.pinned_to), never more than every core but
two (fit_megacube_jax.cpu_budget).

The pool sits behind the two functions every stage of the fit goes through,
fast_cube.fit_queue_fast and fast_cube.run_rows: while a pool is installed
(fast_cube.POOL), they split their spaxels into small chunks, the workers
run the very same function on each chunk, and the results come back in the
spaxels' order. Spaxels are fitted independently, so the fit is the one a
single process or a GPU gives, to rounding; the stars-alone kinematics, the
main fit, the weak lines, the population and nebular fits and the errors all
run on the workers.

Workers start once per cube fit and stay warm between the stages. Arrays a
call shares between its chunks (the templates, the line tables, the nebular
model) are sent to each worker once, by the hash of their contents, and kept
there; only the chunk's own rows travel with every call. A worker is
`python -m brainsp.pool_worker`, never a re-run of the caller's script.

On a machine with several GPUs a single cube is spread over them the same
way (Pool(gpu_slots=...)): one worker per GPU, each running the queue with
the lanes its card takes, while this process, on the CPU, only coordinates
(the cube-wide decisions, the statistics, the megacube).
"""

import hashlib
import os
import pickle
import queue
import struct
import subprocess
import sys
import threading

import numpy as np

# Arrays at least this large are sent once per worker and referred to by
# hash afterwards; smaller ones travel with each call.
SHARE_BYTES = 4096


class _Ref:
    """An array a worker already holds (or is sent alongside), by hash."""
    __slots__ = ("key", "jax")

    def __init__(self, key, jax):
        self.key, self.jax = key, jax

    def __reduce__(self):
        return (_Ref, (self.key, self.jax))


def _is_array(x):
    return isinstance(x, np.ndarray) or type(x).__module__.startswith(
        ("jax", "jaxlib"))

def strip(obj, blobs):
    """
    obj with every large array replaced by a _Ref; the arrays go to `blobs`
    (key -> (array, is_jax)). Tuples (named ones included), lists and dicts
    are walked; anything else is left as it is.
    """
    if _is_array(obj) and hasattr(obj, "shape"):
        is_jax = not isinstance(obj, np.ndarray)
        a = np.asarray(obj)
        if a.nbytes < SHARE_BYTES:
            return _Small(a, is_jax)
        a = np.ascontiguousarray(a)
        h = hashlib.blake2b(a.view(np.uint8).ravel(), digest_size=16)
        h.update(repr((a.shape, a.dtype.str, is_jax)).encode())
        key = h.hexdigest()
        blobs.setdefault(key, (a, is_jax))
        return _Ref(key, is_jax)
    if isinstance(obj, tuple) and hasattr(obj, "_fields"):
        return type(obj)(*[strip(x, blobs) for x in obj])
    if isinstance(obj, (tuple, list)):
        return type(obj)(strip(x, blobs) for x in obj)
    if isinstance(obj, dict):
        return {k: strip(v, blobs) for k, v in obj.items()}
    return obj




class _Small:
    """A small array, carried with the call."""
    __slots__ = ("a", "jax")

    def __init__(self, a, jax):
        self.a, self.jax = a, jax

    def __reduce__(self):
        return (_Small, (self.a, self.jax))

def dress(obj, store, to_jax):
    """strip, undone in the worker: _Ref from `store`, _Small converted."""
    if isinstance(obj, _Ref):
        return store[obj.key]
    if isinstance(obj, _Small):
        return to_jax(obj.a) if obj.jax else obj.a
    if isinstance(obj, tuple) and hasattr(obj, "_fields"):
        return type(obj)(*[dress(x, store, to_jax) for x in obj])
    if isinstance(obj, (tuple, list)):
        return type(obj)(dress(x, store, to_jax) for x in obj)
    if isinstance(obj, dict):
        return {k: dress(v, store, to_jax) for k, v in obj.items()}
    return obj

def send(stream, obj):
    data = pickle.dumps(obj, protocol=pickle.HIGHEST_PROTOCOL)
    stream.write(struct.pack("<Q", len(data)))
    stream.write(data)
    stream.flush()

def receive(stream):
    head = stream.read(8)
    if len(head) < 8:
        raise EOFError("the worker closed its pipe")
    (n,) = struct.unpack("<Q", head)
    data = bytearray()
    while len(data) < n:
        chunk = stream.read(n - len(data))
        if not chunk:
            raise EOFError("the worker closed its pipe")
        data += chunk
    return pickle.loads(bytes(data))




class _Task:
    __slots__ = ("fn", "args", "kwargs", "index", "finished", "result",
                 "error")

    def __init__(self, fn, args, kwargs, index, finished):
        self.fn, self.args, self.kwargs = fn, args, kwargs
        self.index, self.finished = index, finished
        self.result = self.error = None

    def done(self):
        self.finished.put(self)

class Pool:
    """
    n worker processes on the cores of cpu_budget(), one each.

    map(fn, calls) runs ``fn(*args, **kwargs)`` for every (args, kwargs) in
    `calls` on whichever worker is free, and returns the results in order.
    fn is a module-level function, sent by name. Thread-safe: the nebular
    fit runs its bins from several threads at once, and their chunks share
    the workers.
    """

    def __init__(self, n_workers=None, gpu_slots=None):
        """
        n_workers CPU workers, one core each; or, with gpu_slots (a list of
        scheduler.slots()' dicts), one worker per slot on its GPU, the CPU
        cores dealt out among them, and the cards described in self.gpus
        (core_count, memory_gb) for sizing their lanes.
        """
        from brainsp.fit_megacube_jax import pinned_to, worker_cores
        cores = worker_cores()
        base = dict(os.environ, OMP_NUM_THREADS="1", MKL_NUM_THREADS="1",
                    OPENBLAS_NUM_THREADS="1")
        parent = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
        base["PYTHONPATH"] = os.pathsep.join(
            [parent] + [p for p in base.get("PYTHONPATH", "").split(os.pathsep)
                        if p])
        self.kind = "gpu" if gpu_slots else "cpu"
        self.n = len(gpu_slots) if gpu_slots else int(n_workers)
        self._tasks = queue.Queue()
        self._procs, self._threads = [], []
        self._state_version = 0
        self._state = None
        self._closed = False
        self._lock = threading.Lock()
        self.gpus = []
        self.set_state()
        for k in range(self.n):
            if self.kind == "cpu":
                env = dict(base, JAX_PLATFORMS="cpu",
                           BRAIN_PRECOMPILE_THREADS="1")
                mine = {cores[k % len(cores)]}
            else:
                slot = gpu_slots[k]
                env = dict(base, JAX_PLATFORMS="cuda", BRAIN_POOL_HELLO="1",
                           CUDA_VISIBLE_DEVICES=str(slot["gpu"]))
                if slot.get("fraction"):
                    env["XLA_PYTHON_CLIENT_MEM_FRACTION"] = str(
                        slot["fraction"])
                mine = set(cores[k::self.n] or cores)
            proc = subprocess.Popen(
                [sys.executable, "-m", "brainsp.pool_worker"],
                stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                stderr=subprocess.PIPE, env=env,
                preexec_fn=(pinned_to(next(iter(mine))) if len(mine) == 1
                            else (lambda m=mine: os.sched_setaffinity(0, m))))
            self._procs.append(proc)
        for proc in self._procs:
            if self.kind == "gpu":
                # each GPU worker says what its card is before any work
                status, info = receive(proc.stdout)
                if status != "hello":
                    self.close()
                    raise RuntimeError(f"A GPU worker did not start: {info}")
                self.gpus.append(info)
            t = threading.Thread(target=self._feed, args=(proc,), daemon=True)
            t.start()
            self._threads.append(t)

    def lanes(self, n_pix, n_fine, n_ssp, n_gas, n_params):
        """Lanes for each worker's queue: one on a CPU core; on the GPUs,
        what the smallest card takes (estimate_lanes_fast)."""
        if self.kind == "cpu":
            return 1
        import types
        from brainsp.fit_megacube_jax import estimate_lanes_fast
        return min(estimate_lanes_fast(
            g["memory_gb"], n_pix, n_fine, n_ssp, n_gas, n_params,
            device=types.SimpleNamespace(core_count=g["core_count"]))
            for g in self.gpus)

    def set_state(self):
        """
        Send the module state the fit functions read (fast_cube.NATIVE, the
        native-pixel operator; emission_lines.LINE_LSF, the instrument's
        line width) to every worker before its next call. Called whenever
        the fit sets them.
        """
        from brainsp import emission_lines, fast_cube
        nat = fast_cube.NATIVE
        nat = (None if nat is None else
               {k: (np.asarray(v) if hasattr(v, "shape") else v)
                for k, v in nat._asdict().items()})
        lsf = (None if emission_lines.LINE_LSF is None else
               tuple(np.asarray(a) for a in emission_lines.LINE_LSF))
        with self._lock:
            self._state = (nat, lsf)
            self._state_version += 1



    def _feed(self, proc):
        known = set()
        version = 0
        while True:
            task = self._tasks.get()
            if task is None:
                return
            try:
                with self._lock:
                    state, v = self._state, self._state_version
                if v != version:
                    # a new fit: the worker drops the arrays of the last
                    # (they are sent again as needed), so a pool kept over
                    # many cubes does not grow
                    send(proc.stdin, ("state", state))
                    version = v
                    known = set()
                blobs = {}
                args = strip(task.args, blobs)
                kwargs = strip(task.kwargs, blobs)
                new = {k: b for k, b in blobs.items() if k not in known}
                send(proc.stdin, ("call", task.fn.__module__,
                                  task.fn.__name__, args, kwargs, new))
                known.update(new)
                status, value = receive(proc.stdout)
                if status == "ok":
                    task.result = value
                else:
                    task.error = RuntimeError(
                        f"A CPU worker failed in {task.fn.__name__}:\n{value}")
            except Exception as exc:          # the worker died
                err = b""
                try:
                    err = proc.stderr.read()[-3000:]
                except Exception:
                    pass
                task.error = RuntimeError(
                    f"A CPU worker stopped ({exc}).\n"
                    f"{err.decode(errors='replace')}")
                task.done()
                self._fail_rest(task.error)
                return
            task.done()

    def _fail_rest(self, error):
        """A dead worker takes no more tasks; the others go on."""
        alive = sum(t.is_alive() for t in self._threads) - 1
        if alive <= 0:
            while True:
                try:
                    t = self._tasks.get_nowait()
                except queue.Empty:
                    return
                if t is not None:
                    t.error = error
                    t.done()

    def map(self, fn, calls, on_done=None):
        """
        ``fn(*args, **kwargs)`` for each (args, kwargs) of `calls`, on the
        workers; the results in the order of `calls`. on_done(i, result) is
        called here, in the caller's thread, as each call finishes.
        """
        if self._closed:
            raise RuntimeError("the CPU worker pool is closed")
        finished = queue.Queue()
        tasks = [_Task(fn, a, k, i, finished)
                 for i, (a, k) in enumerate(calls)]
        for t in tasks:
            self._tasks.put(t)
        error = None
        for _ in tasks:
            t = finished.get()
            if t.error is not None:
                error = error or t.error
            elif on_done is not None and error is None:
                on_done(t.index, t.result)
        if error is not None:
            raise error
        return [t.result for t in tasks]

    def close(self):
        if self._closed:
            return
        self._closed = True
        for _ in self._procs:
            self._tasks.put(None)
        for proc in self._procs:
            try:
                send(proc.stdin, ("stop",))
            except Exception:
                pass
        for proc in self._procs:
            try:
                proc.wait(timeout=10)
            except Exception:
                proc.kill()

class keep_warm:
    """
    ``with keep_warm():`` -- every cube fitted inside shares one pool, closed
    on leaving. Spaxels are fitted independently, so a cube's fit does not
    depend on what the workers fitted before it.
    """

    def __enter__(self):
        global KEEP
        self._was = KEEP
        KEEP = True
        return self

    def __exit__(self, *exc):
        global KEEP
        KEEP = self._was
        if not KEEP:
            from brainsp import fast_cube
            if fast_cube.POOL is not None:
                fast_cube.POOL.close()
                fast_cube.POOL = None
        return False

def chunks(n, n_workers, largest=8, per_worker=4):
    """
    Row ranges for n spaxels over n_workers: about `per_worker` chunks per
    worker (so the last ones finish together), of at most `largest` rows.
    """
    size = max(1, min(int(largest),
                      -(-n // (per_worker * max(1, n_workers)))))
    return [(i, min(n, i + size)) for i in range(0, n, size)]
