"""
NewBrain I/O Utilities
Loading spectra, SSP libraries, masks, and writing output.

Everything here is file handling and numpy except the resampling of a
template onto the fitted grid, which is `spectral_engine` and therefore JAX.
That import is made inside the two functions that resample, so that asking
this module what a library covers, or what a spectrum contains, does not
start a JAX backend. The desktop interface asks both of those questions
before a device has been chosen, and a backend started in the process that
asks is a backend the fit can no longer choose.
"""

import os
import numpy as np
import logging

logger = logging.getLogger("NEWBRAIN")

def load_spectrum(file_path, has_errors=True):
    """
    Load observed spectrum from text file.

    Supports 2-column (wavelength, flux) or 3-column (wavelength, flux, error).

    Parameters
    ----------
    file_path : str
        Path to spectrum file.
    has_errors : bool
        Whether to look for a 3rd column with errors.

    Returns
    -------
    wavelength : 1D array
    flux : 1D array
    err : 1D array or None
    """
    data = np.loadtxt(file_path)
    wl, flux = data[:, 0], data[:, 1]

    err = None
    flag = None

    if has_errors and data.shape[1] >= 3:
        err = np.abs(data[:, 2])
        bad = err <= 0
        if np.any(bad):
            median_err = np.median(err[~bad]) if np.any(~bad) else 1.0
            err[bad] = median_err * 10.0

    if data.shape[1] >= 4:
        flag = data[:, 3].astype(int)
        n_flagged = np.sum(flag > 0)
        if n_flagged > 0:
            logger.info(f"  Spectrum flags: {n_flagged}/{len(flag)} pixels "
                         f"flagged (flag>0)")

    return wl, flux, err, flag

def load_bases_dir(bases_file):
    """
    Parse a BasesDir-format SSP library file.

    Format:
        Line 1: N_base
        Lines 2+: filename  age_yr  metallicity  short_name

    Parameters
    ----------
    bases_file : str
        Path to BasesDir file.

    Returns
    -------
    entries : list of dict
        Each dict has: 'filename', 'age_gyr', 'metallicity', 'short_name'
    """
    entries = []
    with open(bases_file) as f:
        n_base = int(f.readline().split()[0])
        for line in f:
            parts = line.split()
            if len(parts) >= 4:
                entries.append({
                    'filename': parts[0],
                    'age_gyr': float(parts[1]) / 1e9,
                    'metallicity': float(parts[2]),
                    'short_name': parts[3],
                    'mstars': float(parts[4]) if len(parts) > 4 else 1.0,
                    'yav_flag': int(parts[5]) if len(parts) > 5 else 0
                })
    logger.info(f"  BasesDir: {len(entries)} SSPs from {bases_file}")
    return entries

def scan_ssp_folder(folder, suffix='_baseFe'):
    """
    Scan folder for SSP files matching a suffix (fallback if no BasesDir).

    Returns list of dicts with filename and placeholder metadata.
    """
    files = sorted([f for f in os.listdir(folder) if f.endswith(suffix)])
    entries = [{
        'filename': f,
        'age_gyr': -1.0,
        'metallicity': -1.0,
        'short_name': f[:16],
        'mstars': 1.0,
        'yav_flag': 0
    } for f in files]
    logger.info(f"  Folder scan: {len(entries)} SSPs from {folder}")
    return entries



LYMAN_LIMIT_A = 911.76


def _first_last_wavelength(path):
    """First and last wavelength in a two-column template file, or None."""
    first = last = None
    with open(path) as fh:
        for raw in fh:
            line = raw.split("#")[0].strip()
            if not line:
                continue
            try:
                w = float(line.split()[0])
            except (ValueError, IndexError):
                continue
            if first is None:
                first = w
            last = w
    if first is None:
        return None
    return first, last

def library_coverage(ssp_folder, bases_info):
    """
    What a template library covers, measured from its own files.

    Whether a library can support the nebular modes is arithmetic rather
    than preference. Q(H) is integrated from the raw spectra shortward of
    the Lyman limit, so a library whose files begin redward of it emits, by
    construction, no ionising photons at all, and its nebular continuum
    would be identically zero while the fit ran to completion and looked
    converged. A hardcoded list of which libraries qualify goes stale the
    first time one is regenerated or a variant is shipped, so the question
    is asked of the files.

    Only the first and last wavelength of each file are read, which is
    cheap enough to run before a fit rather than partway through one.

    Only the nebular modes are affected. spop and espop fit the emission
    lines from the catalogue with free amplitudes and predict nothing, so
    they need no ionising photon budget and are available for every
    library.

    Returns a dict with the bluest and reddest wavelength across the
    library, how many of its files reach below the Lyman limit, and the
    modes that are therefore available for it.
    """
    blue = red = None
    n_ionising = 0
    unreadable = []
    for entry in bases_info:
        path = os.path.join(ssp_folder, entry['filename'])
        try:
            span = _first_last_wavelength(path)
        except OSError:
            span = None
        if span is None:
            unreadable.append(entry['filename'])
            continue
        lo, hi = span
        blue = lo if blue is None else min(blue, lo)
        red = hi if red is None else max(red, hi)
        if lo < LYMAN_LIMIT_A:
            n_ionising += 1

    n_read = len(bases_info) - len(unreadable)
    modes = ["spop", "espop"]
    if n_ionising > 0:
        modes += ["nepop", "photopop"]
    return {
        'folder': ssp_folder,
        'blue': blue,
        'red': red,
        'n_files': len(bases_info),
        'n_read': n_read,
        'n_ionising': n_ionising,
        'unreadable': unreadable,
        'modes': modes,
        'partial': 0 < n_ionising < n_read,
    }

def describe_coverage(cov):
    """One or two lines stating what a library covers and what it supports."""
    if cov['blue'] is None:
        return f"  library      : {cov['folder']} — no readable template files"
    lines = [f"  library      : {cov['folder']} covers "
             f"{cov['blue']:.1f}-{cov['red']:.1f} A, "
             f"{cov['n_ionising']}/{cov['n_read']} files reach below "
             f"{LYMAN_LIMIT_A} A",
             f"  modes here   : {', '.join(cov['modes'])}"]
    if 'nepop' not in cov['modes']:
        lines.append(f"                 the nebular modes need coverage below "
                     f"{LYMAN_LIMIT_A} A, which this library does not have")
    if cov['partial']:
        lines.append("                 only some files reach it, so Q(H) "
                     "would be integrated over a biased subset of the library")
    if cov['unreadable']:
        lines.append(f"                 {len(cov['unreadable'])} file(s) could "
                     f"not be read, first: {cov['unreadable'][0]}")
    return "\n".join(lines)

def load_native_template(path):
    """
    Read one template file at the sampling the library wrote it on.

    Nothing is resampled, normalised or trimmed. This is the form a
    line-spread-function correction has to see: the kernel that separates two
    line-spread functions is a width in Angstrom on the grid the library was
    tabulated on, and a template that has already been mapped onto the fitted
    log-lambda grid no longer has that grid to be convolved on.

    Returns
    -------
    wavelength : 1D array
        The template's own wavelength axis, in Angstrom.
    flux : 1D array
        The template's own flux density, in the library's own units.
    """
    data = np.loadtxt(path)
    return data[:, 0], data[:, 1]

def resample_template(ssp_wl, ssp_fl, target_wl):
    """
    Map one template onto the fitted grid, conserving flux.

    Target pixels outside the template's own span stay at exactly zero, which
    is the signal the coverage test downstream reads to decide which pixels no
    template reaches. Because the span is a property of the file and not of
    anything done to its flux, a template convolved before this step covers
    exactly the pixels it covered before.
    """
    from brainsp.spectral_engine import flux_conserving_resample

    in_range = (target_wl >= ssp_wl[0]) & (target_wl <= ssp_wl[-1])
    out = np.zeros(len(target_wl))
    if np.any(in_range):
        out[in_range] = flux_conserving_resample(
            ssp_wl, ssp_fl, target_wl[in_range]
        )
    return out

def load_ssp_templates(ssp_folder, bases_info, target_wl, norm_wavelength,
                       exact_norm=True):
    """
    Load all SSP templates and interpolate them onto the target wavelength grid.

    Parameters
    ----------
    ssp_folder : str
        Path to SSP files.
    bases_info : list of dict
        SSP library entries (from load_bases_dir or scan_ssp_folder).
    target_wl : 1D array
        Target wavelength grid to interpolate onto.
    norm_wavelength : float
        Normalization wavelength.

    Returns
    -------
    matrix : 2D array (N_pix, N_ssp)
        Normalized SSP template matrix.
    names : list of str
        SSP filenames.
    ages : 1D array
        SSP ages in Gyr.
    metals : 1D array
        SSP metallicities.
    short_names : list of str
        Short names for output.
    """
    from brainsp.spectral_engine import normalize_at_wavelength

    matrix = []
    names, ages, metals, short_names = [], [], [], []
    mstars, yav_flags, fbase_norms = [], [], []

    for entry in bases_info:
        fpath = os.path.join(ssp_folder, entry['filename'])
        if not os.path.isfile(fpath):
            logger.warning(f"  SSP not found: {fpath}")
            continue

        ssp_wl, ssp_fl = load_native_template(fpath)
        ssp_interp = resample_template(ssp_wl, ssp_fl, target_wl)

        if exact_norm:
            _i0 = int(np.argmin(np.abs(np.asarray(target_wl) - norm_wavelength)))
            factor = float(ssp_interp[_i0])
            if not np.isfinite(factor) or factor <= 0:
                factor = float(np.median(ssp_interp[ssp_interp > 0])) \
                         if np.any(ssp_interp > 0) else 1.0
            ssp_interp = ssp_interp / factor
        else:
            ssp_interp, factor = normalize_at_wavelength(
                ssp_interp, target_wl, norm_wavelength
            )

        matrix.append(ssp_interp)
        names.append(entry['filename'])
        ages.append(entry['age_gyr'])
        metals.append(entry['metallicity'])
        short_names.append(entry['short_name'])
        
        mstars.append(entry.get('mstars', 1.0))
        yav_flags.append(entry.get('yav_flag', 0))
        fbase_norms.append(factor)

    matrix = np.array(matrix).T
    
    return matrix, names, np.array(ages), np.array(metals), short_names, np.array(mstars), np.array(yav_flags), np.array(fbase_norms)

def read_mask_regions(mask_file):
    """
    The regions of a wavelength mask file, as [(start, end, weight, label)].

    Two layouts are read:

    * The .sm / .gm layout (BRAIN's own when it saves one): a first
      line holding the number of regions N, then N lines of

          start  end  weight  [label ...]

      Exactly N lines are regions. Whatever follows is free text, and
      published mask files use it: the standard SDSS emission-line mask ends with an *example* that gives Ca II K and the
      G band twice the weight. Reading every numeric line, as this function
      once did, applied that example and up-weighted both windows in every
      fit that used the file.

    * A plain list, with no count: every line that starts with two numbers
      is a region. Lines starting with '#' are comments in both layouts.

    The weight is optional and defaults to 0 (the region is excluded). A
    third field that is not a number is the start of the label, and the
    region is excluded as usual.

    Returns (regions, n_skipped): n_skipped counts lines that should have
    been regions and did not start with two numbers.
    """
    with open(mask_file) as handle:
        lines = [l.strip() for l in handle]
    lines = [l for l in lines if l and not l.startswith("#")]
    limit = None
    if lines:
        head = lines[0].split()
        if len(head) == 1:
            try:
                limit = int(float(head[0]))
                lines = lines[1:]
            except ValueError:
                pass
    if limit is not None:
        lines = lines[:limit]

    regions, n_bad = [], 0
    for line in lines:
        parts = line.split()
        if len(parts) < 2:
            n_bad += 1 if limit is not None else 0
            continue
        try:
            w0, w1 = float(parts[0]), float(parts[1])
        except ValueError:
            n_bad += 1
            continue
        weight, rest = 0.0, parts[2:]
        if rest:
            try:
                weight = float(rest[0])
                rest = rest[1:]
            except ValueError:
                pass
        regions.append((min(w0, w1), max(w0, w1), max(weight, 0.0),
                        " ".join(rest)))
    return regions, n_bad

def write_mask_regions(path, regions):
    """
    Write regions as a .sm mask file: the count, then one per line.

    `regions` holds (start, end, weight) or (start, end, weight, label). The
    file reads back the same.
    """
    with open(path, "w") as handle:
        handle.write(f"{len(regions)}\n")
        for region in regions:
            lo, hi, w = region[:3]
            label = region[3] if len(region) > 3 and region[3] else ""
            handle.write(f"{lo:10.2f} {hi:10.2f} {w:6.2f}"
                         + (f"   {label}" if label else "") + "\n")

def load_mask(mask_file, wavelength):
    """
    Load a wavelength mask as a per-pixel weight on `wavelength`.

    The file is read by read_mask_regions, which says which layouts are
    understood. A pixel no region covers has weight 1; a region sets the
    pixels inside it to its weight (0 excludes them, above 1 makes them count
    for more). Later regions override earlier ones.

    Parameters
    ----------
    mask_file : str or None
        Path to mask file.
    wavelength : 1D array
        Wavelength grid.

    Returns
    -------
    weight : 1D float array
    """
    weight = np.ones(len(wavelength), dtype=float)
    if mask_file is None or not os.path.isfile(mask_file):
        return weight

    regions, n_bad = read_mask_regions(mask_file)
    for lo, hi, w, _ in regions:
        weight[(wavelength >= lo) & (wavelength <= hi)] = w
    n_reg = len(regions)

    n_masked = int(np.sum(weight == 0))
    n_up = int(np.sum(weight > 1))
    msg = (f"  Mask: {n_reg} regions, {n_masked}/{len(wavelength)} pixels "
           f"masked ({100 * n_masked / len(wavelength):.1f}%)")
    if n_up:
        msg += f", {n_up} up-weighted"
    logger.info(msg)
    if n_bad:
        logger.warning(
            f"  {n_bad} lines of {mask_file} do not start with two "
            f"wavelengths and were skipped.")
    if n_reg == 0:
        logger.warning(
            f"  {mask_file} produced no regions, so nothing is masked. "
            f"Each region needs a line 'start end' in Angstrom, with an "
            f"optional weight and label; a .sm file starts with the "
            f"number of regions.")
    return weight

def stellar_component_flags(bases_info, names):
    """
    Which of the loaded templates are stellar populations, said by the library.

    The template catalogue's KIND column is the declared answer: 'ssp' for a
    stellar population and something else -- 'powerlaw' -- for a component
    that is not one. brain_settings.read_catalogue carries that word onto
    every entry, and this reads it back. Nothing here looks at a filename, a
    short name, an age or the shape of a spectrum, because all four are
    conventions a new library is free to break, and breaking them would move
    a mass silently.

    The lookup is by filename rather than by position: load_ssp_templates
    skips an entry whose file is missing, so the loaded arrays can be shorter
    than bases_info and a positional flag would then be attached to the wrong
    component. A name that no entry describes is a programming error and is
    raised rather than assumed stellar.

    Libraries loaded through load_bases_dir or scan_ssp_folder carry no KIND
    at all; those formats cannot express a non-stellar component, so every
    entry is a stellar population and the flags come back all True.
    """
    kinds = {entry['filename']: str(entry.get('kind', 'ssp')).lower()
             for entry in bases_info}
    missing = [n for n in names if n not in kinds]
    if missing:
        raise ValueError(
            f"{len(missing)} loaded template(s) are in no catalogue entry, so "
            f"whether they are stellar populations cannot be said: "
            f"{', '.join(missing[:5])}")
    return np.array([kinds[n] == 'ssp' for n in names], dtype=bool)

def population_statistics(weights_star, norm_factor, av, yav, ssp_yav_flags,
                          ssp_mstars, ssp_fbase_norms, k_norm, is_stellar):
    """
    The per-template light and mass quantities, for one spectrum or a batch.

    x_j, Mini_j and Mcor_j are what every population product is built from --
    the summary file's percentage columns, and every population map of a
    megacube. They are computed here, once, so that a map value and that
    spaxel's summary line cannot disagree: they are the same arithmetic on
    the same weights rather than two implementations of the same formula.

    A weight is the contribution of a template to the *reddened* spectrum at
    the normalisation wavelength. Turning it into a mass means undoing the
    extinction that template suffered -- A_V plus YAV for the templates
    flagged as young -- and dividing by the library's flux per unit initial
    mass at that wavelength, which is what ssp_fbase_norms holds. Leaving the
    extinction out makes the dustiest populations look lighter than they are,
    and the error grows with A_V, so it does not show up as an offset.

    `is_stellar` says, per template, whether that division means anything.
    A featureless power law is not a stellar population: it has no stars, no
    surviving stellar fraction and no mass, and its fbase_norm is the flux of
    an arbitrarily scaled file rather than a flux per solar mass. Its Mini_j
    and Mcor_j are therefore exactly zero, it is out of Mini_tot and
    Mcor_tot, and its mass percentage columns read 0. The magnitude of what
    that removes is set by nothing physical: the same fit, with the same
    weights, reports whatever mass the power-law file's normalisation
    happens to imply, so the error it fixes is unbounded rather than small.

    Its LIGHT is untouched and stays in x_j, in `light` and in flux_tot. The
    power law genuinely does contribute light, and a component that is 30 per
    cent of the light and 0 per cent of the mass is the correct reading of an
    AGN continuum sitting on top of a stellar population. The two columns are
    normalised over different sets on purpose: x_j over every component,
    Mini_j and Mcor_j over the stellar ones, which is why a summary file can
    show a non-zero x_j against a zero Mcor_j without either being a bug.

    The ionising photon budget is deliberately not touched here. Q(H)_total
    is built in brainv1 from the same expression, but what it multiplies is
    photons per unit of a template's own normalisation, and a power law does
    ionise. Zeroing a mass is not a statement about photons.

    The leading axis is optional. Pass a (n_ssp,) weight vector and every
    returned array is one-dimensional; pass (n_spec, n_ssp) and the totals
    come back as (n_spec,). That is how the batched cube path and the single
    spectrum share one implementation.
    """
    weights = np.asarray(weights_star, dtype=np.float64)
    flags = np.asarray(ssp_yav_flags) != 0
    mstars = np.asarray(ssp_mstars, dtype=np.float64)
    fbase = np.asarray(ssp_fbase_norms, dtype=np.float64)
    stellar = np.asarray(is_stellar, dtype=bool)

    if stellar.shape != (weights.shape[-1],):
        raise ValueError(
            f"is_stellar has shape {stellar.shape} for {weights.shape[-1]} "
            f"templates; a misaligned flag would zero the mass of the wrong "
            f"component and nothing downstream could tell")

    norm = np.asarray(norm_factor, dtype=np.float64)[..., None]
    av_total = (np.asarray(av, dtype=np.float64)[..., None]
                + np.asarray(yav, dtype=np.float64)[..., None] * flags)
    unreddened = 10.0 ** (0.4 * av_total * float(k_norm))

    mini_j = np.where(stellar,
                      weights * norm * unreddened / np.maximum(fbase, 1e-30),
                      0.0)
    mcor_j = mini_j * mstars

    light = np.sum(weights, axis=-1)
    mini_tot = np.sum(mini_j, axis=-1)
    mcor_tot = np.sum(mcor_j, axis=-1)

    def percent(part, total):
        safe = np.where(total > 0, total, 1.0)
        return np.where(total[..., None] > 0, part / safe[..., None] * 100.0,
                        0.0)

    x_j = percent(weights, light)

    return {
        'x_j': x_j,
        'mini_j': mini_j,
        'mcor_j': mcor_j,
        'mini_pct': percent(mini_j, mini_tot),
        'mcor_pct': percent(mcor_j, mcor_tot),
        'flux_tot': light * np.asarray(norm_factor, dtype=np.float64),
        'mini_tot': mini_tot,
        'mcor_tot': mcor_tot,
    }

def deviation_percent(flux_observed, model, fitted):
    """
    adev: the mean of abs(observed - model) / observed over the fitted pixels.

    A deviation relative to the observed flux is undefined where that flux is
    not positive, so those pixels are left out of the mean and counted
    separately rather than being given an arbitrary value. They are still
    fitted and still counted in the chi-square; this is a reporting decision
    and not a change to what was minimised.

    Works on one spectrum or on a batch: the last axis is wavelength.
    """
    flux = np.asarray(flux_observed, dtype=np.float64)
    resid = flux - np.asarray(model, dtype=np.float64)
    good = np.asarray(fitted, dtype=bool)
    usable = good & (flux > 0)
    n_used = np.sum(usable, axis=-1)
    total = np.sum(np.where(usable, np.abs(resid) / np.where(usable, flux, 1.0),
                            0.0), axis=-1)
    adev = 100.0 * total / np.maximum(n_used, 1)
    return adev, n_used, np.sum(good, axis=-1)

def effective_pixel_count(inv_sigma):
    """
    Nl_eff: the pixels of non-zero weight, the ones that entered the chi2.

    A pixel is in the chi2 exactly when its inverse sigma is above zero, so
    this is the count the summary file divides by and the count the cube's
    Nl_eff plane carries. Works on one spectrum or on a batch: the last axis
    is wavelength.
    """
    return np.sum(np.asarray(inv_sigma) > 0, axis=-1)

def chi2_per_effective_pixel(chi2, n_eff):
    """
    chi2 divided by Nl_eff, which is the convention BRAIN reports.

    The single-spectrum summary, the cube's CHI2 extension and its PoPBins
    ChiSqrt plane all quote this one number, computed here so that they
    cannot drift apart: a reader comparing a spaxel of a cube against the
    same spectrum fitted on its own is comparing the same quantity. Nothing
    is subtracted for the free parameters -- that is the other convention,
    and it is reported separately under its own name.

    Nl_eff is floored at one so that a spectrum which somehow reached this
    point with no fitted pixel returns its raw chi2 instead of an infinity.
    Scalars and batches alike.
    """
    return np.asarray(chi2, dtype=np.float64) / np.maximum(n_eff, 1)





def _ensure_folder(filepath):
    """
    Make the folder an output is about to be written into.

    The output paths carry a folder now rather than naming a bare file in
    whatever directory the process started in, and a folder that has never
    been used does not exist in a fresh checkout. Creating it here rather
    than at every call site means a path that was accepted as a setting is a
    path that can be written.
    """
    folder = os.path.dirname(str(filepath))
    if folder:
        os.makedirs(folder, exist_ok=True)

def write_summary(filepath, config, params, weights, w_errors, n_star,

    Write the fit summary to a text file.

    `vd_min` keeps its historical meaning and its exact value: the dispersion
    the kinematic kernel applied to the template library. Everything the
    instrument contributes to that number is reported separately, in the
    instrumental resolution block, so that no reader has to assume vd_min is
    astrophysical and no reader is left with a bare 0.00 to interpret.

    The gas kinematics are written here too, per kinematic group, because
    until now they reached the log and nothing else: a summary file recording
    an espop fit stated the emission-line fluxes and said nothing about the
    velocity or the width they were fitted with. The two dispersions in this
    file follow DIFFERENT conventions and every line that carries one says
    which: sigma_star is INTRINSIC, the instrument taken out in quadrature,
    while sigma_gas and vd_min are OBSERVED, with the instrument still in
    them. See _write_gas_kinematics_block.

    The emission-line block states the wavelength scale the lines were placed
    on, for the same reason every dispersion in this file states its
    convention: a flux table that does not say whether its rest wavelengths
    were air or vacuum is a table whose gas velocity is ambiguous by 83 km/s,
    and nothing else in the file would reveal it. The scale is a declaration
    about the observed spectrum, made in the settings file and copied here;
    it is not a measurement and not something BRAIN infers.

    `Mini_tot`, `Mcor_tot` and `Q(H)_total` are apparent, not absolute. They
    are the observed flux normalisation divided by the library's flux per
    solar mass, which is a mass only once the flux has been turned into a
    luminosity. No distance enters this code anywhere, so the missing factor
    is flux_unit * 4*pi*d^2 / L_sun and the reader has to supply it. These
    were previously labelled "(Msun)", which was wrong by whatever that factor
    happens to be — about 3.5 dex for a MaNGA spaxel at 100 Mpc, and
    distance-dependent, so not even a fixed offset across a sample.

    The factor carries no template index, so every fraction built from these
    numbers was and remains correct: the per-template percentage columns are
    bit-identical under a rescaling of the input flux. Only the absolute
    normalisation was ever mislabelled, which is why this is a change of
    wording and not of arithmetic.

    x_j, Mini_j(%) and Mcor_j(%) are taken from fit_stats as they were
    computed, not rebuilt here from the weights and the totals. They were
    rebuilt here once, from the same formulae, and a second implementation of
    a formula is a second place for it to change: the whole point of
    population_statistics is that a megacube's population map and this file's
    column are one arithmetic. A power law reaches this table with a non-zero
    x_j and a Mini_j(%) and Mcor_j(%) of exactly zero, because it carries
    light and has no stars; see population_statistics.
    """
    _ensure_folder(filepath)

    av = params[0]
    yav = params[1]
    sig_s = params[2]
    v0_s = params[3]

    weights = np.asarray(weights)
    w_errors = np.asarray(w_errors) if w_errors is not None else None

    chi2_red = fit_stats['chi2_red']
    chi2_data_red = fit_stats.get('chi2_data_red', chi2_red)
    adev = fit_stats['adev']
    flux_tot = fit_stats['flux_tot']
    mini_tot = fit_stats['mini_tot']
    mcor_tot = fit_stats['mcor_tot']

    fracs = np.asarray(fit_stats['x_j'], dtype=np.float64)
    mini_pct = np.asarray(fit_stats['mini_pct'], dtype=np.float64)
    mcor_pct = np.asarray(fit_stats['mcor_pct'], dtype=np.float64)

    with open(filepath, 'w') as f:
        f.write("###################################################################\n")
        f.write("## OUTPUT of BRAIN\n")
        f.write("###################################################################\n\n")

        f.write("## Some input info\n")
        f.write(f"{config.obs_spectrum_file:<40} [arq_obs]\n")
        _base_src = (config.bases_dir_file
                     or (f"{config.ssp_folder} (selected)"
                         if getattr(config, 'bases_entries', None)
                         else 'folder scan'))
        f.write(f"{_base_src:<40} [arq_base]\n")
        f.write(f"{config.mask_file or 'None':<40} [arq_masks]\n")
        f.write(f"{n_star:<40} [N_base]\n")
        f.write(f"{np.sum(ssp_yav_flags):<40} [N_YAV_components = # of components with extra extinction!]\n")
        f.write(f"{config.reddening_law:<40} [red_law_option]\n")
        f.write(f"{config.norm_wavelength:<40g} [normalisation_wavelength (A)]\n")
        # Every run states its mode here. It used to be written only inside
        # the photoionization block, which exists only in nepop and photopop,
        # so an spop or espop summary never said which it was and anything
        # reading the file back -- the plot title among them -- had to guess.
        f.write(f"{config.photoionization_mode:<40} [photoionization_mode]\n")
        ebv = float(getattr(config, 'galactic_ebv', 0.0) or 0.0)
        law = getattr(config, 'galactic_law', 'CCM')
        rv = float(getattr(config, 'galactic_rv', 3.1) or 3.1)
        how = f", {law}, R_V = {rv:g}" if ebv else ""
        f.write(f"{ebv:<40g} [E(B-V) Galactic, removed before fitting{how}]\n")
        z = float(getattr(config, 'spectrum_redshift', 0.0) or 0.0)
        who = "the user" if getattr(config, 'spectrum_in_rest_frame', True) else "BRAIN"
        f.write(f"{who:<40} [removed spectrum_redshift = {z:g}]\n\n")

        f.write("## Synthesis Results - Best model ##\n\n")

        f.write(f" {chi2_red:>12.5E}                             "
                f"[chi2/Nl_eff (data + regularisation + self-consistency rows, "
                f"over the Nl_eff pixels the fit ended on)]\n")
        f.write(f" {chi2_data_red:>12.5E}                             "
                f"[chi2_data/Nl_eff (data rows only, same Nl_eff pixels)]\n")
        f.write(f" {adev:>12.5f}                             [adev (%)]\n\n")
        entries = getattr(config, 'bases_entries', None)
        if entries:
            stellar = stellar_component_flags(entries, list(ssp_names))
            if not np.all(stellar):
                # The AGN templates are nearly collinear in the optical; only
                # their summed light is a measurement, so it is stated once.
                f.write(f" {float(np.sum(fracs[~stellar])):>12.5f}"
                        f"                             [x_AGN (%), light of all "
                        f"{int(np.sum(~stellar))} featureless (AGN) components "
                        f"summed; the split between them is not a measurement]\n\n")

        f.write(f" {flux_tot:>12.5E}                             [Flux_tot (units of input spectrum!): the model's flux at the normalisation wavelength, sum of x_j times fobs_norm]\n")
        if fit_stats.get('fobs_norm') is not None:
            f.write(f" {fit_stats['fobs_norm']:>12.5E}                             [fobs_norm (units of input spectrum): the observed flux the spectrum was divided by; multiply normalised fluxes by it]\n")
        f.write(f" {mini_tot:>12.5E}                             [Mini_tot (apparent; x flux_unit * 4*pi*d_cm^2 / 3.828E33 for Msun)]\n")
        f.write(f" {mcor_tot:>12.5E}                             [Mcor_tot (apparent; x flux_unit * 4*pi*d_cm^2 / 3.828E33 for Msun)]\n\n")

        f.write(f" {v0_s:>7.2f}                                  [v0_min  (km/s)]\n")
        f.write(f" {sig_s:>7.2f}                                  "
                f"[vd_min  (km/s), OBSERVED]\n")
        _write_lsf_block(f, lsf_report)
        f.write(f" {av:>6.4f}                                  [AV_min  (mag)]\n")
        f.write(f" {yav:>6.4f}                                  [YAV_min (mag)]\n\n")

        # The last column says what each component is (ssp, powerlaw,
        # blackbody ...), so a reader need not guess it from the name.
        kinds = {e['filename']: str(e.get('kind', 'ssp')).lower()
                 for e in (getattr(config, 'bases_entries', None) or [])}
        f.write("# j     x_j(%)      Mini_j(%)     Mcor_j(%)     age_j(yr)     Z_j      (L/M)_j    YAV?  Mstars   component_j        a/Fe...       SSP_chi2r SSP_adev(%)   SSP_AV   SSP_x(%)  kind\n")
        for j in range(n_star):
            age_yr = ssp_ages[j] * 1e9
            kind = kinds.get(ssp_names[j], "ssp")
            f.write(f"{j+1:>3}   {fracs[j]:>7.4f}     {mini_pct[j]:>10.4E}    {mcor_pct[j]:>10.4E}    {age_yr:>12.6E}  {ssp_metals[j]:>7.5f}  {ssp_fbase_norms[j]:>9.3E}    {ssp_yav_flags[j]}   {ssp_mstars[j]:>6.4f}   {ssp_short[j]:<20}  0.0000     0.0E+00      0.00     0.00      0.00  {kind}\n")

        _write_gas_kinematics_block(f, gas_report)

        if gas_names:
            f.write(f"\n## Emission Lines ##\n")
            f.write(f" {str(config.wavelength_scale):<40} "
                    f"[wavelength_scale of the OBSERVED spectrum; the rest "
                    f"wavelengths these lines were placed at are on it]\n")
            # What the numbers are. Both facts were true and written nowhere.
            f.write("# Fluxes are in units of the NORMALISED spectrum times "
                    "Angstrom: multiply by fobs_norm (above) for the units of "
                    "the input spectrum.\n")
            f.write("# They are corrected for the stellar extinction AV_min, "
                    "relative to the normalisation wavelength: multiply by "
                    "10**(-0.4*AV_min*(q(lambda)-q(lambda_0))) for the flux "
                    "as observed (brainsp: FitResult.line_fluxes()).\n")
            f.write("# A tied doublet's flux is its strong line's; the weak "
                    "line is fixed at the atomic ratio.\n")
            f.write(f"{'Line/Doublet':<35} | {'Flux':>12} | {'Err':>10}\n")
            for i, name in enumerate(gas_names):
                f.write(f"{name:<35} | {weights[n_star + i]:>12.4e} | {w_errors[n_star + i]:>10.4e}\n")

        if nebular_stats is not None:
            f.write(f"\n## Photoionization Diagnostics ##\n")
            f.write(f" {nebular_stats.get('Q_eff', 0.0):>12.5E}                             [Q_eff (effective ionizing param)]\n")
            f.write(f" {nebular_stats.get('Q_H_total', 0.0):>12.5E}                             [Q(H)_total (apparent, mass-weighted; same scaling as Mcor_tot for photons/s)]\n")
            f.write(f" {nebular_stats.get('tau_ratio', 0.0):>12.5f}                             [tau (predicted/observed Halpha); in photopop mode the penalty rows set this and it is a diagnostic, not a test]\n")
            f.write(f" {nebular_stats.get('Av_neb', 0.0):>12.5f}                             [AV_neb (mag); in photopop mode the value that minimises chi2, see AV_neb status]\n")
            f.write(f" {nebular_stats.get('Te', config.electron_temperature):>12.1f}"
                    f"                             [T_e (K)"
                    f"{'' if nebular_stats.get('Te_measured') else ', default'}]\n")
            f.write(f" {nebular_stats.get('ne', config.electron_density):>12.1f}"
                    f"                             [n_e (cm^-3)"
                    f"{'' if nebular_stats.get('ne_measured') else ', default'}]\n")
            f.write(f" {nebular_stats.get('bpt_flag', -999):>12d}"
                    f"                             [BPT_flag "
                    f"(-999 not defined, 0 SF, 1 composite, 2 LINER, 3 Seyfert)]\n")
            f.write(f" {str(nebular_stats.get('bpt_class', 'undetermined')):<40} [BPT_class]\n")
            f.write(f" {nebular_stats.get('f_neb_norm', 0.0):>12.5f}                             [f_neb (nebular frac at lambda_0)]\n")
            f.write(f" {nebular_stats.get('sc_converged', False)!s:<40} [self_consistency_converged]\n")
            f.write(f" {str(nebular_stats.get('sc_status', 'not applicable')):<40} [AV_neb status: converged, LOWER_BOUND, UPPER_BOUND, BUDGET_EXHAUSTED]\n")
            f.write(f" {nebular_stats.get('sc_iterations', 0):<40} [n_AV_neb refits]\n")
            _dav = nebular_stats.get('sc_d_av_neb', float('nan'))
            f.write(f" {_dav:>12.5E}                             [AV_neb bracket width (mag)]\n")
            f.write(f" {nebular_stats.get('sc_dchi2_range', float('nan')):>12.5E}                             [dchi2 across the whole AV_neb range]\n")
            f.write(f" {nebular_stats.get('sc_one_sigma', float('nan')):>12.5E}                             [AV_neb at dchi2 = 1 (mag); NaN = never reached]\n")

            pred_lines = nebular_stats.get('predicted_lines', {})
            if pred_lines:
                f.write(f"\n## Predicted Emission Lines (from Q(H)) ##\n")
                f.write(f"{'Line':<25} | {'lambda':>8} | {'Predicted':>12} | {'Observed':>12}\n")
                obs_lines = nebular_stats.get('observed_lines', {})
                for name, (wl, flux_pred) in sorted(pred_lines.items(), key=lambda x: x[1][0]):
                    flux_obs = obs_lines.get(name, 0.0)
                    f.write(f"{name:<25} | {wl:>8.1f} | {flux_pred:>12.4e} | {flux_obs:>12.4e}\n")

    logger.info(f"  Summary written to {filepath}")



def _write_lsf_block(f, lsf_report):
    """
    The three dispersions, and where the line-spread functions came from.

    Written inline with the kinematics rather than in a section of its own,
    because sigma_obs is the number directly above it and the difference
    between the two is the whole point. A value that is not measurable is
    written as an upper limit with a '<', never as a number that could be
    mistaken for a measurement.

    Every dispersion here states its convention in its own label, OBSERVED or
    INTRINSIC, rather than leaving it to be inferred from the name. The file
    carries both: sigma_obs and vd_min include the instrument, sigma_star has
    it taken out in quadrature, and the gas block below has its own OBSERVED
    dispersion which will become an intrinsic one when the line-spread
    function reaches the emission line. A reader comparing a stellar and a gas
    dispersion from one summary is comparing two different quantities unless
    the file says so on every line that carries one.
    """
    if lsf_report is None:
        return

    def _line(value, label):
        f.write(f" {value:>7}                                  [{label}]\n")

    _line(f"{lsf_report['sigma_obs']:.2f}",
          "sigma_obs (km/s), OBSERVED, broadening applied to the library")

    if lsf_report['delta_sigma_inst'] is None:
        _why = ("needs instrumental_resolution"
                if lsf_report['data_text'] == "not set"
                else "library resolution unknown")
        _line("unknown", f"delta_sigma_inst (km/s), {_why}")
        _line("unknown", f"sigma_star (km/s), INTRINSIC, {_why}")
    else:
        _line(f"{lsf_report['delta_sigma_inst']:+.2f}",
              "delta_sigma_inst (km/s), signed, data minus library LSF, "
              "flux-weighted")
        if lsf_report['upper_limit']:
            _line(f"<{abs(lsf_report['sigma_star']):.2f}",
                  "sigma_star (km/s), INTRINSIC, UPPER LIMIT")
        else:
            _line(f"{lsf_report['sigma_star']:.2f}",
                  "sigma_star (km/s), INTRINSIC, instrument removed in "
                  "quadrature")

    f.write(f" {lsf_report['data_text']:<40} "
            f"[instrumental_resolution, observed frame]\n")
    f.write(f" {lsf_report['library_text']:<40} [library_resolution]\n")
    f.write(f" {lsf_report['library_source']:<40} "
            f"[library_resolution source]\n")
    f.write(f" {lsf_report['redshift']:>7.5f}                                  "
            f"[spectrum_redshift removed before fitting]\n")
    _state = lsf_report.get('state_text', 'none')
    f.write(f" {_state:<40} [LSF correction applied to the library]\n")
    f.write(f" {lsf_report.get('n_applied', 0):<40} "
            f"[templates pre-convolved]\n")
    f.write(f" {lsf_report.get('n_refused', 0):<40} "
            f"[templates left at library resolution]\n")
    if lsf_report['reason']:
        for chunk in _wrap(lsf_report['reason'], 70):
            f.write(f"# {chunk}\n")


def _write_gas_kinematics_block(f, gas_report):
    """
    The fitted gas velocity and dispersion, per kinematic group.

    These reached the log and nowhere else: a summary file recording a fit
    with emission lines in it listed their fluxes and said nothing about the
    kinematics they were fitted with, so the one number a reader would quote
    was the one number the file did not have.

    It follows _write_lsf_block's conventions line for line, because the two
    blocks sit in one file and a reader should not have to learn two ways of
    being told that something is not measurable. A dispersion on the lower
    bound of gas_dispersion_range is written '<value' and labelled UPPER
    LIMIT, one on the upper bound '>value' and LOWER LIMIT, and a group with
    no detected line prints the word 'unconstrained' in place of a number, the
    way the stellar block prints 'unknown'. What the optimiser actually
    returned for such a group is in the '#' reason beneath, where it cannot be
    read as a result.

    sigma_gas is labelled OBSERVED on every line. The instrumental line-spread
    function is not yet taken out of the emission-line profile, so the number
    includes it; the label states the convention rather than assuming it,
    because it becomes an intrinsic dispersion the day the correction lands
    and the file must not then be quietly wrong. sigma_inst is written beside
    it, per group, so that both halves of the arithmetic are in the file and
    neither is done for the reader -- as recommended by Westfall et al.
    (2019), and for the reason they give: a sigma_obs below sigma_inst is not a measurement of
    zero and must not be clipped to one.
    """
    if not gas_report:
        return

    from brainsp.gas_diagnostics import gas_dispersion_text, gas_velocity_text

    f.write("\n## Gas Kinematics ##\n")

    def _line(value, label):
        f.write(f" {value:>13}                            [{label}]\n")

    for entry in gas_report:
        label = entry["label"]
        status = ("" if entry["quotable"] else f", {entry['status']}")

        _line(gas_dispersion_text(entry),
              f"sigma_gas group {label} (km/s), OBSERVED{status}")
        _line(gas_velocity_text(entry), f"v0_gas group {label} (km/s)")

        if entry["sigma_inst"] is None:
            _line("unknown",
                  f"sigma_inst group {label} (km/s), needs "
                  f"instrumental_resolution")
        else:
            _line(f"{entry['sigma_inst']:.2f}",
                  f"sigma_inst group {label} (km/s), at the group's lines, "
                  f"flux-weighted")

        _line(f"{entry['n_detected']}/{entry['n_lines']}",
              f"lines carrying flux in group {label}")

        if entry["reason"]:
            for chunk in _wrap(entry["reason"], 70):
                f.write(f"# {chunk}\n")


def _wrap(text, width):
    """Fold a sentence onto fixed-width lines without importing textwrap."""
    words, line, out = text.split(), "", []
    for word in words:
        if line and len(line) + 1 + len(word) > width:
            out.append(line)
            line = word
        else:
            line = f"{line} {word}".strip()
    if line:
        out.append(line)
    return out

def write_fitted_spectrum(filepath, wavelength, flux_obs, total_fit,
                          stellar_fit, emission_fit, err, residual,
                          nebular_fit=None):
    """
    Write the fitted spectrum to a text file.
    """
    _ensure_folder(filepath)

    cols = [wavelength, flux_obs, total_fit, stellar_fit, emission_fit]
    header_parts = ["Wavelength", "Flux_Obs", "Total_Fit",
                    "Stellar_Fit", "Emission_Fit"]

    if nebular_fit is not None:
        cols.append(nebular_fit)
        header_parts.append("Nebular_Fit")

    if err is not None:
        cols.append(err)
        header_parts.append("Err")

    cols.append(residual)
    header_parts.append("Resid")

    cols = [np.asarray(c) for c in cols]

    np.savetxt(filepath, np.column_stack(cols),
               header=" ".join(header_parts), fmt='%.6e')

    logger.info(f"  Spectrum written to {filepath}")
