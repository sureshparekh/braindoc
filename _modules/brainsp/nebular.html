"""
BRAIN-v1 Nebular Physics Module
Photoionization-based nebular continuum and emission line predictions.

Implements self-consistent nebular modeling:
- Q(H) integrated from each SSP spectrum below the Lyman limit
- Free-free (bremsstrahlung) emission
- Free-bound (recombination continuum) emission
- Two-photon continuum emission
- Total nebular continuum template
- Case B hydrogen emission line predictions
- Semi-empirical metal emission line predictions
- Self-consistency tau ratio

References:
- Osterbrock & Ferland (2006)
- Hummer & Storey (1987)
- Nussbaumer & Schmutz (1984)
- Ercolano & Storey (2006)
"""
import os

import jax.numpy as jnp

import numpy as np
import logging

from brainsp.atomic_data import (
    h_planck, c_cgs, c_angstrom, k_boltz, m_electron, e_charge,
    Rydberg_erg, a_bohr, alpha_fs,
    LYMAN_LIMIT, HeI_LIMIT, HeII_LIMIT,
    LAMBDA_LYMAN_ALPHA, LAMBDA_HALPHA, LAMBDA_HBETA,
    NU_LYMAN_ALPHA, NU_HALPHA, NU_HBETA,
    alpha_B, alpha_Hbeta_eff, alpha_Halpha_eff, alpha_2S_eff,
    alpha_B_HeI,
    SIGMA_BF_0, chi_n, nu_n,
    two_photon_shape_HI, A_2q_HI, Q_2S_DEEXCIT,
    two_photon_shape_HeI, A_2q_HeI,
    qh_from_spectrum, LYMAN_LIMIT, L_SUN_ERG,
    hydrogen_case_b_ratios, all_hydrogen_ratios,
    predicted_metal_ratios,
)

logger = logging.getLogger("NEWBRAIN")

def gaunt_ff(Te, nu):
    """
    Velocity-averaged free-free Gaunt factor (Sutherland 1998 approximation).

    Parameters
    ----------
    Te : float
        Electron temperature [K].
    nu : array
        Frequency [Hz].

    Returns
    -------
    gff : array
        Free-free Gaunt factor.
    """
    u = h_planck * nu / (k_boltz * Te)
    u = jnp.clip(u, 1e-10, 500.0)

    gff = (1.0
           + 0.1728 * u**(1.0/3.0) * (1.0 + 2.0 / u)
           - 0.0496 * u**(2.0/3.0) * (1.0 + 2.0 / (3.0 * u)))

    return jnp.clip(gff, 1.0, 5.0)

def gamma_ff(wavelength, Te, Z=1):
    """
    Free-free emission coefficient for ion of charge Z.

    Parameters
    ----------
    wavelength : array
        Wavelength grid [Angstroms].
    Te : float
        Electron temperature [K].
    Z : int
        Ion charge (1 for H+, 2 for He++).

    Returns
    -------
    gff_coeff : array
        gamma_ff(nu, Te) [erg cm^3 s^-1 Hz^-1].

    Volume emission coefficient gamma_nu, defined by
        4*pi*j_nu = n_e n_p gamma_nu,
    i.e. integrated over the full sphere, NOT the per-steradian j_nu.
    The emission-line predictions (C_Hbeta, predict_*_lines) and the
    template normalisation gamma_eff/alpha_B are on this convention, so a
    per-steradian coefficient here would put the lines and the continuum
    4*pi apart in scale.

    The constant 6.8416e-38 is the exact theoretical coefficient
    (2^5 pi e^6)/(3 m_e c^3) * sqrt(2 pi /(3 k m_e)), quoted rounded as 6.8e-38
    in Rybicki & Lightman (1979) eq. 5.14b. Their eq. 5.14a, 5.44e-39, is the
    same quantity divided by 4*pi.
    """
    nu = c_cgs / (wavelength * 1e-8)
    gff = gaunt_ff(Te, nu)

    prefactor = 6.8416e-38 * Z**2 * Te**(-0.5)

    u = h_planck * nu / (k_boltz * Te)
    u = jnp.clip(u, 0, 500.0)

    return prefactor * jnp.exp(-u) * gff




def _gaunt_bf(n, nu_ratio):
    """
    Approximate bound-free Gaunt factor.
    nu_ratio = nu / nu_n (must be >= 1 at threshold).
    Kramers (1923) + corrections from Karzas & Latter (1961).
    """
    g = jnp.ones_like(nu_ratio)
    g = 1.0 - 0.3456 * (1.0 / nu_ratio - 0.5) / (nu_ratio**(1.0/3.0) + 0.5)
    return jnp.clip(g, 0.7, 1.3)

def gamma_fb_HI(wavelength, Te, n_max=300):
    """
    Free-bound emission coefficient for H I (Case B: skip n=1).

    Parameters
    ----------
    wavelength : array
        Wavelength grid [Angstroms].
    Te : float
        Electron temperature [K].
    n_max : int
        Maximum principal quantum number for summation.

    Returns
    -------
    gamma_fb : array
        Free-bound emission coefficient [erg cm^3 s^-1 Hz^-1].

    Volume emission coefficient gamma_nu, defined by
        4*pi*j_nu = n_e n_p gamma_nu,
    i.e. integrated over the full sphere, NOT the per-steradian j_nu.
    The emission-line predictions (C_Hbeta, predict_*_lines) and the
    template normalisation gamma_eff/alpha_B are on this convention, so a
    per-steradian coefficient here would put the lines and the continuum
    4*pi apart in scale.

    The prefactor is 8*pi*h*nu^3/c^2, from the Milne relation (Osterbrock &
    Ferland 2006 eq. 2.28) combined with a Maxwellian: 2 h nu^3/c^2 is the
    Planck normalisation of the per-steradian form. Checked against the Case B
    photon identity, integral gamma_fb/(h nu) dnu = alpha_B, which closes to
    1.3% -- the residual is the n > 51 loop break below, not the prefactor.
    """
    nu = c_cgs / (wavelength * 1e-8)
    h_nu = h_planck * nu
    kT = k_boltz * Te

    thermal_factor = (h_planck**2 / (2.0 * jnp.pi * m_electron * kT))**1.5

    prefactor_common = (8.0 * jnp.pi * h_planck / c_cgs**2) * nu**3 * thermal_factor

    gamma_fb = jnp.zeros_like(nu)

    for n in range(2, n_max + 1):
        chi = chi_n(n)
        nu_thresh = chi / h_planck

        above_thresh = nu >= nu_thresh

        if not jnp.any(above_thresh):
            continue

        nu_ratio = jnp.where(above_thresh, nu / nu_thresh, 1.0)
        g_bf = _gaunt_bf(n, nu_ratio)
        sigma = SIGMA_BF_0 * n * (1.0 / nu_ratio)**3 * g_bf

        stat_weight = float(n**2)

        exponent = (chi - h_nu) / kT
        exponent = jnp.clip(exponent, -500.0, 0.0)

        contribution = prefactor_common * sigma * stat_weight * jnp.exp(exponent)
        gamma_fb += jnp.where(above_thresh, contribution, 0.0)

        if chi / kT < 0.01 and n > 50:
            break

    return gamma_fb

def gamma_fb_HeI(wavelength, Te, n_max=100):
    """
    Free-bound emission coefficient for He I (approximate).

    He I is not hydrogenic; we use a scaled version of H I fb emission
    with the He I ionization potential (24.6 eV) and effective charge.
    This is an approximation; detailed tables (Ercolano & Storey 2006)
    would be more accurate.

    Volume emission coefficient gamma_nu, defined by
        4*pi*j_nu = n_e n_p gamma_nu,
    i.e. integrated over the full sphere, NOT the per-steradian j_nu.
    The emission-line predictions (C_Hbeta, predict_*_lines) and the
    template normalisation gamma_eff/alpha_B are on this convention, so a
    per-steradian coefficient here would put the lines and the continuum
    4*pi apart in scale.
    """


    nu = c_cgs / (wavelength * 1e-8)
    h_nu = h_planck * nu
    kT = k_boltz * Te
    thermal_factor = (h_planck**2 / (2.0 * jnp.pi * m_electron * kT))**1.5
    prefactor_common = (8.0 * jnp.pi * h_planck / c_cgs**2) * nu**3 * thermal_factor

    Z_eff = 1.0

    gamma_fb = jnp.zeros_like(nu)

    for n in range(2, min(n_max, 50) + 1):
        chi = Rydberg_erg * Z_eff**2 / n**2
        nu_thresh = chi / h_planck

        above_thresh = nu >= nu_thresh
        if not jnp.any(above_thresh):
            continue

        nu_ratio = jnp.where(above_thresh, nu / nu_thresh, 1.0)
        sigma = SIGMA_BF_0 * Z_eff**4 * n * (1.0 / nu_ratio)**3
        stat_weight = float(n**2)

        exponent = jnp.clip((chi - h_nu) / kT, -500.0, 0.0)
        contribution = prefactor_common * sigma * stat_weight * jnp.exp(exponent)
        gamma_fb += jnp.where(above_thresh, contribution, 0.0)

        if chi / kT < 0.01 and n > 30:
            break

    return gamma_fb

def gamma_2q_HI(wavelength, Te, ne=100.0):
    """
    Two-photon continuum emission coefficient for H I.

    Parameters
    ----------
    wavelength : array
        Wavelength grid [Angstroms].
    Te : float
        Electron temperature [K].
    ne : float
        Electron density [cm^-3].

    Returns
    -------
    gamma_2q : array
        Two-photon emission coefficient [erg cm^3 s^-1 Hz^-1].
    """
    nu = c_cgs / (wavelength * 1e-8)

    x = nu / NU_LYMAN_ALPHA

    valid = (x > 0) & (x < 1.0)

    A_x = two_photon_shape_HI(x)

    a_2S = alpha_2S_eff(Te)

    f_suppress = 1.0 / (1.0 + ne * Q_2S_DEEXCIT / A_2q_HI)

    gamma_2q = jnp.where(valid,
        a_2S * h_planck * nu * A_x / (A_2q_HI * NU_LYMAN_ALPHA) * f_suppress,
        0.0
    )

    return gamma_2q

def gamma_2q_HeI(wavelength, Te, ne=100.0):
    """
    Two-photon continuum emission coefficient for He I.
    From 1s2s ^1S_0 -> 1s^2 ^1S_0 transition.
    Two-photon energy = 20.6 eV (lambda_cutoff ~ 601.4 A).
    Only significant in the UV; negligible at optical wavelengths.
    """
    nu = c_cgs / (wavelength * 1e-8)
    nu_2q_HeI = c_cgs / (601.4e-8)

    x = nu / nu_2q_HeI
    valid = (x > 0) & (x < 1.0)

    A_x = two_photon_shape_HeI(x)

    a_2S_HeI = 0.1 * alpha_B_HeI(Te)

    f_suppress = 1.0 / (1.0 + ne * 1e-5 / A_2q_HeI)

    gamma_2q = jnp.where(valid,
        a_2S_HeI * h_planck * nu * A_x / (A_2q_HeI * nu_2q_HeI) * f_suppress,
        0.0
    )

    return gamma_2q

def gamma_eff(wavelength, Te, ne=100.0, Y_He=0.25):
    """
    Total effective nebular continuum emission coefficient.

    gamma_eff = gamma_HI(ff+fb+2q) + (nHe/nH) * gamma_HeI(ff+fb+2q)

    Parameters
    ----------
    wavelength : array
        Wavelength grid [Angstroms].
    Te : float
        Electron temperature [K].
    ne : float
        Electron density [cm^-3].
    Y_He : float
        Helium mass fraction (0.25 for ~solar).

    Returns
    -------
    gamma : array
        Total emission coefficient [erg cm^3 s^-1 Hz^-1].
    """
    g_ff_HI = gamma_ff(wavelength, Te, Z=1)
    g_fb_HI = gamma_fb_HI(wavelength, Te)
    g_2q_HI = gamma_2q_HI(wavelength, Te, ne)
    gamma_HI = g_ff_HI + g_fb_HI + g_2q_HI

    nHe_nH = Y_He / (4.0 * (1.0 - Y_He))

    g_ff_HeI = gamma_ff(wavelength, Te, Z=1)
    g_fb_HeI = gamma_fb_HeI(wavelength, Te)
    g_2q_HeI = gamma_2q_HeI(wavelength, Te, ne)
    gamma_HeI = g_ff_HeI + g_fb_HeI + g_2q_HeI

    gamma_total = gamma_HI + nHe_nH * gamma_HeI

    return gamma_total

class NebularModel:
    """
    Encapsulates all nebular physics computations.

    Initialize with Te, ne, Y_He and the fitting wavelength grid.
    Pre-computes the normalized nebular continuum template such that:
        f_neb(lambda) = template(lambda) * Q_eff

    where Q_eff = sum_j w_j * unredden_j * qh_ratio_j.
    """

    def __init__(self, wavelength, wavelength_scale, Te=1e4, ne=100.0,
                 Y_He=0.25, Z_gas=0.02, f_esc=0.0):
        """
        Parameters
        ----------
        wavelength : array
            Fitting wavelength grid [Angstroms] (log-lambda grid).
        wavelength_scale : {'air', 'vacuum'}
            The scale the OBSERVED spectrum is on, and therefore the scale
            the predicted line wavelengths are put on. REQUIRED. The
            predicted lines are compared, by name and side by side, against
            the lines the fit measured; putting the two on different scales
            would offset them by 83 km/s with nothing in the output saying
            so. Every mode that builds a NebularModel is a mode that fits
            emission lines, so the setting is always available here.
        Te : float
            Electron temperature [K].
        ne : float
            Electron density [cm^-3].
        Y_He : float
            Helium mass fraction.
        Z_gas : float
            Gas-phase metallicity (for metal line predictions).
        f_esc : float
            Lyman continuum escape fraction [0, 1].
        """
        self.wavelength = jnp.asarray(wavelength)
        self.wavelength_scale = wavelength_scale
        self.Te = Te
        self.ne = ne
        self.Y_He = Y_He
        self.Z_gas = Z_gas
        self.f_esc = f_esc

        self._gamma_eff = gamma_eff(self.wavelength, Te, ne, Y_He)
        self._alpha_B = alpha_B(Te)

        lam_cm = self.wavelength * 1e-8
        self._template = self._gamma_eff / self._alpha_B * c_cgs / lam_cm**2 * 1e-8

        self._H_ratios = all_hydrogen_ratios(Te, wavelength_scale)
        self._metal_ratios = predicted_metal_ratios(Z_gas, wavelength_scale)

        self._alpha_Hbeta = alpha_Hbeta_eff(Te)
        self._alpha_Halpha = alpha_Halpha_eff(Te)
        self._hnu_Hbeta = h_planck * NU_HBETA
        self._hnu_Halpha = h_planck * NU_HALPHA

        self._f_capture = 1.0 - f_esc

        logger.info(f"   NebularModel: Te={Te:.0f} K, ne={ne:.0f} cm^-3, "
                     f"Y_He={Y_He:.3f}, Z_gas={Z_gas:.4f}, f_esc={f_esc:.2f}")

    @property
    def template(self):
        """Normalized nebular continuum template [erg/A per (photon/s)]."""
        return self._template

    def nebular_continuum(self, Q_eff):
        """
        Compute the nebular continuum spectrum for a given effective Q(H).

        Parameters
        ----------
        Q_eff : float
            Effective ionizing photon parameter
            = sum_j w_j * unredden_j * qh_ratio_j
            (in normalized flux units, distance-independent).

        Returns
        -------
        f_neb : array
            Nebular continuum in normalized flux units (same as observed spectrum).
        """
        return self._template * Q_eff * self._f_capture

    def predict_Hbeta_flux(self, Q_eff):
        """
        Predict H-beta integrated flux from Q(H).

        Parameters
        ----------
        Q_eff : float
            Effective ionizing photon parameter.

        Returns
        -------
        f_Hbeta : float
            Predicted H-beta integrated flux (same units as NNLS gas weights,
            i.e., integrated line flux in normalized flux units).
        """
        return (self._alpha_Hbeta / self._alpha_B) * self._hnu_Hbeta * Q_eff * self._f_capture

    def predict_Halpha_flux(self, Q_eff):
        """Predict H-alpha integrated flux from Q(H).

        Uses F_Hbeta * Balmer ratio rather than alpha_Halpha directly,
        ensuring consistency with the Case B line ratio tables.
        """
        f_Hbeta = self.predict_Hbeta_flux(Q_eff)
        return f_Hbeta * self._H_ratios['Halpha'][1]

    def predict_hydrogen_lines(self, Q_eff, wavelength_range=None):
        """
        Predict all hydrogen recombination line fluxes.

        Parameters
        ----------
        Q_eff : float
            Effective ionizing photon parameter.
        wavelength_range : tuple or None
            (wl_min, wl_max) to filter lines. None = return all.

        Returns
        -------
        lines : dict {name: (wavelength, flux)}
        """
        f_Hbeta = self.predict_Hbeta_flux(Q_eff)

        lines = {}
        for name, (wl, ratio) in self._H_ratios.items():
            if wavelength_range is not None:
                if wl < wavelength_range[0] or wl > wavelength_range[1]:
                    continue
            lines[name] = (wl, f_Hbeta * ratio)

        return lines

    def tau_ratio(self, Q_eff, observed_Halpha, Av_neb=0.0,
                  reddening_law='CCM', norm_wavelength=5500.0):
        """
        Compute the self-consistency tau ratio.

        tau = L_predicted(Halpha) / L_observed(Halpha)

        After correcting the prediction for nebular extinction.

        Parameters
        ----------
        Q_eff : float
            Effective ionizing photon parameter.
        observed_Halpha : float
            Observed H-alpha integrated flux (from NNLS gas weight).
        Av_neb : float
            Nebular extinction [mag].
        reddening_law : str
            Reddening law name.
        norm_wavelength : float
            Normalization wavelength [A].

        Returns
        -------
        tau : float
            Self-consistency ratio. tau=1 => self-consistent.
        """
        predicted_Ha = self.predict_Halpha_flux(Q_eff)

        if Av_neb > 0:
            from brainsp.reddening_laws import get_k_lambda
            k_Ha = get_k_lambda(jnp.array([LAMBDA_HALPHA]), reddening_law)[0]
            k_norm = get_k_lambda(jnp.array([norm_wavelength]), reddening_law)[0]
            dk = k_Ha - k_norm
            predicted_Ha *= 10**(-0.4 * Av_neb * dk)

        if observed_Halpha <= 0:
            return jnp.inf

        return predicted_Ha / observed_Halpha

def compute_qh_ratios(ages_gyr, metallicities, fbase_norms,
                      ssp_flux_unit='Lsun', ssp_folder=None, bases_info=None,
                      return_absolute=False, allow_partial=False):
    """
    Q(H) per unit NNLS weight for each SSP, integrated from the SSP spectra.

    Each base spectrum is integrated shortwards of the hydrogen ionisation
    edge to get the ionising photon rate it actually emits, then divided by
    its own normalisation flux so the result multiplies the NNLS weight
    directly.

    This requires the raw library files, not the templates BRAIN fits with:
    those have been resampled onto the observed wavelength grid, which starts
    thousands of Angstrom above the Lyman limit and therefore contains no
    ionising continuum at all.

    Parameters
    ----------
    ages_gyr, metallicities : array
        Per-SSP age and metallicity. Carried for reporting only -- nothing is
        looked up from them.
    fbase_norms : array
        SSP flux at the normalisation wavelength, per Msun.
    ssp_flux_unit : {'Lsun', 'cgs'}
    ssp_folder : str
        Directory holding the library.
    bases_info : list of dict
        Base entries with a 'filename' key, in the same order as the arrays.
    return_absolute : bool
        Also return the absolute rates per Msun, for reporting.

    Returns
    -------
    qh_ratios : jnp array
        Q(H) per unit weight. Zero for SSPs with no ionising output.
    absolute : dict, optional
        {'QHc', 'QHe1', 'QHe2'} arrays in photons/s per Msun.

    Raises
    ------
    ValueError
        If the library does not reach below the Lyman limit. Degrading
        quietly to a zero nebular continuum would leave a fit that looks
        converged and is not physically meaningful.
    """
    if ssp_folder is None or bases_info is None:
        raise ValueError(
            "compute_qh_ratios needs the raw SSP library (ssp_folder and "
            "bases_info) to integrate the ionising continuum.")

    if ssp_flux_unit == 'Lsun':
        unit_conv = L_SUN_ERG
    elif ssp_flux_unit == 'cgs':
        unit_conv = 1.0
    else:
        raise ValueError(f"Unknown ssp_flux_unit: {ssp_flux_unit}")

    n_ssp = len(ages_gyr)
    qh_ratios = np.zeros(n_ssp)
    absolute = {k: np.zeros(n_ssp) for k in ('QHc', 'QHe1', 'QHe2')}
    n_covered = 0
    covered_idx = set()
    bluest = np.inf

    for j in range(n_ssp):
        path = os.path.join(ssp_folder, bases_info[j]['filename'])
        data = np.loadtxt(path)
        wl, fl = data[:, 0], data[:, 1]
        bluest = min(bluest, float(wl.min()))

        q = qh_from_spectrum(wl, fl, ssp_flux_unit=ssp_flux_unit)
        if 'QHc' not in q:
            continue
        n_covered += 1
        covered_idx.add(j)
        for k, v in q.items():
            absolute[k][j] = v

        fbase_cgs = float(fbase_norms[j]) * unit_conv
        if fbase_cgs > 0:
            qh_ratios[j] = q['QHc'] / fbase_cgs

    if n_covered == 0:
        raise ValueError(
            f"The SSP library in {ssp_folder!r} starts at {bluest:.1f} A and "
            f"never reaches the Lyman limit at {LYMAN_LIMIT} A, so its "
            f"ionising photon output cannot be computed.\n"
            f"The nebular continuum would be identically zero and the fit "
            f"would look converged while modelling no gas at all.\n"
            f"Fix it one of two ways:\n"
            f"  - set nebular_mode: spop, and fit the stellar continuum "
            f"only;\n"
            f"  - or point base_folder at a library whose spectra extend "
            f"below {LYMAN_LIMIT} A. What matters is the first wavelength in "
            f"the library files themselves, not the fitted range. "
            f"brainsp libraries info <library> reports what the library you have "
            f"covers and which modes it supports.")

    if n_covered < n_ssp:
        short = [bases_info[j]['filename'] for j in range(n_ssp)
                 if j not in covered_idx][:5]
        if not allow_partial:
            raise ValueError(
                f"Only {n_covered} of {n_ssp} templates in {ssp_folder!r} "
                f"reach below the Lyman limit at {LYMAN_LIMIT} A.\n"
                f"Q(H) would be integrated over that subset alone, so the "
                f"populations that happen to reach it would carry the whole "
                f"ionising budget and the rest would contribute none. The fit "
                f"completes either way and nothing in the output says which "
                f"templates were counted.\n"
                f"First few that fall short: {', '.join(short)}\n"
                f"Either use a library whose templates all reach below "
                f"{LYMAN_LIMIT} A, or set allow_partial_ionising_coverage: "
                f"true to accept a Q(H) integrated over "
                f"{n_covered}/{n_ssp} of them.")
        logger.warning(
            f"   Only {n_covered}/{n_ssp} SSPs reach below the Lyman limit; "
            f"the rest contribute no ionising photons. Q(H) is integrated "
            f"over a biased subset of the library because "
            f"allow_partial_ionising_coverage is true.")

    if return_absolute:
        return jnp.asarray(qh_ratios), absolute
    return jnp.asarray(qh_ratios)

def compute_Q_eff(weights, qh_ratios, ssp_yav_flags, av, yav,
                  k_norm):
    """
    Compute the effective ionizing photon parameter from current fit.

    Q_eff = sum_j w_j * unredden_j * qh_ratio_j

    Parameters
    ----------
    weights : array
        Stellar NNLS weights.
    qh_ratios : array
        Q(H) ratio per SSP (from compute_qh_ratios).
    ssp_yav_flags : array
        YAV flags (1 for young components).
    av : float
        Stellar extinction Av [mag].
    yav : float
        Young AV excess [mag].
    k_norm : float
        k(lambda_0) from the reddening law.

    Returns
    -------
    Q_eff : float
        Effective ionizing photon parameter.
    """
    n_ssp = len(qh_ratios)
    Q_eff = 0.0

    for j in range(n_ssp):
        if qh_ratios[j] <= 0 or weights[j] <= 0:
            continue

        av_tot = av + (yav if ssp_yav_flags[j] else 0.0)
        unredden = 10.0**(0.4 * av_tot * k_norm)
        Q_eff += weights[j] * unredden * qh_ratios[j]

    return Q_eff

def Av_neb_from_balmer_decrement(F_Halpha, F_Hbeta, reddening_law='CCM',
                                  Te=1e4):
    """
    Derive nebular extinction Av from the observed Balmer decrement.

    A_V^neb = 2.5 / (k_Hb - k_Ha) * log10[(F_Ha/F_Hb)_obs / (F_Ha/F_Hb)_CaseB]

    Parameters
    ----------
    F_Halpha : float
        Observed H-alpha flux (integrated, from NNLS weight).
    F_Hbeta : float
        Observed H-beta flux.
    reddening_law : str
        Reddening law name.
    Te : float
        Electron temperature [K] (for intrinsic Balmer ratio).

    Returns
    -------
    Av_neb : float
        Nebular extinction [mag]. Returns 0 if decrement is unphysical.
    """
    if F_Halpha <= 0 or F_Hbeta <= 0:
        return 0.0

    intrinsic_ratio = hydrogen_case_b_ratios(Te)['Halpha']

    observed_ratio = F_Halpha / F_Hbeta

    if observed_ratio <= intrinsic_ratio:
        return 0.0

    from brainsp.reddening_laws import get_k_lambda
    k_Ha = get_k_lambda(jnp.array([LAMBDA_HALPHA]), reddening_law)[0]
    k_Hb = get_k_lambda(jnp.array([LAMBDA_HBETA]), reddening_law)[0]
    dk = k_Hb - k_Ha

    if abs(dk) < 1e-10:
        return 0.0

    Av_neb = 2.5 / dk * jnp.log10(observed_ratio / intrinsic_ratio)
    return max(Av_neb, 0.0)

def extract_line_flux(gas_names, gas_weights, line_key):
    """
    Extract a specific emission line flux from the NNLS gas results.

    Parameters
    ----------
    gas_names : list of str
        Names of emission line templates.
    gas_weights : array
        NNLS weights for gas templates.
    line_key : str
        A catalogue entry name, or a species from LINE_SPECIES.

    Returns
    -------
    flux : float
        Integrated line flux, summed over every entry of that species, or
        0.0 if the catalogue provides none.

    This matched by substring and returned the first hit until the name
    couplings were broken. That returned [S II] 6716 while silently
    discarding 6731, which is half the density diagnostic; returned the
    [O III] 4363 auroral line in place of the 4959+5007 doublet; and
    returned He II when asked for He I, because "He I" is a substring of
    "He II". It did not misfire in practice only because the two keys in
    use, Halpha and Hbeta, are unambiguous in the shipped catalogue.

    A species is not a line. Asking for "[O III]" returns 4363 plus the
    4959+5007 doublet, because all three are [O III], and 4363 comes from a
    different upper level and is not usually wanted alongside the other two.
    Ask for the entry by name when a particular line is meant; an exact name
    always wins over the species sum.
    """
    from brainsp.emission_lines import species_of

    for i, name in enumerate(gas_names):
        if name == line_key:
            return gas_weights[i]

    total, found = 0.0, False
    for i, name in enumerate(gas_names):
        if species_of(name) == line_key:
            total += gas_weights[i]
            found = True
    return total if found else 0.0
