#!/usr/bin/env python3
"""
BRAIN — run it.

    python -m brainsp.run_brain

The original settings-file runner, kept for brain_setup.yaml workflows: it
reads every setting from ./brain_setup.yaml (or the package's template).
`brainsp fit --settings FILE` is the same with more options.

    python -m brainsp.run_brain --check       validate and exit
    python -m brainsp.run_brain --templates   list the templates available
    python -m brainsp.run_brain --list        every setting, with its default
    python -m brainsp.run_brain --clear-cache wipe cached engine + templates
    python -m brainsp.run_brain FILE ...      fit these spectra instead
"""

import glob
import logging
import os
import sys
import time

sys.dont_write_bytecode = True

from brainsp import brain_settings
from brainsp import brain_cache


def _setup_device(S):
    """Honour the 'device' setting before JAX initialises a backend."""
    dev = S.get("device", "auto")
    if dev == "cpu":
        os.environ["JAX_PLATFORMS"] = "cpu"
    elif dev == "gpu":
        os.environ.pop("JAX_PLATFORMS", None)


def _write_settings_record(cfg, path, spectrum):
    """
    Record every setting that produced this result, next to the result.

    A single settings file is a good interface but a poor archive: change one
    number, re-run, and the two output files look identical with nothing to
    say which settings made which. This closes that gap — the fit and the
    inputs that produced it always travel together.
    """
    import dataclasses
    with open(path, "w") as f:
        f.write("# Settings used for this fit. Written by BRAIN; not an input.\n")
        f.write(f"# spectrum: {spectrum}\n#\n")
        entries = getattr(cfg, "bases_entries", None) or []
        if entries:
            ages = sorted({e["age_gyr"] * 1e9 for e in entries})
            zs = sorted({e["metallicity"] for e in entries})
            f.write(f"n_templates                    {len(entries)}\n")
            f.write(f"template_ages_yr               "
                    f"{', '.join(f'{a:.4g}' for a in ages)}\n")
            f.write(f"template_metallicities         "
                    f"{', '.join(f'{z:.6f}' for z in zs)}\n#\n")
        for fld in dataclasses.fields(cfg):
            if fld.name == "bases_entries":
                continue
            f.write(f"{fld.name:<30} {getattr(cfg, fld.name)}\n")


def _banner(msg):
    print(f"\n{'=' * 70}\n  {msg}\n{'=' * 70}")

def main(argv):
    args = [a for a in argv[1:] if not a.startswith("--")]
    flags = {a for a in argv[1:] if a.startswith("--")}

    if "--help" in flags or "-h" in flags:
        print(__doc__)
        return 0

    if "--clear-cache" in flags:
        brain_cache.clear("all")
        print("Caches cleared. The next run will rebuild them.")
        return 0

    if "--list" in flags:
        print("Every setting BRAIN understands, with its default:\n")
        for k, v in brain_settings.DEFAULTS.items():
            print(f"  {k:<38} {v!r}")
        return 0

    S = brain_settings.validate(brain_settings.load())

    if "--templates" in flags:
        print(brain_settings.describe_templates(S["base_folder"]))
        _f, _c = brain_settings.build_base_list(S)
        print(f"\nYour current selection: {len(_c)} templates "
              f"(ages: {S['ages']}, metallicities: {S['metallicities']})")
        return 0

    if "--check" in flags:
        from brainsp import io_utils
        folder, entries = brain_settings.build_base_list(S)
        cov = io_utils.library_coverage(folder, entries)
        print("brain_setup.yaml is valid.")
        print(f"  templates    : {len(entries)} from {folder}")
        print(f"  spectrum     : {S['spectrum']}")
        print(f"  range        : {S['wavelength_min']}-{S['wavelength_max']} A")
        print(f"  nebular mode : {S['nebular_mode']}")
        print(io_utils.describe_coverage(cov))
        if S["nebular_mode"] not in cov["modes"]:
            avail = cov["modes"]
            choice = (f"use {avail[0]}" if len(avail) == 1
                      else f"use one of {', '.join(avail)}")
            print(f"\n  {S['nebular_mode']!r} is not available for this "
                  f"library: {choice}, or point base_folder at a library "
                  f"whose spectra reach below {io_utils.LYMAN_LIMIT_A} A.")
            return 1
        return 0

    _setup_device(S)

    logging.basicConfig(
        level=logging.INFO, format="%(message)s", stream=sys.stdout)

    if S["use_cached_engine"]:
        brain_cache.enable_jax_cache()

    import numpy as np
    import jax
    from brainsp import brainv1
    from brainsp import io_utils

    if S["use_cached_templates"]:
        _raw = io_utils.load_ssp_templates

        _memo = {}

        def _cached(ssp_folder, bases_info, target_wl, norm_wavelength):
            key = brain_cache.ssp_cache_key(
                ssp_folder, bases_info, target_wl, norm_wavelength)
            if key not in _memo:
                _memo[key] = brain_cache.load_ssp_templates_cached(
                    ssp_folder, bases_info, target_wl, norm_wavelength, _raw)
            return _memo[key]

        io_utils.load_ssp_templates = _cached
        brainv1.load_ssp_templates = _cached

    if args:
        targets = []
        for a in args:
            targets.extend(sorted(glob.glob(a)) if any(c in a for c in "*?")
                           else [a])
    elif any(c in str(S["spectrum"]) for c in "*?"):
        targets = sorted(glob.glob(S["spectrum"]))
    else:
        targets = [S["spectrum"]]

    if not targets:
        print(f"No spectra matched {S['spectrum']!r}.")
        return 1

    out_dir = S["output_folder"]
    os.makedirs(out_dir, exist_ok=True)

    _banner(f"BRAIN — {len(targets)} "
            f"{'spectrum' if len(targets) == 1 else 'spectra'}"
            f" | device: {jax.default_backend()} | mode: "
            f"{S['nebular_mode']}")

    t_all = time.time()
    times, failures = [], []

    for i, spec in enumerate(targets, 1):
        name = os.path.splitext(os.path.basename(spec))[0]
        cfg = brain_settings.to_config(S, spectrum_file=spec)
        cfg.output_summary = os.path.join(out_dir, f"{name}_summary.txt")
        cfg.output_spectrum = os.path.join(out_dir, f"{name}_fitted.txt")

        if len(targets) > 1:
            print(f"\n--- [{i}/{len(targets)}] {name} ---")

        _write_settings_record(cfg, os.path.join(
            out_dir, f"{name}_settings.txt"), spec)

        t0 = time.time()
        try:
            brainv1.run(cfg)
            dt = time.time() - t0
            times.append(dt)
            if i == 1 and len(targets) > 1:
                print(f"    ({dt:.1f}s — later spectra reuse the compiled "
                      f"engine and loaded templates, so they are faster)")
        except Exception as exc:
            failures.append((name, str(exc)))
            logging.error(f"    FAILED: {exc}")

    if S["make_plot"] and len(targets) == 1 and not failures:
        try:
            from brainsp.plot_fit import plot_fit
            name = os.path.splitext(os.path.basename(targets[0]))[0]
            plot_fit(os.path.join(out_dir, f"{name}_fitted.txt"),
                     os.path.join(out_dir, f"{name}_summary.txt"),
                     os.path.join(out_dir, f"{name}_fit.pdf"))
            print(f"  Plot: {out_dir}/{name}_fit.pdf")
        except Exception as exc:
            logging.warning(f"  (plot skipped: {exc})")

    total = time.time() - t_all
    _banner("Done")
    print(f"  fitted   : {len(times)}/{len(targets)}")
    print(f"  total    : {total:.1f}s")
    if len(times) > 1:
        import statistics
        print(f"  first    : {times[0]:.1f}s   (includes one-time setup)")
        print(f"  median   : {statistics.median(times[1:]):.1f}s per spectrum")
    print(f"  output   : {out_dir}/")
    if failures:
        print(f"\n  {len(failures)} failed:")
        for n, e in failures:
            print(f"    {n}: {e}")
    return 1 if failures else 0




if __name__ == "__main__":
    sys.exit(main(sys.argv))
