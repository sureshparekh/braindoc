Source code for brainsp.mock

"""
Mock spectra and a mock datacube whose right answer is known.

Every mock is built from the shipped BasesGM templates by a forward model
that is written out here in full, independently of the fitter:

1. a population: light fractions :math:`x_j` of chosen templates, each
   template normalised to 1 at :math:`\\lambda_0` = 5500 A;
2. dust: the mixture is reddened by
   :math:`10^{-0.4 A_V [q(\\lambda) - q(\\lambda_0)]}`, with
   :math:`q = A_\\lambda / A_V` from the chosen law, so the fractions stay
   light fractions at :math:`\\lambda_0`;
3. kinematics: a Gaussian line-of-sight velocity distribution, mean
   :math:`v_\\star` and dispersion :math:`\\sigma_\\star`, applied on a
   logarithmic wavelength grid;
4. emission lines: Gaussians of flux :math:`F_i`, velocity
   :math:`v_{gas}` and dispersion :math:`\\sigma_{gas}`, at vacuum
   wavelengths from BRAIN's line catalogue;
5. redshift: wavelengths multiplied by :math:`1 + z` (flux density not
   rescaled, BRAIN's convention);
6. Galactic reddening at the observed wavelengths, :math:`E(B-V)` with
   :math:`R_V = 3.1` and the CCM law;
7. noise: Gaussian, with signal-to-noise ``sn`` per pixel at
   :math:`\\lambda_0`, scaled as the square root of the flux elsewhere.

The true values are written beside each file (``<name>.truth.json``, and a
``TRUTH`` FITS file for the cube), which is what :mod:`brainsp.selftest`
compares BRAIN's answers with. Everything is seeded, so the same call writes
the same files on every machine.

Typical use::

    import brainsp
    folder = brainsp.mock.write_examples()          # ~/.brainsp/examples
    r = brainsp.fit_spectrum(folder + "/spectra/mock_old_population.spec")
"""

import json
import os

import numpy as np

C_KMS = 299792.458
NORM_WAVELENGTH = 5500.0
# MaNGA's sampling: d(log10 lambda) = 1e-4, 69.0 km/s per pixel; the cube is
# written on it. Models are built on a grid ten times finer, 6.9 km/s, so
# that sampling them onto the output grid does not itself broaden them:
# interpolating from a 69 km/s grid adds about 28 km/s in quadrature, which
# a fit would then (correctly) report as dispersion.
LOG10_STEP = 1e-4
FINE_STEP = 1e-5
REST_RANGE = (3600.0, 7150.0)

__all__ = ["make_spectrum", "write_spectrum", "make_cube", "write_cube",
           "write_examples", "SPECTRA", "LINE_RATIOS"]

_template_cache = {}


def _template(age, metallicity, library="BasesGM"):
    """One template on its native grid, normalised to 1 at 5500 A."""
    key = (library, round(float(age), -3), round(float(metallicity), 6))
    if key in _template_cache:
        return _template_cache[key]
    from brainsp import libraries
    from brainsp.brain_settings import read_catalogue
    folder = libraries.path(library)
    ssp, _ = read_catalogue(folder)
    best = min(ssp, key=lambda e: (abs(np.log10(e["age_gyr"] * 1e9)
                                       - np.log10(age))
                                   + abs(e["metallicity"] - metallicity)))
    data = np.loadtxt(os.path.join(folder, best["filename"]), comments="#")
    wave, flux = data[:, 0], data[:, 1]
    norm = np.interp(NORM_WAVELENGTH, wave, flux)
    out = (wave, flux / norm, best["age_gyr"] * 1e9, best["metallicity"],
           best["short_name"])
    _template_cache[key] = out
    return out


def _power_law(wave, alpha=1.5):
    """F_nu ~ nu^-alpha, i.e. F_lambda ~ lambda^(alpha - 2), 1 at 5500 A."""
    return (wave / NORM_WAVELENGTH) ** (alpha - 2.0)


def _q(wave, law="CCM", rv=3.1):
    """
    q(lambda) = A_lambda / A_V of Cardelli, Clayton & Mathis (1989).

    Written in NumPy here rather than taken from BRAIN's reddening module,
    which is JAX: building a mock must not start JAX, or the device of the
    fit that follows would already be chosen. Same coefficients, same
    branches; the self-test fits these mocks with BRAIN's own copy.
    """
    if str(law).upper() != "CCM":
        raise ValueError("the mocks are reddened with the CCM law")
    x = 10000.0 / np.atleast_1d(np.asarray(wave, dtype=np.float64))
    q = np.zeros_like(x)
    ir = (x >= 0.3) & (x < 1.1)
    q[ir] = 0.574 * x[ir] ** 1.61 - 0.527 * x[ir] ** 1.61 / rv
    opt = (x >= 1.1) & (x < 3.3)
    y = x[opt] - 1.82
    a = (1.0 + 0.17699 * y - 0.50447 * y ** 2 - 0.02427 * y ** 3
         + 0.72085 * y ** 4 + 0.01979 * y ** 5 - 0.77530 * y ** 6
         + 0.32999 * y ** 7)
    b = (1.41338 * y + 2.28305 * y ** 2 + 1.07233 * y ** 3
         - 5.38434 * y ** 4 - 0.62251 * y ** 5 + 5.30260 * y ** 6
         - 2.09002 * y ** 7)
    q[opt] = a + b / rv
    uv = (x >= 3.3) & (x < 8.0)
    xu = x[uv]
    fa = np.where(xu >= 5.9, -0.04473 * (xu - 5.9) ** 2
                  - 0.009779 * (xu - 5.9) ** 3, 0.0)
    fb = np.where(xu >= 5.9, 0.2130 * (xu - 5.9) ** 2
                  + 0.1207 * (xu - 5.9) ** 3, 0.0)
    a = 1.752 - 0.316 * xu - 0.104 / ((xu - 4.67) ** 2 + 0.341) + fa
    b = -3.090 + 1.825 * xu + 1.206 / ((xu - 4.62) ** 2 + 0.263) + fb
    q[uv] = a + b / rv
    return q

def _log_grid(lo, hi, step=LOG10_STEP):
    n = int(np.floor((np.log10(hi) - np.log10(lo)) / step)) + 1
    return 10 ** (np.log10(lo) + step * np.arange(n))


def _broaden(flux, sigma_kms, step=FINE_STEP):
    """Gaussian convolution on a log10 grid of the given step."""
    if sigma_kms <= 0:
        return flux
    kms_per_pixel = C_KMS * np.log(10.0) * step
    s = sigma_kms / kms_per_pixel
    half = int(np.ceil(5 * s))
    x = np.arange(-half, half + 1)
    kernel = np.exp(-0.5 * (x / s) ** 2)
    kernel /= kernel.sum()
    padded = np.pad(flux, half, mode="edge")
    return np.convolve(padded, kernel, mode="valid")


def _continuum(population, av, sigma, agn=0.0, law="CCM", library="BasesGM"):
    """Rest-frame, unshifted model on the fine log grid, and the truth."""
    grid = _log_grid(REST_RANGE[0] - 200.0, REST_RANGE[1] + 200.0, FINE_STEP)
    total = np.zeros_like(grid)
    used = []
    stellar = sum(f for _, _, f in population)
    scale = (1.0 - agn) / stellar
    for age, metallicity, fraction in population:
        wave, flux, true_age, true_z, name = _template(age, metallicity,
                                                       library)
        total += scale * fraction * np.interp(grid, wave, flux)
        used.append({"age_yr": true_age, "Z": true_z, "name": name,
                     "light_fraction": scale * fraction})
    if agn:
        total += agn * _power_law(grid)
        used.append({"age_yr": 0.0, "Z": 0.0, "name": "FC1.50",
                     "light_fraction": agn})
    total *= 10 ** (-0.4 * av * (_q(grid, law) - _q(np.array(
        [NORM_WAVELENGTH]), law)[0]))
    return grid, _broaden(total, sigma), used


def _lines(grid, kind, halpha, v_gas, sigma_gas):
    from brainsp.line_catalogue import EMISSION_LINES_OPTICAL_VACUUM as LINES
    flux = np.zeros_like(grid)
    truth = {}
    for name, ratio in LINE_RATIOS[kind].items():
        rest = LINES[name]
        centre = rest * (1.0 + v_gas / C_KMS)
        width = sigma_gas / C_KMS * centre
        f = halpha * ratio
        flux += f / (np.sqrt(2 * np.pi) * width) * np.exp(
            -0.5 * ((grid - centre) / width) ** 2)
        truth[name] = f
    return flux, truth

def make_spectrum(population, av=0.3, v=0.0, sigma=150.0, sn=30.0, agn=0.0,
                  lines=None, halpha=5.0, v_gas=0.0, sigma_gas=80.0,
                  redshift=0.0, galactic_ebv=0.0, law="CCM", seed=0,
                  library="BasesGM", step=1.0, flag_fraction=0.002):
    """
    Build one mock spectrum.

    Parameters
    ----------
    population : list of (age_yr, Z, light_fraction)
        Templates (the nearest in the library is used) and their share of
        the stellar light at 5500 A; the shares are rescaled to sum to
        ``1 - agn``.
    av : float
        V-band extinction, magnitudes.
    v, sigma : float
        Stellar velocity and dispersion, km/s.
    sn : float
        Signal-to-noise per pixel at 5500 A (rest).
    agn : float
        Light fraction at 5500 A of an F_nu ~ nu^-1.5 power law.
    lines : {"star forming", "seyfert", "liner"} or None
        Emission-line ratios (see :data:`LINE_RATIOS`).
    halpha : float
        H-alpha flux, in units of the continuum at 5500 A times Angstrom
        (so it is roughly the H-alpha equivalent width in Angstrom).
    v_gas, sigma_gas : float
        Gas velocity and dispersion, km/s.
    redshift : float
        Wavelengths are multiplied by 1 + z.
    galactic_ebv : float
        Milky Way reddening applied at observed wavelengths (CCM, R_V=3.1).
    law : str
        The galaxy's own extinction law.
    seed : int
        Random seed of the noise.
    step : float
        Output pixel size in Angstrom (observed frame).
    flag_fraction : float
        Fraction of pixels flagged bad (column 4 = 1), at random.

    Returns
    -------
    wave, flux, error, flag : numpy.ndarray
    truth : dict
        Everything the spectrum was built with.
    """
    rng = np.random.default_rng(seed)
    grid, model, used = _continuum(population, av, sigma, agn, law, library)
    truth_lines = {}
    if lines:
        line_flux, truth_lines = _lines(grid, lines, halpha, v_gas, sigma_gas)
        model = model + line_flux
    lo = REST_RANGE[0] * (1 + redshift)
    hi = REST_RANGE[1] * (1 + redshift)
    wave = np.arange(np.ceil(lo), np.floor(hi), step)
    rest = wave / (1.0 + redshift) / (1.0 + v / C_KMS)
    # The stellar part moves with v; the lines were placed at v_gas already,
    # so they are sampled at the systemic rest wavelength.
    flux = np.interp(rest, grid, model - (line_flux if lines else 0.0))
    if lines:
        flux += np.interp(wave / (1.0 + redshift), grid, line_flux)
    if galactic_ebv:
        flux *= 10 ** (-0.4 * 3.1 * galactic_ebv * _q(wave, "CCM", 3.1))
    level = np.interp(NORM_WAVELENGTH * (1 + redshift), wave, flux)
    error = (level / sn) * np.sqrt(np.clip(flux / level, 0.05, None))
    flux = flux + rng.normal(0.0, 1.0, flux.size) * error
    flag = (rng.random(flux.size) < flag_fraction).astype(int)
    stellar = [u for u in used if u["age_yr"] > 0]
    total = sum(u["light_fraction"] for u in stellar) or 1.0
    logt = sum(u["light_fraction"] * np.log10(u["age_yr"])
               for u in stellar) / total
    zmean = sum(u["light_fraction"] * u["Z"] for u in stellar) / total
    truth = {
        "components": used, "av": av, "v": v, "sigma": sigma, "sn": sn,
        "agn_fraction": agn, "mean_log_age_light": logt,
        "mean_metallicity_light": zmean, "redshift": redshift,
        "galactic_ebv": galactic_ebv, "law": law,
        "lines": truth_lines, "line_ratios": lines, "v_gas": v_gas,
        "sigma_gas": sigma_gas, "wavelength_scale": "vacuum",
        "normalisation_wavelength": NORM_WAVELENGTH, "seed": seed,
    }
    return wave, flux, error, flag, truth

def write_spectrum(path, wave, flux, error, flag, truth=None):
    """Write a spectrum in BRAIN's text format, and its truth beside it."""
    os.makedirs(os.path.dirname(os.path.abspath(path)), exist_ok=True)
    header = ("BRAIN mock spectrum. Columns: wavelength [A]  flux  error  "
              "flag (1 = bad pixel)")
    np.savetxt(path, np.column_stack([wave, flux, error, flag]),
               fmt=["%.3f", "%.6e", "%.6e", "%d"], header=header)
    if truth is not None:
        with open(os.path.splitext(path)[0] + ".truth.json", "w") as handle:
            json.dump(truth, handle, indent=2, default=float)
    return path

def cube_truth(n=16, redshift=0.02):
    """
    The maps the mock cube is built from: a rotating disc around an old
    bulge, with a star-forming ring and a LINER-like nucleus.

    Returns
    -------
    dict of numpy.ndarray, each (n, n)
        ``v`` (km/s, relative to systemic), ``sigma`` (km/s), ``av`` (mag),
        ``old``, ``intermediate``, ``young`` (light fractions at 5500 A),
        ``sn`` (S/N at 5500 A), ``halpha`` (H-alpha flux, units of the
        continuum at 5500 A times A), ``line_kind`` (0 none, 1 star forming,
        2 LINER), ``inside`` (spaxels with data), and the scalars
        ``redshift`` and ``n``.
    """
    y, x = np.mgrid[:n, :n].astype(float)
    cx = cy = (n - 1) / 2.0
    inc = np.radians(55.0)
    pa = np.radians(30.0)
    dx, dy = x - cx, y - cy
    xp = dx * np.cos(pa) + dy * np.sin(pa)
    yp = (-dx * np.sin(pa) + dy * np.cos(pa)) / np.cos(inc)
    r = np.hypot(xp, yp)
    phi = np.arctan2(yp, xp)
    v = 220.0 * np.tanh(r / 2.5) * np.cos(phi) * np.sin(inc)
    sigma = 60.0 + 140.0 * np.exp(-r / 3.0)
    av = 0.1 + 0.7 * np.exp(-((r - 4.5) / 2.0) ** 2)
    old = 0.25 + 0.65 * np.exp(-r / 3.5)
    young = 0.30 * np.exp(-((r - 5.0) / 1.8) ** 2)
    intermediate = 1.0 - old - young
    sn = 3.0 + 55.0 * np.exp(-r / 4.0)
    ring = np.exp(-((r - 5.0) / 1.5) ** 2)
    nucleus = np.exp(-(r / 1.2) ** 2)
    halpha = 12.0 * ring + 2.5 * nucleus
    kind = np.where(ring > 0.3, 1, np.where(nucleus > 0.3, 2, 0))
    inside = np.hypot(dx, dy) <= 0.5 * n - 0.6
    return dict(v=v, sigma=sigma, av=av, old=old, intermediate=intermediate,
                young=young, sn=sn, halpha=halpha, line_kind=kind,
                inside=inside, redshift=redshift, n=n)


CUBE_POPULATIONS = {
    "old": (1.26e10, 0.019),
    "intermediate": (1.0e9, 0.019),
    "young": (1.0e8, 0.007564),
}

def make_cube(n=16, redshift=0.02, seed=1, library="BasesGM"):
    """
    Build the mock datacube, as arrays.

    Returns
    -------
    wave : numpy.ndarray
        Observed wavelengths, log-sampled as MaNGA (d log10 lambda = 1e-4).
    flux, ivar : numpy.ndarray
        (n_wave, n, n), flux density and inverse variance; 0 outside the
        field.
    mask : numpy.ndarray
        (n_wave, n, n) int32; bit 10 (MaNGA's DONOTUSE) outside the field
        and on randomly flagged pixels.
    truth : dict
        :func:`cube_truth`.
    """
    rng = np.random.default_rng(seed)
    t = cube_truth(n, redshift)
    lo = REST_RANGE[0] * (1 + redshift)
    hi = REST_RANGE[1] * (1 + redshift)
    wave = _log_grid(lo, hi)
    fine = _log_grid(REST_RANGE[0] - 200.0, REST_RANGE[1] + 200.0, FINE_STEP)
    from brainsp.line_catalogue import EMISSION_LINES_OPTICAL_VACUUM as LINES
    bases = {}
    for key, (age, metallicity) in CUBE_POPULATIONS.items():
        w, f, *_ = _template(age, metallicity, library)
        bases[key] = np.interp(fine, w, f)
    q = _q(fine) - _q(np.array([NORM_WAVELENGTH]))[0]
    q_obs = None
    flux = np.zeros((wave.size, n, n), dtype=np.float32)
    ivar = np.zeros_like(flux)
    mask = np.zeros(flux.shape, dtype=np.int32)
    donotuse = 1 << 10
    for j in range(n):
        for i in range(n):
            if not t["inside"][j, i]:
                mask[:, j, i] = donotuse
                continue
            model = sum(t[k][j, i] * bases[k] for k in bases)
            model = model * 10 ** (-0.4 * t["av"][j, i] * q)
            model = _broaden(model, t["sigma"][j, i])
            v = t["v"][j, i]
            spec = np.interp(wave / (1 + redshift) / (1 + v / C_KMS),
                             fine, model)
            kind = {1: "star forming", 2: "liner"}.get(int(t["line_kind"][j, i]))
            if kind and t["halpha"][j, i] > 0.2:
                rest = wave / (1 + redshift)
                sg = 50.0 if kind == "star forming" else 150.0
                for name, ratio in LINE_RATIOS[kind].items():
                    centre = LINES[name] * (1 + v / C_KMS)
                    width = sg / C_KMS * centre
                    spec += (t["halpha"][j, i] * ratio
                             / (np.sqrt(2 * np.pi) * width)
                             * np.exp(-0.5 * ((rest - centre) / width) ** 2))
            # Brightness falls off with radius like an exponential disc.
            r = np.hypot(j - (n - 1) / 2, i - (n - 1) / 2)
            bright = 1e-16 * (np.exp(-r / 3.0) + 0.05)
            spec = spec * bright
            level = np.interp(NORM_WAVELENGTH * (1 + redshift), wave, spec)
            err = (level / t["sn"][j, i]) * np.sqrt(
                np.clip(spec / level, 0.05, None))
            flux[:, j, i] = spec + rng.normal(0, 1, wave.size) * err
            ivar[:, j, i] = 1.0 / err ** 2
            bad = rng.random(wave.size) < 0.002
            mask[bad, j, i] = donotuse
            ivar[bad, j, i] = 0.0
    del q_obs
    return wave, flux, ivar, mask, t

def write_cube(path, n=16, redshift=0.02, seed=1):
    """
    Write the mock datacube as a MaNGA-like FITS file, and its truth.

    Extensions ``FLUX`` (1e-17 erg/s/cm^2/A/spaxel), ``IVAR``, ``MASK`` and
    ``WAVE``, with a celestial coordinate system on the two spatial axes
    (0.5 arcsec spaxels). The truth maps go to ``<path minus .fits>
    .truth.fits``, one extension per map.

    Returns
    -------
    str
        `path`.
    """
    from astropy.io import fits
    wave, flux, ivar, mask, t = make_cube(n, redshift, seed)
    scale = 1e17
    head = fits.Header()
    head["BUNIT"] = "1E-17 erg/s/cm^2/Angstrom/spaxel"
    head["CTYPE1"], head["CTYPE2"] = "RA---TAN", "DEC--TAN"
    head["CRPIX1"] = head["CRPIX2"] = (n + 1) / 2.0
    head["CRVAL1"], head["CRVAL2"] = 150.0, 2.2
    head["CDELT1"], head["CDELT2"] = -0.5 / 3600.0, 0.5 / 3600.0
    head["CUNIT1"] = head["CUNIT2"] = "deg"
    head["CTYPE3"] = "WAVE-LOG"
    head["CRPIX3"], head["CRVAL3"] = 1.0, float(wave[0])
    head["CDELT3"] = float(wave[0] * np.log(10) * LOG10_STEP)
    head["CUNIT3"] = "Angstrom"
    primary = fits.PrimaryHDU()
    primary.header["OBJECT"] = "BRAIN mock galaxy"
    primary.header["Z"] = (redshift, "redshift the cube was built at")
    primary.header["COMMENT"] = ("Mock datacube written by brainsp.mock: a "
                                 "rotating disc, old bulge, star-forming "
                                 "ring and LINER-like nucleus.")
    hdus = fits.HDUList([
        primary,
        fits.ImageHDU((flux * scale).astype(np.float32), head, name="FLUX"),
        fits.ImageHDU((ivar / scale ** 2).astype(np.float32), head,
                      name="IVAR"),
        fits.ImageHDU(mask, head, name="MASK"),
        fits.ImageHDU(wave.astype(np.float64), name="WAVE"),
    ])
    os.makedirs(os.path.dirname(os.path.abspath(path)), exist_ok=True)
    hdus.writeto(path, overwrite=True)
    truth = fits.HDUList([fits.PrimaryHDU()])
    for key in ("v", "sigma", "av", "old", "intermediate", "young", "sn",
                "halpha", "line_kind", "inside"):
        truth.append(fits.ImageHDU(np.asarray(t[key], dtype=np.float32),
                                   name=key.upper()))
    truth[0].header["Z"] = redshift
    truth[0].header["COMMENT"] = ("The maps brainsp.mock built the cube "
                                  "from. V is relative to systemic.")
    truth.writeto(path.replace(".fits", ".truth.fits"), overwrite=True)
    return path

def write_examples(folder=None, cube=True, overwrite=False):
    """
    Write every example the documentation and the self-test use.

    Writes ``spectra/`` (five mock spectra, each with ``.truth.json``, and
    ``corrections.csv``), ``masks/emission_lines.sm`` (the standard
    SDSS emission-line windows) and ``cube/mock_cube.fits`` (16 x 16 spaxels,
    with ``mock_cube.truth.fits``).

    Parameters
    ----------
    folder : str, optional
        Where; ``~/.brainsp/examples`` by default.
    cube : bool
        Also write the datacube (about 9 MB).
    overwrite : bool
        Rewrite files that exist.

    Returns
    -------
    str
        The folder.
    """
    from brainsp._paths import EXAMPLES_DIR
    from brainsp.io_utils import STANDARD_OPTICAL_MASK, write_mask_regions
    folder = os.path.abspath(folder or EXAMPLES_DIR)
    spectra = os.path.join(folder, "spectra")
    os.makedirs(spectra, exist_ok=True)
    for k, (name, spec) in enumerate(SPECTRA.items()):
        path = os.path.join(spectra, f"{name}.spec")
        if os.path.isfile(path) and not overwrite:
            continue
        options = {key: value for key, value in spec.items()
                   if key != "about"}
        wave, flux, err, flag, truth = make_spectrum(seed=k, **options)
        truth["about"] = spec["about"]
        write_spectrum(path, wave, flux, err, flag, truth)
    with open(os.path.join(spectra, "corrections.csv"), "w") as handle:
        handle.write("filename,redshift,ebv,rv,law\n")
        for name, spec in SPECTRA.items():
            handle.write(f"{name}.spec,{spec.get('redshift', 0.0)},"
                         f"{spec.get('galactic_ebv', 0.0)},3.1,CCM\n")
    masks = os.path.join(folder, "masks")
    os.makedirs(masks, exist_ok=True)
    write_mask_regions(os.path.join(masks, "emission_lines.sm"),
                       [(lo, hi, 0.0) for lo, hi in STANDARD_OPTICAL_MASK])
    if cube:
        path = os.path.join(folder, "cube", "mock_cube.fits")
        if overwrite or not os.path.isfile(path):
            write_cube(path)
    return folder
