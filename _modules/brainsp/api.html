"""
The brainsp scripting interface: everything the desktop interface does, from
Python.

The four things most people need::

    import brainsp

    # one spectrum
    result = brainsp.fit_spectrum("galaxy.spec", nebular_mode="espop",
                                  wavelength_scale="vacuum")
    print(result.v0, result.sigma, result.av, result.mean_log_age_light)

    # many spectra, each with its own redshift and Galactic reddening
    results = brainsp.fit_spectra("spectra/*.spec",
                                  corrections="corrections.csv")

    # a datacube
    cube = brainsp.fit_cube("manga-8626-12704-LOGCUBE.fits", redshift=0.04724,
                            sn_mask=[3, 5], fit_sn_threshold=3)
    velocity = cube.plane("vrot_star")

    # a datacube pipeline, every stage spelled out
    pipeline = brainsp.CubePipeline(cubes=[...], redshift=..., ...)
    pipeline.run()

Settings travel as a :class:`Settings` object: a dictionary of every BRAIN
setting with its default, which can be read from or written to a YAML file
and whose keys are also attributes. Every function also accepts settings as
keyword arguments, which override the :class:`Settings` given.

JAX picks its device the first time it is used in a process, so the device
has to be chosen before the first fit: pass ``device=`` to the first call, or
call :func:`set_device` first. ``"auto"`` (the default) uses a GPU when JAX
can see one.
"""

class Settings(dict):
    """
    Every BRAIN setting, with its default, as a dictionary.

    Keys are also attributes, so ``s.nebular_mode`` and
    ``s["nebular_mode"]`` are the same thing. An unknown key is refused, with
    the nearest valid name suggested, rather than silently ignored.

    Parameters
    ----------
    source : str, dict or Settings, optional
        A settings file (``.yaml``), or a mapping of settings. Missing keys
        take their defaults.
    **overrides
        Settings to change, by name.

    Examples
    --------
    >>> import brainsp
    >>> s = brainsp.Settings(nebular_mode="espop", wavelength_scale="vacuum")
    >>> s.dust_range
    [0.0, 5.0]
    >>> s.ages = [1e8, 1e9, 1e10]
    >>> s.changed()                       # what differs from the defaults
    {'nebular_mode': 'espop', 'wavelength_scale': 'vacuum', 'ages': [...]}
    >>> s.save("my_settings.yaml")
    >>> brainsp.Settings("my_settings.yaml") == s
    True
    >>> print(brainsp.Settings.describe("dust_range"))
    """

    # -- dictionary with checked keys -------------------------------------
    def _check(self, key):
        if key not in _bs.DEFAULTS:
            near = _bs._closest(key, _bs.DEFAULTS)
            raise KeyError(f"{key!r} is not a BRAIN setting"
                           + (f"; did you mean {near!r}?" if near else ".")
                           + " brainsp.Settings.names() lists them all.")

    def __setitem__(self, key, value):
        self._check(key)
        super().__setitem__(key, value)

    def changed(self):
        """The settings that differ from BRAIN's defaults."""
        return {k: v for k, v in self.items() if _bs.DEFAULTS.get(k) != v}



    def __repr__(self):
        changed = self.changed()
        inner = ", ".join(f"{k}={v!r}" for k, v in changed.items())
        return f"Settings({inner})"

    def save(self, path):
        """Write every setting to a YAML file that :class:`Settings` and
        ``brainsp fit --settings`` read back."""
        import yaml
        with open(path, "w") as handle:
            handle.write("# BRAIN settings, written by brainsp. Every key is "
                         "described in brainsp.Settings.describe(name).\n")
            yaml.safe_dump(dict(self), handle, sort_keys=False)
        return path

    @staticmethod
    def describe(name=None):
        """
        What a setting does, from the documented settings file.

        Parameters
        ----------
        name : str, optional
            A setting. Without it, every setting is described.
        """
        docs = _bs.setting_docs()
        if name is None:
            return "\n\n".join(f"{k} = {_bs.DEFAULTS[k]!r}\n{d['text']}"
                               for k, d in docs.items())
        if name not in _bs.DEFAULTS:
            Settings()._check(name)
        d = docs.get(name, {"text": "", "section": ""})
        return (f"{name} = {_bs.DEFAULTS[name]!r}   [{d['section']}]\n"
                f"{d['text']}")






def _settings(settings, overrides):
    if isinstance(settings, Settings):
        s = settings.copy()
    else:
        s = Settings(settings)
    s.update(overrides)
    return s

def set_device(device="auto"):
    """
    Choose the processor BRAIN fits on, before the first fit.

    Parameters
    ----------
    device : {"auto", "cpu", "gpu"}
        ``"auto"`` lets JAX choose (a GPU when it can see one), unless the
        ``JAX_PLATFORMS`` environment variable already names a platform.

    Raises
    ------
    RuntimeError
        If JAX has already started on a different device in this process.
        JAX chooses its device once per process; start a new Python session
        to change it.
    """
    device = str(device or "auto").lower()
    if device not in ("auto", "cpu", "gpu"):
        raise ValueError(f"device must be auto, cpu or gpu, not {device!r}")
    if "jax" in sys.modules:
        import jax
        try:
            backend = jax.default_backend()
        except Exception:
            backend = None
        if device == "cpu" and backend not in (None, "cpu"):
            raise RuntimeError(
                f"JAX is already running on {backend} in this process; "
                f"device='cpu' has to be chosen before the first fit.")
        if device == "gpu" and backend not in (None, "gpu", "cuda", "rocm"):
            raise RuntimeError(
                f"JAX is already running on {backend} in this process; "
                f"device='gpu' has to be chosen before the first fit.")
        return backend
    platform = _PLATFORM.get(device)
    if platform:
        os.environ["JAX_PLATFORMS"] = platform
    # "auto" leaves the environment as it is: a JAX_PLATFORMS the user set
    # stands, and with none JAX picks a GPU when it can see one.
    return device

def _prepare_engine(s):
    """Turn on the compiled-kernel cache and the template cache, once."""
    global _engine_ready
    if _engine_ready:
        return
    from brainsp import brain_cache, brainv1, io_utils
    import jax
    jax.config.update("jax_enable_x64", True)
    if s.get("use_cached_engine", True):
        brain_cache.enable_jax_cache(verbose=False)
    if s.get("use_cached_templates", True):
        raw = io_utils.load_ssp_templates
        memo = {}

        def cached(ssp_folder, bases_info, target_wl, norm_wavelength):
            key = brain_cache.ssp_cache_key(ssp_folder, bases_info, target_wl,
                                            norm_wavelength)
            if key not in memo:
                memo[key] = brain_cache.load_ssp_templates_cached(
                    ssp_folder, bases_info, target_wl, norm_wavelength, raw)
            return memo[key]

        io_utils.load_ssp_templates = cached
        brainv1.load_ssp_templates = cached
    _engine_ready = True


class _quiet:
    """Lower BRAIN's own logging for the duration of a fit."""

    def __init__(self, quiet):
        self.quiet = quiet

    def __enter__(self):
        if self.quiet:
            self.loggers = [logging.getLogger(n) for n in ("NEWBRAIN",
                                                            "brainsp")]
            self.levels = [lg.level for lg in self.loggers]
            for lg in self.loggers:
                lg.setLevel(logging.WARNING)

    def __exit__(self, *exc):
        if self.quiet:
            for lg, level in zip(self.loggers, self.levels):
                lg.setLevel(level)

class FitResult:
    """
    The result of fitting one spectrum, read from the files BRAIN wrote.

    A fit writes two files: ``<name>_summary.txt`` (the fitted parameters,
    the population vector, the emission lines) and ``<name>_fitted.txt`` (the observed and model spectra, column by column).
    This object reads both.

    Attributes
    ----------
    spectrum : str
        The input spectrum.
    summary_file, fitted_file : str
        The two output files.
    values : dict
        Every ``value [label]`` line of the summary, by label, as text.
    v0, sigma : float
        Stellar velocity and velocity dispersion in km/s (``v0_min``,
        ``vd_min``; the dispersion is OBSERVED, including the instrument,
        unless ``instrumental_resolution`` was given; see
        :attr:`sigma_star`).
    sigma_star : float or None
        The intrinsic stellar dispersion, when the instrumental resolution
        was known.
    av, yav : float
        V-band extinction of the whole population and the extra extinction
        of the youngest components, in magnitudes.
    chi2, chi2_data, adev : float
        Reduced chi-squared (with and without the penalty rows), and the
        mean absolute deviation in per cent.
    populations : numpy.ndarray
        Structured array, one row per template: ``x`` (light fraction, per
        cent, at the normalisation wavelength), ``mini`` and ``mcor``
        (initial and present mass fractions, per cent), ``age`` (yr), ``Z``,
        ``mstars``, ``name``, ``stellar`` (False for an AGN component).
    lines : dict
        Emission-line fluxes and errors, ``{name: (flux, error)}``, in the
        units of the input spectrum times Angstrom; empty in ``spop``.
        Names are BRAIN's (``"Halpha_6563"``, ``"[S II]_6716"`` ...). The
        tied doublets appear as ``"[O III]_Doublet"`` and
        ``"[N II]_Doublet"`` (the strong line's flux) and also under each
        member's own name, at the fixed ratio.
    wavelength, observed, model, stellar, emission, error, residual : ndarray
        The columns of the fitted-spectrum file.

    Examples
    --------
    >>> r = brainsp.fit_spectrum("galaxy.spec")
    >>> r.v0, r.sigma, r.av
    (188.3, 47.55, 0.2171)
    >>> r.mean_log_age_light, r.mean_log_age_mass
    >>> young = r.populations[r.populations["age"] < 1e8]["x"].sum()
    >>> r.plot("galaxy_fit.pdf")
    """

    error_message = None

    def __init__(self, summary_file, fitted_file=None, spectrum=None):
        self.summary_file = str(summary_file)
        self.fitted_file = str(fitted_file) if fitted_file else \
            self.summary_file.replace("_summary.txt", "_fitted.txt")
        self.values = {}
        self.lines = {}
        rows = []
        section = None
        with open(self.summary_file) as handle:
            for raw in handle:
                line = raw.rstrip("\n")
                if line.startswith("## Emission Lines"):
                    section = "lines"
                    continue
                if line.startswith("##"):
                    section = None if "Emission" not in line else section
                    continue
                if section == "lines" and "|" in line:
                    parts = [p.strip() for p in line.split("|")]
                    if len(parts) >= 3 and parts[0] != "Line/Doublet":
                        try:
                            self.lines[parts[0]] = (float(parts[1]),
                                                    float(parts[2]))
                        except ValueError:
                            pass
                    continue
                stripped = line.strip()
                if "[" in stripped and not stripped.startswith("#"):
                    value = stripped.split("[")[0].strip()
                    label = stripped.split("[", 1)[1].rsplit("]", 1)[0]
                    key = label.split("(")[0].strip()
                    self.values.setdefault(key, value)
                    continue
                parts = stripped.split()
                if len(parts) >= 10 and parts[0].isdigit():
                    rows.append(parts)
        # A tied doublet is fitted as one column whose weight is the strong
        # line's flux; its members are listed under their own names too.
        from brainsp.emission_lines import TIED_DOUBLETS
        for doublet, members in TIED_DOUBLETS.items():
            if doublet in self.lines:
                flux, err = self.lines[doublet]
                for member, ratio in members:
                    self.lines.setdefault(member, (flux * ratio, err * ratio))
        self.spectrum = spectrum or self.values.get("arq_obs")
        dtype = [("x", float), ("mini", float), ("mcor", float),
                 ("age", float), ("Z", float), ("mstars", float),
                 ("name", "U40"), ("stellar", bool)]
        table = np.zeros(len(rows), dtype=dtype)
        for i, p in enumerate(rows):
            name = p[9]
            # The summary states each component's kind in its last column;
            # files written before it did are read by the name conventions.
            stellar = (p[15] == "ssp" if len(p) > 15 else
                       not name.startswith(("FC", "AGN", "PowerOpt", "BB")))
            table[i] = (float(p[1]), float(p[2]), float(p[3]), float(p[4]),
                        float(p[5]), float(p[8]), name, stellar)
        self.populations = table
        self._spectrum_columns = None

    def line_fluxes(self, units="input", observed=False):
        """
        Emission-line fluxes, in the units asked for.

        The summary file holds fluxes relative to the normalised spectrum
        (continuum = 1 at the normalisation wavelength) and corrected for the
        stellar extinction relative to that wavelength. This converts them.

        Parameters
        ----------
        units : {"input", "normalised"}
            ``"input"`` multiplies by ``fobs_norm``, giving flux in the units
            of the input spectrum times Angstrom; ``"normalised"`` leaves
            them as written.
        observed : bool
            Undo the stellar-extinction correction, giving the flux as it
            arrives (still after any Galactic dereddening of the input).

        Returns
        -------
        dict
            ``{name: (flux, error)}``, doublets and their members included.
        """
        from brainsp.line_catalogue import (
            EMISSION_LINES_OPTICAL_VACUUM as LINES)
        scale = (self.fobs_norm or 1.0) if units == "input" else 1.0
        out = {}
        for name, (flux, err) in self.lines.items():
            factor = scale
            if observed and self.av:
                rest = LINES.get(name)
                if rest is None:
                    from brainsp.emission_lines import TIED_DOUBLETS
                    members = TIED_DOUBLETS.get(name)
                    rest = LINES.get(members[-1][0]) if members else None
                if rest is not None:
                    factor *= 10 ** (-0.4 * self.av * (
                        self._q(rest) - self._q(self._lambda0())))
            out[name] = (flux * factor, err * factor)
        return out



    def _lambda0(self):
        try:
            return float(self.values.get("normalisation_wavelength", 5500.0))
        except ValueError:
            return 5500.0

    def _q(self, wave):
        law = str(self.values.get("red_law_option", "CCM")).upper()
        if law == "CCM":
            from brainsp.mock import _q
            return float(_q([wave])[0])
        from brainsp.reddening_laws import get_k_lambda
        return float(np.asarray(get_k_lambda(np.array([float(wave)]),
                                             law))[0])

    # -- derived population quantities -----------------------------------
    def _mean(self, quantity, weight):
        p = self.populations[self.populations["stellar"]]
        w = p[weight]
        if w.sum() <= 0:
            return float("nan")
        return float(np.sum(w * quantity(p)) / w.sum())

    @property
    def mean_log_age_light(self):
        """<log t>_L: the light-weighted mean of log10(age / yr)."""
        return self._mean(lambda p: np.log10(p["age"]), "x")

    @property
    def mean_log_age_mass(self):

    @property
    def mean_metallicity_light(self):

    @property
    def mean_metallicity_mass(self):

    @property
    def agn_fraction(self):

def read_corrections_table(path):
    """
    Per-spectrum pre-fit corrections from a CSV file.

    The same table the desktop interface loads: a header row, the spectrum's
    file name in the first column, then any of ``redshift`` (or ``z``),
    ``ebv``, ``rv``, ``law`` and ``rest_frame``. A redshift in the table is
    one BRAIN removes (the spectrum is taken to be in the observed frame)
    unless ``rest_frame`` says otherwise. Empty cells keep the defaults.

    Parameters
    ----------
    path : str
        The CSV file. File names are relative to its folder.

    Returns
    -------
    dict
        ``{absolute spectrum path: {setting: value}}``, ready for
        :func:`fit_spectra`.

    Examples
    --------
    corrections.csv::

        filename,redshift,ebv,rv
        galaxy_a.spec,0.031,0.04,3.1
        galaxy_b.spec,0.0,0.0,
    """
    folder = os.path.dirname(os.path.abspath(path))
    out = {}
    with open(path, newline="") as handle:
        reader = csv.reader(handle)
        header = [h.strip().lower() for h in next(reader)]
        for row in reader:
            if not row or not row[0].strip():
                continue
            name = row[0].strip()
            spec = name if os.path.isabs(name) else os.path.join(folder, name)
            values = {}
            for key, text in zip(header[1:], row[1:]):
                text = text.strip()
                setting = _CORRECTION_KEYS.get(key)
                if setting is None or text == "":
                    continue
                if setting == "spectrum_in_rest_frame":
                    values[setting] = text.lower() in ("1", "true", "yes", "y")
                elif setting == "galactic_law":
                    values[setting] = text.upper()
                else:
                    values[setting] = float(text)
            if "spectrum_redshift" in values and \
                    "spectrum_in_rest_frame" not in values:
                values["spectrum_in_rest_frame"] = \
                    values["spectrum_redshift"] == 0.0
            out[os.path.abspath(spec)] = values
    return out




def _spectrum_path(spectrum, output_dir, name):
    """A file name for `spectrum`, writing arrays to a file when given."""
    if isinstance(spectrum, (str, os.PathLike)):
        return os.path.abspath(str(spectrum))
    arrays = [np.asarray(a, dtype=float) for a in spectrum]
    if len(arrays) not in (2, 3, 4):
        raise ValueError("a spectrum given as arrays is (wavelength, flux), "
                         "(wavelength, flux, error) or (..., flag)")
    os.makedirs(output_dir, exist_ok=True)
    path = os.path.join(output_dir, f"{name or 'spectrum'}.spec")
    np.savetxt(path, np.column_stack(arrays), fmt="%.8e")
    return path

def fit_spectrum(spectrum, settings=None, *, output_dir=None, device=None,
                 name=None, quiet=False, **overrides):
    """
    Fit one spectrum.

    Parameters
    ----------
    spectrum : str or tuple of arrays
        A spectrum file (columns: wavelength in Angstrom, flux, and
        optionally error and flag), or ``(wavelength, flux[, error[,
        flag]])`` arrays, which are written to ``output_dir`` first.
    settings : Settings, dict or str, optional
        The settings to fit with (a :class:`Settings`, a mapping, or a YAML
        file). Defaults to BRAIN's defaults.
    output_dir : str, optional
        Where the results go; ``settings.output_folder`` (``brain_out``) by
        default, relative to the working directory.
    device : {"auto", "cpu", "gpu"}, optional
        See :func:`set_device`. Only the first fit in a process can choose.
    name : str, optional
        Base name of the output files; the spectrum file's name by default.
    quiet : bool
        Show only warnings, not BRAIN's progress log.
    **overrides
        Any setting by name, e.g. ``nebular_mode="espop"``,
        ``spectrum_redshift=0.03``, ``spectrum_in_rest_frame=False``,
        ``galactic_ebv=0.05``, ``mask="lines.sm"``.

    Returns
    -------
    FitResult

    Examples
    --------
    >>> import brainsp
    >>> r = brainsp.fit_spectrum("galaxy.spec", nebular_mode="espop",
    ...                          wavelength_scale="vacuum")
    >>> r.lines["Halpha_6563"]
    """
    s = _settings(settings, overrides)
    if device is not None:
        set_device(device)
        s["device"] = device
    elif "jax" not in sys.modules:
        set_device(s.get("device", "auto"))
    out = os.path.abspath(output_dir or s["output_folder"] or "brain_out")
    s["output_folder"] = out
    path = _spectrum_path(spectrum, out, name)
    s["spectrum"] = path
    s.validate()
    from brainsp import brainv1
    from brainsp.run_brain import _write_settings_record
    stem = name or os.path.splitext(os.path.basename(path))[0]
    cfg = _bs.to_config(dict(s), spectrum_file=path)
    cfg.output_summary = os.path.join(out, f"{stem}_summary.txt")
    cfg.output_spectrum = os.path.join(out, f"{stem}_fitted.txt")
    os.makedirs(out, exist_ok=True)
    _prepare_engine(s)
    _write_settings_record(cfg, os.path.join(out, f"{stem}_settings.txt"),
                           path)
    with _quiet(quiet):
        brainv1.run(cfg)
    result = FitResult(cfg.output_summary, cfg.output_spectrum, path)
    if s.get("make_plot"):
        try:
            result.plot()
        except Exception as exc:
            logger.warning(f"plot not written: {exc}")
    return result




def _fit_spectra_workers(jobs, s, out, n_workers):
    """Fit (index, path, extra) jobs on CPU worker processes; results by index."""
    import pickle
    import subprocess
    import tempfile
    import threading
    env = dict(os.environ, JAX_PLATFORMS="cpu", OMP_NUM_THREADS="1",
               XLA_FLAGS="--xla_cpu_multi_thread_eigen=false "
                         "intra_op_parallelism_threads=1")
    parent = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
    env["PYTHONPATH"] = os.pathsep.join(
        [parent] + [p for p in env.get("PYTHONPATH", "").split(os.pathsep)
                    if p])
    folder = tempfile.mkdtemp(prefix="brainsp-spectra-")
    results, lock = {}, threading.Lock()
    procs = []
    try:
        for k in range(n_workers):
            share = jobs[k::n_workers]
            path = os.path.join(folder, f"job{k}.pkl")
            with open(path, "wb") as handle:
                pickle.dump(dict(settings=dict(s), output_dir=out,
                                 spectra=share), handle)
            procs.append(subprocess.Popen(
                [sys.executable, "-m", "brainsp.spectra_worker", path],
                stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, env=env,
                text=True))

        def read(proc):
            for line in proc.stdout:
                kind, _, rest = line.rstrip("\n").partition(" ")
                index, _, value = rest.partition(" ")
                if kind not in ("DONE", "FAILED") or not index.isdigit():
                    continue
                with lock:
                    results[int(index)] = (kind, value)
                    n = len(results)
                logger.info(f"[{n}/{len(jobs)}] {kind.lower()}: "
                            f"{os.path.basename(str(value))[:60]}")

        threads = [threading.Thread(target=read, args=(p,), daemon=True)
                   for p in procs]
        for t in threads:
            t.start()
        for p in procs:
            p.wait()
        for t in threads:
            t.join(timeout=5)
    finally:
        for p in procs:
            if p.poll() is None:
                p.kill()
        import shutil
        shutil.rmtree(folder, ignore_errors=True)
    return results

def fit_spectra(spectra, settings=None, *, corrections=None, output_dir=None,
                device=None, quiet=False, stop_on_error=False, workers=None,
                **overrides):
    """
    Fit several spectra, each with its own pre-fit corrections if given.

    Templates and the compiled engine are loaded once and reused, so a batch
    costs far less than the same fits run one at a time.

    Parameters
    ----------
    spectra : str or list of str
        Files, or a glob pattern such as ``"spectra/*.spec"``. May be omitted
        (``None``) when `corrections` is a table: its spectra are fitted.
    settings : Settings, dict or str, optional
        Settings shared by every spectrum.
    corrections : str or dict, optional
        A CSV table (see :func:`read_corrections_table`) or a mapping
        ``{spectrum: {setting: value}}`` of per-spectrum settings, typically
        ``spectrum_redshift``, ``spectrum_in_rest_frame``, ``galactic_ebv``,
        ``galactic_rv``, ``galactic_law``.
    output_dir, device, quiet, **overrides
        As for :func:`fit_spectrum`.
    stop_on_error : bool
        Raise on the first failure. Otherwise a failed spectrum gives a
        result whose ``error_message`` says why, and the batch carries on.
    workers : int or "auto", optional
        On a CPU, fit the spectra on this many worker processes at once,
        each with one thread and each keeping its own compiled fit warm.
        ``"auto"`` (the default on a CPU) uses every core but one, as many
        as the free memory allows; ``1`` fits them all in this process.
        On a GPU the spectra are fitted one after another in this process,
        which keeps the device busy on its own.

    Returns
    -------
    list of FitResult
        In the order the spectra were given.
    """
    table = {}
    if isinstance(corrections, (str, os.PathLike)):
        table = read_corrections_table(str(corrections))
    elif corrections:
        table = {os.path.abspath(str(k)): dict(v)
                 for k, v in corrections.items()}
    if spectra is None:
        paths = list(table)
    elif isinstance(spectra, (str, os.PathLike)):
        pattern = str(spectra)
        paths = sorted(glob.glob(pattern)) if any(c in pattern for c in "*?[") \
            else [pattern]
    else:
        paths = [str(p) for p in spectra]
    if not paths:
        raise ValueError(f"no spectra matched {spectra!r}")
    s0 = _settings(settings, overrides)
    if device is not None:
        set_device(device)
    elif "jax" not in sys.modules:
        set_device(s0.get("device", "auto"))
    n_workers = 1
    if len(paths) > 1 and not stop_on_error:
        backend = "cpu" if os.environ.get("JAX_PLATFORMS") == "cpu" else None
        if backend is None and "jax" in sys.modules:
            import jax
            backend = jax.default_backend()
        if backend is None:
            import jax
            backend = jax.default_backend()
        if backend == "cpu":
            from brainsp.fit_megacube_jax import cpu_worker_count
            requested = workers if workers is not None else \
                s0.get("cpu_workers", "auto")
            # Each worker compiles the fit once, a few seconds on one
            # thread, so each is given at least 8 spectra to spread it over.
            n_workers = cpu_worker_count(requested,
                                         max(1, len(paths) // 8))
    if n_workers > 1:
        out = os.path.abspath(output_dir or s0["output_folder"] or "brain_out")
        jobs = [(i, os.path.abspath(p), table.get(os.path.abspath(p), {}))
                for i, p in enumerate(paths)]
        logger.info(f"{len(paths)} spectra on {n_workers} CPU workers")
        found = _fit_spectra_workers(jobs, s0, out, n_workers)
        results = []
        for i, path in enumerate(paths):
            kind, value = found.get(i, ("FAILED", "the worker fitting it "
                                                  "stopped"))
            results.append(FitResult(value, spectrum=path) if kind == "DONE"
                           else _FailedFit(path, value))
        return results
    results = []
    for i, path in enumerate(paths, 1):
        extra = table.get(os.path.abspath(path), {})
        logger.info(f"[{i}/{len(paths)}] {os.path.basename(path)}")
        try:
            results.append(fit_spectrum(
                path, settings, output_dir=output_dir,
                device=device if i == 1 else None, quiet=quiet,
                **dict(overrides, **extra)))
        except (Exception, SystemExit) as exc:
            if stop_on_error:
                raise
            logger.error(f"{os.path.basename(path)} failed: {exc}")
            results.append(_FailedFit(path, str(exc)))
    return results

class CubePipeline:
    """
    A datacube run: the preparation stages, the fit, and its outputs.

    The stages run in this order, each only when asked for: Galactic dereddening, spatial binning, resolution degrading,
    spectral resampling, Butterworth filter, signal-to-noise mask, fit. The
    fit writes one megacube per cube, ``<cube>-BRAIN_MEGACUBE.fits``.

    Parameters
    ----------
    cubes : str or list of str
        Datacube FITS files (or a glob pattern).
    redshift : float or None
        The galaxy's redshift; BRAIN divides each cube's wavelengths by
        1 + z. Required, unless every cube has one in `per_cube`; 0 for a
        cube already in the rest frame.
    settings : Settings, dict or str, optional
        The physics of the fit (templates, mode, dust, kinematics ...), as
        for a spectrum. ``spectrum`` is ignored.
    output_dir : str
        Where megacubes are written.
    flux_hdu, error_hdu, mask_hdu : str
        Names of the cube's flux, uncertainty and bad-pixel extensions
        (MaNGA: ``FLUX``, ``IVAR``, ``MASK``). `mask_hdu` may be None.
    error_type : str
        What `error_hdu` holds: ``"inv_variance"``, ``"variance"``,
        ``"error"`` or ``"inv_error"``.
    wavelength_scale : {"air", "vacuum"}, optional
        The cube's wavelength scale. Required for ``espop``, ``nepop`` and
        ``photopop`` (MaNGA and SDSS are vacuum; MUSE and CALIFA air).
    galactic_ebv, galactic_rv, galactic_law : float, float, str
        Milky Way reddening to remove before anything else; 0 turns the
        stage off.
    spatial_binning : int, optional
        Bin spaxels n x n.
    degrade_resolution : dict, optional
        ``{"input type": "R", "input value": 2200, "output type": "R",
        "output value": 1000}``; resolutions as ``R``, ``FWHM`` or ``sigma``
        (Angstrom).
    spectral_binning : float, optional
        Resample the wavelength axis by this factor.
    butterworth : dict or bool, optional
        Spatial low-pass filter; ``True`` for the defaults or
        ``{"order": 3, "range": 0.3}``.
    sn_mask : list of float or dict, optional
        Signal-to-noise thresholds to compute masks for (e.g. ``[1, 3, 5,
        10]``), or ``{"thresholds": [...], "sn window": [4000, 6000],
        "method": "ivar"}``.
    fit_sn_threshold : float, optional
        Fit only the spaxels with S/N above this threshold (one of the
        `sn_mask` thresholds).
    wavelength_mask : str or list, optional
        A mask file (``.sm``, ``.gm`` or a list of ``start end
        [weight]``), or regions ``[(start, end, weight), ...]``; applied to
        every spaxel.
    batch_size : int, optional
        Spaxels fitted at once (automatic by default).
    population_bins : list, optional
        ``[(name, youngest_yr, oldest_yr), ...]`` for the PoPBins maps
        (min exclusive, max inclusive). Bins may overlap, so a total such as
        ``("xy_Tot", 10, 2e8)`` is a bin of its own.
    sfr_bins : list, optional
        ``[(name, min_yr, max_yr), ...]``: star-formation-rate maps, the mass
        formed in the bin over its width, Msun/yr. Need `galaxy_distance`.
    fc_bins, bb_bins : list, optional
        ``[(name, min, max), ...]``: the light (per cent) of the featureless
        power laws with min < exponent <= max, or blackbodies with min <
        temperature (K) <= max, e.g. ``[("FC_Tot", 0.245, 0.75)]``,
        ``[("BB_warm", 699, 1000)]``.
    galaxy_distance : float, optional
        Distance in Mpc; with it the megacube carries Mstar and Mpcross (Msun)
        and the SFR maps. Per cube: a ``galaxy distance`` column of
        `per_cube`.
    flux_scale : float, optional
        The number the cube's flux unit carries (1e-17 for MaNGA); read from
        BUNIT when not given.
    per_cube : str or list of dict, optional
        Per-cube overrides: a CSV file (first column the cube, then any of
        ``redshift``, ``ebv``, ``rv``, ``law``, ``mask file``, ...) or a list
        of ``{"target": path, "redshift": ...}``.
    overwrite : bool
        Re-run a cube whose megacube exists.
    fit : bool
        Run the fit (False runs the preparation stages only).

    Examples
    --------
    >>> import brainsp
    >>> p = brainsp.CubePipeline(
    ...     "manga-8626-12704-LOGCUBE.fits", redshift=0.04724,
    ...     settings=brainsp.Settings(nebular_mode="espop"),
    ...     wavelength_scale="vacuum", sn_mask=[3, 5], fit_sn_threshold=3,
    ...     wavelength_mask="emission_lines.sm")
    >>> p.validate()
    >>> outputs = p.run(device="gpu")
    >>> p.export_script("run_8626.py")    # the same run, as one Python file
    """

    @classmethod
    def from_config(cls, cfg, settings=None):
        """
        A pipeline from the dictionary BRAIN's datacube runner reads (what
        the desktop interface builds, or ``brain_ifu.load_config`` reads).

        The fit's settings come from the file `cfg` names unless `settings`
        is given; a wavelength-mask file is read into regions, so the
        pipeline no longer depends on it.
        """
        from brainsp.io_utils import read_mask_regions
        fit = cfg["fit"]
        if settings is None:
            path = fit.get("settings")
            settings = Settings(path) if path and os.path.isfile(str(path)) \
                else Settings()
        targets = cfg["targets"]
        columns = list(targets["columns"])
        rows = [r for r in targets["rows"] if r and str(r[0]).strip()]
        cubes = [os.path.abspath(os.path.join(targets["dir"] or "",
                                              str(r[0]).strip()))
                 for r in rows]
        per_cube = None
        if len(columns) > 1:
            per_cube = []
            for cube, r in zip(cubes, rows):
                entry = {"target": cube}
                for key, value in zip(columns[1:], r[1:]):
                    if str(value).strip():
                        entry[key] = value
                per_cube.append(entry)
        mask = fit.get("mask file")
        if mask and os.path.isfile(str(mask)):
            regions, _ = read_mask_regions(str(mask))
            mask = [(lo, hi, w) for lo, hi, w, _ in regions]
        opts = dict(
            output_dir=cfg["output"]["save_path_root"],
            flux_hdu=cfg["input"]["hdu flux"],
            error_hdu=cfg["input"]["hdu stat"],
            error_type=cfg["input"]["stat type"],
            mask_hdu=cfg["input"].get("hdu mask"),
            wavelength_scale=fit.get("wavelength scale"),
            fit_sn_threshold=fit.get("flag threshold"),
            wavelength_mask=mask or None,
            batch_size=fit.get("batch size"),
            population_bins=fit.get("population bins"),
            flxobs_step=fit.get("flxobs step"),
            sfr_bins=fit.get("sfr bins") or None,
            fc_bins=fit.get("fc bins") or None,
            bb_bins=fit.get("bb bins") or None,
            galaxy_distance=fit.get("galaxy distance"),
            flux_scale=fit.get("flux scale"),
            per_cube=per_cube,
            overwrite=cfg["output"].get("overwrite", True),
            save_config=cfg["output"].get("save_config", True),
            fit=fit.get("enabled", True))
        if cfg["spatial"]["enabled"]:
            opts["spatial_binning"] = cfg["spatial"]["resample size"]
        if cfg["resolution"]["enabled"]:
            opts["degrade_resolution"] = {k: v for k, v in
                                          cfg["resolution"].items()
                                          if k != "enabled"}
        if cfg["spectral_bin"]["enabled"]:
            opts["spectral_binning"] = cfg["spectral_bin"]["resample size"]
        if cfg["filter"]["enabled"]:
            opts["butterworth"] = {"order": cfg["filter"]["order"],
                                   "range": cfg["filter"]["range"]}
        if cfg["dereddening"]["enabled"]:
            d = cfg["dereddening"]
            opts.update(galactic_ebv=d["ebv"], galactic_rv=d["rv"],
                        galactic_law=d["law"])
        if cfg["sn_mask"]["enabled"]:
            m = cfg["sn_mask"]
            opts["sn_mask"] = {"thresholds": list(m["thresholds"]),
                               "sn window": list(m["sn window"]),
                               "method": m["method"]}
        return cls(cubes, fit.get("redshift"), settings, **opts)

    def _per_cube_rows(self):
        from brainsp import brain_ifu
        if not self.per_cube:
            return ["target"], [[c] for c in self.cubes]
        if isinstance(self.per_cube, (str, os.PathLike)):
            columns, rows = brain_ifu.read_targets_csv(str(self.per_cube))
            folder = os.path.dirname(os.path.abspath(str(self.per_cube)))
            rows = [[os.path.join(folder, r[0])] + list(r[1:]) for r in rows]
            return columns, rows
        columns = ["target"]
        for entry in self.per_cube:
            for key in entry:
                if key != "target" and key not in columns:
                    columns.append(key)
        rows = [[os.path.abspath(str(e["target"]))]
                + ["" if e.get(k) is None else str(e.get(k))
                   for k in columns[1:]] for e in self.per_cube]
        return columns, rows

    def _mask_file(self, out):

        if not self.wavelength_mask:

                   for r in self.wavelength_mask]
        path = os.path.join(out, "brain_wavelength_mask.sm")
        write_mask_regions(path, regions)
        return path

    def config(self):
        """
        The pipeline as the dictionary BRAIN's datacube runner reads.

        Writes the fit's settings, and a wavelength mask given as regions,
        into the output folder, because the runner reads both from files;
        the megacube then records exactly what it was fitted with.
        """
        from brainsp import brain_ifu
        cfg = brain_ifu.default_config()
        out = os.path.abspath(self.output_dir)
        os.makedirs(out, exist_ok=True)
        settings_path = os.path.join(out, "brain_settings_used.yaml")
        self.settings.save(settings_path)
        cfg["input"] = {"hdu flux": self.flux_hdu, "hdu stat": self.error_hdu,
                        "stat type": self.error_type,
                        "hdu mask": self.mask_hdu}
        if self.spatial_binning:
            cfg["spatial"] = {"enabled": True,
                              "resample size": int(self.spatial_binning)}
        if self.degrade_resolution:
            cfg["resolution"].update(dict(self.degrade_resolution),
                                     enabled=True)
        if self.spectral_binning:
            cfg["spectral_bin"] = {"enabled": True,
                                   "resample size": float(self.spectral_binning)}
        if self.butterworth:
            options = {} if self.butterworth is True else dict(self.butterworth)
            cfg["filter"].update(options, enabled=True)
        if self.galactic_ebv:
            cfg["dereddening"] = {"enabled": True, "law": self.galactic_law,
                                  "ebv": float(self.galactic_ebv),
                                  "rv": float(self.galactic_rv)}
        if self.sn_mask:
            options = ({"thresholds": list(self.sn_mask)}
                       if isinstance(self.sn_mask, (list, tuple))
                       else dict(self.sn_mask))
            cfg["sn_mask"].update(options, enabled=True)
            cfg["sn_mask"]["redshift"] = float(self.redshift or 0.0)
        cfg["fit"].update({
            "enabled": bool(self.fit), "settings": settings_path,
            "flag threshold": self.fit_sn_threshold,
            "batch size": self.batch_size,
            "redshift": None if self.redshift is None else float(self.redshift),
            "mask file": self._mask_file(out),
            "wavelength scale": self.wavelength_scale})
        if self.population_bins:
            cfg["fit"]["population bins"] = [list(b)
                                             for b in self.population_bins]
        if self.flxobs_step:
            cfg["fit"]["flxobs step"] = float(self.flxobs_step)
        for key, value in (("sfr bins", self.sfr_bins), ("fc bins", self.fc_bins),
                           ("bb bins", self.bb_bins)):
            if value:
                cfg["fit"][key] = [list(b) for b in value]
        cfg["fit"]["galaxy distance"] = self.galaxy_distance
        cfg["fit"]["flux scale"] = self.flux_scale
        columns, rows = self._per_cube_rows()
        cfg["targets"] = {"dir": "", "columns": columns, "rows": rows}
        cfg["output"] = {"save_path_root": out,
                         "save_config": bool(self.save_config),
                         "overwrite": bool(self.overwrite)}
        return cfg

    def validate(self):
        """
        Check the whole pipeline without reading a spaxel.

        Raises
        ------
        ValueError
            Listing every problem, each with the stage it belongs to.
        """
        from brainsp import brain_ifu
        problems = []
        if self.fit and self.redshift is None:
            columns, rows = self._per_cube_rows()
            if "redshift" not in [c.lower() for c in columns] or not all(
                    str(r[[c.lower() for c in columns].index("redshift")])
                    .strip() for r in rows):
                problems.append(("redshift", "give the galaxy's redshift (0 "
                                 "for a rest-frame cube), here or per cube"))
        problems += brain_ifu.validate_detailed(self.config())
        if problems:
            raise ValueError("This pipeline will not run:\n" + "\n".join(
                f"  [{section}] {message}" for section, message in problems))
        return True

    def run(self, device=None, progress=None, log=None):
        """
        Run the pipeline on every cube.

        Parameters
        ----------
        device : {"auto", "cpu", "gpu"}, optional
            See :func:`set_device`.
        progress : callable, optional
            Called as ``progress(fraction, message)`` as the run advances.
        log : callable, optional
            Called with each line of the run's log; ``print`` by default.

        Returns
        -------
        list of str
            The megacubes written.
        """
        from brainsp import brain_ifu
        set_device(device or self.settings.get("device", "auto"))
        self.validate()
        _prepare_engine(self.settings)
        summaries = brain_ifu.run(self.config(), progress=progress,
                                  log=log or print)
        return [s["output"] for s in summaries
                if s.get("state") == "done" and s.get("output")]

    def export_script(self, path, device="auto"):
        """
        Write this pipeline as one self-contained Python file.

        The file holds every setting and every stage, as Python, and runs
        the same pipeline with ``python <file>``: on a server, with no
        display, or as the record of how a result was made.
        """
        from brainsp.scripts import cube_script
        with open(path, "w") as handle:
            handle.write(cube_script(
                self, device=device,

        return path

def fit_cube(cube, redshift, settings=None, *, device=None, **options):
    """
    Fit one datacube and open the megacube it produces.

    Parameters
    ----------
    cube : str
        A datacube FITS file.
    redshift : float
        The galaxy's redshift (0 for a rest-frame cube).
    settings : Settings, dict or str, optional
        The physics of the fit.
    device : {"auto", "cpu", "gpu"}, optional
        See :func:`set_device`.
    **options
        Any :class:`CubePipeline` option (``output_dir``,
        ``wavelength_scale``, ``sn_mask``, ``fit_sn_threshold``,
        ``wavelength_mask``, ``galactic_ebv`` ...).

    Returns
    -------
    brainsp.megacube.Megacube
    """
    outputs = CubePipeline([cube], redshift, settings, **options).run(device)
    if not outputs:
        raise RuntimeError(f"{cube} was not fitted; see the log above")
    return read_megacube(outputs[0])
