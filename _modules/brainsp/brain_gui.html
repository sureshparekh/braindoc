def enable(on, reason, *widgets):
    """
    Enable or disable a group of controls, carrying the reason on each.

    A control that is live but whose value the run discards is the failure
    this interface is most exposed to, because the fit still finishes and
    the number still looks ordinary. Greying the control removes the
    possibility, and the tooltip answers the question the greying provokes.
    """
    for w in widgets:
        w.setEnabled(on)
        w.setToolTip("" if on else reason)

def coverage_sentence(cov):
    """
    Why a library does or does not support the nebular modes, in a sentence.

    Every number here is measured by io_utils.library_coverage from the
    template files themselves; this only chooses the words. Its own
    describe_coverage renders the same facts as an aligned block for a
    terminal, which does not fit a panel four hundred pixels wide.
    """
    if cov is None:
        return ""
    name = os.path.basename(str(cov["folder"]).rstrip("/")) or cov["folder"]
    if cov["blue"] is None:
        return f"{name} has no readable template files."
    limit = cov["lyman_limit"]
    head = (f"{name} covers {cov['blue']:,.0f} to {cov['red']:,.0f} Å; "
            f"{cov['n_ionising']} of {cov['n_read']} files reach the Lyman "
            f"limit at {limit:g} Å")
    if "nepop" not in cov["modes"]:
        return (f"{head}, so the nebular continuum would be identically "
                f"zero. nepop and photopop are unavailable with this "
                f"library.")
    if cov["partial"]:
        return (f"{head}. Only some of them do, so Q(H) would be integrated "
                f"over a biased subset of the library.")
    return f"{head}. All four modes are available."




def _focused_window_centre():
    """
    Centre of the window that currently has focus, in desktop coordinates.

    When BRAIN is started from a shell that window is the terminal, which is
    the best evidence we have about which monitor the user is looking at.
    X11 only; returns None anywhere else, including Wayland, where clients
    are deliberately not allowed to ask.
    """
    if sys.platform != "linux" or not os.environ.get("DISPLAY"):
        return None
    try:
        wid = os.environ.get("WINDOWID")
        if not wid:
            out = subprocess.run(["xprop", "-root", "_NET_ACTIVE_WINDOW"],
                                 capture_output=True, text=True,
                                 timeout=2).stdout
            m = re.search(r"(0x[0-9a-fA-F]+)", out)
            if not m:
                return None
            wid = m.group(1)
        out = subprocess.run(["xwininfo", "-id", str(wid)],
                             capture_output=True, text=True, timeout=2).stdout
        g = {k: int(v) for k, v in re.findall(
            r"(Absolute upper-left X|Absolute upper-left Y|Width|Height):"
            r"\s+(-?\d+)", out)}
        if len(g) < 4 or g["Width"] < 2:
            return None
        return QPoint(g["Absolute upper-left X"] + g["Width"] // 2,
                      g["Absolute upper-left Y"] + g["Height"] // 2)
    except Exception:
        return None

def launch_screen():
    """
    Pick the monitor to open on.

    Qt defaults to the primary screen, which on a multi-monitor desk is
    routinely not the one in front of the user. Prefer the screen holding
    the launching terminal, then the screen under the pointer, and only
    fall back to primary when neither can be determined.
    """
    for point in (_focused_window_centre(), QCursor.pos()):
        if point is not None:
            screen = QGuiApplication.screenAt(point)
            if screen is not None:
                return screen
    return QGuiApplication.primaryScreen()

def centre_on(widget, screen):
    """Centre a window on a screen, shrinking it if the screen is small."""
    if screen is None:
        return
    area = screen.availableGeometry()
    w = min(widget.width(), area.width() - 40)
    h = min(widget.height(), area.height() - 60)
    if (w, h) != (widget.width(), widget.height()):
        widget.resize(max(w, widget.minimumWidth()),
                      max(h, widget.minimumHeight()))
    widget.move(area.center().x() - widget.width() // 2,
                area.center().y() - widget.height() // 2)

class FitThread(ChildRun):
    """
    One fit, in a process of its own, watched from a thread of its own.

    The interface used to run the fit in this thread and set JAX_PLATFORMS
    just before it, which chose nothing: by then JAX had been used in this
    process and its backend was already fixed. The configuration now goes to
    a fresh interpreter whose environment names the platform before it
    starts, which is the only moment the variable is read, and that
    interpreter refuses to fit at all if it did not get the device asked for.

    The two caching settings are carried here rather than read from the
    settings file, because this is the only place in the interface that could
    honour them and a caching tick box that the fit ignored would be another
    control with no effect. They are honoured in the child, which is where the
    loader they replace now lives.

    Whatever the fit prints comes back as log messages and is written to this
    process's own output, so a BRAIN started from a terminal still shows its
    progress there.
    """

    status = Signal(str)
    failed = Signal(str)
    done = Signal(str, str)

    def __init__(self, cfg, device, cache_engine=True, cache_templates=True):
        super().__init__({"job": "spectrum",
                          "config": dataclasses.asdict(cfg),
                          "cache_engine": bool(cache_engine),
                          "cache_templates": bool(cache_templates)}, device)
        self.cfg = cfg

    def _load_spec_table(self):
        """
        A table of spectra, each with its own corrections, as the
        targets table is for cubes: filename first, then any of redshift,
        ebv, rv and law. A redshift in the table is one BRAIN removes: the
        spectra are taken to be in the observed frame unless a rest_frame
        column says true.
        """
        import csv
        path, _ = QFileDialog.getOpenFileName(
            self, "Open a table of spectra", os.getcwd(), "CSV (*.csv *.txt);;All files (*)")
        if not path:
            return
        base = os.path.dirname(os.path.abspath(path))
        table, missing = {}, []
        with open(path, newline="") as fh:
            reader = csv.reader(fh)
            header = [h.strip().lower() for h in next(reader)]
            for raw in reader:
                if not raw or not raw[0].strip() or raw[0].strip().startswith("#"):
                    continue
                cells = dict(zip(header, (c.strip() for c in raw)))
                name = raw[0].strip()
                full = name if os.path.isabs(name) else os.path.join(base, name)
                if not os.path.isfile(full):
                    missing.append(name)
                    continue

                def num(*keys, default=0.0):
                    for k in keys:
                        if cells.get(k):
                            return float(cells[k])
                    return default
                rest = str(cells.get("rest_frame", "")).lower() in ("1", "true", "yes")
                table[full] = dict(z=num("redshift", "z"), ebv=num("ebv", "e(b-v)"),
                                   rv=num("rv", "r_v", default=3.1),
                                   law=(cells.get("law") or "CCM").upper(),
                                   rest=rest or not num("redshift", "z"))
        self.spec_table = table
        self.spectra = list(table)
        self._filter()
        self.lbl_table.setText(
            f"{len(table)} spectra from {os.path.basename(path)}"
            + (f"; {len(missing)} not found: {', '.join(missing[:3])}" if missing else ""))
        self.btn_fit_all.setVisible(bool(table))
        self.btn_fit_all.setText(f"Fit all {len(table)} spectra in the table")
        if self.lst.count():
            self.lst.setCurrentRow(0)

    def _fit_all(self):
        """
        Fit every spectrum in the table, in one process.

        Each spectrum's configuration is built here, with its own row of the
        table applied, and validated before anything runs; a row that would
        be refused is reported and left out. The rest go to one child
        process, which compiles the fit once and keeps it warm.
        """
        if self.thread is not None and self.thread.isRunning():
            return
        configs, refused = [], []
        for path in self.spec_table:
            self.spec_path = path
            self._apply_table_row(path)
            try:
                configs.append(self._config())
            except (SystemExit, Exception) as exc:
                refused.append(f"{os.path.basename(path)}: {exc}")
        if refused:
            QMessageBox.warning(self, APP_NAME, "Left out of the run:\n\n"
                                + "\n\n".join(refused)[:1800])
        if not configs:
            return
        self._batch = {"n": len(configs), "done": 0, "failed": [],
                       "seconds": []}
        self.btn_fit.setEnabled(False)
        self.btn_fit_all.setEnabled(False)
        self.btn_fit.setText("  Fitting the table…")
        self.thread = BatchFitThread(configs, self.cb_device.currentText())
        self.thread.status.connect(self.lbl_status.setText)
        self.thread.item_done.connect(self._batch_item_done)
        self.thread.item_failed.connect(self._batch_item_failed)
        self.thread.all_done.connect(self._batch_finished)
        self.thread.failed.connect(self._failed)
        self.thread.start()

    def _batch_item_done(self, spec_file, summary_file, seconds):
        self._batch["done"] += 1
        self._batch["seconds"].append(seconds)
        self._show_fit(spec_file, summary_file)

    def _batch_item_failed(self, spectrum, text):
        self._batch["failed"].append((spectrum, text))

    def _batch_finished(self):
        b = self._batch
        self._reset_button()
        self.btn_fit_all.setEnabled(True)
        warm = b["seconds"][1:]
        speed = (f"; first {b['seconds'][0]:.1f} s, then "
                 f"{sum(warm) / len(warm):.2f} s each" if warm else "")
        self.lbl_status.setText(f"table done: {b['done']} of {b['n']} "
                                f"fitted{speed}")
        if b["failed"]:
            failure_dialog(self, "\n\n".join(
                f"{os.path.basename(s)}:\n{t}" for s, t in b["failed"]),
                self._out_dir(), APP_NAME)

    def _fit_next(self):
        if not self._queue:
            self.lbl_status.setText("table done")
            return
        if self.thread is not None and self.thread.isRunning():
            # "done" arrives while the finished fit's thread is still
            # winding down; starting now would be refused by _fit's guard
            # after this spectrum had already left the queue.
            QTimer.singleShot(100, self._fit_next)
            return
        path = self._queue.pop(0)
        self.spec_path = path
        self._apply_table_row(path)
        if path in self._shown:
            self.lst.blockSignals(True)
            self.lst.setCurrentRow(self._shown.index(path))
            self.lst.blockSignals(False)
        self._fit()

    def _update_steps(self):
        """
        Each step's one line of state, and whether it is ready.

        Read straight from the widgets, so the rail cannot say something the
        fit would not do.
        """
        from brainsp.brain_shell import READY, TODO, ATTENTION
        if not hasattr(self, "ws_spec"):
            return
        ws = self.ws_spec
        name = os.path.basename(self.spec_path) if self.spec_path else ""
        ws.set_state(0, READY if name else TODO, name or "no spectrum chosen")
        if self._scale_needed() and self.seg_scale.value() is None:

    def _plot(self):
        """
        The spectrum stage.

        One wavelength coordinate runs the full width and everything that has
        an opinion about wavelength registers to it: the spectrum, the
        residual, and a spine carrying the fitting range and the masks. The
        three sit in one figure sharing an x axis, so the registration is
        exact by construction rather than by two widgets agreeing on a
        margin.

        Axes are positioned in pixels on every resize rather than in figure
        fractions, because the spine and the tick strip are chrome and should
        not grow with the window while the data does.
        """
        w = QWidget()
        w.setObjectName("Stage")
        v = QVBoxLayout(w)
        v.setContentsMargins(0, 0, 0, 0)
        v.setSpacing(0)

        self.fig = Figure(figsize=(9, 7.4), dpi=100)
        self.ax = self.fig.add_axes([0.06, 0.30, 0.93, 0.68])
        self.ax_r = self.fig.add_axes([0.06, 0.16, 0.93, 0.12],
                                      sharex=self.ax)
        self.ax_s = self.fig.add_axes([0.06, 0.05, 0.93, 0.08],
                                      sharex=self.ax)
        self.canvas = FigureCanvasQTAgg(self.fig)
        self.canvas.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Expanding)
        self.canvas.setFocusPolicy(Qt.ClickFocus)

        self._nav = NavigationToolbar2QT(self.canvas, w)
        self._nav.hide()
        self._show_residual = False
        self._spans = []
        self._drag_edge = None

        well = QWidget()
        well.setObjectName("Well")
        wv = QVBoxLayout(well)
        wv.setContentsMargins(0, 0, 0, 0)
        wv.addWidget(self.canvas)

        v.addWidget(self._toolbar())
        v.addWidget(well, 1)

        self.canvas.mpl_connect("button_press_event", self._plot_click)
        self.canvas.mpl_connect("scroll_event", self._plot_scroll)
        self.canvas.mpl_connect("resize_event", lambda _: self._layout_axes())
        self.canvas.mpl_connect(
            "button_release_event",
            lambda ev: setattr(self, "_user_view", True)
            if str(self._nav.mode) else None)
        self._build_selectors()
        return w

    def _build_selectors(self):
        """
        Drag a region to mark it.

        A span selector on the spectrum and another on the spine, both
        writing the same mask, both blitting so the rubber band follows the
        pointer without repainting six thousand points behind it. They stand
        down while pan or zoom is armed, so one drag never means two things.
        """
        from matplotlib.widgets import SpanSelector
        for ax in (self.ax, self.ax_s):
            s = SpanSelector(
                ax, self._span_done, "horizontal", useblit=True,
                button=1, minspan=0.5, interactive=False,
                props=dict(facecolor=self.C["accent"], alpha=0.22),
                handle_props=dict(color=self.C["accent"]),
                drag_from_anywhere=True)
            self._spans.append(s)

    def _selectors_live(self, on):
        for s in self._spans:
            s.set_active(bool(on))

    def _span_done(self, lo, hi):
        lo, hi = sorted((float(lo), float(hi)))
        if hi - lo <= 0.5:
            return
        self.masks.append((lo, hi, WEIGHTS[self._weight_index()]))
        self._refresh_masks()

    def _layout_axes(self):
        """
        Place the three axes in pixels.

        The spine and the tick strip are fixed-height chrome; the residual
        takes a fixed share when there is one and no height at all when there
        is not, so an empty panel never sits on the screen waiting to be
        filled.
        """
        if not hasattr(self, "ax_s"):
            return
        W = max(self.fig.get_figwidth() * self.fig.dpi, 200.0)
        H = max(self.fig.get_figheight() * self.fig.dpi, 200.0)
        left, right = 58.0 / W, 1.0 - 12.0 / W
        width = right - left
        ticks, spine, gap, top_pad = 24.0 / H, 20.0 / H, 7.0 / H, 10.0 / H

        y_spine = ticks
        resid = max(0.14, 110.0 / H) if self._show_residual else 0.0
        y_resid = y_spine + spine + gap
        y_spec = y_resid + (resid + gap if resid else 0.0)
        h_spec = max(1.0 - top_pad - y_spec, 0.05)

        self.ax.set_position([left, y_spec, width, h_spec])
        self.ax_r.set_position([left, y_resid, width, max(resid, 1e-4)])
        self.ax_r.set_visible(self._show_residual)
        self.ax_s.set_position([left, y_spine, width, spine])
        self.canvas.draw_idle()

    def _toolbar(self):
        """

        What is on screen, said once, above the well: the file, how many

        """

        bar.setFixedHeight(30)
        h = QHBoxLayout(bar)
        h.setContentsMargins(14, 0, 12, 0)
        h.setSpacing(10)

        def tool(kind, tip, slot, checkable=False):
            b = QPushButton()

            b.setFixedSize(32, 32)
            b.setCheckable(checkable)
            b.setToolTip(tip)
            b.clicked.connect(slot)
            h.addWidget(b)
            self._tools.append((b, kind))
            return b

        self._tools = []

                             lambda: self._redraw(reset_view=True))

                             self._toggle_zoom, True)

        return bar

    def _plot_motion(self, ev):
        if ev.inaxes is None or ev.xdata is None:
            self.lbl_cursor.setText("")
        elif ev.inaxes is self.ax_s:
            hit = [(lo, hi, w) for lo, hi, w in self.masks
                   if lo <= ev.xdata <= hi]
            self.lbl_cursor.setText(
                f"{hit[0][0]:.1f} to {hit[0][1]:.1f} A   weight {hit[0][2]:g}"
                if hit else f"{ev.xdata:.1f} A")
        else:
            self.lbl_cursor.setText(f"{ev.xdata:9.2f} A  {ev.ydata:9.4f}")

    def _sync_modes(self):
        """
        Offer only the modes this library can actually run, and say why.

        A library whose spectra stop redward of the Lyman limit emits no
        ionising photons, so its nebular continuum is identically zero and
        nepop and photopop would run to completion and return a fit with
        nothing nebular in it. That is the failure this code is most prone
        to, so the two modes are struck through and disabled rather than
        left live, and the measured reason sits under them.

        A mode already chosen that the new library cannot run falls back to
        spop, which every library supports.
        """
        cov = self.coverage
        if cov is None:
            return
        modes = cov["modes"]
        reason = coverage_sentence(cov)
        for name in NEBULAR_MODES:
            self.seg_mode.set_available(name, name in modes, reason)
        self.lbl_cov.setText(reason)
        if self.seg_mode.value() is None:
            self.seg_mode.setValue("spop")
        self._mode(self.seg_mode.value())

    def _template_span(self, folder, entries):
        """Wavelength range common to every template, cached beside them."""
        cache = os.path.join(folder, ".coverage.json")
        try:
            if os.path.isfile(cache):
                d = json.load(open(cache))
                if d.get("n") == len(entries):
                    return d["lo"], d["hi"]
        except Exception:
            pass
        los, his = [], []
        for e in entries:
            p = os.path.join(folder, e["filename"])
            try:
                rows = [l for l in open(p) if l.strip()
                        and not l.lstrip().startswith("#")]
                los.append(float(rows[0].split()[0]))
                his.append(float(rows[-1].split()[0]))
            except Exception:
                continue
        if not los:
            return None, None
        lo, hi = max(los), min(his)
        try:
            json.dump({"lo": lo, "hi": hi, "n": len(entries)}, open(cache, "w"))
        except Exception:
            pass
        return lo, hi

    def _data_span(self):
        """
        The wavelengths the open workspace's data cover, and what they are.

        The Model panel is shared, so on the Datacube page this is the
        selected cube (rest frame) and on the Spectrum page the spectrum.
        """
        if getattr(self, "stack", None) is not None \
                and self.stack.currentIndex() == 1:
            if self._cube_span is not None:
                return self._cube_span, "cube"
            return None, "cube"
        if self.spec_data is not None:
            wl = self.spec_data[:, 0]
            return (float(wl.min()), float(wl.max())), "spectrum"
        return None, "spectrum"

    def _sync_wave_bounds(self, whole=False):
        """
        Hold the fitting range inside what both the data and the templates
        can support.

        Outside the template coverage every column is zero, so the model is
        identically zero there and the fit reproduces nothing at all; outside
        the data there is nothing to fit. Whatever the library spans, the
        range is bounded by the data. With `whole` (a cube was just selected)
        the range becomes all of it; otherwise a range that still fits is
        kept, and one that does not becomes all of the new span.
        """
        if self.tpl_lo is None:
            return
        lo, hi = self.tpl_lo, self.tpl_hi
        span, what = self._data_span()
        note = f"templates cover {lo:,.0f} to {hi:,.0f} Å"
        if span is not None:
            lo, hi = max(lo, span[0]), min(hi, span[1])
            if hi <= lo:
                self.lbl_wave.setText(
                    f"the {what} and the templates do not overlap")
                return
            if span[0] > self.tpl_lo or span[1] < self.tpl_hi:
                note = (f"limited by the {what} ({span[0]:,.0f} to "
                        f"{span[1]:,.0f} Å), templates reach "
                        f"{self.tpl_lo:,.0f} to {self.tpl_hi:,.0f} Å")
            else:
                note = (f"limited by the templates, the {what} spans "
                        f"{span[0]:,.0f} to {span[1]:,.0f} Å")
        cur_lo, cur_hi = self.sl_wave.value()
        keep = not whole and lo <= cur_lo < cur_hi <= hi
        if whole and not (lo <= self.sp_norm.value() <= hi):
            self._norm_auto = True      # a typed lambda_0 the new data lacks
        self._wave_setting = True
        self.sl_wave.setBounds(lo, hi, keep=keep)
        self._wave_setting = False
        for sp in (self.sp_wave_lo, self.sp_wave_hi):
            sp.blockSignals(True)
            sp.setRange(lo, hi)
            sp.blockSignals(False)
        self.lbl_wave.setText(note)
        self._show_wave()

    def _show_wave(self):
        """The typed fields, and lambda_0, follow the slider."""
        if not hasattr(self, "sp_norm"):

        lo, hi = self.sl_wave.value()

    def _identify(self, wl):
        """
        Say what is on screen in the title bar, not on the plot.

        The file, its pixel count and its span used to sit in a strip above
        the data along with a key and a hint about the cursor. That is five
        pieces of text arranged around a picture whose whole job is to be
        looked at. The window already has a bar whose purpose is to name what

        if wl is None or self.spec_path is None:
            self.setWindowTitle(f"{APP_NAME}  stellar population fitting")
            return
        self.setWindowTitle(
            f"{os.path.basename(self.spec_path)}    {len(wl)} px    "

    def _draw_spectrum(self):
        if self.spec_data is None:
            return self._blank()
        C = self.C
        wl, fl = self.spec_data[:, 0], self.spec_data[:, 1]
        lo, hi = self.sl_wave.value()
        m = (wl >= lo) & (wl <= hi)

        self.ax.clear()
        self.ax_r.clear()
        self._style_axes(residual=False)
        self.ax.plot(wl, fl, color=C["p_faint"], lw=0.6, zorder=1)
        if m.sum() > 2:
            self.ax.plot(wl[m], fl[m], color=C["p_obs"], lw=0.8, zorder=2)
        self._draw_spine()

        y = fl[m] if m.sum() > 10 else fl
        y = y[np.isfinite(y)]
        if y.size:
            q1, q99 = np.percentile(y, 1), np.percentile(y, 99)
            pad = 0.35 * (q99 - q1 + 1e-30)
            self.ax.set_ylim(q1 - pad, q99 + pad)
        self.ax.set_xlim(wl.min(), wl.max())
        self._identify(wl)
        self.canvas.draw_idle()
        self.lbl_info.setText("")

        self.ax_s.set_yticks([])

        for name, sp in self.ax_s.spines.items():
            sp.set_visible(name == "bottom")
            sp.set_color(C["border_2"])

                              width=1.0, colors=C["text_mute"], labelsize=9,
                              pad=3)
        for lab in self.ax_s.get_xticklabels():
            lab.set_fontfamily("monospace")

        lo, hi = self.sl_wave.value()
        x0, x1 = self.ax.get_xlim()

    def _mask_list_height(self):
        """
        Height enough for the rows there are, asked of the list itself.

        The height used to be rows times a constant measured against a face
        this interface no longer uses, so the last region was sliced in half
        by whatever sat under it. A list knows how tall its own row is.
        """
        if not self.masks:
            return 0
        row_h = self.lst_mask.sizeHintForRow(0)
        if row_h <= 0:
            row_h = TH.ROW
        frame = 2 * self.lst_mask.frameWidth() + 4
        return min(4 * row_h + frame, len(self.masks) * row_h + frame)

    def _scale_changed(self):
        """
        Show the wavelength scale as a question only where it is answerable.

        The setting describes the observed spectrum, and BRAIN reads it only
        when it is about to place emission lines. In spop no rest wavelength
        is ever read, so the control is put out of the way rather than asked
        for; in a line-fitting mode it has to be answered and the run is
        refused without it, so an unanswered control is marked as the error it
        will become rather than failing in a dialog after the button is
        pressed.

        The caption says what the wrong answer costs, because nothing in the
        output reveals it: the two scales differ by about 83 km/s, very nearly
        constant in velocity across the optical, so the difference is absorbed
        into the fitted gas velocity and chi2 barely moves.
        """
        needed = self._scale_needed()
        answered = self.seg_scale.value()
        self.lbl_scale_t.setVisible(needed)
        self.seg_scale.setVisible(needed)
        self.lbl_scale.setVisible(needed)
        # Its heading too, or spop leaves a title over nothing.
        if "Wavelength scale" in getattr(self, "secs", {}):
            self.secs["Wavelength scale"].setVisible(needed)
        if not needed:
            return
        self.seg_scale.setProperty("error", answered is None)
        self.seg_scale.style().unpolish(self.seg_scale)
        self.seg_scale.style().polish(self.seg_scale)
        if answered is None:
            self.lbl_scale.setText(
                "Needed before this mode can run. It is a fact about the "
                "survey, not about the galaxy: SDSS and MaNGA are vacuum, "
                "most ground-based optical spectra are air. BRAIN will not "
                "guess, because the wrong answer runs to completion and "
                "shifts the gas velocity by about 83 km/s with nothing in "
                "the output to show for it.")
        else:
            self.lbl_scale.setText(
                f"Rest wavelengths will be placed on the {answered} scale, "
                f"to match your spectrum.")

    def _model_settings(self):
        """
        The shared Model panel as settings: the physics both workspaces fit
        with. Read in one place so a spectrum and a cube cannot be given two
        different answers to the same question.

        lo, hi = self.sl_wave.value()

        sig_lo, sig_hi = self.sl_sig.value()

            wavelength_max=float(hi),
            normalisation_wavelength=float(self.sp_norm.value()),
            base_folder=self._library_folder(),
            ages=[a for a in self.ages if a0 - 1 <= a <= a1 + 1],
            metallicities=[z for z in self.metals
                           if z0 - 1e-9 <= z <= z1 + 1e-9],
            include_power_law=self.ck_pl.isChecked(),
            reddening_law=self.cb_law.currentData(),
            dust_range=list(self.sl_av.value()),
            velocity_range=list(self.sl_vel.value()),
            dispersion_range=[max(sig_lo, 1.0), sig_hi],
            nebular_mode=self.seg_mode.value(),
            smoothness=self.sp_smooth.value(),
            error_samples=self.sp_err.value(),
            reject_outliers=self.ck_out.isChecked(),
            device=self.cb_device.currentText(),
        )

    def _settings(self):
        """Every setting of this fit, as brain_settings reads them."""
        from brainsp import brain_settings as bs
        s = dict(bs.DEFAULTS)

        os.makedirs(self._out_dir(), exist_ok=True)
        mask_path = None
        if self.masks:
            mask_path = os.path.join(self._out_dir(), "_gui_mask.txt")
            with open(mask_path, "w") as f:
                for a, b, w in self.masks:
                    f.write(f"{a:.2f} {b:.2f} {w:.2f}\n")

        s.update(self._model_settings())
        s.update(
            spectrum=self.spec_path,
            mask=mask_path,
            spectrum_redshift=float(self.sp_z.value()),
            spectrum_in_rest_frame=bool(self.ck_rest.isChecked()),
            galactic_ebv=float(self.sp_ebv.value()),
            galactic_rv=float(self.sp_rv.value()),
            galactic_law=str(self.cb_glaw.currentData()),
            wavelength_scale=(self.seg_scale.value()
                              if self._scale_needed() else None),
            smoothness=self.sp_smooth.value(),
            error_samples=self.sp_err.value(),
            reject_outliers=self.ck_out.isChecked(),
            make_plot=self.ck_plot.isChecked(),
            output_folder=self._out_dir(),
            device=self.cb_device.currentText(),
        )
        return s

    def _show_fit(self, spec_file, summary_file):
        """Draw a finished fit and put its numbers in the readout."""
        try:
            d = np.loadtxt(spec_file)
        except Exception as exc:
            self.lbl_info.setText(f"cannot read the result: {exc}")
            return
        self._result = {
            "wl": d[:, 0], "obs": d[:, 1], "tot": d[:, 2], "star": d[:, 3],
            "emis": d[:, 4] if d.shape[1] > 4 else np.zeros(len(d)),
            "title": os.path.basename(spec_file),
        }
        self._redraw(reset_view=True)
        self._set_readout(summary_file)

    def _draw_result(self):
        C = self.C
        r = self._result

                                    r["emis"])

        self.ax.plot(wl, tot, color=C["p_fit"], lw=1.3, zorder=6)
        keys = [("observed", C["p_obs"]),

            self.ax.plot(wl, star, color=C["p_star"], lw=0.9, alpha=0.95,

            pad = 0.30 * (q99 - q1 + 1e-30)

        self._identify(wl)
