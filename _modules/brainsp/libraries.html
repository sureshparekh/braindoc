Source code for brainsp.libraries

"""
Template libraries: where they are, what is in them, how to add one.

A *library* is a folder of single-stellar-population (SSP) spectra with a
``CATALOGUE.txt`` that lists them, one per line::

    # AGE_YR      Z         FILENAME                 SHORT_NAME   MSTARS   KIND
    1.0e+06       0.019     bc03_z019_t1e6.dat       t1e6_z019    1.0      ssp

Each template file holds two columns, wavelength in Angstrom (rest frame,
air or vacuum as the library states) and flux per unit wavelength for the
population, normalised as the library normalises it. ``MSTARS`` is the
fraction of the initial mass still in stars (and remnants) at that age, used
to turn light fractions into stellar masses; ``KIND`` is ``ssp`` for a stellar
population and ``powerlaw`` for a featureless component such as an AGN
continuum. Lines starting with ``#`` are comments.

Libraries are looked up by name, in this order:

1. a folder path given directly (absolute, or relative to the working
   directory);
2. every folder in the ``BRAINSP_LIBRARY_PATH`` environment variable
   (separated by ``:`` on Linux and macOS, ``;`` on Windows);
3. ``~/.brainsp/libraries`` (or ``$BRAINSP_HOME/libraries``), where
   :func:`install` puts libraries added after installation;
4. the libraries shipped inside the package (``BasesGM``, ``M13`` and
   ``BC03``, the library the nebular mode nespop is validated with).

So a library added later, however large, does not have to live inside the
package: install it once with :func:`install`, or point
``BRAINSP_LIBRARY_PATH`` at the folder that holds it, and refer to it by its
folder name everywhere (``base_folder="MILES"`` in the settings, the Library
menu in the interface).

The ``AGN`` folder shipped with the package is not a library: it holds the
seven featureless continua that ``include_power_law`` adds to a library, and
it is never offered as one. A library can carry its own AGN set instead, in
an ``AGN`` sub-folder with a ``CATALOGUE.txt`` of the same layout (KIND
``powerlaw``, ``blackbody``, ...): ``include_power_law`` then adds that set,
and not the package's seven (:func:`agn_folder`). M13, for example, ships
power laws and hot-dust blackbodies built for it.
"""

import os
import shutil

from brainsp._paths import LIBRARY_DIR, USER_LIBRARY_DIR

CATALOGUE_NAME = "CATALOGUE.txt"
AGN_NAME = "AGN"
DEFAULT_LIBRARY = "BasesGM"

# The nebular mode (nespop) needs a library whose spectra reach below the
# Lyman limit (911.76 A): the populations' ionising photons are integrated
# from there, and set the nebular continuum and the Case B lines. BC03 is
# the library nespop has been validated with (independent PyNeb mocks, FADO
# on the same base, MaNGA 8626-12704 and 8566-1901), and the one the
# interface selects for it. Another library that reaches the Lyman limit can
# be used, with a warning that it has not been.
NEBULAR_LIBRARY = "BC03"
NEBULAR_VALIDATED = frozenset((NEBULAR_LIBRARY,))

def nebular_status(name_or_folder):
    """
    Whether a library can run the nebular mode, nespop.

    Returns
    -------
    dict
        ``status``: ``"validated"`` (it reaches the Lyman limit and nespop
        has been validated with it), ``"unvalidated"`` (it reaches the Lyman
        limit; nespop runs, untested with it) or ``"unable"`` (it does not:
        its nebular continuum would be identically zero); ``name``;
        ``reason``, a sentence saying which and why.
    """
    from brainsp import io_utils
    from brainsp.brain_settings import read_catalogue
    folder = path(name_or_folder)
    name = os.path.basename(os.path.normpath(folder))
    ssp, _ = read_catalogue(folder)
    cov = io_utils.library_coverage(folder, ssp)
    if "nespop" not in cov["modes"]:
        return {"status": "unable", "name": name, "reason": (
            f"{name} does not reach the Lyman limit at "
            f"{io_utils.LYMAN_LIMIT_A:g} A, so it has no ionising photons "
            f"and its nebular continuum would be zero. nespop needs "
            f"{NEBULAR_LIBRARY} (or another library reaching below "
            f"{io_utils.LYMAN_LIMIT_A:g} A).")}
    if name in NEBULAR_VALIDATED:
        return {"status": "validated", "name": name, "reason": (
            f"{name} reaches the Lyman limit; nespop is validated with it.")}
    return {"status": "unvalidated", "name": name, "reason": (
        f"{name} reaches the Lyman limit, so nespop runs, but it has been "
        f"validated only with {NEBULAR_LIBRARY}: its ionising photons, and "
        f"so the nebular continuum and the Case B lines, are untested "
        f"with {name}.")}

def agn_folder(library=None):
    """
    The AGN templates ``include_power_law`` adds to `library`.

    The library's own set, ``<library>/AGN/CATALOGUE.txt``, when it has one;
    otherwise the seven continua shipped with the package. With no library
    given, the package's set.
    """
    if library is not None:
        own = os.path.join(path(library), AGN_NAME)
        if os.path.isfile(os.path.join(own, CATALOGUE_NAME)):
            return own
    return os.path.join(LIBRARY_DIR, AGN_NAME)

def search_path():
    """
    The folders searched for libraries, first match winning.

    Returns
    -------
    list of str
        ``BRAINSP_LIBRARY_PATH`` entries, then the per-user library folder,
        then the package's own.
    """
    folders = []
    env = os.environ.get("BRAINSP_LIBRARY_PATH", "")
    folders += [p for p in env.split(os.pathsep) if p]
    folders += [USER_LIBRARY_DIR, LIBRARY_DIR]
    return folders




def _is_library(folder):
    """True when `folder` has a catalogue listing at least one SSP."""
    if os.path.basename(os.path.normpath(folder)) == AGN_NAME \
            and os.path.abspath(os.path.dirname(os.path.normpath(folder))) \
            == os.path.abspath(LIBRARY_DIR):
        return False
    if not os.path.isfile(os.path.join(folder, CATALOGUE_NAME)):
        return False
    from brainsp.brain_settings import read_catalogue
    try:
        ssp, _ = read_catalogue(folder)
    except SystemExit:
        return False
    return bool(ssp)

def available():
    """
    The names of every library that can be used, in search order.

    Returns
    -------
    list of str
        Folder names, e.g. ``["BasesGM"]``. A name found in two places is
        listed once, for the place that wins.

    Examples
    --------
    >>> import brainsp
    >>> brainsp.libraries.available()
    ['BasesGM', 'M13']
    """
    names = []
    for root in search_path():
        if not os.path.isdir(root):
            continue
        for name in sorted(os.listdir(root)):
            if name in names:
                continue
            if _is_library(os.path.join(root, name)):
                names.append(name)
    return names

def path(name_or_folder=DEFAULT_LIBRARY):
    """
    The folder of a library, from its name or a path to it.

    Parameters
    ----------
    name_or_folder : str
        A library name such as ``"BasesGM"`` (a trailing ``/`` is ignored),
        or a path to a folder holding a ``CATALOGUE.txt``.

    Returns
    -------
    str
        Absolute path of the library folder.

    Raises
    ------
    SystemExit
        When no folder of that name holds a catalogue. The message lists the
        libraries that are available and where they were looked for.
    """
    given = str(name_or_folder or DEFAULT_LIBRARY)
    if os.path.isfile(os.path.join(given, CATALOGUE_NAME)):
        return os.path.abspath(given)
    name = os.path.basename(os.path.normpath(given))
    for root in search_path():
        folder = os.path.join(root, name)
        if os.path.isfile(os.path.join(folder, CATALOGUE_NAME)):
            return os.path.abspath(folder)
    raise SystemExit(
        f"\nNo template library {given!r}.\n"
        f"Available: {', '.join(available()) or 'none'}.\n"
        f"Looked in: {given!r}, then " + ", ".join(search_path()) + ".\n"
        f"Add a library with brainsp.libraries.install(folder), or set "
        f"BRAINSP_LIBRARY_PATH to the folder that holds it.\n")

def info(name_or_folder=DEFAULT_LIBRARY):
    """
    What a library holds.

    Parameters
    ----------
    name_or_folder : str
        Library name or folder.

    Returns
    -------
    dict
        ``folder``; ``n_ssp``; ``ages_yr`` and ``metallicities`` (sorted);
        ``extra`` (names of non-SSP components the library carries);
        ``agn`` (the components ``include_power_law`` adds) and
        ``agn_folder`` (where they are: the library's own ``AGN`` set, or
        the package's);
        ``wavelength_min`` and ``wavelength_max`` in Angstrom (the range
        every template covers); ``reaches_lyman`` (True when every template
        extends below the Lyman limit at 911.76 A, which the nebular
        continuum mode ``nespop`` needs).
    """
    from brainsp.brain_settings import read_catalogue
    import numpy as np
    folder = path(name_or_folder)
    ssp, extra = read_catalogue(folder)
    lo, hi = [], []
    for e in ssp:
        data = np.loadtxt(os.path.join(folder, e["filename"]), comments="#",
                          usecols=(0,))
        lo.append(float(data.min()))
        hi.append(float(data.max()))
    agn_dir = agn_folder(folder)
    _, agn = read_catalogue(agn_dir)
    return {
        "folder": folder,
        "n_ssp": len(ssp),
        "agn": [e["short_name"] for e in agn],
        "agn_folder": agn_dir,
        "ages_yr": sorted({e["age_gyr"] * 1e9 for e in ssp}),
        "metallicities": sorted({e["metallicity"] for e in ssp}),
        "extra": [e["short_name"] for e in extra],
        "wavelength_min": max(lo) if lo else None,
        "wavelength_max": min(hi) if hi else None,
        "reaches_lyman": bool(lo) and max(lo) <= 911.76,
    }

def install(folder, name=None, overwrite=False):
    """
    Copy a library into the per-user library folder, so it can be named.

    Parameters
    ----------
    folder : str
        A folder with a ``CATALOGUE.txt`` and the template files it lists.
    name : str, optional
        The name to install it under; the folder's own name by default.
    overwrite : bool
        Replace a library already installed under that name.

    Returns
    -------
    str
        Where the library now is (``~/.brainsp/libraries/<name>``).

    Raises
    ------
    SystemExit
        If the folder is not a library, a file the catalogue names is
        missing, or the name is taken and `overwrite` is False.
    """
    from brainsp.brain_settings import read_catalogue
    folder = os.path.abspath(folder)
    ssp, extra = read_catalogue(folder)
    if not ssp:
        raise SystemExit(f"\n{folder} lists no SSP in its catalogue.\n")
    missing = [e["filename"] for e in ssp + extra
               if not os.path.isfile(os.path.join(folder, e["filename"]))]
    if missing:
        raise SystemExit(f"\n{len(missing)} file(s) the catalogue names are "
                         f"missing from {folder}: {missing[:5]}\n")
    name = name or os.path.basename(os.path.normpath(folder))
    target = os.path.join(USER_LIBRARY_DIR, name)
    if os.path.exists(target):
        if not overwrite:
            raise SystemExit(f"\nA library {name!r} is already installed at "
                             f"{target}; pass overwrite=True to replace it.\n")
        shutil.rmtree(target)
    os.makedirs(USER_LIBRARY_DIR, exist_ok=True)
    shutil.copytree(folder, target)
    return target
