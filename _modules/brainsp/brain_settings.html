Source code for brainsp.brain_settings

"""
BRAIN — reads brain_setup.yaml and turns it into a validated fit configuration.

You do not need to open this file. Edit brain_setup.yaml instead.

    python brain_settings.py            show the templates available
    python brain_settings.py --check    validate brain_setup.yaml

WHY YAML RATHER THAN PYTHON
---------------------------
The settings used to be a Python file that BRAIN imported. That works, but it
means a stray typo anywhere in it stops the program with a Python traceback
instead of a useful message, and a settings file that can execute arbitrary
code is a poor thing to email around. This file is *parsed*, never executed,
so the worst a malformed settings file can do is produce a clear error with a
line number.
"""

import os
import re

CATALOGUE_NAME = "CATALOGUE.txt"
from brainsp._paths import LIBRARY_DIR as _LIBRARY_DIR

AGN_FOLDER = os.path.join(_LIBRARY_DIR, "AGN")
"""
The AGN continuum templates, shared by every stellar library.

An AGN's featureless continuum has nothing to do with which stellar library
describes the host, so it is not a member of any library: `include_power_law`
adds this one set to whichever library is chosen. Seven templates: power laws
F_nu ~ nu^-alpha with alpha = 0.5, 1.0, 1.5, 2.0, and
Cloudy's "AGN" incident continuum for big-bump temperatures 1e5, 1.5e5 and
2e5 K. Only their summed light is a measurement: in the optical the three
Cloudy continua are power laws of alpha 0.74, 0.64 and 0.59 to within 1.4 per
cent, so the split between neighbouring templates is set by noise.
"""
DEFAULT_SETTINGS_FILE = "brain_setup.yaml"

MATCH_TOLERANCE = 2e-2


def _yaml():
    try:
        import yaml
    except ImportError:
        raise SystemExit(
            "\nBRAIN needs PyYAML to read brain_setup.yaml.\n"
            "Install it with:   pip install pyyaml\n")
    return yaml


def _loader():
    """
    A YAML loader that reads scientific notation the way people write it.

    YAML 1.1 only treats a number as a float if it has both a decimal point
    and a signed exponent, so `1.0e+6` is a float but `1e6`, `5.01e9` and
    `1e-5` all silently come through as *strings*. Nobody writing an age in
    years expects that, and a string where a number belongs would either crash
    somewhere far away or, worse, be quietly coerced. So we widen the float
    pattern to cover the ordinary forms.
    """
    yaml = _yaml()

    class Loader(yaml.SafeLoader):
        pass

    Loader.add_implicit_resolver(
        "tag:yaml.org,2002:float",
        re.compile(r"""^(?:
             [-+]?[0-9][0-9_]*\.[0-9_]*(?:[eE][-+]?[0-9]+)?
            |[-+]?\.[0-9_]+(?:[eE][-+]?[0-9]+)?
            |[-+]?[0-9][0-9_]*(?:[eE][-+]?[0-9]+)
            |[-+]?\.(?:inf|Inf|INF)
            |\.(?:nan|NaN|NAN))$""", re.X),
        list("-+0123456789."))
    return Loader

def load(path=DEFAULT_SETTINGS_FILE):
    """
    Read the settings file, fill in defaults, reject unknown keys.

    A setting that BRAIN used to have and no longer does is answered from
    RETIRED, with the reason it went, rather than through the unknown-key
    path. A settings file written for an older BRAIN is a record of a fit
    somebody ran; the person reading the error is entitled to know what
    changed underneath them, and 'did you mean ...?' against the nearest
    surviving name would be worse than silence. Neither is it ignored: a
    setting that is parsed and does nothing is the defect this project keeps
    finding, and the fit it silently changes is one nobody asked for.
    """
    yaml = _yaml()
    if path == DEFAULT_SETTINGS_FILE and not os.path.isfile(path):
        # No settings file where we were started: the package's own copy,
        # which holds every default and documents every setting.
        from brainsp._paths import SETTINGS_TEMPLATE
        path = SETTINGS_TEMPLATE
    if not os.path.isfile(path):
        raise SystemExit(
            f"\nSettings file not found: {path!r}\n"
            f"BRAIN expects a {DEFAULT_SETTINGS_FILE} in the folder you run "
            f"it from.\n")
    try:
        with open(path) as fh:
            raw = yaml.load(fh, Loader=_loader()) or {}
    except yaml.YAMLError as exc:
        mark = getattr(exc, "problem_mark", None)
        where = f" (line {mark.line + 1}, column {mark.column + 1})" if mark else ""
        raise SystemExit(
            f"\n{path} is not valid YAML{where}.\n"
            f"  {getattr(exc, 'problem', exc)}\n\n"
            f"Common causes: a missing colon after a name, inconsistent "
            f"indentation, or a stray tab character (YAML needs spaces).\n")

    if not isinstance(raw, dict):
        raise SystemExit(f"\n{path} must be a list of 'name: value' lines.\n")

    retired = sorted(set(raw) & set(RETIRED))
    if retired:
        blocks = []
        for k in retired:
            body = "\n".join(f"    {chunk}" for chunk in _wrap(RETIRED[k], 68))
            blocks.append(f"  * {k!r} was removed.\n{body}")
        raise SystemExit(
            f"\n{path} sets {len(retired)} setting(s) BRAIN no longer has:\n"
            + "\n\n".join(blocks)
            + "\n\nRemove the line(s) and re-run.\n")

    unknown = sorted(set(raw) - set(DEFAULTS))
    if unknown:
        lines = []
        for k in unknown:
            near = _closest(k, DEFAULTS)
            lines.append(f"  * {k!r}" + (f"  — did you mean {near!r}?" if near else ""))
        raise SystemExit(
            f"\n{path} contains setting(s) BRAIN does not recognise:\n"
            + "\n".join(lines)
            + "\n\nRun  brainsp settings  to see every valid "
              "setting name.\n")

    s = dict(DEFAULTS)
    s.update(raw)
    return s

def setting_docs(path=None):
    """
    Each setting's documentation, read from the documented settings file.

    The template's layout: a setting's description is the comment block
    directly above it; a comment block that follows a setting after a blank
    line continues that setting (notes, examples); a block at the start of a
    section, before its first setting, introduces the section and is put in
    front of the first setting's own text. A comment on the setting's own
    line is added at the end. The numbered banner it sits under is its
    section.

    Returns
    -------
    dict
        ``{name: {"section": str, "text": str}}`` for every setting the file
        documents, in the file's order.
    """
    from brainsp._paths import SETTINGS_TEMPLATE
    docs, section, block, intro, last = {}, "", [], [], None
    key_line = re.compile(r"^([a-z_0-9]+):(.*)$")
    with open(path or SETTINGS_TEMPLATE) as handle:
        lines = handle.read().splitlines()

    def flush():
        nonlocal block, intro
        text = "\n".join(block).strip()
        block = []
        if not text:
            return
        if text.startswith("--- "):
            return
        if last is None:
            intro.append(text)
        else:
            docs[last]["text"] = (docs[last]["text"] + "\n\n" + text).strip()

    i = 0
    while i < len(lines):
        line = lines[i]
        banner = re.match(r"^#\s+(\d+[a-z]?\.\s+.*)$", line)
        if banner and i > 0 and lines[i - 1].startswith("# ====="):
            section = banner.group(1).strip()
            block, intro, last = [], [], None
        elif line.startswith("# =====") or line.startswith("#  ==="):
            block = []
        elif line.startswith("#"):
            text = line[1:]
            if text.startswith(" "):
                text = text[1:]
            block.append(text)
        elif not line.strip():
            flush()
        else:
            m = key_line.match(line)
            if m:
                name = m.group(1)
                inline = m.group(2).split("#", 1)[1].strip() \
                    if "#" in m.group(2) else ""
                own = "\n".join(block).strip()
                if own.startswith("--- "):
                    own = own.split("\n", 1)[1].strip() if "\n" in own \
                        else ""
                text = "\n\n".join([t for t in intro + [own] if t])
                if inline:
                    text = (text + "\n" + inline).strip()
                docs[name] = {"section": section, "text": text}
                block, intro, last = [], [], name
                while i + 1 < len(lines) and lines[i + 1].startswith(
                        (" ", "\t")) and lines[i + 1].strip():
                    i += 1
        i += 1
    flush()
    return docs




def _wrap(text, width):
    """
    Fold a sentence onto fixed-width lines.

    io_utils carries the same six lines, and they are deliberately not shared:
    this module reads a settings file and imports nothing beyond the standard
    library and PyYAML, so that a malformed setting is reported before numpy,
    JAX or a template library is ever touched.
    """
    words, line, out = text.split(), "", []
    for word in words:
        if line and len(line) + 1 + len(word) > width:
            out.append(line)
            line = word
        else:
            line = f"{line} {word}".strip()
    if line:
        out.append(line)
    return out


def _closest(word, options):
    """Cheap typo suggestion — no dependency on difflib subtleties."""
    import difflib
    m = difflib.get_close_matches(word, list(options), n=1, cutoff=0.7)
    return m[0] if m else None

def read_catalogue(folder):
    """
    Parse <folder>/CATALOGUE.txt.

    Comment lines start with #. Columns:
        AGE_YR  Z  FILENAME  SHORT_NAME  MSTARS  KIND
    KIND is 'ssp' or 'powerlaw'. It is carried onto every entry, because a
    component that is not an SSP has no line-spread function of its own and
    the instrumental-resolution module has to be able to tell it apart from a
    template that simply forgot to state one.
    """
    path = os.path.join(folder, CATALOGUE_NAME)
    if not os.path.isfile(path):
        raise SystemExit(
            f"\nNo template catalogue at {path!r}.\n"
            f"'base_folder' must point at a folder containing "
            f"{CATALOGUE_NAME}.\n")

    ssp, extra = [], []
    with open(path) as fh:
        for lineno, raw in enumerate(fh, 1):
            line = raw.split("#")[0].strip()
            if not line:
                continue
            p = line.split()
            if len(p) < 4:
                raise SystemExit(
                    f"\n{path} line {lineno}: expected at least 4 columns "
                    f"(AGE_YR Z FILENAME SHORT_NAME), found {len(p)}.\n")
            kind = p[5].lower() if len(p) > 5 else "ssp"
            entry = {
                "filename": p[2],
                "age_gyr": float(p[0]) / 1e9,
                "metallicity": float(p[1]),
                "short_name": p[3],
                "mstars": float(p[4]) if len(p) > 4 else 1.0,
                "yav_flag": 0,
                "kind": kind,
            }
            (extra if kind != "ssp" else ssp).append(entry)
    return ssp, extra

def stellar_libraries(root=None):
    """
    The template libraries under `root`: folders whose catalogue lists at
    least one stellar population.

    A folder with a catalogue is not necessarily a library. AGN/ has one and
    holds only featureless continua, and offered as a library it fits a
    spectrum with no stars in it at all.
    """
    if root is None:
        from brainsp import libraries
        return libraries.available()
    found = []
    for name in sorted(os.listdir(root)):
        folder = os.path.join(root, name)
        if not os.path.isfile(os.path.join(folder, CATALOGUE_NAME)):
            continue
        if os.path.abspath(folder) == os.path.abspath(AGN_FOLDER):
            continue
        try:
            ssp, _ = read_catalogue(folder)
        except SystemExit:
            continue
        if ssp:
            found.append(name)
    return found




def _match(requested, offered, label, folder, unit=""):
    fmt = (lambda v: f"{v:.4g}") if unit else (lambda v: f"{v:.4f}")
    hits = [o for o in offered
            if abs(requested - o) <= MATCH_TOLERANCE * max(abs(o), 1e-30)]
    if len(hits) == 1:
        return hits[0]
    if len(hits) > 1:
        raise SystemExit(
            f"\n{label} {fmt(requested)}{unit} is ambiguous — it matches "
            f"{len(hits)} catalogue values: "
            f"{', '.join(fmt(v) + unit for v in hits)}.\n"
            f"  Write it more precisely to say which you mean.\n")
    near = sorted(offered, key=lambda o: abs(o - requested))[:3]
    raise SystemExit(
        f"\n{label} {fmt(requested)}{unit} is not in this template library.\n"
        f"  Nearest available: {', '.join(fmt(v) + unit for v in near)}\n"
        f"  Full list: {', '.join(fmt(v) for v in offered)}\n"
        f"  (see {os.path.join(folder, CATALOGUE_NAME)}, or run "
        f"python brain_settings.py)\n")


def _is_all(v):
    return isinstance(v, str) and v.strip().lower() == "all"


def _numlist(v, name):
    if v is None:
        raise SystemExit(f"\n'{name}' cannot be empty. Give a list of values, "
                         f"or the word  all.\n")
    seq = v if isinstance(v, (list, tuple)) else [v]
    out = []
    for x in seq:
        try:
            out.append(float(x))
        except (TypeError, ValueError):
            raise SystemExit(
                f"\n'{name}' contains {x!r}, which is not a number.\n"
                f"  Write ages as plain numbers, e.g.  [1e6, 1e7, 1e8]\n")
    return out

def build_base_list(s):
    """Turn ages / metallicities / include_power_law into a template list.

    `base_folder` may name a library ("BasesGM") or give a folder; names are
    looked up by brainsp.libraries.path.
    """
    from brainsp import libraries
    folder = libraries.path(s["base_folder"])
    ssp, extra = read_catalogue(folder)
    all_ages = sorted({e["age_gyr"] * 1e9 for e in ssp})
    all_zs = sorted({e["metallicity"] for e in ssp})

    want_ages = all_ages if _is_all(s["ages"]) else sorted(
        {_match(a, all_ages, "Age", folder, " yr")
         for a in _numlist(s["ages"], "ages")})
    want_zs = all_zs if _is_all(s["metallicities"]) else sorted(
        {_match(z, all_zs, "Metallicity Z =", folder)
         for z in _numlist(s["metallicities"], "metallicities")})

    chosen = [e for e in ssp
              if e["age_gyr"] * 1e9 in want_ages and e["metallicity"] in want_zs]
    if not chosen:
        raise SystemExit(
            f"\nYour 'ages' and 'metallicities' select no templates at all "
            f"({len(want_ages)} ages x {len(want_zs)} metallicities requested, "
            f"none of those combinations exist).\n")

    expected = len(want_ages) * len(want_zs)
    if len(chosen) < expected:
        print(f"  note: {expected - len(chosen)} of {expected} requested "
              f"(age, Z) combinations do not exist in this library; "
              f"fitting with the {len(chosen)} that do.")

    if s["include_power_law"]:
        # The AGN set: the library's own (<library>/AGN), when it ships one,
        # else the package's seven. Featureless components listed in the
        # library's main catalogue are left out either way: BasesGM's
        # PowerOpt_150 is the same curve as FC1.50 to 5e-11, and fitting both
        # would split one component's light between two identical columns.
        agn_dir = libraries.agn_folder(folder)
        _, agn = read_catalogue(agn_dir)
        for e in agn:
            e["filename"] = os.path.join(agn_dir, e["filename"])
        chosen = chosen + agn

    chosen.sort(key=lambda e: (e["metallicity"], e["age_gyr"]))

    missing = [e["filename"] for e in chosen
               if not os.path.isfile(os.path.join(folder, e["filename"]))]
    if missing:
        raise SystemExit(
            f"\n{len(missing)} template file(s) named in the catalogue are "
            f"missing from {folder!r}:\n  {', '.join(missing[:5])}"
            f"{' ...' if len(missing) > 5 else ''}\n")
    return folder, chosen




def _line_group_problems(groups):
    """
    Everything wrong with a 'line_groups' setting, in words a user can act on.

    Checking only that it is a dict let three different mistakes through to
    the fit, where each one is silent:

      - A species name that does not exist. Group membership is looked up by
        exact species, so 'ONIII]' or the '[NII]' spelling that the settings
        file's own example used to show simply never matches, and every line
        of that species quietly stays in the default group.
      - A species listed in two groups. assign_kinematic_groups raises on
        this, but only once a fit is already running.
      - Group labels that cannot be put in order, which is how the parameter
        vector is laid out, so the failure is a TypeError from inside the
        optimizer rather than a sentence about the settings file.

    A label may be any value, and need not be contiguous or start at zero;
    the labels only have to be orderable among themselves, because
    emission_lines.group_positions sorts them to fix each group's slot in
    the parameter vector.
    """
    if groups is None:
        return []

    shape = ("'line_groups' must be blank, or a mapping of group label to a "
             "list of species, like {0: ['Halpha', 'Hbeta'], 1: ['[O III]']}.")
    if not isinstance(groups, dict):
        return [shape]
    if not groups:
        return [f"'line_groups' is empty. {shape}"]

    from brainsp.line_catalogue import LINE_SPECIES

    known = sorted(set(LINE_SPECIES.values()))
    problems = []

    try:
        sorted(groups)
    except TypeError:
        problems.append(
            f"'line_groups' labels {list(groups)!r} cannot be put in order, "
            f"and BRAIN orders them to decide which pair of fitted gas "
            f"parameters belongs to which group. Use labels of one type.")

    seen = {}
    for label, species_list in groups.items():
        if isinstance(species_list, str) or not isinstance(
                species_list, (list, tuple)):
            problems.append(
                f"'line_groups' group {label!r} is {species_list!r}; it must "
                f"be a list of species names.")
            continue
        if not species_list:
            problems.append(
                f"'line_groups' group {label!r} is empty; a group with no "
                f"species costs two fitted parameters and constrains nothing.")
            continue
        for sp in species_list:
            if not isinstance(sp, str):
                problems.append(
                    f"'line_groups' group {label!r} lists {sp!r}, which is "
                    f"not a species name.")
                continue
            if sp not in known:
                hint = _closest(sp, known)
                problems.append(
                    f"'line_groups' group {label!r} lists the species {sp!r}, "
                    f"which is not one of BRAIN's emission-line species"
                    + (f". Did you mean {hint!r}?" if hint else
                       f": {', '.join(known)}"))
                continue
            if sp in seen and seen[sp] != label:
                problems.append(
                    f"'line_groups' puts the species {sp!r} in group "
                    f"{seen[sp]!r} and again in group {label!r}. A species "
                    f"has one velocity and one dispersion, so it belongs to "
                    f"exactly one group.")
                continue
            seen[sp] = label

    return problems


def _wavelength_scale_problems(s):
    """
    Refuse an unset 'wavelength_scale' when, and only when, it changes a number.

    The setting describes the OBSERVED spectrum. It is required when emission
    lines are actually being fitted, which is 'fit_emission_lines' true AND a
    'nebular_mode' that fits lines, and it is not required otherwise. In
    'spop' the emission-line catalogue is never built, no rest wavelength is
    ever read, and demanding a declaration that cannot reach a number would
    be noise in the one mode most people run first.

    It is refused rather than defaulted because this is the shape of error
    this project keeps finding: a wrong answer here runs to completion and
    produces a plausible fit. The two scales differ by about 83 km/s, the
    difference is very nearly constant in velocity across the optical, so it
    is absorbed almost entirely into the fitted gas velocity; chi2 moves by
    under 0.05 per cent, with a sign that depends on the spectrum. Nothing in
    the output looks wrong. The only symptom is a gas-to-stars velocity
    offset of 83 km/s that was never there.

    The message says how to answer, because "air or vacuum?" is a question
    about the survey, not about the galaxy, and the answer is a fact the user
    can look up rather than a judgement they have to make.
    """
    from brainsp.config import LINE_MODES

    if not (s["fit_emission_lines"] and s["nebular_mode"] in LINE_MODES):
        return []
    if s["wavelength_scale"] is not None:
        return []

    return [
        "'wavelength_scale' is not set, and this fit needs it: "
        f"'nebular_mode' is {s['nebular_mode']!r} and 'fit_emission_lines' "
        "is true, so BRAIN is about to place emission lines in your "
        "spectrum and has to know which wavelength scale your spectrum is "
        "on. Set it to 'air' or 'vacuum'.\n"
        "     BRAIN will not guess, because a wrong guess is invisible. "
        "Air and vacuum wavelengths differ by about 83 km/s across the "
        "optical (85.2 km/s at [O II] 3726, 82.7 at [S II] 6731), and the "
        "offset is very nearly constant in velocity, so it lands almost "
        "entirely on the fitted gas velocity. chi2 changes by less than "
        "0.05 per cent, with a sign that depends on the spectrum. The fit "

def validate(s, path=DEFAULT_SETTINGS_FILE, for_cube=False):
    p = []

    def _rng(name):
        v = s[name]
        if not isinstance(v, (list, tuple)) or len(v) != 2:
            p.append(f"'{name}' must be two numbers, e.g. [0.0, 5.0]; "
                     f"got {v!r}.")
            return None
        try:
            lo, hi = float(v[0]), float(v[1])
        except (TypeError, ValueError):
            p.append(f"'{name}' must be two numbers; got {v!r}.")
            return None
        if lo > hi:
            p.append(f"'{name}' is [{lo}, {hi}] — the first value must be "
                     f"the smaller one.")
            return None
        return lo, hi

    # A cube run takes its data from the cube, so 'spectrum' is not asked
    # for there: a settings file written for a datacube has none to give.
    if for_cube:
        pass
    elif not s["spectrum"]:
        p.append("'spectrum' must name a file to fit.")
    elif not any(c in str(s["spectrum"]) for c in "*?") \
            and not os.path.exists(str(s["spectrum"])):
        p.append(f"'spectrum' file not found: {s['spectrum']!r}")

    if s["mask"] and not os.path.exists(str(s["mask"])):
        p.append(f"'mask' file not found: {s['mask']!r} "
                 f"(leave it blank for no mask)")

    try:
        from brainsp import libraries
        libraries.path(s["base_folder"])
    except SystemExit:
        from brainsp import libraries
        p.append(f"'base_folder' not found: {s['base_folder']!r}. Available "
                 f"libraries: {', '.join(libraries.available()) or 'none'}")

    from brainsp.reddening_catalogue import LAW_NAMES
    from brainsp.line_catalogue import WAVELENGTH_SCALES
    from brainsp.config import FIT_MODES, LINE_MODES
    for name, allowed in (("reddening_law", LAW_NAMES),
                          ("nebular_mode", FIT_MODES),
                          ("ssp_flux_unit", ("Lsun", "cgs")),
                          ("linear_solver", ("active_set", "fista")),
                          ("cube_fitter", ("fast", "reference")),
                          ("missing_error", ("exclude", "interpolate")),
                          ("outlier_scale", ("error", "typical")),
                          ("device", ("auto", "cpu", "gpu")),
                          ("wavelength_scale", WAVELENGTH_SCALES)):
        if name == "wavelength_scale" and s[name] is None:
            continue
        if s[name] not in allowed:
            p.append(f"'{name}' is {s[name]!r}; must be one of "
                     f"{', '.join(repr(a) for a in allowed)}.")

    p.extend(_wavelength_scale_problems(s))

    workers = s.get("cpu_workers", "auto")
    if str(workers).lower() != "auto" and not (
            isinstance(workers, int) and not isinstance(workers, bool)
            and workers >= 1):
        p.append(f"'cpu_workers' is {workers!r}; it is 'auto' or a whole "
                 f"number of worker processes, at least 1.")

    if s["smoothness_order"] not in (1, 2):
        p.append(f"'smoothness_order' is {s['smoothness_order']!r}; "
                 f"must be 1 or 2.")

    if not s["fit_emission_lines"] and s["nebular_mode"] == "espop":
        p.append(
            "'nebular_mode' is 'espop' but 'fit_emission_lines' is false. "
            "espop is spop plus the emission lines; with the lines off it "
            "is spop by a longer name. Set fit_emission_lines: true, or "
            "nebular_mode: spop.")

    if not s["fit_emission_lines"] and s["nebular_mode"] == "photopop":
        p.append(
            "'fit_emission_lines' is false but 'nebular_mode' is "
            "'photopop'. photopop derives the ionizing photon budget Q(H) "
            "from the fitted hydrogen line amplitudes, so it cannot run "
            "without them. Use nebular_mode 'nepop' to fit the nebular "
            "continuum with no lines, or 'spop' for a purely stellar fit.")

    ranges = {}
    for name in ("dust_range", "young_extra_dust_range", "velocity_range",
                 "dispersion_range", "gas_velocity_range",
                 "gas_dispersion_range", "nebular_dust_range"):
        r = _rng(name)
        if r:
            ranges[name] = r

    for start, rng in (("av_start", "dust_range"),
                       ("dispersion_start", "dispersion_range"),
                       ("gas_dispersion_start", "gas_dispersion_range"),
                       ("nebular_dust_start", "nebular_dust_range")):
        if rng in ranges and not (ranges[rng][0] <= s[start] <= ranges[rng][1]):
            p.append(f"'{start}' = {s[start]} lies outside "
                     f"'{rng}' {list(ranges[rng])}.")

    wmin, wmax = s["wavelength_min"], s["wavelength_max"]
    if wmin is not None and wmax is not None:
        if wmin >= wmax:
            p.append("'wavelength_min' must be below 'wavelength_max'.")
        elif not (wmin <= s["normalisation_wavelength"] <= wmax):
            p.append(f"'normalisation_wavelength' "
                     f"({s['normalisation_wavelength']}) must lie inside the "
                     f"fitting range ({wmin}-{wmax}).")

    for name, lo in (("smoothness", 0), ("error_samples", 0),
                     ("nebular_iterations", 1), ("max_iterations", 1),
                     ("gas_temperature", 1), ("gas_density", 0)):
        try:
            if float(s[name]) < lo:
                p.append(f"'{name}' cannot be below {lo}; got {s[name]}.")
        except (TypeError, ValueError):
            p.append(f"'{name}' must be a number; got {s[name]!r}.")

    from brainsp.instrumental_lsf import parse_resolution
    for name in ("instrumental_resolution", "library_resolution"):
        try:
            parse_resolution(s[name], name)
        except SystemExit as exc:
            p.append(str(exc).strip().replace("\n", " "))

    try:
        z = float(s["spectrum_redshift"] or 0.0)
        if z <= -1.0:
            p.append(f"'spectrum_redshift' is {z}; a redshift cannot be at or "
                     f"below -1.")
    except (TypeError, ValueError):
        p.append(f"'spectrum_redshift' must be a number; got "
                 f"{s['spectrum_redshift']!r}.")

    try:
        zc = float(s["cube_redshift"] or 0.0)
        if zc <= -1.0:
            p.append(f"'cube_redshift' is {zc}; a redshift cannot be at or "
                     f"below -1.")
        if zc != 0.0 and float(s["spectrum_redshift"] or 0.0) != 0.0:
            p.append(
                f"'cube_redshift' is {zc} and 'spectrum_redshift' is "
                f"{s['spectrum_redshift']}, and they do opposite jobs. "
                f"'cube_redshift' asks BRAIN to shift a cube into the rest "
                f"frame; 'spectrum_redshift' tells BRAIN a shift has already "
                f"been applied. Setting both describes a cube that is "
                f"de-redshifted twice. Set whichever one is true and leave "
                f"the other at 0.")
    except (TypeError, ValueError):
        p.append(f"'cube_redshift' must be a number; got "
                 f"{s['cube_redshift']!r}.")

    try:
        ebv = float(s["galactic_ebv"] or 0.0)
        if ebv < 0.0:
            p.append(f"'galactic_ebv' is {ebv}; a colour excess removed before "
                     f"fitting cannot be negative.")
    except (TypeError, ValueError):
        p.append(f"'galactic_ebv' must be a number; got {s['galactic_ebv']!r}.")
    try:
        if float(s["galactic_rv"]) <= 0.0:
            p.append(f"'galactic_rv' must be positive; got {s['galactic_rv']!r}.")
    except (TypeError, ValueError):
        p.append(f"'galactic_rv' must be a number; got {s['galactic_rv']!r}.")
    if str(s["galactic_law"]).upper() not in LAW_NAMES:
        p.append(f"'galactic_law' {s['galactic_law']!r} is not a law BRAIN "
                 f"knows; choose one of: {', '.join(LAW_NAMES)}.")
    if not isinstance(s["spectrum_in_rest_frame"], bool):
        p.append(f"'spectrum_in_rest_frame' must be true or false; got "
                 f"{s['spectrum_in_rest_frame']!r}.")

    p.extend(_line_group_problems(s["line_groups"]))

    if p:
        raise SystemExit(f"\nPlease fix these settings in {path}:\n\n"
                         + "\n".join(f"  * {x}" for x in p) + "\n")
    return s

def to_config(s, spectrum_file=None):
    """
    Translate validated settings into the object BRAIN runs on.

    The two output paths are named here rather than left at the bare file
    names `NewBrainConfig` declares. Those names have no folder in them, so a
    caller that builds a config and fits without overriding them writes its
    result into whatever directory the process happened to start in, which
    for anyone running from a checkout is the source tree. Every caller that
    cared already overrode them; the ones that did not were the ones leaving
    files in the repository.
    """
    from brainsp.config import NewBrainConfig, LINE_MODES

    folder, entries = build_base_list(s)
    spectrum = spectrum_file or s.get("spectrum")
    stem = os.path.splitext(os.path.basename(str(spectrum)))[0] if spectrum \
        else "brain_fit"
    out_dir = str(s.get("output_folder") or DEFAULTS["output_folder"])
    return NewBrainConfig(
        obs_spectrum_file=spectrum_file or s["spectrum"],
        output_summary=os.path.join(out_dir, f"{stem}_summary.txt"),
        output_spectrum=os.path.join(out_dir, f"{stem}_fitted.txt"),

def describe_templates(folder):
    from brainsp import libraries
    folder = libraries.path(folder)
    ssp, extra = read_catalogue(folder)
    ages = sorted({e["age_gyr"] * 1e9 for e in ssp})
    zs = sorted({e["metallicity"] for e in ssp})
    grid = {(round(e["age_gyr"] * 1e9, 6), e["metallicity"]) for e in ssp}
    out = [f"Template library: {folder}",
           f"  {len(ssp)} stellar templates"
           + (f" + {len(extra)} extra component(s)" if extra else ""), "",
           "  AGES available (years) — copy into 'ages':"]
    for i in range(0, len(ages), 5):
        out.append("    " + ", ".join(f"{a:>9.4g}" for a in ages[i:i + 5]))
    out += ["", "  METALLICITIES available — copy into 'metallicities':",
            "    " + ", ".join(f"{z:.4f}" for z in zs)
            + "    (0.0190 = solar)", "",
            "  Coverage  (X = template exists):",
            "      age_yr  " + "".join(f"{z:>9.4f}" for z in zs)]
    for a in ages:
        out.append(f"    {a:>9.4g}  " + "".join(
            "        X" if (round(a, 6), z) in grid else "        ."
            for z in zs))
    if extra:
        out += ["", "  Extra components (include_power_law: true):"]
        out += [f"    {e['short_name']:<14} {e['filename']}" for e in extra]
    return "\n".join(out)
