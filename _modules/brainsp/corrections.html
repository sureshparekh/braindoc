"""
BRAIN: corrections applied to a spectrum before it is fitted.

Two, in the order the light met them on its way to the telescope, undone in
reverse:

1. Galactic (Milky Way) reddening. The foreground dust acts on the light as
   it arrives, so it is removed at OBSERVED wavelengths: the flux is
   multiplied by 10^(0.4 A_lambda), with A_lambda = R_V E(B-V) q(lambda) and
   q = A_lambda / A_V from BRAIN's law registry. The error spectrum is scaled
   by the same factor, so the signal-to-noise of every pixel is unchanged.
   It is the datacube's dereddening stage applied to a single spectrum.

2. Redshift. When the spectrum is still in the observed frame
   (spectrum_in_rest_frame: false), the wavelength axis is divided by 1 + z,
   as for a datacube. The flux is not rescaled by (1 + z): BRAIN fits a
   normalised spectrum and a cube is not rescaled either, so doing it here
   would make a spectrum and a cube of the same galaxy disagree.

The galaxy's own dust is NOT corrected here. It is a fitted parameter (A_V),
and removing any of it beforehand would be undone or double-counted by the
fit.

A spectrum handed over already in the rest frame (the default) still has its
Galactic reddening removed at the observed wavelength, lambda_rest (1 + z),
which is why spectrum_redshift matters to this correction either way.
"""

def galactic_factor(wavelength_observed, ebv, rv=3.1, law="CCM"):
    """10^(0.4 A_lambda) at the observed wavelengths: the dereddening factor."""
    if not ebv:
        return np.ones_like(np.asarray(wavelength_observed, dtype=np.float64))
    import jax
    jax.config.update("jax_enable_x64", True)
    from brainsp.reddening_laws import get_k_lambda
    q = np.asarray(get_k_lambda(np.asarray(wavelength_observed, dtype=np.float64),
                                str(law).upper(), rv=float(rv)), dtype=np.float64)
    return 10.0 ** (0.4 * float(rv) * float(ebv) * q)

def apply_pre_fit_corrections(wavelength, flux, error, cfg):
    """
    The spectrum with Galactic reddening and, if asked, the redshift removed.

    Returns (wavelength, flux, error, notes). `error` may be None. `notes` are
    sentences for the log saying what was done, empty when nothing was.
    """
    z = float(getattr(cfg, "spectrum_redshift", 0.0) or 0.0)
    rest = bool(getattr(cfg, "spectrum_in_rest_frame", True))
    ebv = float(getattr(cfg, "galactic_ebv", 0.0) or 0.0)
    rv = float(getattr(cfg, "galactic_rv", 3.1) or 3.1)
    law = str(getattr(cfg, "galactic_law", "CCM") or "CCM")

    wavelength = np.asarray(wavelength, dtype=np.float64)
    flux = np.asarray(flux, dtype=np.float64)
    notes = []

    observed = wavelength * (1.0 + z) if rest else wavelength
    if ebv:
        factor = galactic_factor(observed, ebv, rv, law)
        flux = flux * factor
        if error is not None:
            error = np.asarray(error, dtype=np.float64) * factor
        notes.append(
            f"Galactic reddening removed: E(B-V) = {ebv:g}, R_V = {rv:g}, "
            f"{law}; flux and errors scaled by {factor.min():.4f} to "
            f"{factor.max():.4f} at observed wavelengths "
            f"{observed[0]:.0f}-{observed[-1]:.0f} A")

    if not rest and z:
        wavelength = wavelength / (1.0 + z)
        notes.append(
            f"Rest frame: wavelengths divided by 1+z with z = {z:g} "
            f"({observed[0]:.0f}-{observed[-1]:.0f} A observed is now "
            f"{wavelength[0]:.0f}-{wavelength[-1]:.0f} A rest)")
    return wavelength, flux, error, notes
